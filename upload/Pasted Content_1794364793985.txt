/* ---------- bake: authored keys -> 32 frames / attack ---------- */
const EASE = {
  lin: t=>t,
  io:  t=>t*t*(3-2*t),
  in:  t=>Math.pow(t,2.6),
  out: t=>1-Math.pow(1-t,2.4)
};
const NA = ATTACKS.length, N = NA*FR;
const B = new Float32Array(N*NP);
const KEYF = new Uint8Array(N);
const WGT = new Float32Array(N).fill(1);
const XCH = new Set([I.x,I.bfx,I.ffx]);
const EVENTS = [];
const INVSET = new Set();

(function bake(){
  let offX = 0;
  ATTACKS.forEach((A,ai)=>{
    const full = [];
    let prev = null;
    A.keys.forEach(k=>{
      const o = prev ? Object.assign({}, prev, k.o) : Object.assign({}, k.o);
      full.push({f:k.f,e:k.e,v:o}); prev = o;
    });
    for(let fr=0; fr<FR; fr++){
      let s=0; while(s<full.length-2 && full[s+1].f<=fr) s++;
      const a=full[s], b=full[s+1];
      let t = (fr-a.f)/(b.f-a.f); t = Math.max(0,Math.min(1,t));
      t = EASE[b.e||'lin'](t);
      const gi=(ai*FR+fr)*NP;
      for(let k=0;k<NP;k++){
        const n=PN[k];
        let v = a.v[n] + (b.v[n]-a.v[n])*t;
        if(XCH.has(k)) v += offX;
        B[gi+k]=v;
      }
      if(full.some(q=>q.f===fr)) KEYF[ai*FR+fr]=1;
    }
    A.keyCount = full.length-1;
    offX += full[full.length-1].v.x;
    Object.keys(A.slow||{}).forEach(f=>{ WGT[ai*FR+(+f)] = A.slow[f]; });
  });
})();

/* ---------- interpolator (Catmull-Rom across baked frames) ---------- */
function paramAt(f,k){
  if(f<0) f=0; if(f>N-1) f=N-1;
  const i=Math.floor(f), t=f-i;
  const p1=B[i*NP+k];
  if(t<1e-6) return p1;
  const p0=B[Math.max(i-1,0)*NP+k], p2=B[Math.min(i+1,N-1)*NP+k], p3=B[Math.min(i+2,N-1)*NP+k];
  const t2=t*t, t3=t2*t;
  return 0.5*((2*p1)+(-p0+p2)*t+(2*p0-5*p1+4*p2-p3)*t2+(-p0+3*p1-3*p2+p3)*t3);
}
function sampleP(f){ const o=new Array(NP); for(let k=0;k<NP;k++) o[k]=paramAt(f,k); return o; }

function arm(sh,u,f){
  const ua=u*RAD, fa=f*RAD;
  const e=[sh[0]+UA*Math.sin(ua), sh[1]+UA*Math.cos(ua)];
  const h=[e[0]+FA*Math.sin(fa), e[1]+FA*Math.cos(fa)];

function leg(hip,tx,ty){
  let vx=tx-hip[0], vy=ty-hip[1], d=Math.hypot(vx,vy);
  const mx=(TH+SH_)*0.9985;
  if(d>mx){ vx*=mx/d; vy*=mx/d; d=mx; }
  if(d<2){ d=2; }
  const mxp=hip[0]+vx/2, myp=hip[1]+vy/2;
  const h=Math.sqrt(Math.max(0,TH*TH-d*d/4));
  const p1=[-vy/d,vx/d], p2=[vy/d,-vx/d];
  const s1=p1[0]*1+p1[1]*-0.25, s2=p2[0]*1+p2[1]*-0.25;
  const pp = s1>s2?p1:p2;
  return {k:[mxp+pp[0]*h, myp+pp[1]*h], f:[hip[0]+vx, hip[1]+vy]};
}
function tailPts(f,hip,pwv){

}

    ctx.beginPath(); ctx.moveTo(mn,9); ctx.lineTo(mx,9); ctx.stroke();
    phys.unstable = (c[0]<mn-2 || c[0]>mx+2);
    phys.air=false;
  } else { phys.air=true; phys.unstable=false; }
  ctx.setLineDash([2,3]); ctx.strokeStyle='rgba(20,115,230,.7)'; ctx.lineWidth=1;
  ctx.beginPath(); ctx.moveTo(c[0],c[1]); ctx.lineTo(c[0],14); ctx.stroke(); ctx.setLineDash([]);
  ctx.fillStyle=phys.unstable?'#e5231b':'#1473e6';
  ctx.beginPath(); ctx.moveTo(c[0],11); ctx.lineTo(c[0]-5,19); ctx.lineTo(c[0]+5,19); ctx.closePath(); ctx.fill();
  ctx.strokeStyle='#1473e6'; ctx.lineWidth=1.6;
  ctx.beginPath(); ctx.arc(c[0],c[1],6,0,Math.PI*2); ctx.stroke();
  ctx.beginPath(); ctx.moveTo(c[0]-9,c[1]); ctx.lineTo(c[0]+9,c[1]); ctx.moveTo(c[0],c[1]-9); ctx.lineTo(c[0],c[1]+9); ctx.stroke();
  ctx.restore();
}
function physicsVectors(dp,r,draw){
  const a=rigAt(dp-0.25,true), b=rigAt(dp+0.25,true);
  const vel=n=>[(b[n][0]-a[n][0])/0.5,(b[n][1]-a[n][1])/0.5];
  const cA=comOf(a), cB=comOf(b), cc=comOf(r);
  const vc=[(cB[0]-cA[0])/0.5,(cB[1]-cA[1])/0.5];
  const items=[
    {n:'lh',v:vel('lh'),pt:r.lh,col:'#111',s:2.6,min:2},
    {n:'bh',v:vel('bh'),pt:r.bh,col:'#111',s:2.6,min:2},
    {n:'tip',v:vel('tip'),pt:r.tip,col:'#0a9396',s:1.6,min:2.5},
    {n:'com',v:vc,pt:cc,col:'#1473e6',s:4,min:0.8}
  ];
  const labels=[];
  items.forEach(it=>{
    const sp=Math.hypot(it.v[0],it.v[1]), mps=sp*BASE_FPS/100;
    if(it.n==='lh') phys.vl=mps; if(it.n==='bh') phys.vb=mps; if(it.n==='tip') phys.vt=mps; if(it.n==='com') phys.vc=mps;
    if(!draw || mps<it.min) return;
    let len=Math.min(140,sp*it.s);
    const ux=it.v[0]/sp, uy=it.v[1]/sp;
    arrow(it.pt[0],it.pt[1],it.pt[0]+ux*len,it.pt[1]+uy*len,it.col,2);
    labels.push({x:it.pt[0]+ux*len,y:it.pt[1]+uy*len,t:mps.toFixed(1)+' m/s',c:it.col});
  });
  phys.vh=Math.max(phys.vl||0,phys.vb||0);
  phys.ke=0.5*2.4*phys.vh*phys.vh;
  const cm=comOf(rigAt(dp-1,false)), cp=comOf(rigAt(dp+1,false));
  const ax=cp[0]-2*cc[0]+cm[0], ay=cp[1]-2*cc[1]+cm[1];
  phys.aCom=Math.hypot(ax,ay)*BASE_FPS*BASE_FPS/100;
  phys.ay=ay*BASE_FPS*BASE_FPS/100;
  if(draw && phys.air){
    arrow(cc[0]+16,cc[1],cc[0]+16,cc[1]+28,'#e5231b',1.6);
    labels.push({x:cc[0]+16,y:cc[1]+36,t:'g 9.8',c:'#e5231b'});
  }
  return labels;
}

/* ---------- camera ---------- */
function camera(dp){
  let sx=0,w=0,hyA=0;
  for(let k=0;k<=8;k++){
    const ww=1-k*0.09, f=dp-k*1.5;
    sx+=paramAt(f,I.x)*ww; hyA+=paramAt(f,I.hy)*ww; w+=ww;
  }
  const cx=sx/w+28, hy=hyA/w;
  const camY=-100+Math.min(0,(-hy+75))*0.55;
  let zoom=1-clamp((hy-85)/420,0,0.17);
  let shx=0,shy=0;
  EVENTS.forEach((e,i)=>{
    const age=dp-e.gf;
    if(age>=0&&age<14){
      zoom+=e.p*0.06*Math.exp(-age/4);
      const amp=e.p*6*Math.exp(-age/3);
      shx+=amp*Math.sin(age*4.1+i*2); shy+=amp*Math.cos(age*5.3+i);
    }
  });
  return {cx,camY,zoom,shx,shy};
}

/* ---------- loop ---------- */
function getDisplayP(){
  if(sampling===60) return p;
  if(!playing) return Math.floor(p);
  const step=BASE_FPS/sampling;
  return Math.floor(p/step)*step;
}
function loop(now){
  const raw=(now-lastT)/1000; lastT=now;
  if(raw>0){ hzAcc.push(1/raw); if(hzAcc.length>30) hzAcc.shift(); hz=hzAcc.reduce((a,b)=>a+b,0)/hzAcc.length; }
  const dt=Math.min(0.1,Math.max(0,raw));
  if(playing){
    const w=WGT[Math.min(N-1,Math.floor(p))];
    p+=dt*BASE_FPS*speed/w;
    if(p>=N) p-=N;
  }
  render(getDisplayP());
  requestAnimationFrame(loop);
}
