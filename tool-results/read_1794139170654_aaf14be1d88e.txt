    35→    35→ * fighter (H ≈ 104 local units ≈ 137px at SCALE 1.32). Every joint offset
    36→    36→ * below derives from this table; nothing is placed by eye:
    37→    37→ *
    38→    38→ *   region            fraction    local units
    39→    39→ *   ──────────────────────────────────────────────
    40→    40→ *   skull height      1.28 head   17.6 (r 8.8 — stylized readability)
    41→    41→ *   neck column       0.35 head    4.6 (C7 → skull base)
    42→    42→ *   hip→shoulder line 0.355 H     37   (waist split 0.57/0.43)
    43→    43→ *   femur head offset             ±~2  (front hip +1.7, rear −2.1)
    44→    44→ *   glenohumeral jt   0.031 H      3.2 fwd × 3.3 drop (near shoulder)
    45→    45→ *   humerus           0.20 H      21
    46→    46→ *   forearm           0.18 H      19   (humerus:forearm = 1.1:1 ✓)
    47→    47→ *   femur             0.24 H      25
    48→    48→ *   tibia             0.23 H      24   (femur:tibia = 1.04:1 ✓)
    49→    49→ *   foot length       0.10 H      10
    50→    50→ *   blade 56 ≈ 54% of height + tsuka ≈ 70% — legit long-katana read
    51→    51→ *
    52→    52→ * Leg chain 49 keeps HIP_H = 64 world EXACTLY stable (ground contact,

    99→    99→export function makeSkelGeom(): SkelGeom {
   100→   100→  const z = (): Vec2 => ({ x: 0, y: 0 })
   101→   101→  return {
   102→   102→    hip: z(), hipPtR: z(), hipPtL: z(), waist: z(), chest: z(), neckTop: z(),
   103→   103→    headC: z(), headAng: 0, shoulderR: z(), shoulderL: z(),
   104→   104→    kneeL: z(), ankleL: z(), toeL: z(),
   105→   105→    kneeR: z(), ankleR: z(), toeR: z(),
   106→   106→    elbL: z(), handL: z(), elbR: z(), handR: z(),
   107→   107→    bladeAngle: 0, torsoAngle: 0,
   108→   108→  }
   109→   109→}
   110→   110→
   111→   111→/** Solve joint positions. Reuses `out`.
   112→   112→ *
   113→   113→ * SPINE CHAIN (four articulated regions, parented in order):
   114→   114→ *   hip → waist (pelvis/abdomen, 57% along `rot`) → chest (ribcage +
   115→   115→ *   shoulder line, 43% along `rot + spine`) → neck column (along the head
   116→   116→ *   channel) → skull. The waist joint makes the torso BEND instead of
   117→   117→ *   shear — the lumbar and thoracic curves separate.
   118→   118→ *
   119→   119→ * SHOULDER GIRDLE (the arms-from-the-neck fix): the glenohumeral joints
   120→   120→ * solve as chest + (shoulderFwd, shoulderDrop) in the chest's frame —
   121→   121→ * below the shoulder line and slightly anterior, like a real skeleton.
   122→   122→ * The far shoulder sits behind and lower (2.8D depth read). Arms chain
   123→   123→ * from THESE, never from the spine top.
   124→   124→ *
   125→   125→ * LEGS chain from offset femur-head joints (front hip slightly anterior,
   126→   126→ * rear hip behind) — the pelvis reads as a real basin, and stride scissor
   127→   127→ * from two distinct roots. */
   128→   128→export function solveSkeleton(pose: Pose, spec: SkelSpec, out: SkelGeom): SkelGeom {
   129→   129→  const rot = pose[2]
   130→   130→  const spine = pose[3]
   131→   131→  const head = pose[4]
   132→   132→
   133→   133→  // spine: hip → waist → chest
   134→   134→  out.hip.x = pose[0]
   135→   135→  out.hip.y = pose[1]
   136→   136→  const t1 = rot
   137→   137→  out.waist.x = out.hip.x + Math.sin(t1) * spec.torso * 0.57
   138→   138→  out.waist.y = out.hip.y - Math.cos(t1) * spec.torso * 0.57
   139→   139→  const t2 = rot + spine
   140→   140→  out.chest.x = out.waist.x + Math.sin(t2) * spec.torso * 0.43
   141→   141→  out.chest.y = out.waist.y - Math.cos(t2) * spec.torso * 0.43
   142→   142→  out.torsoAngle = t2
   143→   143→
   144→   144→  // neck column + skull — a REAL neck: full spec.neck of visible column
   145→   145→  // between the shoulder line and the skull base
   146→   146→  const ha = t2 + head
   147→   147→  out.neckTop.x = out.chest.x + Math.sin(ha) * spec.neck
   148→   148→  out.neckTop.y = out.chest.y - Math.cos(ha) * spec.neck
   149→   149→  out.headC.x = out.neckTop.x + Math.sin(ha) * spec.headR * 0.9
   150→   150→  out.headC.y = out.neckTop.y - Math.cos(ha) * spec.headR * 0.9
   151→   151→  out.headAng = ha
   152→   152→
   153→   153→  // ---- SHOULDER GIRDLE: the arm roots ----
   154→   154→  // fwd(t) = (cos t, sin t); down-along-torso = (−sin t, cos t)
   155→   155→  const sf = spec.shoulderFwd
   156→   156→  const sd = spec.shoulderDrop
   157→   157→  out.shoulderR.x = out.chest.x + Math.cos(t2) * sf - Math.sin(t2) * sd
   158→   158→  out.shoulderR.y = out.chest.y + Math.sin(t2) * sf + Math.cos(t2) * sd
   159→   159→  const sfL = -sf * 0.62
   160→   160→  const sdL = sd * 1.18
   161→   161→  out.shoulderL.x = out.chest.x + Math.cos(t2) * sfL - Math.sin(t2) * sdL
   162→   162→  out.shoulderL.y = out.chest.y + Math.sin(t2) * sfL + Math.cos(t2) * sdL
   163→   163→
   164→   164→  // ---- femur heads ----
   165→   165→  out.hipPtR.x = out.hip.x + Math.cos(t1) * 1.7
   166→   166→  out.hipPtR.y = out.hip.y + Math.sin(t1) * 1.7
   167→   167→  out.hipPtL.x = out.hip.x + Math.cos(t1) * -2.1
   168→   168→  out.hipPtL.y = out.hip.y + Math.sin(t1) * -2.1
   169→   169→
   170→   170→  // legs (angle from straight-down, positive = forward)
   171→   171→  const hl = pose[5]
   172→   172→  const kl = pose[6]
   173→   173→  const fl = pose[7]
   174→   174→  out.kneeL.x = out.hipPtL.x + Math.sin(hl) * spec.thigh
   175→   175→  out.kneeL.y = out.hipPtL.y + Math.cos(hl) * spec.thigh
   176→   176→  const al = hl + kl
   177→   177→  out.ankleL.x = out.kneeL.x + Math.sin(al) * spec.shin
   178→   178→  out.ankleL.y = out.kneeL.y + Math.cos(al) * spec.shin
   179→   179→  // foot roll (Williams, Animator's Survival Kit): the toe defaults to FLAT
   180→   180→  // on the ground line; the foot channel is heel/toe flex (±) with a whisper
   181→   181→  // of shin influence so airborne tucks still point. Authoring foot angles
   182→   182→  // relative to the shin pointed the toe nearly straight down, planting
   183→   183→  // ~12px of rigid foot below the ground in every grounded pose.
   184→   184→  const tl = Math.PI / 2 + al * 0.35 + fl * 0.5
   185→   185→  out.toeL.x = out.ankleL.x + Math.sin(tl) * spec.foot
   186→   186→  out.toeL.y = out.ankleL.y + Math.cos(tl) * spec.foot
   187→   187→
   188→   188→  const hr = pose[8]
   189→   189→  const kr = pose[9]
   190→   190→  const fr = pose[10]
   191→   191→  out.kneeR.x = out.hipPtR.x + Math.sin(hr) * spec.thigh
   192→   192→  out.kneeR.y = out.hipPtR.y + Math.cos(hr) * spec.thigh
   193→   193→  const ar = hr + kr
   194→   194→  out.ankleR.x = out.kneeR.x + Math.sin(ar) * spec.shin
   195→   195→  out.ankleR.y = out.kneeR.y + Math.cos(ar) * spec.shin
   196→   196→  const tr = Math.PI / 2 + ar * 0.35 + fr * 0.5
   197→   197→  out.toeR.x = out.ankleR.x + Math.sin(tr) * spec.foot
   198→   198→  out.toeR.y = out.ankleR.y + Math.cos(tr) * spec.foot
   199→   199→
   200→   200→  // arms — FROM THE SHOULDERS (angle from straight-down, positive = forward)
   201→   201→  const sl = pose[11]
   202→   202→  const el = pose[12]
   203→   203→  out.elbL.x = out.shoulderL.x + Math.sin(sl) * spec.upperArm
   204→   204→  out.elbL.y = out.shoulderL.y + Math.cos(sl) * spec.upperArm
   205→   205→  const wl = sl + el
   206→   206→  out.handL.x = out.elbL.x + Math.sin(wl) * spec.foreArm
   207→   207→  out.handL.y = out.elbL.y + Math.cos(wl) * spec.foreArm
   208→   208→
   209→   209→  const sr = pose[14]
   210→   210→  const er = pose[15]
   211→   211→  out.elbR.x = out.shoulderR.x + Math.sin(sr) * spec.upperArm
   212→   212→  out.elbR.y = out.shoulderR.y + Math.cos(sr) * spec.upperArm
   213→   213→  const wr = sr + er
   214→   214→  out.handR.x = out.elbR.x + Math.sin(wr) * spec.foreArm
   215→   215→  out.handR.y = out.elbR.y + Math.cos(wr) * spec.foreArm
   216→   216→
   217→   217→  // blade: forearm direction rotated by wrist. forearm angle-from-+X = π/2 − (shR+elR);
   218→   218→  // positive wrist swings the tip toward facing (+X).
   219→   219→  out.bladeAngle = Math.PI / 2 - wr - pose[16]
   220→   220→  return out
   221→   221→}

   270→   270→function limb(ctx: CanvasRenderingContext2D, ax: number, ay: number, bx: number, by: number, cx: number, cy: number): void {
   271→   271→  ctx.beginPath()
   272→   272→  ctx.moveTo(ax, ay)
   273→   273→  ctx.lineTo(bx, by)
   274→   274→  ctx.lineTo(cx, cy)
   275→   275→  ctx.stroke()
   276→   276→}
   277→   277→
   278→   278→/** Fills a foot wedge (ankle → toe) with heel/toe widths. */
   279→   279→function fillFoot(
   280→   280→  ctx: CanvasRenderingContext2D,
   281→   281→  ax: number, ay: number, bx: number, by: number,
   282→   282→  color: string,
   283→   283→  alpha = 1,
   284→   284→): void {
   285→   285→  const dx = bx - ax
   286→   286→  const dy = by - ay
   287→   287→  const len = Math.hypot(dx, dy) || 1
   288→   288→  const nx = -dy / len
   289→   289→  const ny = dx / len
   290→   290→  ctx.save()
   291→   291→  ctx.globalAlpha *= alpha
   292→   292→  ctx.fillStyle = color
   293→   293→  ctx.beginPath()
   294→   294→  ctx.moveTo(ax + nx * 3.4, ay + ny * 3.4)
   295→   295→  ctx.lineTo(bx + nx * 2.5, by + ny * 2.5)
   296→   296→  ctx.lineTo(bx - nx * 2.5, by - ny * 2.5)
   297→   297→  ctx.lineTo(ax - nx * 3.4, ay - ny * 3.4)
   298→   298→  ctx.closePath()
   299→   299→  ctx.fill()
   300→   300→  ctx.beginPath()
   301→   301→  ctx.arc(ax, ay, 3.4, 0, Math.PI * 2)
   302→   302→  ctx.fill()
   303→   303→  ctx.beginPath()
   304→   304→  ctx.arc(bx, by, 2.3, 0, Math.PI * 2)
   305→   305→  ctx.fill()
   306→   306→  ctx.restore()
   307→   307→}

   309→   309→/**
   310→   310→ * Draws the stickman. The ctx is expected to be pre-translated to the hip
   311→   311→ * position, flipped horizontally for facing, and scaled — geom is local space.
   312→   312→ */
   313→   313→export function drawStick(
   314→   314→  ctx: CanvasRenderingContext2D,
   315→   315→  g: SkelGeom,
   316→   316→  style: StickStyle,
   317→   317→  opts: DrawOpts = {},
   318→   318→): void {
   319→   319→  const alpha = opts.alpha ?? 1
   320→   320→  if (alpha <= 0.01) return
   321→   321→  const t = opts.time ?? 0
   322→   322→  const ghost = opts.ghost
   323→   323→  const lineW = style.lineW
   324→   324→
   325→   325→  ctx.save()
   326→   326→  ctx.globalAlpha *= alpha
   327→   327→  ctx.lineCap = 'round'
   328→   328→  ctx.lineJoin = 'round'
   329→   329→
   330→   330→  if (ghost) {
   331→   331→    // ---- afterimage pass: flat, cheap, gorgeous ----
   332→   332→    ctx.strokeStyle = ghost
   333→   333→    ctx.lineWidth = lineW * 0.85
   334→   334→    const ga = opts.ghostAlpha ?? 1
   335→   335→    ctx.globalAlpha *= ga
   336→   336→    limb(ctx, g.hipPtL.x, g.hipPtL.y, g.kneeL.x, g.kneeL.y, g.ankleL.x, g.ankleL.y)
   337→   337→    limb(ctx, g.hipPtR.x, g.hipPtR.y, g.kneeR.x, g.kneeR.y, g.ankleR.x, g.ankleR.y)
   338→   338→    limb(ctx, g.shoulderL.x, g.shoulderL.y, g.elbL.x, g.elbL.y, g.handL.x, g.handL.y)
   339→   339→    limb(ctx, g.shoulderR.x, g.shoulderR.y, g.elbR.x, g.elbR.y, g.handR.x, g.handR.y)
   340→   340→    ctx.beginPath()
   341→   341→    ctx.moveTo(g.hip.x, g.hip.y)
   342→   342→    ctx.lineTo(g.waist.x, g.waist.y)
   343→   343→    ctx.lineTo(g.chest.x, g.chest.y)
   344→   344→    ctx.stroke()
   345→   345→    ctx.beginPath()
   346→   346→    ctx.arc(g.headC.x, g.headC.y, style === PLAYER_STYLE ? 8.8 : 8, 0, Math.PI * 2)
   347→   347→    ctx.stroke()
   348→   348→    ctx.restore()
   349→   349→    return
   350→   350→  }
   351→   351→
   352→   352→  // ---- full-black humanoid silhouette (player) ----
   353→   353→  if (style.silhouette) {
   354→   354→    drawSilhouette(ctx, g, style, opts, t, lineW)
   355→   355→    ctx.restore()
   356→   356→    return
   357→   357→  }
   358→   358→
   359→   359→  // ---- 1. outline pass (readability against the neon city) ----
   360→   360→  if (!opts.simple) {
   361→   361→    ctx.strokeStyle = style.outline
   362→   362→    ctx.lineWidth = lineW + 3.6
   363→   363→    limb(ctx, g.hipPtL.x, g.hipPtL.y, g.kneeL.x, g.kneeL.y, g.ankleL.x, g.ankleL.y)
   364→   364→    limb(ctx, g.hipPtR.x, g.hipPtR.y, g.kneeR.x, g.kneeR.y, g.ankleR.x, g.ankleR.y)
   365→   365→    limb(ctx, g.shoulderL.x, g.shoulderL.y, g.elbL.x, g.elbL.y, g.handL.x, g.handL.y)
   366→   366→    limb(ctx, g.shoulderR.x, g.shoulderR.y, g.elbR.x, g.elbR.y, g.handR.x, g.handR.y)
   367→   367→    ctx.beginPath()
   368→   368→    ctx.moveTo(g.hip.x, g.hip.y)
   369→   369→    ctx.lineTo(g.waist.x, g.waist.y)
   370→   370→    ctx.lineTo(g.chest.x, g.chest.y)
   371→   371→    ctx.stroke()
   372→   372→  }
   373→   373→
   374→   374→  // ---- 2. back limbs ----
   375→   375→  ctx.strokeStyle = style.dark
   376→   376→  ctx.lineWidth = lineW
   377→   377→  limb(ctx, g.hipPtL.x, g.hipPtL.y, g.kneeL.x, g.kneeL.y, g.ankleL.x, g.ankleL.y)
   378→   378→  ctx.strokeStyle = rgba(style.dark, 0.82)
   379→   379→  limb(ctx, g.shoulderL.x, g.shoulderL.y, g.elbL.x, g.elbL.y, g.handL.x, g.handL.y)
   380→   380→
   381→   381→  // ---- 3. torso (thicker, armored core) ----
   382→   382→  ctx.strokeStyle = style.body
   383→   383→  ctx.lineWidth = lineW * 1.42
   384→   384→  ctx.beginPath()
   385→   385→  ctx.moveTo(g.hip.x, g.hip.y)
   386→   386→  ctx.lineTo(g.waist.x, g.waist.y)
   387→   387→  ctx.lineTo(g.chest.x, g.chest.y)
   388→   388→  ctx.stroke()
   389→   389→
   390→   390→  // ---- 4. front limbs ----
   391→   391→  ctx.strokeStyle = style.body
   392→   392→  ctx.lineWidth = lineW
   393→   393→  limb(ctx, g.hipPtR.x, g.hipPtR.y, g.kneeR.x, g.kneeR.y, g.ankleR.x, g.ankleR.y)
   394→   394→  limb(ctx, g.shoulderR.x, g.shoulderR.y, g.elbR.x, g.elbR.y, g.handR.x, g.handR.y)
   395→   395→
   396→   396→  // ---- 5. feet ----
   397→   397→  ctx.strokeStyle = style.body
   398→   398→  ctx.lineWidth = lineW * 0.72
   399→   399→  ctx.beginPath()
   400→   400→  ctx.moveTo(g.ankleR.x, g.ankleR.y)
   401→   401→  ctx.lineTo(g.toeR.x, g.toeR.y)
   402→   402→  ctx.moveTo(g.ankleL.x, g.ankleL.y)
   403→   403→  ctx.lineTo(g.toeL.x, g.toeL.y)
   404→   404→  ctx.stroke()
   405→   405→
   406→   406→  // ---- 6. mechanical joints ----
   407→   407→  const jr = lineW * 0.52
   408→   408→  ctx.fillStyle = style.outline
   409→   409→  for (const j of [g.kneeL, g.kneeR, g.elbL, g.elbR]) {
   410→   410→    ctx.beginPath()
   411→   411→    ctx.arc(j.x, j.y, jr + 1.2, 0, Math.PI * 2)
   412→   412→    ctx.fill()
   413→   413→  }
   414→   414→  ctx.fillStyle = style.dark
   415→   415→  for (const j of [g.kneeL, g.kneeR, g.elbL, g.elbR]) {
   416→   416→    ctx.beginPath()
   417→   417→    ctx.arc(j.x, j.y, jr, 0, Math.PI * 2)
   418→   418→    ctx.fill()
   419→   419→  }
   420→   420→  if (!opts.simple) {
   421→   421→    ctx.fillStyle = rgba(style.glow, 0.85)
   422→   422→    for (const j of [g.kneeR, g.elbR]) {
   423→   423→      ctx.beginPath()
   424→   424→      ctx.arc(j.x, j.y, jr * 0.42, 0, Math.PI * 2)
   425→   425→      ctx.fill()
   426→   426→    }
   427→   427→  }
   428→   428→
   429→   429→  // ---- 7. head ----
   430→   430→  const hr = 8.8
   431→   431→  ctx.fillStyle = style.outline
   432→   432→  ctx.beginPath()
   433→   433→  ctx.arc(g.headC.x, g.headC.y, hr + 1.4, 0, Math.PI * 2)
   434→   434→  ctx.fill()
   435→   435→  ctx.fillStyle = style.dark
   436→   436→  ctx.beginPath()
   437→   437→  ctx.arc(g.headC.x, g.headC.y, hr, 0, Math.PI * 2)
   438→   438→  ctx.fill()
   439→   439→  // visor slit (facing side) — glowing
   440→   440→  const va = style.visorAlpha ?? 1
   441→   441→  const vx = g.headC.x + Math.cos(g.headAng) * 0
   442→   442→  const vy = g.headC.y
   443→   443→  const flick = 0.85 + 0.15 * Math.sin(t * 11.3)
   444→   444→  ctx.strokeStyle = rgba(style.glow, va * flick)
   445→   445→  ctx.lineWidth = 2.6
   446→   446→  ctx.beginPath()
   447→   447→  ctx.moveTo(vx + 1.5, vy - 3.1)
   448→   448→  ctx.lineTo(vx + hr - 0.6, vy - 2.2)
   449→   449→  ctx.stroke()
   450→   450→  if (!opts.simple) {
   451→   451→    // helmet crest
   452→   452→    ctx.strokeStyle = rgba(style.body, 0.9)
   453→   453→    ctx.lineWidth = 2.2
   454→   454→    ctx.beginPath()
   455→   455→    ctx.moveTo(g.headC.x - 4.5, g.headC.y - hr + 0.8)
   456→   456→    ctx.lineTo(g.headC.x - 2.5, g.headC.y - hr - 3.4)
   457→   457→    ctx.stroke()
   458→   458→  }
   459→   459→
   460→   460→  // ---- 8. cybernetic armor (player / elite / heavy) ----
   461→   461→  if (style.armor && !opts.simple) {
   462→   462→    const ca = Math.sin(t * 6.2) * 0.5 + 0.5
   463→   463→    // shoulder pauldron on sword arm
   464→   464→    const shx = g.chest.x
   465→   465→    const shy = g.chest.y
   466→   466→    ctx.fillStyle = rgba(style.body, 0.94)
   467→   467→    ctx.strokeStyle = style.outline
   468→   468→    ctx.lineWidth = 1.4
   469→   469→    ctx.beginPath()
   470→   470→    ctx.moveTo(shx - 2, shy - 3.4)
   471→   471→    ctx.lineTo(shx + 9.5, shy - 5.6)
   472→   472→    ctx.lineTo(shx + 11.5, shy + 2.2)
   473→   473→    ctx.lineTo(shx - 1, shy + 3.4)
   474→   474→    ctx.closePath()
   475→   475→    ctx.fill()
   476→   476→    ctx.stroke()
   477→   477→    // chest core (pulsing energy)
   478→   478→    if (style.core) {
   479→   479→      const cxp = (g.hip.x + g.chest.x) * 0.5
   480→   480→      const cyp = (g.hip.y + g.chest.y) * 0.5
   481→   481→      ctx.fillStyle = rgba(style.core, 0.55 + ca * 0.4)
   482→   482→      ctx.beginPath()
   483→   483→      ctx.arc(cxp, cyp, 3.1 + ca * 0.9, 0, Math.PI * 2)
   484→   484→      ctx.fill()
   485→   485→      ctx.strokeStyle = rgba(style.core, 0.9)
   486→   486→      ctx.lineWidth = 1
   487→   487→      ctx.beginPath()
   488→   488→      ctx.arc(cxp, cyp, 5.6 + ca * 1.4, 0, Math.PI * 2)
   489→   489→      ctx.stroke()
   490→   490→    }
   491→   491→    // thigh plate on front leg
   492→   492→    ctx.strokeStyle = rgba(style.body, 0.85)
   493→   493→    ctx.lineWidth = lineW * 0.5
   494→   494→    ctx.beginPath()
   495→   495→    ctx.moveTo((g.hip.x + g.kneeR.x) / 2, (g.hip.y + g.kneeR.y) / 2)
   496→   496→    ctx.lineTo((g.hip.x + g.kneeR.x) / 2 + 1.2, (g.hip.y + g.kneeR.y) / 2 - 2.6)
   497→   497→    ctx.stroke()
   498→   498→  }
   499→   499→
   500→   500→  // ---- 9. hit flash overlay ----
   501→   501→  const flash = clamp01(opts.flash ?? 0)
   502→   502→  if (flash > 0.02) {
   503→   503→    ctx.strokeStyle = rgba(opts.flashColor ?? '#ffffff', flash)
   504→   504→    ctx.lineWidth = lineW
   505→   505→    limb(ctx, g.hipPtL.x, g.hipPtL.y, g.kneeL.x, g.kneeL.y, g.ankleL.x, g.ankleL.y)
   506→   506→    limb(ctx, g.hipPtR.x, g.hipPtR.y, g.kneeR.x, g.kneeR.y, g.ankleR.x, g.ankleR.y)
   507→   507→    limb(ctx, g.shoulderL.x, g.shoulderL.y, g.elbL.x, g.elbL.y, g.handL.x, g.handL.y)
   508→   508→    limb(ctx, g.shoulderR.x, g.shoulderR.y, g.elbR.x, g.elbR.y, g.handR.x, g.handR.y)
   509→   509→    ctx.lineWidth = lineW * 1.4
   510→   510→    ctx.beginPath()
   511→   511→    ctx.moveTo(g.hip.x, g.hip.y)
   512→   512→    ctx.lineTo(g.waist.x, g.waist.y)
   513→   513→    ctx.lineTo(g.chest.x, g.chest.y)
   514→   514→    ctx.stroke()
   515→   515→  }
   516→   516→
   517→   517→  ctx.restore()
   518→   518→}

   524→   524→/** Fist: a short capsule perpendicular to the grip axis — knuckles read. */
   525→   525→function fillFist(
   526→   526→  ctx: CanvasRenderingContext2D,
   527→   527→  x: number, y: number,
   528→   528→  ax: number, ay: number, // grip axis (unit)
   529→   529→  size: number,
   530→   530→  color: string,
   531→   531→  alpha = 1,
   532→   532→): void {
   533→   533→  // perpendicular of grip axis
   534→   534→  const px = -ay
   535→   535→  const py = ax
   536→   536→  ctx.save()
   537→   537→  ctx.globalAlpha *= alpha
   538→   538→  ctx.fillStyle = color
   539→   539→  ctx.beginPath()
   540→   540→  ctx.moveTo(x + px * size, y + py * size)
   541→   541→  ctx.lineTo(x + ax * size * 0.55 + px * size * 0.8, y + ay * size * 0.55 + py * size * 0.8)
   542→   542→  ctx.lineTo(x + ax * size * 0.55 - px * size * 0.8, y + ay * size * 0.55 - py * size * 0.8)
   543→   543→  ctx.lineTo(x - px * size, y - py * size)
   544→   544→  ctx.closePath()
   545→   545→  ctx.fill()
   546→   546→  ctx.beginPath()
   547→   547→  ctx.arc(x, y, size * 0.92, 0, Math.PI * 2)
   548→   548→  ctx.fill()
   549→   549→  ctx.restore()
   550→   550→}
   551→   551→
   552→   552→/** Rounded wedge between two points with per-end widths — torso/lats. */
   553→   553→function fillTaper(
   554→   554→  ctx: CanvasRenderingContext2D,
   555→   555→  ax: number, ay: number, bx: number, by: number,
   556→   556→  wa: number, wb: number,
   557→   557→  color: string,
   558→   558→  alpha = 1,
   559→   559→): void {
   560→   560→  const dx = bx - ax
   561→   561→  const dy = by - ay
   562→   562→  const len = Math.hypot(dx, dy) || 1
   563→   563→  const nx = -dy / len
   564→   564→  const ny = dx / len
   565→   565→  ctx.save()
   566→   566→  ctx.globalAlpha *= alpha
   567→   567→  ctx.fillStyle = color
   568→   568→  ctx.beginPath()
   569→   569→  ctx.moveTo(ax + nx * wa, ay + ny * wa)
   570→   570→  ctx.lineTo(bx + nx * wb, by + ny * wb)
   571→   571→  ctx.lineTo(bx - nx * wb, by - ny * wb)
   572→   572→  ctx.lineTo(ax - nx * wa, ay - ny * wa)
   573→   573→  ctx.closePath()
   574→   574→  ctx.fill()
   575→   575→  ctx.beginPath()
   576→   576→  ctx.arc(ax, ay, wa, 0, Math.PI * 2)
   577→   577→  ctx.fill()
   578→   578→  ctx.beginPath()
   579→   579→  ctx.arc(bx, by, wb, 0, Math.PI * 2)
   580→   580→  ctx.fill()
   581→   581→  ctx.restore()
   582→   582→}
   583→   583→
   584→   584→/**
   585→   585→ * Limbo/INSIDE-style silhouette, modeled on the articulated skeleton:
   586→   586→ * pelvis basin → abdominal taper → ribcage flare → trapezius → neck column
   587→   587→ * → skull; deltoid caps AT the glenohumeral joints; arms chain shoulder →
   588→   588→ * elbow → fist. Draw order back-to-front: rim → back limbs → torso → front
   589→   589→ * limbs → hands/feet → head → eyes → hit flash. Local space, ctx pre-flipped.
   590→   590→ */
   591→   591→function drawSilhouette(
   592→   592→  ctx: CanvasRenderingContext2D,
   593→   593→  g: SkelGeom,
   594→   594→  style: StickStyle,
   595→   595→  opts: DrawOpts,
   596→   596→  t: number,
   597→   597→  lineW: number,
   598→   598→): void {
   599→   599→  const black = style.body
   600→   600→  const rim = style.outline
   601→   601→  // back-limb depth shade: still silhouette-dark, but separable from the
   602→   602→  // pure-black mass (the old 82%-alpha-black vanished against the torso)
   603→   603→  const back = style.dark !== black ? style.dark : rgba(black, 0.82)
   604→   604→
   605→   605→  // ---- 0. rim pass (separates pure black from the night city) ----
   606→   606→  if (!opts.simple) {
   607→   607→    ctx.save()
   608→   608→    ctx.globalAlpha *= 0.78
   609→   609→    ctx.strokeStyle = rim
   610→   610→    ctx.lineWidth = lineW + 3.6
   611→   611→    ctx.lineCap = 'round'
   612→   612→    ctx.lineJoin = 'round'
   613→   613→    limb(ctx, g.hipPtL.x, g.hipPtL.y, g.kneeL.x, g.kneeL.y, g.ankleL.x, g.ankleL.y)
   614→   614→    limb(ctx, g.hipPtR.x, g.hipPtR.y, g.kneeR.x, g.kneeR.y, g.ankleR.x, g.ankleR.y)
   615→   615→    limb(ctx, g.shoulderL.x, g.shoulderL.y, g.elbL.x, g.elbL.y, g.handL.x, g.handL.y)
   616→   616→    limb(ctx, g.shoulderR.x, g.shoulderR.y, g.elbR.x, g.elbR.y, g.handR.x, g.handR.y)
   617→   617→    ctx.lineWidth = (lineW + 3.6) * 1.5
   618→   618→    ctx.beginPath()
   619→   619→    ctx.moveTo(g.hip.x, g.hip.y)
   620→   620→    ctx.lineTo(g.waist.x, g.waist.y)
   621→   621→    ctx.lineTo(g.chest.x, g.chest.y)
   622→   622→    ctx.lineTo(g.neckTop.x, g.neckTop.y)
   623→   623→    ctx.stroke()
   624→   624→    ctx.restore()
   625→   625→  }
   626→   626→
   627→   627→  // ---- 1. back limbs (recessed — depth cue), with anatomy ----
   628→   628→  // back leg from the far femur head: thigh (quad mass) → calf taper
   629→   629→  fillTaper(ctx, g.hipPtL.x, g.hipPtL.y, g.kneeL.x, g.kneeL.y, 5.8, 4.3, back)
   630→   630→  fillTaper(ctx, g.kneeL.x, g.kneeL.y, g.ankleL.x, g.ankleL.y, 4.0, 2.8, back)
   631→   631→  fillFoot(ctx, g.ankleL.x, g.ankleL.y, g.toeL.x, g.toeL.y, back)
   632→   632→  // knee cap (back) — centers the joint inside the calf silhouette
   633→   633→  ctx.fillStyle = back
   634→   634→  ctx.beginPath()
   635→   635→  ctx.arc(g.kneeL.x, g.kneeL.y, 4.2, 0, Math.PI * 2)
   636→   636→  ctx.fill()
   637→   637→  // rear deltoid cap — the back arm's root reads at the far shoulder
   638→   638→  ctx.beginPath()
   639→   639→  ctx.arc(g.shoulderL.x, g.shoulderL.y, 4.6, 0, Math.PI * 2)
   640→   640→  ctx.fill()
   641→   641→  // back arm: triceps taper + forearm, from the far glenohumeral joint
   642→   642→  fillTaper(ctx, g.shoulderL.x, g.shoulderL.y, g.elbL.x, g.elbL.y, 4.4, 3.3, back)
   643→   643→  fillTaper(ctx, g.elbL.x, g.elbL.y, g.handL.x, g.handL.y, 3.2, 2.5, back)
   644→   644→
   645→   645→  // ---- 2. torso: pelvis basin + abdomen + ribcage + trapezius + neck ----
   646→   646→  // pelvis: WIDE at the hip (iliac crest), tapering hard to the waist —
   647→   647→  // strong region contrast so the pelvis reads as a basin
   648→   648→  fillTaper(ctx, g.hip.x, g.hip.y, g.waist.x, g.waist.y, 9.6, 6.6, black)
   649→   649→  // abdomen → ribcage: narrow at the navel, flaring WIDE at the chest —
   650→   650→  // the V-taper of the latissimus into the ribcage
   651→   651→  fillTaper(ctx, g.waist.x, g.waist.y, g.chest.x, g.chest.y, 6.6, 10.4, black)
   652→   652→  // trapezius: the shoulder girdle — slopes from the skull base out to BOTH
   653→   653→  // deltoid caps (near + far). This polygon is what ties arms to shoulders.
   654→   654→  {
   655→   655→    const ndx = g.neckTop.x - g.chest.x
   656→   656→    const ndy = g.neckTop.y - g.chest.y
   657→   657→    const nl = Math.hypot(ndx, ndy) || 1
   658→   658→    const px = -ndy / nl
   659→   659→    const py = ndx / nl
   660→   660→    ctx.save()
   661→   661→    ctx.fillStyle = black
   662→   662→    ctx.beginPath()
   663→   663→    ctx.moveTo(g.neckTop.x + px * 3.4, g.neckTop.y + py * 3.4)
   664→   664→    ctx.lineTo(g.neckTop.x - px * 3.4, g.neckTop.y - py * 3.4)
   665→   665→    // out to the FAR shoulder cap (behind + lower — 2.8D depth read)
   666→   666→    ctx.lineTo(g.shoulderL.x + px * 4.8, g.shoulderL.y + py * 4.8)
   667→   667→    ctx.lineTo(g.shoulderL.x - px * 3.4, g.shoulderL.y - py * 3.4)
   668→   668→    // through the shoulder line to the NEAR shoulder cap
   669→   669→    ctx.lineTo(g.shoulderR.x - px * 5.2, g.shoulderR.y - py * 5.2)
   670→   670→    ctx.lineTo(g.shoulderR.x + px * 5.8, g.shoulderR.y + py * 5.8)
   671→   671→    ctx.closePath()
   672→   672→    ctx.fill()
   673→   673→    ctx.restore()
   674→   674→  }
   675→   675→  // neck column: chest (shoulder line) up to the skull base — a real neck,
   676→   676→  // wide at the base (sternocleidomastoid + trap roots) tapering to atlas
   677→   677→  fillTaper(ctx, g.chest.x, g.chest.y, g.neckTop.x, g.neckTop.y, 4.6, 3.3, black)
   678→   678→
   679→   679→  // ---- 3. front limbs (solid black, anatomical) ----
   680→   680→  // front leg from the near femur head: quad → knee → calf
   681→   681→  fillTaper(ctx, g.hipPtR.x, g.hipPtR.y, g.kneeR.x, g.kneeR.y, 6.5, 4.7, black)
   682→   682→  fillTaper(ctx, g.kneeR.x, g.kneeR.y, g.ankleR.x, g.ankleR.y, 4.4, 2.9, black)
   683→   683→  fillFoot(ctx, g.ankleR.x, g.ankleR.y, g.toeR.x, g.toeR.y, black)
   684→   684→  // knee cap — centers the joint in the front calf silhouette
   685→   685→  ctx.fillStyle = black
   686→   686→  ctx.beginPath()
   687→   687→  ctx.arc(g.kneeR.x, g.kneeR.y, 4.6, 0, Math.PI * 2)
   688→   688→  ctx.fill()
   689→   689→  // deltoid cap ON the glenohumeral joint — the arm reads from the shoulder
   690→   690→  ctx.beginPath()
   691→   691→  ctx.arc(g.shoulderR.x, g.shoulderR.y, 5.7, 0, Math.PI * 2)
   692→   692→  ctx.fill()
   693→   693→  // bicep → forearm taper
   694→   694→  fillTaper(ctx, g.shoulderR.x, g.shoulderR.y, g.elbR.x, g.elbR.y, 4.9, 3.6, black)
   695→   695→  fillTaper(ctx, g.elbR.x, g.elbR.y, g.handR.x, g.handR.y, 3.5, 2.6, black)
   696→   696→  // elbow sphere (silhouette joint)
   697→   697→  ctx.beginPath()
   698→   698→  ctx.arc(g.elbR.x, g.elbR.y, 3.7, 0, Math.PI * 2)
   699→   699→  ctx.fill()
   700→   700→
   701→   701→  // ---- 4. hands: fists oriented on the blade axis (grip anatomy) ----
   702→   702→  const bax = Math.cos(g.bladeAngle)
   703→   703→  const bay = Math.sin(g.bladeAngle)
   704→   704→  // off-hand fist (rear grip — back shade, under the sword hand)
   705→   705→  fillFist(ctx, g.handL.x, g.handL.y, bax, bay, 2.9, back)
   706→   706→  // sword hand — big knuckle fist across the tsuka
   707→   707→  fillFist(ctx, g.handR.x, g.handR.y, bax, bay, 3.4, black)
   708→   708→  // grip readability: a whisper of rim on the sword fist so the grip reads
   709→   709→  // against the black mass (playtest note: hands vanished at gameplay zoom)
   710→   710→  if (!opts.simple) {
   711→   711→    ctx.save()
   712→   712→    ctx.strokeStyle = rgba(rim, 0.55)
   713→   713→    ctx.lineWidth = 1.2
   714→   714→    ctx.beginPath()
   715→   715→    ctx.arc(g.handR.x, g.handR.y, 3.6, -1.2, 0.9)
   716→   716→    ctx.stroke()
   717→   717→    ctx.restore()
   718→   718→  }
   719→   719→
   720→   720→  // ---- 4.5 BLADE-CAST RIM LIGHT (2.8D dynamic lighting): the glowing
   721→   721→  // edge lights the sword-side of the body — arm, torso leading edge,
   722→   722→  // front thigh. An additive whisper that separates the carrier from the
   723→   723→  // night and sells the blade as a real 3D light source in the scene. ----
   724→   724→  const el = opts.edgeLight
   725→   725→  if (el && el.a > 0.03) {
   726→   726→    ctx.save()
   727→   727→    ctx.globalCompositeOperation = 'lighter'
   728→   728→    ctx.lineCap = 'round'
   729→   729→    ctx.strokeStyle = rgba(el.color, Math.min(0.85, el.a))
   730→   730→    ctx.lineWidth = 2
   731→   731→    // sword arm chain (shoulder → elbow → hand) — closest to the source
   732→   732→    ctx.beginPath()
   733→   733→    ctx.moveTo(g.shoulderR.x, g.shoulderR.y)
   734→   734→    ctx.lineTo(g.elbR.x, g.elbR.y)
   735→   735→    ctx.lineTo(g.handR.x, g.handR.y)
   736→   736→    ctx.stroke()
   737→   737→    // torso leading edge (the lit side of the ribcage)
   738→   738→    ctx.beginPath()
   739→   739→    ctx.moveTo(g.hip.x + 5.2, g.hip.y - 1)
   740→   740→    ctx.quadraticCurveTo(g.waist.x + 6.5, g.waist.y - 1, g.chest.x + 8.2, g.chest.y + 1.5)
   741→   741→    ctx.stroke()
   742→   742→    // front thigh quad — the forward leg catches the spill
   743→   743→    ctx.beginPath()
   744→   744→    ctx.moveTo(g.hipPtR.x + 4.4, g.hipPtR.y + 1)
   745→   745→    ctx.lineTo(g.kneeR.x + 3.4, g.kneeR.y - 1)
   746→   746→    ctx.stroke()
   747→   747→    // off-hand knuckles kiss the light when gripping the tsuka
   748→   748→    ctx.lineWidth = 1.4
   749→   749→    ctx.beginPath()
   750→   750→    ctx.arc(g.handL.x, g.handL.y, 2.6, 0, Math.PI * 2)
   751→   751→    ctx.stroke()
   752→   752→    ctx.restore()
   753→   753→  }

   786→   786→    ctx.stroke()

   788→   788→  }
   789→   789→
   790→   790→  // ---- 6. glowing eyes (the only light on the body) ----
