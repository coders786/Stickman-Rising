   187→   187→    const n = this.scaleCount(count)

   192→   192→      const sp = speed * rand(0.45, 1.25)
   193→   193→      p.x = x; p.y = y
   194→   194→      p.vx = Math.cos(a) * sp
   195→   195→      p.vy = Math.sin(a) * sp
   196→   196→      p.age = 0; p.life = life * rand(0.6, 1.3)

   288→   288→  /** jagged lightning bolt between two points */
   289→   289→  lightningArc(x0: number, y0: number, x1: number, y1: number, color: string, segments = 6): void {
   290→   290→    const p = this.nextParticle()
   291→   291→    if (!p) return
   292→   292→    const pts: number[] = [x0, y0]
   293→   293→    for (let i = 1; i < segments; i++) {
   294→   294→      const t = i / segments
   295→   295→      const jitter = 14 * (1 - Math.abs(t - 0.5) * 2 + 0.2)
   296→   296→      pts.push(
   297→   297→        x0 + (x1 - x0) * t + rand(-jitter, jitter),
   298→   298→        y0 + (y1 - y0) * t + rand(-jitter, jitter),
   299→   299→      )
   300→   300→    }
   301→   301→    pts.push(x1, y1)
   302→   302→    p.x = x0; p.y = y0
   303→   303→    p.vx = 0; p.vy = 0
   304→   304→    p.age = 0; p.life = 0.22
   305→   305→    p.size = 2.4
   306→   306→    p.color = color
   307→   307→    p.type = 'arc'
   308→   308→    p.grav = 0; p.drag = 0
   309→   309→    p.rot = 0; p.vrot = 0
   310→   310→    p.pts = pts
   311→   311→    // spark burst at both ends
   312→   312→    this.sparks(x1, y1, Math.atan2(y1 - y0, x1 - x0), 0.7, 5, color, 240, 0.3)
   313→   313→  }

   541→   541→  update(dt: number): void {
   542→   542→    // particles
   543→   543→    for (let i = this.particles.length - 1; i >= 0; i--) {
   544→   544→      const p = this.particles[i]
   545→   545→      if (!p.active) continue
   546→   546→      p.age += dt
   547→   547→      if (p.age >= p.life) {
   548→   548→        p.active = false
   549→   549→        this.pFree.push(p)
   550→   550→        continue
   551→   551→      }
   552→   552→      p.vy += p.grav * dt
   553→   553→      const dr = 1 - p.drag * dt
   554→   554→      p.vx *= dr > 0 ? dr : 0
   555→   555→      p.vy *= dr > 0 ? dr : 0
   556→   556→      p.x += p.vx * dt
   557→   557→      p.y += p.vy * dt
   558→   558→    }
   559→   559→    // trails
   560→   560→    for (let i = this.trails.length - 1; i >= 0; i--) {
   561→   561→      const t = this.trails[i]
   562→   562→      if (!t.active) continue
   563→   563→      if (!t.emitting) {
   564→   564→        t.age += dt
   565→   565→        if (t.age >= t.life) {
   566→   566→          t.active = false
   567→   567→          this.tFree.push(t)
   568→   568→        }
   569→   569→      }
   570→   570→    }
   571→   571→    // afterimages
   572→   572→    for (let i = this.afterimages.length - 1; i >= 0; i--) {
   573→   573→      const a = this.afterimages[i]
   574→   574→      a.age += dt
   575→   575→      if (a.age >= a.life) {
   576→   576→        a.active = false
   577→   577→        this.aFree.push(a)
   578→   578→      }
   579→   579→    }
   580→   580→    // rings
   581→   581→    for (let i = this.rings.length - 1; i >= 0; i--) {
   582→   582→      const r = this.rings[i]
   583→   583→      r.age += dt
   584→   584→      if (r.age >= r.life) {
   585→   585→        r.active = false
   586→   586→        this.rFree.push(r)
   587→   587→      }
   588→   588→    }
   589→   589→    // cracks
   590→   590→    for (let i = this.cracks.length - 1; i >= 0; i--) {
   591→   591→      const c = this.cracks[i]
   592→   592→      c.age += dt
   593→   593→      if (c.age >= c.life) {
   594→   594→        c.active = false
   595→   595→        this.cFree.push(c)
   596→   596→      }
   597→   597→    }
   598→   598→    // texts
   599→   599→    for (let i = this.texts.length - 1; i >= 0; i--) {
   600→   600→      const t = this.texts[i]
   601→   601→      t.age += dt
   602→   602→      if (t.age >= t.life) {
   603→   603→        t.active = false
   604→   604→        this.txtFree.push(t)
   605→   605→      }
   606→   606→      t.vy += 320 * dt
   607→   607→      t.x += t.vx * dt
   608→   608→      t.y += t.vy * dt
   609→   609→    }
   610→   610→    // debris
   611→   611→    for (let i = this.debris.length - 1; i >= 0; i--) {
   612→   612→      const d = this.debris[i]
   613→   613→      d.age += dt
   614→   614→      if (d.age >= d.life) {
   615→   615→        d.active = false
   616→   616→        this.dFree.push(d)
   617→   617→        continue
   618→   618→      }
   619→   619→      d.vy += 900 * dt
   620→   620→      d.x += d.vx * dt
   621→   621→      d.y += d.vy * dt
   622→   622→      d.rot += d.vrot * dt
   623→   623→      if (d.y > 0) {
   624→   624→        d.y = 0
   625→   625→        d.vy *= -0.42
   626→   626→        d.vx *= 0.6
   627→   627→        d.vrot *= 0.5
   628→   628→        if (Math.abs(d.vy) < 40) d.vy = 0
   629→   629→      }
   630→   630→    }
   631→   631→    // flash
   632→   632→    if (this.flashAlpha > 0) {
   633→   633→      this.flashAlpha -= this.flashDecay * dt
   634→   634→      if (this.flashAlpha < 0) this.flashAlpha = 0
   635→   635→    }
   636→   636→    // chromatic strain decay
   637→   637→    if (this.caIntensity > 0) {
   638→   638→      this.caIntensity = Math.max(0, this.caIntensity - dt * 3.4)
   639→   639→    }
   640→   640→    // impact frames
   641→   641→    for (let i = this.impactFrames.length - 1; i >= 0; i--) {
   642→   642→      const f = this.impactFrames[i]
   643→   643→      f.age += dt
   644→   644→      if (f.age >= f.life) {
   645→   645→        f.active = false
   646→   646→        this.impactFrames.splice(i, 1)
   647→   647→      }
   648→   648→    }
   649→   649→  }

   884→   884→function pushPolar(t: Trail, x: number, y: number, a: number, r0: number, r1: number): void {
   885→   885→  const i = t.head * 4
   886→   886→  t.pts[i] = x + Math.cos(a) * r0
   887→   887→  t.pts[i + 1] = y + Math.sin(a) * r0
   888→   888→  t.pts[i + 2] = x + Math.cos(a) * r1
   889→   889→  t.pts[i + 3] = y + Math.sin(a) * r1
   890→   890→  t.head = (t.head + 1) % t.N
   891→   891→  if (t.count < t.N) t.count++
   892→   892→}

   897→   897→function drawTrail(ctx: CanvasRenderingContext2D, t: Trail, fade: number): void {
   898→   898→  const n = t.count
   899→   899→  const pts = t.pts
   900→   900→  // render oldest → newest
   901→   901→  const start = (t.head - n + t.N + t.N) % t.N
   902→   902→
