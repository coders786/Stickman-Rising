    57→    57→  draw(ctx: CanvasRenderingContext2D, game: Game, time: number): void {
    58→    58→    if (this.mode === 'off') return
    59→    59→    const p = game.player
    60→    60→    ctx.save()
    61→    61→    ctx.font = '600 12px ui-monospace, monospace'

    63→    63→
    64→    64→    if (this.mode === 'solo') {

    66→    66→    } else if (this.mode === 'sheet') {
    67→    67→      this.drawSheets(ctx)
    68→    68→    }
    69→    69→
    70→    70→    // state readout (uses public player surface + bracket access for privates)

    79→    79→      `postAtk=${Number(priv.postAttackT ?? 0).toFixed(2)}  turn=${Number(priv.turnT ?? 0).toFixed(2)}`,

    88→    88→      ctx.fillText(ln, 13, y)
    89→    89→      y += 17
    90→    90→    }
    91→    91→    ctx.restore()
    92→    92→  }

    94→    94→  /** big live character + skeleton overlay, centered-left — drawn on a
    95→    95→   * light diagnostic panel so the black silhouette reads at full contrast */
    96→    96→  private drawSolo(ctx: CanvasRenderingContext2D, game: Game, time: number): void {
    97→    97→    const p = game.player
    98→    98→    const dpr = window.devicePixelRatio || 1
    99→    99→    const Z = PSCALE / 1.32 // zoom factor over the in-game world scale
   100→   100→    const cx = (ctx.canvas.width / dpr) * 0.26
   101→   101→    const groundY = (ctx.canvas.height / dpr) * 0.82
   102→   102→    const hipScreenY = groundY + p.hipY * Z
   103→   103→    // light diagnostic panel behind the figure (contrast for the black body)
   104→   104→    ctx.save()
   105→   105→    ctx.fillStyle = '#dfe7ee'
   106→   106→    ctx.beginPath()
   107→   107→    ctx.roundRect(cx - 250, groundY - 330, 500, 348, 14)
   108→   108→    ctx.fill()
   109→   109→    ctx.strokeStyle = 'rgba(40,70,100,0.35)'
   110→   110→    ctx.lineWidth = 1.5
   111→   111→    ctx.stroke()
   112→   112→    ctx.restore()
   113→   113→    // body: local geom → screen at PSCALE
   114→   114→    ctx.save()
   115→   115→    ctx.translate(cx, hipScreenY)
   116→   116→    ctx.scale(p.face * PSCALE, PSCALE)
   117→   117→    drawStick(ctx, p.geom, PLAYER_STYLE, { time })
   118→   118→    this.drawSkeleton(ctx, p.geom)
   119→   119→    ctx.restore()
   120→   120→    // ground line + foot contact markers at their true screen positions
   121→   121→    ctx.save()
   122→   122→    ctx.strokeStyle = 'rgba(30,90,140,0.65)'
   123→   123→    ctx.lineWidth = 2
   124→   124→    ctx.setLineDash([7, 5])
   125→   125→    ctx.beginPath()
   126→   126→    ctx.moveTo(cx - 240, groundY)
   127→   127→    ctx.lineTo(cx + 240, groundY)
   128→   128→    ctx.stroke()
   129→   129→    ctx.setLineDash([])
   130→   130→    if (this.showFeet) {
   131→   131→      const ptScreen = (j: { x: number; y: number }) => ({
   132→   132→        x: cx + p.face * PSCALE * j.x,
   133→   133→        y: hipScreenY + PSCALE * j.y,
   134→   134→      })
   135→   135→      const rA = ptScreen(p.geom.ankleR)
   136→   136→      const lA = ptScreen(p.geom.ankleL)
   137→   137→      // planted = lowest foot within reach of the ground line
   138→   138→      const rPlanted = p.grounded && p.geom.ankleR.y >= p.geom.ankleL.y
   139→   139→      const lPlanted = p.grounded && p.geom.ankleL.y > p.geom.ankleR.y
   140→   140→      this.drawFootMarker(ctx, rA.x, rA.y, rPlanted)
   141→   141→      this.drawFootMarker(ctx, lA.x, lA.y, lPlanted)
   142→   142→    }
   143→   143→    ctx.restore()
   144→   144→  }

   153→   153→  /** bones + joints + labels on any solved geom (local space) */
   154→   154→  private drawSkeleton(ctx: CanvasRenderingContext2D, g: SkelGeom): void {
   155→   155→    if (!this.showBones) return
   156→   156→    ctx.save()
   157→   157→    ctx.strokeStyle = 'rgba(65,246,255,0.55)'
   158→   158→    ctx.lineWidth = 1.1
   159→   159→    const seg = (a: { x: number; y: number }, b: { x: number; y: number }) => {
   160→   160→      ctx.beginPath()
   161→   161→      ctx.moveTo(a.x, a.y)
   162→   162→      ctx.lineTo(b.x, b.y)
   163→   163→      ctx.stroke()
   164→   164→    }
   165→   165→    seg(g.hip, g.waist); seg(g.waist, g.chest); seg(g.chest, g.neckTop); seg(g.neckTop, g.headC)
   166→   166→    seg(g.hipPtR, g.kneeR); seg(g.kneeR, g.ankleR); seg(g.ankleR, g.toeR)
   167→   167→    seg(g.hipPtL, g.kneeL); seg(g.kneeL, g.ankleL); seg(g.ankleL, g.toeL)
   168→   168→    seg(g.shoulderR, g.elbR); seg(g.elbR, g.handR)
   169→   169→    seg(g.shoulderL, g.elbL); seg(g.elbL, g.handL)
   170→   170→    // shoulder girdle line (the anatomy fix, visualized)
   171→   171→    ctx.strokeStyle = 'rgba(255,120,180,0.8)'
   172→   172→    ctx.lineWidth = 1.3
   173→   173→    ctx.beginPath()
   174→   174→    ctx.moveTo(g.shoulderL.x, g.shoulderL.y)
   175→   175→    ctx.lineTo(g.neckTop.x, g.neckTop.y)
   176→   176→    ctx.lineTo(g.shoulderR.x, g.shoulderR.y)
   177→   177→    ctx.stroke()
   178→   178→    // joints
   179→   179→    ctx.fillStyle = '#ffd24a'
   180→   180→    const joints = [
   181→   181→      g.hip, g.hipPtR, g.hipPtL, g.waist, g.chest, g.neckTop, g.headC,
   182→   182→      g.shoulderR, g.shoulderL, g.elbR, g.elbL, g.handR, g.handL,
   183→   183→      g.kneeR, g.kneeL, g.ankleR, g.ankleL,
   184→   184→    ]
   185→   185→    for (const j of joints) {
   186→   186→      ctx.beginPath()
   187→   187→      ctx.arc(j.x, j.y, 1.7, 0, Math.PI * 2)
   188→   188→      ctx.fill()
   189→   189→    }
   190→   190→    // THE proof joints: shoulders in pink
   191→   191→    ctx.fillStyle = '#ff6fb5'
   192→   192→    for (const j of [g.shoulderR, g.shoulderL]) {
   193→   193→      ctx.beginPath()
   194→   194→      ctx.arc(j.x, j.y, 3.0, 0, Math.PI * 2)
   195→   195→      ctx.fill()
   196→   196→    }
   197→   197→    if (this.showLabels) {
   198→   198→      ctx.fillStyle = 'rgba(160,240,255,0.95)'
   199→   199→      ctx.font = '600 4.2px ui-monospace, monospace'
   200→   200→      const lbl = (j: { x: number; y: number }, s: string) => ctx.fillText(s, j.x + 2.4, j.y - 2.4)
   201→   201→      lbl(g.headC, 'skull')
   202→   202→      lbl(g.neckTop, 'neckTop')
   203→   203→      lbl(g.shoulderR, 'SHOULDER-R')
   204→   204→      lbl(g.shoulderL, 'shoulder-l')
   205→   205→      lbl(g.chest, 'chest')
   206→   206→      lbl(g.waist, 'waist')
   207→   207→      lbl(g.hip, 'pelvis')
   208→   208→      lbl(g.kneeR, 'kneeR')
   209→   209→      lbl(g.ankleR, 'ankleR')
   210→   210→      lbl(g.elbR, 'elbR')
   211→   211→      lbl(g.handR, 'handR')
   212→   212→    }
   213→   213→    ctx.restore()
   214→   214→  }

   216→   216→  /** contact sheets: every locomotion cycle, 8 frames each, real solve —
   217→   217→   * light diagnostic background, per-cell ground lines + foot-contact
   218→   218→   * markers (green = support, orange = swing), 2 clip-groups per band */
   219→   219→  private drawSheets(ctx: CanvasRenderingContext2D): void {
   220→   220→    const rows: Array<[string, typeof WALK_CLIP]> = [
   221→   221→      ['WALK', WALK_CLIP],
   222→   222→      ['RUN', RUN_CLIP],
   223→   223→      ['SPRINT', SPRINT_CLIP],
   224→   224→      ['COMBAT STEP', COMBAT_STEP_CLIP],
   225→   225→      ['COMBAT-IDLE (rooted guard)', COMBAT_IDLE_CLIP],
   226→   226→      ['BACKPEDAL', BACKPEDAL_CLIP],
   227→   227→      ['IDLE', IDLE_CLIP],
   228→   228→      ['JUMP', JUMP_CLIP],
   229→   229→      ['DASH LAUNCH (iai)', DASH_LAUNCH_CLIP],
   230→   230→      ['DASH CARRY (shunshin)', DASH_CARRY_CLIP],
   231→   231→      ['DASH HUNT (gap-close)', DASH_HUNT_CLIP],
   232→   232→      ['DASH PIERCE (lunge)', DASH_PIERCE_CLIP],
   233→   233→      ['DASH RETREAT (guard)', DASH_RETREAT_CLIP],
   234→   234→      ['DASH CARVE (breakaway)', DASH_CARVE_CLIP],
   235→   235→    ]
   236→   236→    const dpr = window.devicePixelRatio || 1
   237→   237→    const w = ctx.canvas.width / dpr
   238→   238→    const h = ctx.canvas.height / dpr
   239→   239→    const cols = 8
   240→   240→    const groupsPerRow = 2
   241→   241→    const bands = Math.ceil(rows.length / groupsPerRow) // 3 bands
   242→   242→    const cellW = w / (cols * groupsPerRow)
   243→   243→    const rowH = h / (bands + 0.25)
   244→   244→    // figure scale: fit character height inside a band minus label space
   245→   245→    const S = Math.min(1.0, (rowH - 34) / 138)
   246→   246→    ctx.save()
   247→   247→    // light diagnostic backdrop — black silhouette at full contrast
   248→   248→    ctx.fillStyle = '#e9eef3'
   249→   249→    ctx.fillRect(0, 0, w, h)
   250→   250→    ctx.font = '700 12px ui-monospace, monospace'
   251→   251→    for (let r = 0; r < rows.length; r++) {
   252→   252→      const band = Math.floor(r / groupsPerRow)
   253→   253→      const group = r % groupsPerRow
   254→   254→      const [label, clip] = rows[r]
   255→   255→      const baseY = rowH * (band + 1)
   256→   256→      const labelX = group * (w / groupsPerRow) + 10
   257→   257→      ctx.fillStyle = '#1a3a52'
   258→   258→      ctx.fillText(`${label} · ${clip.dur.toFixed(2)}s cycle`, labelX, baseY - rowH + 16)
   259→   259→      for (let c = 0; c < cols; c++) {
   260→   260→        const t = (clip.dur * c) / cols
   261→   261→        sampleClip(clip, t, this.scratchPose)
   262→   262→        const g = solveSkeleton(this.scratchPose, PLAYER_SPEC, this.scratchGeom)
   263→   263→        // normalize: lowest foot contact sits ON this cell's ground line
   264→   264→        const lowContact = Math.max(g.ankleL.y, g.ankleR.y, g.toeL.y, g.toeR.y)
   265→   265→        const cx = group * (w / groupsPerRow) + cellW * (c + 0.5)
   266→   266→        const groundY = baseY - 14
   267→   267→        ctx.save()
   268→   268→        ctx.translate(cx, groundY - (lowContact + 2.4) * S)
   269→   269→        ctx.scale(S, S)
   270→   270→        drawStick(ctx, g, PLAYER_STYLE, { time: t, simple: true })
   271→   271→        ctx.restore()
   272→   272→        // per-cell ground line
   273→   273→        ctx.strokeStyle = 'rgba(40,80,110,0.45)'
   274→   274→        ctx.lineWidth = 1
   275→   275→        ctx.beginPath()
   276→   276→        ctx.moveTo(cx - cellW * 0.42, groundY)
   277→   277→        ctx.lineTo(cx + cellW * 0.42, groundY)
   278→   278→        ctx.stroke()
   279→   279→        // foot-contact markers: green = support (near ground), orange = swing
   280→   280→        const hipY = groundY - (lowContact + 2.4) * S
   281→   281→        const markFoot = (j: { x: number; y: number }) => {
   282→   282→          const airborne = (lowContact - j.y) > 4 // clearly lifted vs support
   283→   283→          ctx.fillStyle = airborne ? 'rgba(255,140,30,0.9)' : '#18c25a'
   284→   284→          ctx.beginPath()
   285→   285→          ctx.arc(cx + j.x * S, hipY + j.y * S, 3, 0, Math.PI * 2)
   286→   286→          ctx.fill()
   287→   287→        }
   288→   288→        markFoot(g.ankleR)
   289→   289→        markFoot(g.ankleL)
   290→   290→        // frame tick
   291→   291→        ctx.fillStyle = 'rgba(30,70,100,0.55)'
   292→   292→        ctx.font = '600 9px ui-monospace, monospace'
   293→   293→        ctx.fillText(`f${c}`, cx - 5, baseY - 3)
   294→   294→      }
   295→   295→    }
   296→   296→    ctx.restore()
   297→   297→  }
   298→   298→}

   304→   304→interface QaCheck {
   305→   305→  name: string
   306→   306→  pass: boolean
   307→   307→  detail: string
   308→   308→}
   309→   309→
   310→   310→const sleep = (ms: number) => new Promise<void>((r) => setTimeout(r, ms))
   311→   311→
   312→   312→/** Scripted end-to-end verification. Parks the report on
   313→   313→ * window.__qaResult. Drive from the browser console / agent-browser. */
   314→   314→export async function runQA(game: Game): Promise<void> {
   315→   315→  const checks: QaCheck[] = []
   316→   316→  const log: string[] = []
   317→   317→  const p = game.player
   318→   318→  const input = game.input
   319→   319→
   320→   320→  const samplePose = () => {
   321→   321→    let bad = 0
   322→   322→    for (let i = 0; i < 17; i++) if (!Number.isFinite(p.pose[i])) bad++
   323→   323→    return bad
   324→   324→  }
   325→   325→
   326→   326→  try {
   327→   327→    // ---- setup: controlled empty arena (waves suppressed, player alive) ----
   328→   328→    game.qaMode = true
   329→   329→    game.phase = 'playing'
   330→   330→    game.enemies.length = 0
   331→   331→    game.waves.reset()
   332→   332→    // release every virtual input left held by prior console driving —
   333→   333→    // stale held buttons contaminate the whole scripted sequence
   334→   334→    input.setVirtualAxis(0, 0)
   335→   335→    for (const a of ['left', 'right', 'up', 'down', 'light', 'heavy', 'dash', 'jump', 'parry', 'ability', 'ult', 'execute', 'sheathe'] as const) {
   336→   336→      input.setVirtual(a, false)
   337→   337→    }
   338→   338→    p.reset(400)
   339→   339→    input.setEnabled(true)
   340→   340→    log.push('setup ok')
   341→   341→
   342→   342→    // ---- 1. anatomy invariants (the arms-from-the-neck fix) ----
   343→   343→    {
   344→   344→      const g = p.geom
   345→   345→      const spec = PLAYER_SPEC
   346→   346→      const armLen = Math.hypot(g.elbR.x - g.shoulderR.x, g.elbR.y - g.shoulderR.y)
   347→   347→      // local y grows DOWN: shoulder below neck-top means shoulderR.y GREATER
   348→   348→      const shoulderBelowNeck = g.shoulderR.y > g.neckTop.y + 1.5
   349→   349→      const armFromShoulder = Math.abs(armLen - spec.upperArm) < 0.6
   350→   350→      // neck column rises above chest: neckTop.y LESS (more negative) than chest
   351→   351→      const neckVisible = g.neckTop.y < g.chest.y - 2.5
   352→   352→      checks.push({
   353→   353→        name: 'anatomy: shoulder joints below neck top',
   354→   354→        pass: shoulderBelowNeck,
   355→   355→        detail: `shoulderR.y=${g.shoulderR.y.toFixed(1)} neckTop.y=${g.neckTop.y.toFixed(1)}`,
   356→   356→      })
   357→   357→      checks.push({
   358→   358→        name: 'anatomy: arm chains FROM shoulder (len=upperArm)',
   359→   359→        pass: armFromShoulder,
   360→   360→        detail: `len=${armLen.toFixed(2)} spec=${spec.upperArm}`,
   361→   361→      })
   362→   362→      checks.push({
   363→   363→        name: 'anatomy: neck column visible above chest',
   364→   364→        pass: neckVisible,
   365→   365→        detail: `neckTop.y=${g.neckTop.y.toFixed(1)} chest.y=${g.chest.y.toFixed(1)}`,
   366→   366→      })
   367→   367→    }
   368→   368→
   369→   369→    // ---- 2. walk → run escalation (Condition A, analog speed mapping) ----
   370→   370→    input.setVirtualAxis(0.55, 0)
   371→   371→    await sleep(700)
   372→   372→    const earlyMode = p.loco
   373→   373→    const earlyVx = Math.abs(p.vx)
   374→   374→    input.setVirtualAxis(1, 0)
   375→   375→    await sleep(1500)
   376→   376→    const fullMode = p.loco
   377→   377→    input.setVirtualAxis(0, 0)
   378→   378→    await sleep(500)
   379→   379→    checks.push({
   380→   380→      name: 'locomotion: WALK at partial input',
   381→   381→      pass: earlyMode === 'walk' && earlyVx > 60 && earlyVx < 210,
   382→   382→      detail: `mode@0.7s=${earlyMode} vx=${earlyVx.toFixed(0)}`,
   383→   383→    })
   384→   384→    checks.push({
   385→   385→      name: 'locomotion: RUN at full input',
   386→   386→      pass: fullMode === 'run',
   387→   387→      detail: `mode@2.2s=${fullMode}`,
   388→   388→    })
   389→   389→
