    21→    21→  /** glenohumeral joint forward of the spine-top, along the torso frame */

    35→    35→ * fighter (H ≈ 104 local units ≈ 137px at SCALE 1.32). Every joint offset
    36→    36→ * below derives from this table; nothing is placed by eye:
    37→    37→ *
    38→    38→ *   region            fraction    local units
    39→    39→ *   ──────────────────────────────────────────────
    40→    40→ *   skull height      1.28 head   17.6 (r 8.8 — stylized readability)
    41→    41→ *   neck column       0.55 head    7.2 (C7 → skull base — stylized long
    42→    42→ *                                    so the column READS at silhouette
    43→    43→ *                                    scale: a real 0.35-head neck reads
    44→    44→ *                                    as fused at gameplay zoom)
    45→    45→ *   hip→shoulder line 0.355 H     37   (waist split 0.57/0.43)
    46→    46→ *   femur head offset             ±~2  (front hip +1.7, rear −2.1)
    47→    47→ *   glenohumeral jt   0.031 H      3.2 fwd × 3.3 drop (near shoulder)
    48→    48→ *   humerus           0.212 H      22

    50→    50→ *                                      Drillis–Contini intermembral ratio,
    51→    51→ *                                      upper end: joint caps + bent-arm
    52→    52→ *                                      foreshortening eat ~10% of the
    53→    53→ *                                      perceived humerus at render time)
    54→    54→ *   hand              0.062 H     6.5
    55→    55→ *   femur             0.24 H      25

   109→   109→export function makeSkelGeom(): SkelGeom {
   110→   110→  const z = (): Vec2 => ({ x: 0, y: 0 })
   111→   111→  return {
   112→   112→    hip: z(), hipPtR: z(), hipPtL: z(), waist: z(), chest: z(), neckTop: z(),
   113→   113→    headC: z(), headAng: 0, shoulderR: z(), shoulderL: z(),
   114→   114→    kneeL: z(), ankleL: z(), toeL: z(),
   115→   115→    kneeR: z(), ankleR: z(), toeR: z(),
   116→   116→    elbL: z(), handL: z(), elbR: z(), handR: z(),
   117→   117→    bladeAngle: 0, torsoAngle: 0,
   118→   118→  }
   119→   119→}
   120→   120→
   121→   121→/** Solve joint positions. Reuses `out`.
   122→   122→ *
   123→   123→ * SPINE CHAIN (four articulated regions, parented in order):
   124→   124→ *   hip → waist (pelvis/abdomen, 57% along `rot`) → chest (ribcage +
   125→   125→ *   shoulder line, 43% along `rot + spine`) → neck column (along the head
   126→   126→ *   channel) → skull. The waist joint makes the torso BEND instead of
   127→   127→ *   shear — the lumbar and thoracic curves separate.
   128→   128→ *
   129→   129→ * SHOULDER GIRDLE (the arms-from-the-neck fix): the glenohumeral joints
   130→   130→ * solve as chest + (shoulderFwd, shoulderDrop) in the chest's frame —
   131→   131→ * below the shoulder line and slightly anterior, like a real skeleton.
   132→   132→ * The far shoulder sits behind and lower (2.8D depth read). Arms chain
   133→   133→ * from THESE, never from the spine top.
   134→   134→ *
   135→   135→ * LEGS chain from offset femur-head joints (front hip slightly anterior,
   136→   136→ * rear hip behind) — the pelvis reads as a real basin, and stride scissor
   137→   137→ * from two distinct roots. */
   138→   138→export function solveSkeleton(pose: Pose, spec: SkelSpec, out: SkelGeom): SkelGeom {
   139→   139→  const rot = pose[2]
   140→   140→  const spine = pose[3]
   141→   141→  const head = pose[4]
   142→   142→
   143→   143→  // spine: hip → waist → chest
   144→   144→  out.hip.x = pose[0]
   145→   145→  out.hip.y = pose[1]
   146→   146→  const t1 = rot
   147→   147→  out.waist.x = out.hip.x + Math.sin(t1) * spec.torso * 0.57
   148→   148→  out.waist.y = out.hip.y - Math.cos(t1) * spec.torso * 0.57
   149→   149→  const t2 = rot + spine
   150→   150→  out.chest.x = out.waist.x + Math.sin(t2) * spec.torso * 0.43
   151→   151→  out.chest.y = out.waist.y - Math.cos(t2) * spec.torso * 0.43
   152→   152→  out.torsoAngle = t2
   153→   153→
   154→   154→  // neck column + skull — a REAL neck: full spec.neck of visible column
   155→   155→  // between the shoulder line and the skull base
   156→   156→  const ha = t2 + head
   157→   157→  out.neckTop.x = out.chest.x + Math.sin(ha) * spec.neck
   158→   158→  out.neckTop.y = out.chest.y - Math.cos(ha) * spec.neck
   159→   159→  out.headC.x = out.neckTop.x + Math.sin(ha) * spec.headR * 0.9
   160→   160→  out.headC.y = out.neckTop.y - Math.cos(ha) * spec.headR * 0.9
   161→   161→  out.headAng = ha
   162→   162→
   163→   163→  // ---- SHOULDER GIRDLE: the arm roots ----
   164→   164→  // fwd(t) = (cos t, sin t); down-along-torso = (−sin t, cos t)
   165→   165→  const sf = spec.shoulderFwd
   166→   166→  const sd = spec.shoulderDrop
   167→   167→  out.shoulderR.x = out.chest.x + Math.cos(t2) * sf - Math.sin(t2) * sd
   168→   168→  out.shoulderR.y = out.chest.y + Math.sin(t2) * sf + Math.cos(t2) * sd
   169→   169→  const sfL = -sf * 0.62
   170→   170→  const sdL = sd * 1.18
   171→   171→  out.shoulderL.x = out.chest.x + Math.cos(t2) * sfL - Math.sin(t2) * sdL
   172→   172→  out.shoulderL.y = out.chest.y + Math.sin(t2) * sfL + Math.cos(t2) * sdL
   173→   173→
   174→   174→  // ---- femur heads ----
   175→   175→  out.hipPtR.x = out.hip.x + Math.cos(t1) * 1.7
   176→   176→  out.hipPtR.y = out.hip.y + Math.sin(t1) * 1.7
   177→   177→  out.hipPtL.x = out.hip.x + Math.cos(t1) * -2.1
   178→   178→  out.hipPtL.y = out.hip.y + Math.sin(t1) * -2.1
   179→   179→
   180→   180→  // legs (angle from straight-down, positive = forward)
   181→   181→  const hl = pose[5]
   182→   182→  const kl = pose[6]
   183→   183→  const fl = pose[7]
   184→   184→  out.kneeL.x = out.hipPtL.x + Math.sin(hl) * spec.thigh
   185→   185→  out.kneeL.y = out.hipPtL.y + Math.cos(hl) * spec.thigh
   186→   186→  const al = hl + kl
   187→   187→  out.ankleL.x = out.kneeL.x + Math.sin(al) * spec.shin
   188→   188→  out.ankleL.y = out.kneeL.y + Math.cos(al) * spec.shin
   189→   189→  // foot roll (Williams, Animator's Survival Kit): the toe defaults to FLAT
   190→   190→  // on the ground line; the foot channel is heel/toe flex (±) with a whisper
   191→   191→  // of shin influence so airborne tucks still point. Authoring foot angles
   192→   192→  // relative to the shin pointed the toe nearly straight down, planting
   193→   193→  // ~12px of rigid foot below the ground in every grounded pose.
   194→   194→  const tl = Math.PI / 2 + al * 0.35 + fl * 0.5
   195→   195→  out.toeL.x = out.ankleL.x + Math.sin(tl) * spec.foot
   196→   196→  out.toeL.y = out.ankleL.y + Math.cos(tl) * spec.foot
   197→   197→
   198→   198→  const hr = pose[8]
   199→   199→  const kr = pose[9]
   200→   200→  const fr = pose[10]
   201→   201→  out.kneeR.x = out.hipPtR.x + Math.sin(hr) * spec.thigh
   202→   202→  out.kneeR.y = out.hipPtR.y + Math.cos(hr) * spec.thigh
   203→   203→  const ar = hr + kr
   204→   204→  out.ankleR.x = out.kneeR.x + Math.sin(ar) * spec.shin
   205→   205→  out.ankleR.y = out.kneeR.y + Math.cos(ar) * spec.shin
   206→   206→  const tr = Math.PI / 2 + ar * 0.35 + fr * 0.5
   207→   207→  out.toeR.x = out.ankleR.x + Math.sin(tr) * spec.foot
   208→   208→  out.toeR.y = out.ankleR.y + Math.cos(tr) * spec.foot
   209→   209→
   210→   210→  // arms — FROM THE SHOULDERS (angle from straight-down, positive = forward)
   211→   211→  const sl = pose[11]
   212→   212→  const el = pose[12]
   213→   213→  out.elbL.x = out.shoulderL.x + Math.sin(sl) * spec.upperArm
   214→   214→  out.elbL.y = out.shoulderL.y + Math.cos(sl) * spec.upperArm
   215→   215→  const wl = sl + el
   216→   216→  out.handL.x = out.elbL.x + Math.sin(wl) * spec.foreArm
   217→   217→  out.handL.y = out.elbL.y + Math.cos(wl) * spec.foreArm
   218→   218→
   219→   219→  const sr = pose[14]
   220→   220→  const er = pose[15]
   221→   221→  out.elbR.x = out.shoulderR.x + Math.sin(sr) * spec.upperArm
   222→   222→  out.elbR.y = out.shoulderR.y + Math.cos(sr) * spec.upperArm
   223→   223→  const wr = sr + er
   224→   224→  out.handR.x = out.elbR.x + Math.sin(wr) * spec.foreArm
   225→   225→  out.handR.y = out.elbR.y + Math.cos(wr) * spec.foreArm
   226→   226→
   227→   227→  // blade: forearm direction rotated by wrist. forearm angle-from-+X = π/2 − (shR+elR);
   228→   228→  // positive wrist swings the tip toward facing (+X).
   229→   229→  out.bladeAngle = Math.PI / 2 - wr - pose[16]
   230→   230→  return out
   231→   231→}

   280→   280→function limb(ctx: CanvasRenderingContext2D, ax: number, ay: number, bx: number, by: number, cx: number, cy: number): void {
   281→   281→  ctx.beginPath()
   282→   282→  ctx.moveTo(ax, ay)
   283→   283→  ctx.lineTo(bx, by)
   284→   284→  ctx.lineTo(cx, cy)
   285→   285→  ctx.stroke()
   286→   286→}
   287→   287→
   288→   288→/** Fills a foot wedge (ankle → toe) with heel/toe widths. */
   289→   289→function fillFoot(
   290→   290→  ctx: CanvasRenderingContext2D,
   291→   291→  ax: number, ay: number, bx: number, by: number,
   292→   292→  color: string,
   293→   293→  alpha = 1,
   294→   294→): void {
   295→   295→  const dx = bx - ax
   296→   296→  const dy = by - ay
   297→   297→  const len = Math.hypot(dx, dy) || 1
   298→   298→  const nx = -dy / len
   299→   299→  const ny = dx / len
   300→   300→  ctx.save()
   301→   301→  ctx.globalAlpha *= alpha
   302→   302→  ctx.fillStyle = color
   303→   303→  ctx.beginPath()
   304→   304→  ctx.moveTo(ax + nx * 3.4, ay + ny * 3.4)
   305→   305→  ctx.lineTo(bx + nx * 2.5, by + ny * 2.5)
   306→   306→  ctx.lineTo(bx - nx * 2.5, by - ny * 2.5)
   307→   307→  ctx.lineTo(ax - nx * 3.4, ay - ny * 3.4)
   308→   308→  ctx.closePath()
   309→   309→  ctx.fill()
   310→   310→  ctx.beginPath()
   311→   311→  ctx.arc(ax, ay, 3.4, 0, Math.PI * 2)
   312→   312→  ctx.fill()
   313→   313→  ctx.beginPath()
   314→   314→  ctx.arc(bx, by, 2.3, 0, Math.PI * 2)
   315→   315→  ctx.fill()
   316→   316→  ctx.restore()
   317→   317→}
   318→   318→
   319→   319→/**
   320→   320→ * Draws the stickman. The ctx is expected to be pre-translated to the hip
   321→   321→ * position, flipped horizontally for facing, and scaled — geom is local space.
   322→   322→ */
   323→   323→export function drawStick(
   324→   324→  ctx: CanvasRenderingContext2D,
   325→   325→  g: SkelGeom,
   326→   326→  style: StickStyle,
   327→   327→  opts: DrawOpts = {},
   328→   328→): void {
   329→   329→  const alpha = opts.alpha ?? 1
   330→   330→  if (alpha <= 0.01) return
   331→   331→  const t = opts.time ?? 0
   332→   332→  const ghost = opts.ghost
   333→   333→  const lineW = style.lineW
   334→   334→
   335→   335→  ctx.save()
   336→   336→  ctx.globalAlpha *= alpha
   337→   337→  ctx.lineCap = 'round'
   338→   338→  ctx.lineJoin = 'round'
   339→   339→
   340→   340→  if (ghost) {
   341→   341→    // ---- afterimage pass: flat, cheap, gorgeous ----
   342→   342→    ctx.strokeStyle = ghost
   343→   343→    ctx.lineWidth = lineW
   344→   344→    const ga = opts.ghostAlpha ?? 1
   345→   345→    ctx.globalAlpha *= ga
   346→   346→    limb(ctx, g.hipPtL.x, g.hipPtL.y, g.kneeL.x, g.kneeL.y, g.ankleL.x, g.ankleL.y)
   347→   347→    limb(ctx, g.hipPtR.x, g.hipPtR.y, g.kneeR.x, g.kneeR.y, g.ankleR.x, g.ankleR.y)
   348→   348→    limb(ctx, g.shoulderL.x, g.shoulderL.y, g.elbL.x, g.elbL.y, g.handL.x, g.handL.y)
   349→   349→    limb(ctx, g.shoulderR.x, g.shoulderR.y, g.elbR.x, g.elbR.y, g.handR.x, g.handR.y)
   350→   350→    // feet + spine THROUGH the neck to the skull: the ghost reads as a
   351→   351→    // full connected figure (the old ghost stopped the spine at the chest,
   352→   352→    // so the stroked head circle floated above the body — "floating head";
   353→   353→    // the line now pierces the ring to its far edge so head + neck weld)
   354→   354→    ctx.beginPath()
   355→   355→    ctx.moveTo(g.ankleL.x, g.ankleL.y)
   356→   356→    ctx.lineTo(g.toeL.x, g.toeL.y)
   357→   357→    ctx.moveTo(g.ankleR.x, g.ankleR.y)
   358→   358→    ctx.lineTo(g.toeR.x, g.toeR.y)
   359→   359→    ctx.stroke()
   360→   360→    ctx.beginPath()
   361→   361→    ctx.moveTo(g.hip.x, g.hip.y)
   362→   362→    ctx.lineTo(g.waist.x, g.waist.y)
   363→   363→    ctx.lineTo(g.chest.x, g.chest.y)
   364→   364→    ctx.lineTo(g.neckTop.x, g.neckTop.y)
   365→   365→    ctx.lineTo(g.headC.x, g.headC.y)
   366→   366→    // extend THROUGH the skull ring (one headR past the center along the
   367→   367→    // neck axis) — the line reaches the ring's far edge: welded, not
   368→   368→    // dangling inside the circle
   369→   369→    {
   370→   370→      const hdx = g.headC.x - g.neckTop.x
   371→   371→      const hdy = g.headC.y - g.neckTop.y
   372→   372→      const hl = Math.hypot(hdx, hdy) || 1
   373→   373→      const hr = style === PLAYER_STYLE ? 8.8 : 8
   374→   374→      ctx.lineTo(g.headC.x + (hdx / hl) * hr, g.headC.y + (hdy / hl) * hr)
   375→   375→    }
   376→   376→    ctx.stroke()
   377→   377→    ctx.beginPath()
   378→   378→    ctx.arc(g.headC.x, g.headC.y, style === PLAYER_STYLE ? 8.8 : 8, 0, Math.PI * 2)
   379→   379→    ctx.stroke()
   380→   380→    ctx.restore()
   381→   381→    return
   382→   382→  }
   383→   383→
   384→   384→  // ---- full-black humanoid silhouette (player) ----
   385→   385→  if (style.silhouette) {
   386→   386→    drawSilhouette(ctx, g, style, opts, t, lineW)
   387→   387→    ctx.restore()
   388→   388→    return
   389→   389→  }
   390→   390→
   391→   391→  // ---- 1. outline pass (readability against the neon city) ----
   392→   392→  if (!opts.simple) {
   393→   393→    ctx.strokeStyle = style.outline
   394→   394→    ctx.lineWidth = lineW + 3.6
   395→   395→    limb(ctx, g.hipPtL.x, g.hipPtL.y, g.kneeL.x, g.kneeL.y, g.ankleL.x, g.ankleL.y)
   396→   396→    limb(ctx, g.hipPtR.x, g.hipPtR.y, g.kneeR.x, g.kneeR.y, g.ankleR.x, g.ankleR.y)
   397→   397→    limb(ctx, g.shoulderL.x, g.shoulderL.y, g.elbL.x, g.elbL.y, g.handL.x, g.handL.y)

   556→   556→/** Fist: a short capsule perpendicular to the grip axis — knuckles read. */
   557→   557→function fillFist(
   558→   558→  ctx: CanvasRenderingContext2D,
   559→   559→  x: number, y: number,
   560→   560→  ax: number, ay: number, // grip axis (unit)
   561→   561→  size: number,
   562→   562→  color: string,
   563→   563→  alpha = 1,
   564→   564→): void {
   565→   565→  // perpendicular of grip axis
   566→   566→  const px = -ay
   567→   567→  const py = ax
   568→   568→  ctx.save()
   569→   569→  ctx.globalAlpha *= alpha
   570→   570→  ctx.fillStyle = color
   571→   571→  ctx.beginPath()
   572→   572→  ctx.moveTo(x + px * size, y + py * size)
   573→   573→  ctx.lineTo(x + ax * size * 0.55 + px * size * 0.8, y + ay * size * 0.55 + py * size * 0.8)
   574→   574→  ctx.lineTo(x + ax * size * 0.55 - px * size * 0.8, y + ay * size * 0.55 - py * size * 0.8)
   575→   575→  ctx.lineTo(x - px * size, y - py * size)
   576→   576→  ctx.closePath()
   577→   577→  ctx.fill()
   578→   578→  ctx.beginPath()
   579→   579→  ctx.arc(x, y, size * 0.92, 0, Math.PI * 2)
   580→   580→  ctx.fill()
   581→   581→  ctx.restore()
   582→   582→}
   583→   583→
   584→   584→/** Rounded wedge between two points with per-end widths — torso/lats. */
   585→   585→function fillTaper(
   586→   586→  ctx: CanvasRenderingContext2D,
   587→   587→  ax: number, ay: number, bx: number, by: number,
   588→   588→  wa: number, wb: number,
   589→   589→  color: string,
   590→   590→  alpha = 1,
   591→   591→): void {
   592→   592→  const dx = bx - ax
   593→   593→  const dy = by - ay
   594→   594→  const len = Math.hypot(dx, dy) || 1
   595→   595→  const nx = -dy / len
   596→   596→  const ny = dx / len
   597→   597→  ctx.save()
   598→   598→  ctx.globalAlpha *= alpha
   599→   599→  ctx.fillStyle = color
   600→   600→  ctx.beginPath()
   601→   601→  ctx.moveTo(ax + nx * wa, ay + ny * wa)
   602→   602→  ctx.lineTo(bx + nx * wb, by + ny * wb)
   603→   603→  ctx.lineTo(bx - nx * wb, by - ny * wb)
   604→   604→  ctx.lineTo(ax - nx * wa, ay - ny * wa)
   605→   605→  ctx.closePath()
   606→   606→  ctx.fill()
   607→   607→  ctx.beginPath()
   608→   608→  ctx.arc(ax, ay, wa, 0, Math.PI * 2)
   609→   609→  ctx.fill()
   610→   610→  ctx.beginPath()
   611→   611→  ctx.arc(bx, by, wb, 0, Math.PI * 2)
   612→   612→  ctx.fill()
   613→   613→  ctx.restore()
   614→   614→}
   615→   615→
   616→   616→/**
   617→   617→ * Limbo/INSIDE-style silhouette, modeled on the articulated skeleton:
   618→   618→ * pelvis basin → abdominal taper → ribcage flare → trapezius → neck column
   619→   619→ * → skull; deltoid caps AT the glenohumeral joints; arms chain shoulder →
   620→   620→ * elbow → fist. Draw order back-to-front: rim → back limbs → torso → front
   621→   621→ * limbs → hands/feet → head → eyes → hit flash. Local space, ctx pre-flipped.
   622→   622→ */
   623→   623→function drawSilhouette(
   624→   624→  ctx: CanvasRenderingContext2D,
   625→   625→  g: SkelGeom,
   626→   626→  style: StickStyle,
   627→   627→  opts: DrawOpts,
   628→   628→  t: number,
   629→   629→  lineW: number,
   630→   630→): void {
   631→   631→  const black = style.body
   632→   632→  const rim = style.outline
   633→   633→  // back-limb depth shade: still silhouette-dark, but separable from the
   634→   634→  // pure-black mass (the old 82%-alpha-black vanished against the torso)
   635→   635→  const back = style.dark !== black ? style.dark : rgba(black, 0.82)
   636→   636→
   637→   637→  // ---- 0. rim pass (separates pure black from the night city) ----
   638→   638→  if (!opts.simple) {
   639→   639→    ctx.save()
   640→   640→    ctx.globalAlpha *= 0.78
   641→   641→    ctx.strokeStyle = rim
   642→   642→    ctx.lineWidth = lineW + 3.6
   643→   643→    ctx.lineCap = 'round'
   644→   644→    ctx.lineJoin = 'round'
   645→   645→    limb(ctx, g.hipPtL.x, g.hipPtL.y, g.kneeL.x, g.kneeL.y, g.ankleL.x, g.ankleL.y)
   646→   646→    limb(ctx, g.hipPtR.x, g.hipPtR.y, g.kneeR.x, g.kneeR.y, g.ankleR.x, g.ankleR.y)
   647→   647→    limb(ctx, g.shoulderL.x, g.shoulderL.y, g.elbL.x, g.elbL.y, g.handL.x, g.handL.y)
   648→   648→    limb(ctx, g.shoulderR.x, g.shoulderR.y, g.elbR.x, g.elbR.y, g.handR.x, g.handR.y)
   649→   649→    ctx.lineWidth = (lineW + 3.6) * 1.5
   650→   650→    ctx.beginPath()
   651→   651→    ctx.moveTo(g.hip.x, g.hip.y)
   652→   652→    ctx.lineTo(g.waist.x, g.waist.y)
   653→   653→    ctx.lineTo(g.chest.x, g.chest.y)
   654→   654→    ctx.lineTo(g.neckTop.x, g.neckTop.y)
   655→   655→    ctx.stroke()
   656→   656→    ctx.restore()
   657→   657→  }
   658→   658→
   659→   659→  // ---- 1. back limbs (recessed — depth cue), with anatomy ----
   660→   660→  // back leg from the far femur head: thigh (quad mass) → calf taper
   661→   661→  fillTaper(ctx, g.hipPtL.x, g.hipPtL.y, g.kneeL.x, g.kneeL.y, 5.8, 4.3, back)
   662→   662→  fillTaper(ctx, g.kneeL.x, g.kneeL.y, g.ankleL.x, g.ankleL.y, 4.0, 2.8, back)
   663→   663→  fillFoot(ctx, g.ankleL.x, g.ankleL.y, g.toeL.x, g.toeL.y, back)
   664→   664→  // knee cap (back) — centers the joint inside the calf silhouette
   665→   665→  ctx.fillStyle = back
   666→   666→  ctx.beginPath()
   667→   667→  ctx.arc(g.kneeL.x, g.kneeL.y, 4.2, 0, Math.PI * 2)
   668→   668→  ctx.fill()
   669→   669→  // rear deltoid cap — the back arm's root reads at the far shoulder
   670→   670→  ctx.beginPath()
   671→   671→  ctx.arc(g.shoulderL.x, g.shoulderL.y, 4.6, 0, Math.PI * 2)
   672→   672→  ctx.fill()
   673→   673→  // back arm: triceps taper + forearm, from the far glenohumeral joint
   674→   674→  fillTaper(ctx, g.shoulderL.x, g.shoulderL.y, g.elbL.x, g.elbL.y, 4.9, 3.1, back)
   675→   675→  fillTaper(ctx, g.elbL.x, g.elbL.y, g.handL.x, g.handL.y, 2.9, 1.8, back)
   676→   676→
   677→   677→  // ---- 2. torso: pelvis basin + abdomen + ribcage + trapezius + neck ----
   678→   678→  // pelvis: WIDE at the hip (iliac crest), tapering hard to the waist —
   679→   679→  // strong region contrast so the pelvis reads as a basin
   680→   680→  fillTaper(ctx, g.hip.x, g.hip.y, g.waist.x, g.waist.y, 9.6, 6.6, black)
   681→   681→  // abdomen → ribcage: narrow at the navel, flaring WIDE at the chest —
   682→   682→  // the V-taper of the latissimus into the ribcage
   683→   683→  fillTaper(ctx, g.waist.x, g.waist.y, g.chest.x, g.chest.y, 6.6, 10.4, black)
   684→   684→  // trapezius: the shoulder girdle — slopes from the skull base out to BOTH
   685→   685→  // deltoid caps (near + far). This polygon is what ties arms to shoulders.
   686→   686→  {
   687→   687→    const ndx = g.neckTop.x - g.chest.x
   688→   688→    const ndy = g.neckTop.y - g.chest.y
   689→   689→    const nl = Math.hypot(ndx, ndy) || 1
   690→   690→    const px = -ndy / nl
   691→   691→    const py = ndx / nl
   692→   692→    ctx.save()
   693→   693→    ctx.fillStyle = black
   694→   694→    ctx.beginPath()
   695→   695→    ctx.moveTo(g.neckTop.x + px * 3.4, g.neckTop.y + py * 3.4)
   696→   696→    ctx.lineTo(g.neckTop.x - px * 3.4, g.neckTop.y - py * 3.4)
   697→   697→    // out to the FAR shoulder cap (behind + lower — 2.8D depth read)
   698→   698→    ctx.lineTo(g.shoulderL.x + px * 4.8, g.shoulderL.y + py * 4.8)
   699→   699→    ctx.lineTo(g.shoulderL.x - px * 3.4, g.shoulderL.y - py * 3.4)
   700→   700→    // through the shoulder line to the NEAR shoulder cap
   701→   701→    ctx.lineTo(g.shoulderR.x - px * 5.2, g.shoulderR.y - py * 5.2)
   702→   702→    ctx.lineTo(g.shoulderR.x + px * 5.8, g.shoulderR.y + py * 5.8)
   703→   703→    ctx.closePath()
   704→   704→    ctx.fill()
   705→   705→    ctx.restore()
   706→   706→  }
   707→   707→  // neck column: chest (shoulder line) up to the skull base — a real neck,
   708→   708→  // wide at the base (sternocleidomastoid + trap roots) tapering to atlas
   709→   709→  fillTaper(ctx, g.chest.x, g.chest.y, g.neckTop.x, g.neckTop.y, 4.6, 3.3, black)
   710→   710→
   711→   711→  // ---- 3. front limbs (solid black, anatomical) ----
   712→   712→  // front leg from the near femur head: quad → knee → calf
   713→   713→  fillTaper(ctx, g.hipPtR.x, g.hipPtR.y, g.kneeR.x, g.kneeR.y, 6.5, 4.7, black)
   714→   714→  fillTaper(ctx, g.kneeR.x, g.kneeR.y, g.ankleR.x, g.ankleR.y, 4.4, 2.9, black)
   715→   715→  fillFoot(ctx, g.ankleR.x, g.ankleR.y, g.toeR.x, g.toeR.y, black)
   716→   716→  // knee cap — centers the joint in the front calf silhouette
   717→   717→  ctx.fillStyle = black
   718→   718→  ctx.beginPath()
   719→   719→  ctx.arc(g.kneeR.x, g.kneeR.y, 4.6, 0, Math.PI * 2)
   720→   720→  ctx.fill()
   721→   721→  // deltoid cap ON the glenohumeral joint — the arm reads from the shoulder
   722→   722→  ctx.beginPath()
   723→   723→  ctx.arc(g.shoulderR.x, g.shoulderR.y, 5.7, 0, Math.PI * 2)
   724→   724→  ctx.fill()
   725→   725→  // bicep → forearm taper (anatomical: deltoid .055H → elbow .035H →
   726→   726→  // wrist .020H — the forearm tapers STEEPER than the humerus)
   727→   727→  fillTaper(ctx, g.shoulderR.x, g.shoulderR.y, g.elbR.x, g.elbR.y, 5.7, 3.6, black)
   728→   728→  fillTaper(ctx, g.elbR.x, g.elbR.y, g.handR.x, g.handR.y, 3.4, 2.1, black)
   729→   729→  // elbow sphere (silhouette joint)
   730→   730→  ctx.beginPath()
   731→   731→  ctx.arc(g.elbR.x, g.elbR.y, 3.7, 0, Math.PI * 2)
   732→   732→  ctx.fill()
   733→   733→
   734→   734→  // ---- 4. hands: fists oriented on the blade axis (grip anatomy) ----
   735→   735→  const bax = Math.cos(g.bladeAngle)
   736→   736→  const bay = Math.sin(g.bladeAngle)
   737→   737→  // off-hand fist (rear grip — back shade, under the sword hand)
   738→   738→  fillFist(ctx, g.handL.x, g.handL.y, bax, bay, 2.9, back)
   739→   739→  // sword hand — big knuckle fist across the tsuka
   740→   740→  fillFist(ctx, g.handR.x, g.handR.y, bax, bay, 3.4, black)
   741→   741→  // grip readability: a whisper of rim on the sword fist so the grip reads
   742→   742→  // against the black mass (playtest note: hands vanished at gameplay zoom)
   743→   743→  if (!opts.simple) {
   744→   744→    ctx.save()
   745→   745→    ctx.strokeStyle = rgba(rim, 0.55)
   746→   746→    ctx.lineWidth = 1.2
   747→   747→    ctx.beginPath()
   748→   748→    ctx.arc(g.handR.x, g.handR.y, 3.6, -1.2, 0.9)
   749→   749→    ctx.stroke()
   750→   750→    ctx.restore()
   751→   751→  }
   752→   752→
   753→   753→  // ---- 4.5 BLADE-CAST RIM LIGHT (2.8D dynamic lighting): the glowing
   754→   754→  // edge lights the sword-side of the body — arm, torso leading edge,
   755→   755→  // front thigh. An additive whisper that separates the carrier from the
   756→   756→  // night and sells the blade as a real 3D light source in the scene. ----
   757→   757→  const el = opts.edgeLight
   758→   758→  if (el && el.a > 0.03) {
   759→   759→    ctx.save()
   760→   760→    ctx.globalCompositeOperation = 'lighter'
   761→   761→    ctx.lineCap = 'round'
   762→   762→    ctx.strokeStyle = rgba(el.color, Math.min(0.85, el.a))
   763→   763→    ctx.lineWidth = 2
   764→   764→    // sword arm chain (shoulder → elbow → hand) — closest to the source
   765→   765→    ctx.beginPath()
   766→   766→    ctx.moveTo(g.shoulderR.x, g.shoulderR.y)
   767→   767→    ctx.lineTo(g.elbR.x, g.elbR.y)
   768→   768→    ctx.lineTo(g.handR.x, g.handR.y)
   769→   769→    ctx.stroke()
   770→   770→    // torso leading edge (the lit side of the ribcage)
   771→   771→    ctx.beginPath()
   772→   772→    ctx.moveTo(g.hip.x + 5.2, g.hip.y - 1)
   773→   773→    ctx.quadraticCurveTo(g.waist.x + 6.5, g.waist.y - 1, g.chest.x + 8.2, g.chest.y + 1.5)
   774→   774→    ctx.stroke()
   775→   775→    // front thigh quad — the forward leg catches the spill
   776→   776→    ctx.beginPath()
   777→   777→    ctx.moveTo(g.hipPtR.x + 4.4, g.hipPtR.y + 1)
   778→   778→    ctx.lineTo(g.kneeR.x + 3.4, g.kneeR.y - 1)
   779→   779→    ctx.stroke()
   780→   780→    // off-hand knuckles kiss the light when gripping the tsuka
   781→   781→    ctx.lineWidth = 1.4
   782→   782→    ctx.beginPath()
   783→   783→    ctx.arc(g.handL.x, g.handL.y, 2.6, 0, Math.PI * 2)
   784→   784→    ctx.stroke()
   785→   785→    ctx.restore()
   786→   786→  }
   787→   787→
   788→   788→  // ---- 5. head: skull dome + jaw taper + swept hood crest ----
   789→   789→  ctx.save()
   790→   790→  ctx.fillStyle = black
   791→   791→  // skull — egg tilted with headAng (slightly taller than wide)
   792→   792→  ctx.translate(g.headC.x, g.headC.y)
   793→   793→  ctx.rotate(g.headAng * 0.35)
   794→   794→  ctx.beginPath()
   795→   795→  ctx.ellipse(0, -0.4, 8.2, 9.5, 0, 0, Math.PI * 2)
   796→   796→  ctx.fill()
   797→   797→  // jaw taper — chin mass forward-low (reads as a jaw, not a ball)
   798→   798→  ctx.beginPath()
   799→   799→  ctx.moveTo(2.6, 5.6)
   800→   800→  ctx.quadraticCurveTo(1.2, 8.6, -1.8, 8.2)
   801→   801→  ctx.quadraticCurveTo(-4.2, 7.6, -4.6, 4.4)
   802→   802→  ctx.closePath()
   803→   803→  ctx.fill()
   804→   804→  // hood/kabuto crest — swept back over the crown
   805→   805→  ctx.beginPath()
   806→   806→  ctx.moveTo(-2.0, -8.2)
   807→   807→  ctx.quadraticCurveTo(-7.5, -12.6, -12.6, -9.4)
   808→   808→  ctx.quadraticCurveTo(-8.8, -6.4, -5.2, -5.2)
   809→   809→  ctx.closePath()
   810→   810→  ctx.fill()
   811→   811→  ctx.restore()
   812→   812→  // rim arc on the back of the skull — the one lit edge of the head
   813→   813→  if (!opts.simple) {
   814→   814→    ctx.save()
   815→   815→    ctx.strokeStyle = rgba(rim, 0.5)
   816→   816→    ctx.lineWidth = 1.6
   817→   817→    ctx.beginPath()
   818→   818→    ctx.arc(g.headC.x - 1.5, g.headC.y - 0.5, 8.6, Math.PI * 0.62, Math.PI * 1.32)
   819→   819→    ctx.stroke()
   820→   820→    ctx.restore()
   821→   821→  }
   822→   822→
   823→   823→  // ---- 6. glowing eyes (the only light on the body) ----
   824→   824→  if (!opts.simple) {
   825→   825→    const blink = 0.85 + 0.15 * Math.sin(t * 3.1)
   826→   826→    ctx.save()
   827→   827→    ctx.globalCompositeOperation = 'lighter'
   828→   828→    ctx.fillStyle = rgba(style.glow, 0.95 * blink)
   829→   829→    // two eye slits on the forward face, angled with intent
   830→   830→    ctx.beginPath()
   831→   831→    ctx.ellipse(g.headC.x + 3.4, g.headC.y - 1.4, 2.0, 1.15, -0.18, 0, Math.PI * 2)
   832→   832→    ctx.fill()
   833→   833→    ctx.beginPath()
   834→   834→    ctx.ellipse(g.headC.x + 6.4, g.headC.y - 2.2, 1.25, 0.9, -0.18, 0, Math.PI * 2)
   835→   835→    ctx.fill()
   836→   836→    ctx.restore()
   837→   837→  }
   838→   838→
   839→   839→  // ---- 7. hit flash overlay (whole body flashes) ----
   840→   840→  const flash = clamp01(opts.flash ?? 0)
   841→   841→  if (flash > 0.02) {
   842→   842→    const fc = opts.flashColor ?? '#ffffff'
   843→   843→    fillTaper(ctx, g.hipPtR.x, g.hipPtR.y, g.kneeR.x, g.kneeR.y, 6.5, 4.7, fc, flash)
   844→   844→    fillTaper(ctx, g.kneeR.x, g.kneeR.y, g.ankleR.x, g.ankleR.y, 4.4, 2.9, fc, flash)
   845→   845→    fillTaper(ctx, g.hipPtL.x, g.hipPtL.y, g.kneeL.x, g.kneeL.y, 5.8, 4.3, fc, flash * 0.8)
   846→   846→    fillTaper(ctx, g.shoulderR.x, g.shoulderR.y, g.elbR.x, g.elbR.y, 4.9, 3.6, fc, flash)
   847→   847→    fillTaper(ctx, g.elbR.x, g.elbR.y, g.handR.x, g.handR.y, 3.5, 2.6, fc, flash)
   848→   848→    fillTaper(ctx, g.waist.x, g.waist.y, g.chest.x, g.chest.y, 6.6, 10.4, fc, flash)
   849→   849→    fillTaper(ctx, g.hip.x, g.hip.y, g.waist.x, g.waist.y, 9.6, 6.6, fc, flash)
   850→   850→    ctx.fillStyle = fc
   851→   851→    ctx.save()
   852→   852→    ctx.globalAlpha *= flash
   853→   853→    ctx.beginPath()
   854→   854→    ctx.ellipse(g.headC.x, g.headC.y - 0.4, 8.2, 9.5, 0, 0, Math.PI * 2)
   855→   855→    ctx.fill()
   856→   856→    ctx.restore()
   857→   857→  }
   858→   858→}
