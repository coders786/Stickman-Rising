     1→/** STICKMAN RISING — the modular combo moveset.
     2→ *
     3→ * Each attack = a keyframed clip (anticipation → strike → follow-through →
     4→ * recovery) + hitbox window (the physical blade sweep is the hitbox) + trail
     5→ * window + cancel window + legal follow-up chain. Directional inputs,
     6→ * dash-cancels and buffers are resolved by the Player combo machine.
     7→ *
     8→ * Angle conventions (see pose.ts): arm angles measured from straight-down,
     9→ * positive swings toward facing. Blade direction = π/2 − (shR+elR+wrR).
    10→ */
    11→
    12→import { Clip, keyframe, makeClip } from '../anim/animator'
    13→import type { SfxName } from '../audio/audio'
    14→
    15→export type AttackKind =
    16→  | 'light' | 'heavy' | 'launcher' | 'air' | 'slam' | 'dash'
    17→  | 'spin' | 'charged' | 'riposte' | 'finisher' | 'ability'
    18→
    19→export interface AttackHit {
    20→  start: number
    21→  end: number
    22→  dmg: number
    23→  stagger: number
    24→  kb: number
    25→  kbY?: number
    26→  launch?: number
    27→  juggle?: number
    28→  hitstop: number
    29→  shake: number
    30→  heavy: boolean
    31→  aoe?: number
    32→  multiHit?: number
    33→  breakGuard?: boolean
    34→}
    35→
    36→export interface AttackDef {
    37→  id: string
    38→  label: string
    39→  kind: AttackKind
    40→  clip: Clip
    41→  hit: AttackHit
    42→  trail: { start: number; end: number }
    43→  swingSfx: { t: number; name: SfxName }
    44→  cancel: number
    45→  chain: string[]
    46→  moveX?: number
    47→  moveXAt?: number
    48→  moveY?: number
    49→  moveYAt?: number
    50→  requires: 'ground' | 'air' | 'any'
    51→  /** gravity scale while the attack is active (air-combo float) */
    52→  float?: number
    53→  /** AoE + fx fire when the player touches ground mid-attack (slams / dive) */
    54→  slamImpact?: { r: number; dmg: number; launch: number }
    55→}

   769→/** Runtime info passed to the world when a blade connects. */
   770→export interface AttackRuntimeInfo {
   771→  id: string
   772→  dmg: number
   773→  heavy: boolean
   774→  stagger: number
   775→  kb: number
   776→  kbY: number
   777→  launch: number
   778→  juggle: number
   779→  hitstop: number
   780→  shake: number
   781→  aoe: number
   782→  breakGuard: boolean
   783→  kind: AttackKind
   784→}
   785→
   786→export function makeRuntimeInfo(def: AttackDef, dmgMult: number): AttackRuntimeInfo {
   787→  return {
   788→    id: def.id,
   789→    dmg: def.hit.dmg * dmgMult,
   790→    heavy: def.hit.heavy,
   791→    stagger: def.hit.stagger,
   792→    kb: def.hit.kb,
   793→    kbY: def.hit.kbY ?? 0,
   794→    launch: def.hit.launch ?? 0,
   795→    juggle: def.hit.juggle ?? 0,
   796→    hitstop: def.hit.hitstop,
   797→    shake: def.hit.shake,
   798→    aoe: def.hit.aoe ?? 0,
   799→    breakGuard: def.hit.breakGuard ?? false,
   800→    kind: def.kind,
   801→  }
   802→}
