    15→/** circle vs swept blade quad (prev segment + current segment) — prevents tunneling */
    16→function circleHitsSweptQuad(
    17→  cx: number, cy: number, r: number,
    18→  pbx: number, pby: number, ptx: number, pty: number,
    19→  bx: number, by: number, tx: number, ty: number,
    20→): boolean {
    21→  // current + previous blade segments
    22→  if (segHitsCircle(bx, by, tx, ty, cx, cy, r)) return true
    23→  if (segHitsCircle(pbx, pby, ptx, pty, cx, cy, r)) return true
    24→  // swept edges (tip path + base path)
    25→  if (segHitsCircle(ptx, pty, tx, ty, cx, cy, r)) return true
    26→  if (segHitsCircle(pbx, pby, bx, by, cx, cy, r)) return true
    27→  // circle center inside the swept quad (spins / wide arcs)
    28→  if (pointInTri(cx, cy, pbx, pby, ptx, pty, tx, ty)) return true
    29→  if (pointInTri(cx, cy, pbx, pby, tx, ty, bx, by)) return true
    30→  return false
    31→}

    33→function pointInTri(px: number, py: number, x1: number, y1: number, x2: number, y2: number, x3: number, y3: number): boolean {
    34→  const d1 = (px - x2) * (y1 - y2) - (x1 - x2) * (py - y2)
    35→  const d2 = (px - x3) * (y2 - y3) - (x2 - x3) * (py - y3)
    36→  const d3 = (px - x1) * (y3 - y1) - (x3 - x1) * (py - y1)
    37→  const hasNeg = d1 < 0 || d2 < 0 || d3 < 0
    38→  const hasPos = d1 > 0 || d2 > 0 || d3 > 0
    39→  return !(hasNeg && hasPos)
    40→}

   206→  update(dt: number, w: WorldRef): void {
   207→    this.iframes = Math.max(0, this.iframes - dt)
   208→    this.flashT = Math.max(0, this.flashT - dt * 5)
   209→    this.swingGlow = Math.max(0, this.swingGlow - dt * 5)
   210→    this.riposteT = Math.max(0, this.riposteT - dt)
   211→    this.energy = Math.min(this.maxEnergy, this.energy + dt * 3)
   212→
   213→    switch (this.state) {
   214→      case 'move': this.updateMove(dt, w); break
   215→      case 'attack': this.updateAttack(dt, w); break
   216→      case 'dash': this.updateDash(dt, w); break
   217→      case 'parry': this.updateParry(dt, w); break
   218→      case 'charge': this.updateCharge(dt, w); break
   219→      case 'hurt': this.updateHurt(dt, w); break
   220→      case 'dead': this.updateDead(dt, w); break
   221→      case 'scripted': this.updateScripted(dt); break
   222→    }
   223→
   224→    // physics integration (shared)
   225→    if (this.state !== 'dash') {
   226→      const g = this.state === 'attack' && this.action && this.action.def.float
   227→        ? GRAVITY * this.action.def.float
   228→        : GRAVITY
   229→      this.vy += g * dt
   230→    }
   231→    this.x += this.vx * dt
   232→    this.y += this.vy * dt
   233→
   234→    // ground
   235→    const wasAir = !this.grounded
   236→    if (this.y >= 0) {
   237→      this.y = 0
   238→      if (this.vy > 140) {
   239→        this.landSpring = clamp(this.vy / 700, 0, 1)
   240→        if (wasAir) this.onLand(w)
   241→      } else if (wasAir) {
   242→        this.landSpring = 0.4
   243→        this.onLand(w)
   244→      }
   245→      this.vy = 0
   246→      this.grounded = true
   247→      this.jumpsUsed = 0
   248→      this.airDashUsed = false
   249→    } else {
   250→      this.grounded = false
   251→    }
   252→    // arena bounds
   253→    if (this.x < w.arenaMinX + 20) {
   254→      this.x = w.arenaMinX + 20
   255→      if (this.vx < -200) w.fx.dust(this.x - 12, 0, 5)
   256→      this.vx = Math.max(0, this.vx)
   257→    }
   258→    if (this.x > w.arenaMaxX - 20) {
   259→      this.x = w.arenaMaxX - 20
   260→      if (this.vx > 200) w.fx.dust(this.x + 12, 0, 5)
   261→      this.vx = Math.min(0, this.vx)
   262→    }
   263→
   264→    this.dashCd = Math.max(0, this.dashCd - dt)
   265→    this.dashRecharge = Math.max(0, this.dashRecharge - dt)
   266→    if (this.dashRecharge <= 0) this.dashConsec = 0
   267→
   268→    // ability script
   269→    if (this.ability) this.updateAbility(dt, w)
   270→
   271→    // animation + solve
   272→    this.animator.update(dt)
   273→    this.applyProcedural(dt)
   274→    solveSkeleton(this.pose, PLAYER_SPEC, this.geom)
   275→    this.updateBladeWorld()
   276→  }

   278→  private onLand(w: WorldRef): void {
   279→    w.audio.play('land', { volume: 0.5 })
   280→    w.fx.dust(this.x, 0, 7)
   281→    // slam impact while attacking
   282→    const a = this.action
   283→    if (a && a.def.slamImpact && !a.impactDone) {
   284→      a.impactDone = true
   285→      this.slamAoe(w, a.def.slamImpact.r, a.def.slamImpact.dmg, a.def.slamImpact.launch)
   286→    }
   287→    if (this.state === 'move' && this.lastGrounded === false) {
   288→      this.animator.play(LAND_CLIP, { blend: 0.03 })
   289→    }
   290→    this.airIndex = 0
   291→  }

   293→  private slamAoe(w: WorldRef, r: number, dmg: number, launch: number): void {
   294→    const info: AttackRuntimeInfo = {
   295→      id: 'slam', dmg: dmg * this.sword.stats.dmg, heavy: true,
   296→      stagger: 0.5, kb: 240, kbY: 0, launch, juggle: 0,
   297→      hitstop: 0.11, shake: 0.45, aoe: r, breakGuard: true, kind: 'heavy',
   298→    }
   299→    const colors = this.sword.colors
   300→    w.fx.shockwave(this.x, 0, r, colors.glow, 0.5, 8)
   301→    w.fx.groundCrack(this.x, 0, r * 0.6)
   302→    w.fx.burst(this.x, -6, 14, colors.glow, 340, 0.5, 'spark')
   303→    w.fx.screenFlash(colors.trail, 0.16, 5)
   304→    w.addHitstop(0.09, 0.06)
   305→    w.cam.addTrauma(0.4)
   306→    w.audio.play('slam', { intensity: 1 })
   307→    const bx = this.x
   308→    for (const e of w.enemies) {
   309→      if (e.dead || e.spawning) continue
   310→      const dx = e.x - bx
   311→      if (Math.abs(dx) < r && e.y > -140) {
   312→        w.hitEnemy(e.id, { bx: bx - 20, by: -20, tx: bx + 20, ty: -20 }, info)
   313→      }
   314→    }
   315→  }

   321→  private updateMove(dt: number, w: WorldRef): void {
   322→    const input = w.input
   323→    const ax = input.axisX()
   324→    const ay = input.axisY()
   325→
   326→    // facing follows input
   327→    if (ax !== 0) this.face = ax > 0 ? 1 : -1
   328→
   329→    // horizontal movement
   330→    if (ax !== 0) {
   331→      const accel = this.grounded ? ACCEL : AIR_ACCEL
   332→      this.vx += ax * accel * dt
   333→      this.vx = clamp(this.vx, -MOVE_MAX, MOVE_MAX)
   334→    } else if (this.grounded) {
   335→      const f = FRICTION * dt
   336→      this.vx = Math.abs(this.vx) <= f ? 0 : this.vx - Math.sign(this.vx) * f
   337→    } else {
   338→      this.vx = damp(this.vx, 0, 0.4, dt)
   339→    }
   340→
   341→    // locomotion clip selection
   342→    const speed = Math.abs(this.vx)
   343→    if (this.grounded) {
   344→      if (speed > 40) {
   345→        this.animator.play(RUN_CLIP, { blend: 0.09, speed: clamp(speed / MOVE_MAX, 0.65, 2.1) })
   346→      } else if (this.vy === 0) {
   347→        this.animator.play(IDLE_CLIP, { blend: 0.16 })
   348→      }
   349→    } else if (this.vy < -60) {
   350→      this.animator.play(JUMP_CLIP, { blend: 0.06 })
   351→    } else {
   352→      this.animator.play(FALL_CLIP, { blend: 0.12 })
   353→    }
   354→
   355→    // ---- actions ----
   356→    if (this.tryCombatInputs(w, ax, ay)) return
   357→  }

   359→  /** returns true if a new state was entered */
   360→  private tryCombatInputs(w: WorldRef, ax: number, ay: number): boolean {
   361→    const input = w.input
   362→
   363→    // dash (anytime, from move)
   364→    if ((input.pressed('dash') || input.buffered('dash')) && this.canDash()) {
   365→      this.startDash(w, ax, ay)
   366→      return true
   367→    }
   368→    // jump (grounded = jump 1, plus one air jump)
   369→    if (input.pressed('jump') || input.buffered('jump', 0.18)) {
   370→      if (this.jumpsUsed < 2) {
   371→        this.doJump(w)
   372→        return false
   373→      }
   374→    }
   375→    // parry
   376→    if (input.pressed('parry')) {
   377→      this.enterParry(w)
   378→      return true
   379→    }
   380→    // ability
   381→    if (input.pressed('ability')) {
   382→      if (this.energy >= this.sword.ability.cost) {
   383→        this.startAbility(w)
   384→        return true
   385→      }
   386→    }
   387→    // ult is handled by Game (cinematic)
   388→
   389→    // heavy: charge vs tap
   390→    if (input.down('heavy') && this.grounded) {
   391→      this.enterCharge(w)
   392→      return true
   393→    }
   394→    if (input.pressed('heavy') || input.buffered('heavy')) {
   395→      input.takeBuffered('heavy')
   396→      this.startAttack(this.pickHeavy(w), w)
   397→      return true
   398→    }
   399→    // light
   400→    if (input.pressed('light') || input.buffered('light')) {
   401→      input.takeBuffered('light')
   402→      const id = this.riposteT > 0 ? 'riposte' : this.pickLight(w, ax, ay)
   403→      this.startAttack(ATTACKS[id], w)
   404→      return true
   405→    }
   406→    return false
   407→  }

   409→  private doJump(w: WorldRef): void {
   410→    w.input.takeBuffered('jump')
   411→    this.jumpsUsed++
   412→    this.vy = this.jumpsUsed === 1 ? JUMP_V : AIR_JUMP_V
   413→    this.grounded = false
   414→    w.audio.play('jump', { volume: 0.4 })
   415→    w.fx.dust(this.x, 0, 4)
   416→    if (this.jumpsUsed === 2) {
   417→      w.fx.burst(this.x, this.y - 20, 6, '#9fb4c4', 160, 0.4, 'dot')
   418→    }
   419→  }

   425→  private canDash(): boolean {
   426→    if (this.dashCd > 0 || this.dashRecharge > 0) return false
   427→    if (!this.grounded && this.airDashUsed) return false
   428→    return true
   429→  }
   430→
   431→  startDash(w: WorldRef, ax: number, ay: number): void {
   432→    w.input.takeBuffered('dash')
   433→    let dx = ax
   434→    let dy = ay
   435→    if (dx === 0 && dy === 0) dx = this.face
   436→    const len = Math.hypot(dx, dy) || 1
   437→    dx /= len
   438→    dy /= len
   439→    if (dx !== 0) this.face = dx > 0 ? 1 : -1
   440→    this.dashDx = dx
   441→    this.dashDy = dy
   442→    this.state = 'dash'
   443→    this.dashT = DASH_TIME
   444→    this.dashCd = DASH_CD
   445→    this.dashConsec++
   446→    if (this.dashConsec >= 2) this.dashRecharge = DASH_RECHARGE
   447→    if (!this.grounded) this.airDashUsed = true
   448→    this.iframes = Math.max(this.iframes, 0.12)
   449→    this.action = null
   450→    this.charging = false
   451→    this.vx = dx * DASH_SPEED
   452→    this.vy = dy * DASH_SPEED * (this.grounded && dy < -0.4 ? 0.55 : 1)
   453→    this.animator.play(this.grounded ? DASH_CLIP : DASH_AIR_CLIP, { blend: 0.03 })
   454→    w.audio.play('dash', { pitch: rand(0.95, 1.1) })
   455→    const colors = this.sword.colors
   456→    w.fx.speedStreaks(this.x, this.y - 40, dx * DASH_SPEED * 0.9, 6, rgba(colors.trail, 1))
   457→    w.fx.dust(this.x, this.grounded ? 0 : this.y, 5)
   458→    w.cam.addTrauma(0.06)
   459→  }
   460→
   461→  private updateDash(dt: number, w: WorldRef): void {
   462→    this.dashT -= dt
   463→    // afterimages
   464→    this.afterimgTimer -= dt
   465→    if (this.afterimgTimer <= 0) {
   466→      this.afterimgTimer = 0.026
   467→      w.fx.afterimage(this.pose, this.x, this.hipY, this.face, this.sword.colors.trail, 0.3, SCALE)
   468→    }
   469→    // maintain velocity with slight decay
   470→    const k = 1 - this.dashT / DASH_TIME
   471→    const sp = DASH_SPEED * (1 - k * 0.25)
   472→    this.vx = this.dashDx * sp
   473→    this.vy = this.dashDy * sp * (this.grounded ? (this.dashDy < 0 ? 0.4 : 1) : 1)
   474→
   475→    // dash attack
   476→    const input = w.input
   477→    if (input.pressed('light') || input.buffered('light')) {
   478→      input.takeBuffered('light')
   479→      this.startAttack(ATTACKS.dashSlash, w)
   480→      return
   481→    }
   482→    if (input.pressed('heavy') || input.buffered('heavy')) {
   483→      if (!this.grounded) {
   484→        input.takeBuffered('heavy')
   485→        this.startAttack(ATTACKS.slam, w)
   486→        return
   487→      }
   488→      input.takeBuffered('heavy')
   489→      this.startAttack(ATTACKS.dashHeavy, w)
   490→      return
   491→    }
   492→    if (this.dashT <= 0) {
   493→      this.state = 'move'
   494→      this.vx *= 0.55
   495→      if (!this.grounded) this.vy *= 0.5
   496→    }
   497→  }

   503→  private pickLight(w: WorldRef, ax: number, ay: number): string {
   504→    // contextual branches
   505→    if (ay < -0.4 && this.grounded) return 'launcher'
   506→    if (ay > 0.4 && this.grounded) return 'spin'
   507→    if (this.grounded && ax !== 0 && Math.sign(ax) !== this.face) return 'retreat'
   508→    if (!this.grounded) {
   509→      if (ay > 0.4) return 'dive'
   510→      this.airIndex = (this.airIndex % 3) + 1
   511→      return `a${this.airIndex}`
   512→    }
   513→    // ground light chain (with time decay)
   514→    if (w.time - this.lastLightAt > 1.1) this.lightIndex = 0
   515→    this.lightIndex = Math.min(this.lightIndex + 1, 5)
   516→    return `l${Math.min(this.lightIndex, 5)}`
   517→  }

   523→  startAttack(def: AttackDef, w: WorldRef, fromParry = false): void {
   524→    if (!def) return
   525→    // requirement checks
   526→    if (def.requires === 'ground' && !this.grounded) def = ATTACKS.a1
   527→    if (def.requires === 'air' && this.grounded) def = ATTACKS.l1
   528→
   529→    // face toward held direction at attack start
   530→    const ax = w.input.axisX()
   531→    if (ax !== 0 && !fromParry) this.face = ax > 0 ? 1 : -1
   532→
   533→    const trail = w.fx.startTrail(this.sword.colors.trail, this.sword.colors.trail2, this.sword.trailWidth)
   534→    this.hasPrevBlade = false
   535→    this.action = {
   536→      def,
   537→      animT: 0,
   538→      hitMap: new Map(),
   539→      trail,
   540→      sfxDone: false,
   541→      moveDone: false,
   542→      moveYDone: false,
   543→      impactDone: false,
   544→      fromParry,
   545→    }
   546→    if (fromParry) this.riposteT = 0
   547→    this.state = 'attack'
   548→    this.charging = false
   549→    if (def === ATTACKS.l1 || def === ATTACKS.l2 || def === ATTACKS.l3 || def === ATTACKS.l4 || def === ATTACKS.l5) {
   550→      this.lastLightAt = w.time
   551→    }
   552→    this.animator.play(def.clip, { blend: 0.045, speed: this.atkSpeed })
   553→    // slight ground friction during attacks
   554→  }
   555→
   556→  private updateAttack(dt: number, w: WorldRef): void {
   557→    const a = this.action
   558→    if (!a) {
   559→      this.state = 'move'
   560→      return
   561→    }
   562→    const def = a.def
   563→    a.animT = this.animator.getTime()
   564→    const animT = a.animT
   565→    const spd = this.atkSpeed
   566→
   567→    // ground friction / air drag during attack
   568→    if (this.grounded) {
   569→      const f = 900 * dt
   570→      this.vx = Math.abs(this.vx) <= f ? 0 : this.vx - Math.sign(this.vx) * f
   571→    }
   572→
   573→    // sfx event
   574→    if (!a.sfxDone && animT >= def.swingSfx.t) {
   575→      a.sfxDone = true
   576→      w.audio.play(def.swingSfx.name, { pitch: rand(0.94, 1.08) })
   577→    }
   578→    // forward impulse
   579→    if (!a.moveDone && def.moveX !== undefined && animT >= (def.moveXAt ?? def.hit.start)) {
   580→      a.moveDone = true
   581→      this.vx = this.face * def.moveX * (def.kind === 'dash' ? 1 : 1)
   582→    }
   583→    // vertical impulse (launcher hop / slam plunge)
   584→    if (!a.moveYDone && def.moveY !== undefined && animT >= (def.moveYAt ?? 0)) {
   585→      a.moveYDone = true
   586→      this.vy = def.moveY
   587→      if (def.moveY < 0) {
   588→        this.grounded = false
   589→        w.fx.dust(this.x, 0, 6)
   590→      }
   591→    }
   592→
   593→    // hit window: sample blade segment & test enemies (swept, anti-tunneling)
   594→    const h = def.hit
   595→    if (animT >= h.start && animT <= h.end) {
   596→      this.swingGlow = 1
   597→      const info = makeRuntimeInfo(def, this.sword.stats.dmg)
   598→      for (const e of w.enemies) {
   599→        if (e.dead || e.spawning) continue
   600→        const last = a.hitMap.get(e.id)
   601→        const reArm = h.multiHit ? (last === undefined || animT - last >= h.multiHit) : last === undefined
   602→        if (!reArm) continue
   603→        const cy = e.y - 34 * e.scale
   604→        const r = 26 * e.scale + 6
   605→        const hit = this.hasPrevBlade
   606→          ? circleHitsSweptQuad(e.x, cy, r, this.prevBlade.bx, this.prevBlade.by, this.prevBlade.tx, this.prevBlade.ty, this.blade.bx, this.blade.by, this.blade.tx, this.blade.ty)
   607→          : segHitsCircle(this.blade.bx, this.blade.by, this.blade.tx, this.blade.ty, e.x, cy, r)
   608→        if (hit) {
   609→          a.hitMap.set(e.id, animT)
   610→          w.hitEnemy(e.id, this.blade, info)
   611→          this.onLandedHit(w, info, e.x, cy)
   612→        }
   613→      }
   614→      // remember this frame's blade for the next sweep test
   615→      this.prevBlade.bx = this.blade.bx
   616→      this.prevBlade.by = this.blade.by
   617→      this.prevBlade.tx = this.blade.tx
   618→      this.prevBlade.ty = this.blade.ty
   619→      this.hasPrevBlade = true
   620→      // aoe shockwave at strike moment for heavy hits
   621→      if (h.aoe && h.aoe > 0 && !a.impactDone && this.grounded && animT >= h.start + 0.02) {
   622→        a.impactDone = true
   623→        w.fx.shockwave(this.x + this.face * 30, 0, h.aoe, this.sword.colors.glow, 0.4, 6)
   624→        if (this.grounded) w.fx.groundCrack(this.x + this.face * 34, 0, h.aoe * 0.45)
   625→      }
   626→    }
   627→
   628→    // trail emission
   629→    if (animT >= def.trail.start && animT <= def.trail.end) {
   630→      w.fx.pushTrail(a.trail, this.blade.bx, this.blade.by, this.blade.tx, this.blade.ty)
   631→    } else if (animT > def.trail.end && a.trail.emitting) {
   632→      w.fx.endTrail(a.trail)
   633→    }
   634→
   635→    // cancel window → allow chaining / dash-cancel / jump-cancel
   636→    const dashCancel = animT >= h.end * 0.6
   637→    if (animT >= def.cancel || dashCancel) {
   638→      const input = w.input
   639→      if ((input.pressed('dash') || input.buffered('dash')) && this.canDash()) {
   640→        w.fx.endTrail(a.trail)
   641→        this.startDash(w, input.axisX(), input.axisY())
   642→        return
   643→      }
   644→      if (input.pressed('parry')) {
   645→        w.fx.endTrail(a.trail)
   646→        this.enterParry(w)
   647→        return
   648→      }
   649→      if (animT >= def.cancel) {
   650→        // chain next attack
   651→        const ax = input.axisX()
   652→        const ay = input.axisY()
   653→        let next: string | null = null
   654→        if (input.buffered('light')) {
   655→          input.takeBuffered('light')
   656→          next = this.riposteT > 0 && this.grounded ? 'riposte' : this.pickLight(w, ax, ay)
   657→        } else if (input.buffered('heavy')) {
   658→          input.takeBuffered('heavy')
   659→          if (!this.grounded) {
   660→            next = 'slam'
   661→          } else if (input.down('heavy')) {
   662→            // still holding → convert to charge
   663→            this.enterCharge(w)
   664→            w.fx.endTrail(a.trail)
   665→            return
   666→          } else {
   667→            next = def.chain.find((c) => c.startsWith('h')) ?? 'h1'
   668→          }
   669→        }
   670→        if (next) {
   671→          // validate chain (or allow from-parry riposte)
   672→          if (fromChain(def, next, this)) {
   673→            w.fx.endTrail(a.trail)
   674→            this.startAttack(ATTACKS[next], w)
   675→            return
   676→          }
   677→        }
   678→        // jump cancel on ground
   679→        if (this.grounded && (input.pressed('jump') || input.buffered('jump', 0.15))) {
   680→          w.fx.endTrail(a.trail)
   681→          this.doJump(w)
   682→          this.state = 'move'
   683→          return
   684→        }
   685→      }
   686→    }
   687→
   688→    // finish
   689→    if (this.animator.isDone()) {
   690→      w.fx.endTrail(a.trail, 0.18)
   691→      this.action = null
   692→      this.state = 'move'
   693→    }
   694→  }

  1104→  private collectNearest(w: WorldRef, n: number, range: number): EnemyLike[] {
  1105→    const list: EnemyLike[] = []
  1106→    for (const e of w.enemies) {
  1107→      if (e.dead || e.spawning) continue
  1108→      const d = Math.hypot(e.x - this.x, e.y - this.y)
  1109→      if (d < range) list.push({ id: e.id, x: e.x, y: e.y, d })
  1110→    }
  1111→    list.sort((a, b) => (a.d ?? 0) - (b.d ?? 0))
  1112→    return list.slice(0, n)
  1113→  }

  1115→  private findDensest(w: WorldRef, range: number): EnemyLike | null {
  1116→    let best: EnemyLike | null = null
  1117→    let bestScore = -1
  1118→    for (const e of w.enemies) {
  1119→      if (e.dead || e.spawning) continue
  1120→      let score = 0
  1121→      for (const o of w.enemies) {
  1122→        if (o.dead || o.spawning) continue
  1123→        if (Math.abs(o.x - e.x) < range) score++
  1124→      }
  1125→      if (score > bestScore) {
  1126→        bestScore = score
  1127→        best = { id: e.id, x: e.x, y: e.y, d: 0 }
  1128→      }
  1129→    }
  1130→    return best
  1131→  }

  1151→    const speedLean = clamp((this.vx * this.face) / MOVE_MAX, -1, 1)

  1167→    // hurt shake

  1251→function fromChain(def: AttackDef, next: string, player: Player): boolean {
  1252→  if (next === 'riposte') return true
  1253→  // launcher chains naturally into air attacks
  1254→  if (def.id === 'launcher' && (next.startsWith('a') || next === 'dive' || next === 'slam')) return true
  1255→  if (def.chain.includes(next)) return true
  1256→  // any light may restart the chain if the window allows
  1257→  if (player.grounded && next.startsWith('l') && def.chain.some((c) => c.startsWith('l'))) return true
  1258→  if (!player.grounded && (next.startsWith('a') || next === 'dive' || next === 'slam') && def.chain.some((c) => c.startsWith('a') || c === 'dive' || c === 'slam')) return true
  1259→  return false
  1260→}
