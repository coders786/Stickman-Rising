     1→/** STICKMAN RISING — animation inspection lab + QA harness (spec §12).
     2→ *
     3→ * A dedicated diagnostic environment rendered ON TOP of the game:
     4→ *  · SOLO view     — the live character, zoomed, skeleton overlay with
     5→ *                    joint pivots + anatomical labels + foot-contact
     6→ *                    markers + ground line + animation state readout
     7→ *  · SHEET view    — 8-frame contact sheets of every locomotion cycle
     8→ *                    (walk / run / sprint / combat / backpedal / idle)
     9→ *                    side by side for reference comparison, each frame
    10→ *                    solved through the REAL skeleton — inspect the
    11→ *                    actual rendered poses, not code-level assertions
    12→ *  · QA harness    — `window.__srQA.run()`: scripted input sequences
    13→ *                    (walk→run→sprint→combat→backpedal→attacks) with
    14→ *                    anatomy invariants, NaN scan, foot-lock check, loco
    15→ *                    transition log, and combo verification. Result lands
    16→ *                    on `window.__qaResult`.
    17→ */

    38→export type InspectMode = 'off' | 'solo' | 'sheet'

    40→export class Inspect {
    41→  mode: InspectMode = 'off'
    42→  showBones = true
    43→  showLabels = true
    44→  showFeet = true
    45→
    46→  private scratchPose: Pose = makePose()
    47→  private scratchGeom: FighterGeom = makeFighterGeom()
    48→
    49→  setMode(m: InspectMode): void {
    50→    this.mode = m
    51→  }
    52→
    53→  toggleBones(): void {
    54→    this.showBones = !this.showBones
    55→  }
    56→
    57→  /** bottom-left HUD readout (always on while mode != off) */
    58→  draw(ctx: CanvasRenderingContext2D, game: Game, time: number): void {
    59→    if (this.mode === 'off') return
    60→    const p = game.player
    61→    ctx.save()
    62→    ctx.font = '600 12px ui-monospace, monospace'
    63→    ctx.textBaseline = 'top'
    64→
    65→    if (this.mode === 'solo') {
    66→      this.drawSolo(ctx, game, time)
    67→    } else if (this.mode === 'sheet') {
    68→      this.drawSheets(ctx)
    69→    }
    70→
    71→    // state readout (uses public player surface + bracket access for privates)
    72→    const dark = this.mode === 'sheet' // light backdrop → dark text
    73→    const priv = p as unknown as Record<string, unknown>
    74→    const lines = [
    75→      `INSPECT: ${this.mode}${this.showBones ? ' +bones' : ''}`,
    76→      `state=${p.state}  loco=${p.loco}  locoT=${Number(priv.locoT ?? 0).toFixed(2)}s`,
    77→      `dash=${p.dashVariant}${p.dashHandoff ? ' (handoff)' : ''}  cd=${Number(priv.dashCd ?? 0).toFixed(2)}`,
    78→      `vx=${p.vx.toFixed(0)}  vy=${p.vy.toFixed(0)}  grounded=${p.grounded}  face=${p.face}`,
    79→      `enemyW=${p.enemyNearW.toFixed(2)}  threat=${p.hasThreat ? 'yes' : 'no'}  surrounded=${Boolean(priv.surrounded)}`,
    80→      `postAtk=${Number(priv.postAttackT ?? 0).toFixed(2)}  turn=${Number(priv.turnT ?? 0).toFixed(2)}`,
    81→      `footLock=${Number(priv.footLock ?? 0).toFixed(1)}  hipY=${p.hipY.toFixed(1)}`,
    82→    ]
    83→    let y = 10
    84→    for (const ln of lines) {
    85→      ctx.fillStyle = dark ? 'rgba(240,246,250,0.85)' : 'rgba(2,8,14,0.72)'
    86→      const wdt = ctx.measureText(ln).width
    87→      ctx.fillRect(8, y - 2, wdt + 10, 15)
    88→      ctx.fillStyle = dark ? '#12354e' : '#8ff3ff'
    89→      ctx.fillText(ln, 13, y)
    90→      y += 17
    91→    }
    92→    ctx.restore()
    93→  }

    95→  /** big live character + skeleton overlay, centered-left — drawn on a
    96→   * light diagnostic panel so the black silhouette reads at full contrast */
    97→  private drawSolo(ctx: CanvasRenderingContext2D, game: Game, time: number): void {
    98→    const p = game.player
    99→    const dpr = window.devicePixelRatio || 1
   100→    const Z = PSCALE / 1.32 // zoom factor over the in-game world scale
   101→    const cx = (ctx.canvas.width / dpr) * 0.26
   102→    const groundY = (ctx.canvas.height / dpr) * 0.82
   103→    const hipScreenY = groundY + p.hipY * Z
   104→    // light diagnostic panel behind the figure (contrast for the black body)
   105→    ctx.save()
   106→    ctx.fillStyle = '#dfe7ee'
   107→    ctx.beginPath()
   108→    ctx.roundRect(cx - 250, groundY - 330, 500, 348, 14)
   109→    ctx.fill()
   110→    ctx.strokeStyle = 'rgba(40,70,100,0.35)'
   111→    ctx.lineWidth = 1.5
   112→    ctx.stroke()
   113→    ctx.restore()
   114→    // body: local geom → screen at PSCALE
   115→    ctx.save()
   116→    ctx.translate(cx, hipScreenY)
   117→    ctx.scale(p.face * PSCALE, PSCALE)
   118→    drawFighter3D(ctx, p.geom, PLAYER_STYLE, { time })
   119→    this.drawSkeleton(ctx, p.geom)
   120→    ctx.restore()
   121→    // ground line + foot contact markers at their true screen positions
   122→    ctx.save()
   123→    ctx.strokeStyle = 'rgba(30,90,140,0.65)'
   124→    ctx.lineWidth = 2
   125→    ctx.setLineDash([7, 5])
   126→    ctx.beginPath()
   127→    ctx.moveTo(cx - 240, groundY)
   128→    ctx.lineTo(cx + 240, groundY)
   129→    ctx.stroke()
   130→    ctx.setLineDash([])
   131→    if (this.showFeet) {
   132→      const ptScreen = (j: { x: number; y: number }) => ({
   133→        x: cx + p.face * PSCALE * j.x,
   134→        y: hipScreenY + PSCALE * j.y,
   135→      })
   136→      const rA = ptScreen(p.geom.ankleR)
   137→      const lA = ptScreen(p.geom.ankleL)
   138→      // planted = lowest foot within reach of the ground line
   139→      const rPlanted = p.grounded && p.geom.ankleR.y >= p.geom.ankleL.y
   140→      const lPlanted = p.grounded && p.geom.ankleL.y > p.geom.ankleR.y
   141→      this.drawFootMarker(ctx, rA.x, rA.y, rPlanted)
   142→      this.drawFootMarker(ctx, lA.x, lA.y, lPlanted)
   143→    }
   144→    ctx.restore()
   145→  }

   147→  private drawFootMarker(ctx: CanvasRenderingContext2D, x: number, y: number, planted: boolean): void {
   148→    ctx.fillStyle = planted ? '#51ff8e' : 'rgba(255,170,60,0.8)'
   149→    ctx.beginPath()
   150→    ctx.arc(x, y, 4.5, 0, Math.PI * 2)
   151→    ctx.fill()
   152→  }

   154→  /** bones + joints + labels on any solved geom (local space) */
   155→  private drawSkeleton(ctx: CanvasRenderingContext2D, g: SkelGeom): void {
   156→    if (!this.showBones) return
   157→    ctx.save()
   158→    ctx.strokeStyle = 'rgba(65,246,255,0.55)'
   159→    ctx.lineWidth = 1.1
   160→    const seg = (a: { x: number; y: number }, b: { x: number; y: number }) => {
   161→      ctx.beginPath()
   162→      ctx.moveTo(a.x, a.y)
   163→      ctx.lineTo(b.x, b.y)
   164→      ctx.stroke()
   165→    }
   166→    seg(g.hip, g.waist); seg(g.waist, g.chest); seg(g.chest, g.neckTop); seg(g.neckTop, g.headC)
   167→    seg(g.hipPtR, g.kneeR); seg(g.kneeR, g.ankleR); seg(g.ankleR, g.toeR)
   168→    seg(g.hipPtL, g.kneeL); seg(g.kneeL, g.ankleL); seg(g.ankleL, g.toeL)
   169→    seg(g.shoulderR, g.elbR); seg(g.elbR, g.handR)
   170→    seg(g.shoulderL, g.elbL); seg(g.elbL, g.handL)
   171→    // shoulder girdle line (the anatomy fix, visualized)
   172→    ctx.strokeStyle = 'rgba(255,120,180,0.8)'
   173→    ctx.lineWidth = 1.3
   174→    ctx.beginPath()
   175→    ctx.moveTo(g.shoulderL.x, g.shoulderL.y)
   176→    ctx.lineTo(g.neckTop.x, g.neckTop.y)
   177→    ctx.lineTo(g.shoulderR.x, g.shoulderR.y)
   178→    ctx.stroke()
   179→    // joints
   180→    ctx.fillStyle = '#ffd24a'
   181→    const joints = [
   182→      g.hip, g.hipPtR, g.hipPtL, g.waist, g.chest, g.neckTop, g.headC,
   183→      g.shoulderR, g.shoulderL, g.elbR, g.elbL, g.handR, g.handL,
   184→      g.kneeR, g.kneeL, g.ankleR, g.ankleL,
   185→    ]
   186→    for (const j of joints) {
   187→      ctx.beginPath()
   188→      ctx.arc(j.x, j.y, 1.7, 0, Math.PI * 2)
   189→      ctx.fill()
   190→    }
   191→    // THE proof joints: shoulders in pink
   192→    ctx.fillStyle = '#ff6fb5'
   193→    for (const j of [g.shoulderR, g.shoulderL]) {
   194→      ctx.beginPath()
   195→      ctx.arc(j.x, j.y, 3.0, 0, Math.PI * 2)
   196→      ctx.fill()
   197→    }
   198→    if (this.showLabels) {
   199→      ctx.fillStyle = 'rgba(160,240,255,0.95)'
   200→      ctx.font = '600 4.2px ui-monospace, monospace'
   201→      const lbl = (j: { x: number; y: number }, s: string) => ctx.fillText(s, j.x + 2.4, j.y - 2.4)
   202→      lbl(g.headC, 'skull')
   203→      lbl(g.neckTop, 'neckTop')
   204→      lbl(g.shoulderR, 'SHOULDER-R')
   205→      lbl(g.shoulderL, 'shoulder-l')
   206→      lbl(g.chest, 'chest')
   207→      lbl(g.waist, 'waist')
   208→      lbl(g.hip, 'pelvis')
   209→      lbl(g.kneeR, 'kneeR')
   210→      lbl(g.ankleR, 'ankleR')
   211→      lbl(g.elbR, 'elbR')
   212→      lbl(g.handR, 'handR')
   213→    }
   214→    ctx.restore()
   215→  }

   217→  /** contact sheets: every locomotion cycle, 8 frames each, real solve —
   218→   * light diagnostic background, per-cell ground lines + foot-contact
   219→   * markers (green = support, orange = swing), 2 clip-groups per band */
   220→  private drawSheets(ctx: CanvasRenderingContext2D): void {
   221→    const rows: Array<[string, typeof WALK_CLIP]> = [
   222→      ['WALK', WALK_CLIP],
   223→      ['RUN', RUN_CLIP],
   224→      ['SPRINT', SPRINT_CLIP],
   225→      ['COMBAT STEP', COMBAT_STEP_CLIP],
   226→      ['COMBAT-IDLE (rooted guard)', COMBAT_IDLE_CLIP],
   227→      ['BACKPEDAL', BACKPEDAL_CLIP],
   228→      ['IDLE', IDLE_CLIP],
   229→      ['JUMP', JUMP_CLIP],
   230→      ['DASH LAUNCH (iai)', DASH_LAUNCH_CLIP],
   231→      ['DASH CARRY (momentum)', DASH_CARRY_CLIP],
   232→      ['DASH HUNT (gap-close)', DASH_HUNT_CLIP],
   233→      ['DASH PIERCE (lunge)', DASH_PIERCE_CLIP],
   234→      ['DASH RETREAT (guard)', DASH_RETREAT_CLIP],
   235→      ['DASH CARVE (breakaway)', DASH_CARVE_CLIP],
   236→    ]
   237→    const dpr = window.devicePixelRatio || 1
   238→    const w = ctx.canvas.width / dpr
   239→    const h = ctx.canvas.height / dpr
   240→    const cols = 8
   241→    const groupsPerRow = 2
   242→    const bands = Math.ceil(rows.length / groupsPerRow) // 3 bands
   243→    const cellW = w / (cols * groupsPerRow)
   244→    const rowH = h / (bands + 0.25)
   245→    // figure scale: fit character height inside a band minus label space
   246→    const S = Math.min(1.0, (rowH - 34) / 138)
   247→    ctx.save()
   248→    // light diagnostic backdrop — black silhouette at full contrast
   249→    ctx.fillStyle = '#e9eef3'
   250→    ctx.fillRect(0, 0, w, h)
   251→    ctx.font = '700 12px ui-monospace, monospace'
   252→    for (let r = 0; r < rows.length; r++) {
   253→      const band = Math.floor(r / groupsPerRow)
   254→      const group = r % groupsPerRow
   255→      const [label, clip] = rows[r]
   256→      const baseY = rowH * (band + 1)
   257→      const labelX = group * (w / groupsPerRow) + 10
   258→      ctx.fillStyle = '#1a3a52'
   259→      ctx.fillText(`${label} · ${clip.dur.toFixed(2)}s cycle`, labelX, baseY - rowH + 16)
   260→      for (let c = 0; c < cols; c++) {
   261→        const t = (clip.dur * c) / cols
   262→        sampleClip(clip, t, this.scratchPose)
   263→        const g = solveFighter3D(this.scratchPose, PLAYER_SPEC, this.scratchGeom, PLAYER_SPEC.blade)
   264→        // normalize: lowest foot contact sits ON this cell's ground line
   265→        const lowContact = Math.max(g.ankleL.y, g.ankleR.y, g.toeL.y, g.toeR.y)
   266→        const cx = group * (w / groupsPerRow) + cellW * (c + 0.5)
   267→        const groundY = baseY - 14
   268→        ctx.save()
   269→        ctx.translate(cx, groundY - (lowContact + 2.4) * S)
   270→        ctx.scale(S, S)
   271→        drawFighter3D(ctx, g, PLAYER_STYLE, { time: t })
   272→        ctx.restore()
   273→        // per-cell ground line
   274→        ctx.strokeStyle = 'rgba(40,80,110,0.45)'
   275→        ctx.lineWidth = 1
   276→        ctx.beginPath()
   277→        ctx.moveTo(cx - cellW * 0.42, groundY)
   278→        ctx.lineTo(cx + cellW * 0.42, groundY)
   279→        ctx.stroke()
   280→        // foot-contact markers: green = support (near ground), orange = swing
   281→        const hipY = groundY - (lowContact + 2.4) * S
   282→        const markFoot = (j: { x: number; y: number }) => {
   283→          const airborne = (lowContact - j.y) > 4 // clearly lifted vs support
   284→          ctx.fillStyle = airborne ? 'rgba(255,140,30,0.9)' : '#18c25a'
   285→          ctx.beginPath()
   286→          ctx.arc(cx + j.x * S, hipY + j.y * S, 3, 0, Math.PI * 2)
   287→          ctx.fill()
   288→        }
   289→        markFoot(g.ankleR)
   290→        markFoot(g.ankleL)
   291→        // frame tick
   292→        ctx.fillStyle = 'rgba(30,70,100,0.55)'
   293→        ctx.font = '600 9px ui-monospace, monospace'
   294→        ctx.fillText(`f${c}`, cx - 5, baseY - 3)
   295→      }
   296→    }
   297→    ctx.restore()
   298→  }
   299→}

   302→/* QA harness                                                          */

   304→
   305→interface QaCheck {
   306→  name: string
   307→  pass: boolean
   308→  detail: string
   309→}
   310→
   311→const sleep = (ms: number) => new Promise<void>((r) => setTimeout(r, ms))
   312→
   313→/** Scripted end-to-end verification. Parks the report on
   314→ * window.__qaResult. Drive from the browser console / agent-browser. */
   315→export async function runQA(game: Game): Promise<void> {
   316→  const checks: QaCheck[] = []
   317→  const log: string[] = []
   318→  const p = game.player
   319→  const input = game.input
   320→
   321→  const samplePose = () => {
   322→    let bad = 0
   323→    for (let i = 0; i < 17; i++) if (!Number.isFinite(p.pose[i])) bad++
   324→    return bad
   325→  }
   326→
   327→  try {
   328→    // ---- setup: controlled empty arena (waves suppressed, player alive) ----
   329→    game.qaMode = true
   330→    game.phase = 'playing'
   331→    game.enemies.length = 0
   332→    game.waves.reset()
   333→    // release every virtual input left held by prior console driving —
   334→    // stale held buttons contaminate the whole scripted sequence
   335→    input.setVirtualAxis(0, 0)
   336→    for (const a of ['left', 'right', 'up', 'down', 'light', 'heavy', 'dash', 'jump', 'parry', 'ability', 'ult', 'execute', 'sheathe'] as const) {
   337→      input.setVirtual(a, false)
   338→    }
   339→    p.reset(400)
   340→    input.setEnabled(true)
   341→    log.push('setup ok')
   342→
   343→    // ---- 1. anatomy invariants (the arms-from-the-neck fix) ----
   344→    {
   345→      const g = p.geom
   346→      const spec = PLAYER_SPEC
   347→      const armLen = Math.hypot(g.elbR.x - g.shoulderR.x, g.elbR.y - g.shoulderR.y)
   348→      // local y grows DOWN: shoulder below neck-top means shoulderR.y GREATER
   349→      const shoulderBelowNeck = g.shoulderR.y > g.neckTop.y + 1.5
   350→      const armFromShoulder = Math.abs(armLen - spec.upperArm) < 0.6
   351→      // neck column rises above chest: neckTop.y LESS (more negative) than chest
   352→      const neckVisible = g.neckTop.y < g.chest.y - 2.5
   353→      checks.push({
   354→        name: 'anatomy: shoulder joints below neck top',
   355→        pass: shoulderBelowNeck,
   356→        detail: `shoulderR.y=${g.shoulderR.y.toFixed(1)} neckTop.y=${g.neckTop.y.toFixed(1)}`,
   357→      })
   358→      checks.push({
   359→        name: 'anatomy: arm chains FROM shoulder (len=upperArm)',
   360→        pass: armFromShoulder,
   361→        detail: `len=${armLen.toFixed(2)} spec=${spec.upperArm}`,
   362→      })
   363→      checks.push({
   364→        name: 'anatomy: neck column visible above chest',
   365→        pass: neckVisible,
   366→        detail: `neckTop.y=${g.neckTop.y.toFixed(1)} chest.y=${g.chest.y.toFixed(1)}`,
   367→      })
   368→    }
   369→
   370→    // ---- 2. walk → run escalation (Condition A, analog speed mapping) ----
   371→    input.setVirtualAxis(0.55, 0)
   372→    await sleep(700)
   373→    const earlyMode = p.loco
   374→    const earlyVx = Math.abs(p.vx)
   375→    input.setVirtualAxis(1, 0)
   376→    await sleep(1500)
   377→    const fullMode = p.loco
   378→    input.setVirtualAxis(0, 0)
   379→    await sleep(500)
   380→    checks.push({
   381→      name: 'locomotion: WALK at partial input',
   382→      pass: earlyMode === 'walk' && earlyVx > 60 && earlyVx < 210,
   383→      detail: `mode@0.7s=${earlyMode} vx=${earlyVx.toFixed(0)}`,
   384→    })
   385→    checks.push({
   386→      name: 'locomotion: RUN at full input',
   387→      pass: fullMode === 'run',
   388→      detail: `mode@2.2s=${fullMode}`,
   389→    })
   390→
   391→    // ---- 3. sprint latch via double-tap (button events → press stamps) ----
   392→    input.setVirtual('right', true)
   393→    await sleep(120)
   394→    input.setVirtual('right', false)
   395→    await sleep(60)
   396→    input.setVirtual('right', true)
   397→    await sleep(900)
   398→    const sprintMode = p.loco
   399→    const sprintVx = Math.abs(p.vx)
   400→    input.setVirtual('right', false)
   401→    await sleep(400)
   402→    checks.push({
   403→      name: 'locomotion: sprint after double-tap',
   404→      pass: sprintMode === 'sprint' && sprintVx > 380,
   405→      detail: `loco=${sprintMode} vx=${sprintVx.toFixed(0)}`,
   406→    })
   407→
   408→    // ---- 4. combat mode when enemy near (Condition B) ----
   409→    const { Enemy } = await import('../entities/enemy')
   410→    const foe = new Enemy('grunt', p.x + 240, 1)
   411→    foe.spawning = false
   412→    foe.state = 'stun' // stun STATE + huge timer = truly frozen (stunT alone
   413→    foe.stunT = 999   // does nothing from 'enter' — the foe chased & hit us)
   414→    game.enemies.push(foe)
   415→    input.setVirtualAxis(0.5, 0)
   416→    await sleep(800)
   417→    const combatMode = p.loco
   418→    const enemyW = p.enemyNearW
   419→    checks.push({
   420→      name: 'context: combat locomotion near enemy',
   421→      pass: combatMode === 'combat' && enemyW > 0.5,
   422→      detail: `loco=${combatMode} enemyW=${enemyW.toFixed(2)}`,
   423→    })
   424→
   425→    // ---- 5. backpedal away from threat (facing HOLDS toward the enemy) ----
   426→    input.setVirtualAxis(-0.5, 0) // partial retreat: stays under the escape speed
   427→    await sleep(600)
   428→    const backMode = p.loco
   429→    const backFace = p.face
   430→    input.setVirtualAxis(0, 0)
   431→    await sleep(300)
   432→    checks.push({
   433→      name: 'context: backpedal retreating vs threat',
   434→      pass: backMode === 'backpedal' && backFace === 1,
   435→      detail: `loco=${backMode} face=${backFace}`,
   436→    })
   437→
   438→    // ---- 5b. THE CONDITIONED DASH SYSTEM ----
   439→    // One movement, nine authored animations: the variant must follow the
   440→    // CONDITIONS at the input instant (momentum, threat geometry, chain).
   441→    // (a) LAUNCH: stationary + no threat → the iai explosion
   442→    {
   443→      game.enemies.length = 0
   444→      p.reset(400)
   445→      input.setVirtualAxis(0, 0)
   446→      await sleep(500)
   447→      const x0 = p.x
   448→      input.setVirtualAxis(1, 0) // HOLD the direction → handoff at exit
   449→      input.setVirtual('dash', true)
   450→      await sleep(50)
   451→      const midVariant = p.dashVariant
   452→      const midState = p.state
   453→      input.setVirtual('dash', false)
   454→      // poll for the exit frame — sample the handed-off velocity exactly at
   455→      // the dash→move transition (before locomotion friction bleeds it)
   456→      let tailVx = 0
   457→      for (let i = 0; i < 30; i++) {
   458→        await sleep(16)
   459→        if (p.state === 'move') { tailVx = Math.abs(p.vx); break }
   460→      }
   461→      await sleep(250) // dash (0.17s) + exit bleed
