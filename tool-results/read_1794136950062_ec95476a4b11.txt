    27→class ComboTracker {
    28→  hits = 0
    29→  timer = 0
    30→  timerMax = 2.6
    31→  mult = 1
    32→  best = 0
    33→  bestRank = ''
    34→  rankUpT = 0
    35→  lastRank = ''
    36→
    37→  onHit(): void {
    38→    this.hits++
    39→    this.timer = this.timerMax
    40→    this.mult = Math.min(1 + Math.floor(this.hits / 5) * 0.5, 5)
    41→    if (this.hits > this.best) this.best = this.hits
    42→    const { rank } = rankForHits(this.hits)
    43→    if (rank && rank !== this.lastRank) {
    44→      this.lastRank = rank
    45→      if (rank.length > this.bestRank.length) this.bestRank = rank
    46→      this.rankUpT = 0.7
    47→    }
    48→  }
    49→
    50→  update(dt: number): void {
    51→    this.rankUpT = Math.max(0, this.rankUpT - dt)
    52→    if (this.hits > 0) {
    53→      this.timer -= dt
    54→      if (this.timer <= 0) {
    55→        this.hits = 0
    56→        this.mult = 1
    57→        this.lastRank = ''
    58→      }
    59→    }
    60→  }
    61→
    62→  reset(): void {
    63→    this.hits = 0
    64→    this.timer = 0
    65→    this.mult = 1
    66→    this.best = 0
    67→    this.bestRank = ''
    68→    this.rankUpT = 0
    69→    this.lastRank = ''
    70→  }
    71→}

   139→  /* ---- PERFORMANCE GOVERNOR (the "hangs when enemies appear" fix) ----
   140→   * Frame cost must NOT grow when the frame rate falls. Three defenses:
   141→   *   1. substeps are CAPPED (3 × 1/60) — a slow frame simulates at most
   142→   *      50ms of world, never 5× full-cost steps (the old death spiral:

   145→   *      not per substep — rain never integrates at 450Hz
   146→   *   3. render resolution (DPR) auto-scales to the measured frame rate */

   258→  /** resolution governor: every 0.8s of PLAYING, compare the damped frame
   259→   * rate against thresholds and step the render DPR down (or back up) with
   260→   * hysteresis. If DPR 1 still can't hold ~40fps, fall a quality tier
   261→   * (rain/mote/fog/mist/reflection density all scale with tier). */
   262→  private govern(dt: number): void {
   263→    if (!this.autoQuality || this.phase !== 'playing') return
   264→    this.govT += dt
   265→    this.govCool = Math.max(0, this.govCool - dt)
   266→    if (this.govT < 0.8) return
   267→    this.govT = 0
   268→    if (this.govCool > 0) return
   269→    const cap = Math.min(
   270→      this.settings.quality === 'high' ? 1.5 : this.settings.quality === 'medium' ? 1.25 : 1,
   271→      this.dprCeil,
   272→    )
   273→    if (this.fpsAvg < 44 && this.dprTarget > 1) {
   274→      this.dprTarget = Math.max(1, this.dprTarget - 0.25)
   275→      this.lastDropAt = this.wall
   276→      this.govCool = 0.8
   277→      this.resize()
   278→    } else if (this.fpsAvg < 36 && this.dprTarget <= 1) {
   279→      // resolution floor reached and still drowning → drop a quality tier
   280→      const order: GameSettings['quality'][] = ['high', 'medium', 'low']
   281→      const i = order.indexOf(this.settings.quality)
   282→      if (i < order.length - 1) {
   283→        this.applySettings({ ...this.settings, quality: order[i + 1] })
   284→        this.govCool = 4
   285→      }
   286→    } else if (
   287→      this.fpsAvg > 57
   288→      && this.dprTarget < cap
   289→      && this.wall - this.lastDropAt > 12 // respect every drop for 12s
   290→      && this.wall - this.lastRaiseAt > 3 // one raise at a time
   291→      && this.enemies.length < 3          // never raise into a wave
   292→    ) {
   293→      this.dprTarget = Math.min(cap, this.dprTarget + 0.25)
   294→      this.lastRaiseAt = this.wall
   295→      this.govCool = 3
   296→      this.resize()
   297→    }
   298→  }
   299→
   300→  /** CLIFF DETECTOR — the hang killer. rAF is vsync-quantized (16.7 /
   301→   * 33.3 / 49.9ms steps), so the damped fpsAvg reacts sluggishly: by the
   302→   * time the 0.8s polling cadence notices, the player has already eaten
   303→   * 2–5 seconds of 20fps. A real cliff instead shows as a RUN of frames at
   304→   * ≥1.5× the 60Hz budget — react within ~0.15s by stepping the render DPR
   305→   * down immediately. If the cliff lands within 6s of a raise, that raise
   306→   * proved the level unstable on this display: pin the session ceiling at
   307→   * the pre-raise DPR so the governor stops gambling on it. */
   308→  private detectCliff(dtRaw: number): void {
   309→    if (this.phase === 'paused' || document.visibilityState !== 'visible') {
   310→      this.overRun = 0
   311→      return
   312→    }
   313→    if (dtRaw > 0.025) this.overRun++
   314→    else if (dtRaw < 0.019) this.overRun = 0
   315→    if (this.overRun < 8) return
   316→    this.overRun = 0
   317→    if (this.dprTarget <= 1) return
   318→    if (this.wall - this.lastRaiseAt < 6) {
   319→      // the recent raise was the cause → pin the ceiling at the pre-raise level
   320→      this.dprCeil = Math.max(1, this.dprTarget - 0.25)
   321→    }
   322→    this.dprTarget = Math.max(1, this.dprTarget - 0.25)
   323→    this.lastDropAt = this.wall
   324→    this.govCool = Math.max(this.govCool, 1)
   325→    this.resize()
   326→  }

   435→  private loop = (now: number): void => {
   436→    if (this.destroyed) return
   437→    this.raf = requestAnimationFrame(this.loop)
   438→    let dtRaw = (now - this.lastFrame) / 1000
   439→    this.lastFrame = now
   440→    if (dtRaw > 0.05) dtRaw = 0.05
   441→    if (dtRaw <= 0) dtRaw = 0.0001
   442→    this.fps = damp(this.fps, 1 / dtRaw, 3, dtRaw)
   443→    this.fpsAvg = damp(this.fpsAvg, 1 / dtRaw, 1.2, dtRaw)
   444→    this.wall = now / 1000
   445→    this.govern(dtRaw)
   446→    this.detectCliff(dtRaw)
   447→
   448→    const ctx = this.ctx
   449→    if (!ctx) return
   450→
   451→    if (this.phase === 'paused') {
   452→      // frozen frame, HUD still drawn
   453→    } else {
   454→      // --- timing: hitstop & slow-mo ---
   455→      let simDt = dtRaw
   456→      if (this.hitstopT > 0) {
   457→        this.hitstopT -= dtRaw
   458→        simDt = dtRaw * 0.05
   459→        this.timeScale = 0.05 // world consumers (blade impact deflection) must SEE the freeze
   460→      } else if (this.slowmoT > 0) {
   461→        this.slowmoT -= dtRaw
   462→        simDt = dtRaw * this.slowmoScale
   463→        this.timeScale = this.slowmoScale
   464→        if (this.slowmoT <= 0) this.timeScale = 1
   465→      } else {
   466→        this.timeScale = 1
   467→      }
   468→
   469→      // --- cinematics run on real time ---
   470→      this.updateCinematics(dtRaw)
   471→
   472→      // --- capped substeps for stable physics (NO death spiral: cost of a
   473→      // frame is bounded no matter how slow the machine runs) ---
   474→      let remaining = simDt
   475→      let substeps = 0
   476→      while (remaining > 1e-6 && substeps < 3) {
   477→        const stepDt = Math.min(remaining, 1 / 60)
   478→        this.step(stepDt)
   479→        // EDGE LIFETIME = ONE SUBSTEP. pressed()/released() edges must be
   480→        // visible to exactly ONE simulation step: clearing them only once
   481→        // per RENDERED frame let a single jump press fire doJump in BOTH
   482→        // substeps of a slow frame (jump then airJump in the same frame —
   483→        // the −950 impulse was stomped to −820 mid-rise and the leap clip
   484→        // was overwritten by airJump). Input events can only arrive BETWEEN
   485→        // rAF callbacks, so no press is ever missed; presses that land while
   486→        // a state ignores them are still caught by the combo BUFFER, which
   487→        // is pressTime-based and deliberately spans substeps and frames.
   488→        this.input.endFrame(stepDt)
   489→        remaining -= stepDt
   490→        substeps++
   491→      }
   492→      if (remaining > 1e-6) {
   493→        // very slow frame: consume the backlog as one clamped catch-up tick
   494→        // so the world keeps roughly real-time without multiplying cost
   495→        this.step(Math.min(remaining, 0.05))
   496→        this.input.endFrame(0.05)
   497→      }
   498→      // --- visual systems tick ONCE per frame (never per substep) ---
   499→      this.fx.update(simDt)
   500→      this.bg.update(simDt)
   501→      this.updateCamera(simDt)
   502→    }
   503→
   504→    // --- render ---
   505→    ctx.setTransform(this.dpr, 0, 0, this.dpr, 0, 0)
   506→    const world: RenderWorld = {
   507→      cam: this.cam,
   508→      fx: this.fx,
   509→      bg: this.bg,
   510→      enemies: this.enemies,
   511→      player: this.player,
   512→      projectiles: this.projectiles,
   513→      hud: this.hud,
   514→      hudState: this.buildHudState(),
   515→      time: this.time,
   516→    }
   517→    this.renderer.draw(ctx, world, this.cssW, this.cssH, dtRaw)
   518→    // inspection overlay (solo/sheet views + state readout)
   519→    this.inspect.draw(ctx, this, this.time)
   520→
   521→    this.input.endFrame(dtRaw)
   522→  }
   523→
   524→  private step(dt: number): void {
   525→    this.time += dt
   526→    this.worldRef.time = this.time
   527→    this.worldRef.timeScale = this.timeScale
   528→    this.worldRef.quality = this.settings.quality
   529→    this.worldRef.gore = this.settings.gore
   530→    PLAYER_X_REF.x = this.player.x
   531→    PLAYER_X_REF.y = this.player.y
   532→    PLAYER_POS.x = this.player.x
   533→    PLAYER_POS.y = this.player.y
   534→
   535→    if (this.phase === 'playing' || this.phase === 'gameover') {
   536→      if (this.phase === 'playing') {
   537→        this.handleMetaInput()
   538→      }
   539→      this.player.update(dt, this.worldRef)
   540→      for (const e of this.enemies) {
   541→        e.update(dt, this.worldRef)
   542→      }
   543→      // purge dead
   544→      for (let i = this.enemies.length - 1; i >= 0; i--) {
   545→        if (this.enemies[i].dead) this.enemies.splice(i, 1)
   546→      }
   547→      this.projectiles.update(dt, this.worldRef)
   548→
   549→      if (this.phase === 'playing') {
   550→        if (!this.qaMode) this.waves.update(dt, this.enemies.length)
   551→        this.combo.update(dt)
   552→        this.runTime += dt
   553→        this.stats.timeAlive = this.runTime
   554→        // death → game over
   555→        if (!this.qaMode && this.player.isDead && this.player.deadTime > 1.4) {
   556→          this.gameOver()
   557→        }
   558→      }
   559→    } else if (this.phase === 'attract') {
   560→      // idle hero breathing on the rooftop (framed below the title logo)
   561→      this.player.update(dt, this.worldRef)
   562→      this.cam.focus(this.player.x + Math.sin(this.time * 0.14) * 140, this.player.y - 110, 0, 0)
   563→    }
   564→  }

   592→  private handleMetaInput(): void {
   593→    // pause
   594→    if (this.input.pressed('pause')) {
   595→      this.pause()
   596→      return
   597→    }
   598→    // sword switching
   599→    const swordIds: SwordId[] = ['katana', 'inferno', 'thunder', 'void', 'gravity', 'phantom']
   600→    const swordActions = ['sword1', 'sword2', 'sword3', 'sword4', 'sword5', 'sword6'] as const
   601→    for (let i = 0; i < 6; i++) {
   602→      if (this.input.pressed(swordActions[i])) {
   603→        this.trySwitchSword(swordIds[i])
   604→      }
   605→    }
   606→    if (this.input.pressed('cycleSword')) {
   607→      const idx = SWORD_ORDER.indexOf(this.player.sword.id)
   608→      for (let i = 1; i <= 6; i++) {
   609→        const next = SWORD_ORDER[(idx + i) % 6]
   610→        if (this.unlocked.includes(next)) {
   611→          this.trySwitchSword(next)
   612→          break
   613→        }
   614→      }
   615→    }
   616→    // ultimate
   617→    if (this.input.pressed('ult') && this.player.meter >= 100 && !this.ult && !this.exec) {
   618→      this.startUlt()
   619→    }
   620→    // execution
   621→    if (this.input.pressed('execute') && !this.exec && !this.ult) {
   622→      const target = this.findExecuteTarget()
   623→      if (target) this.startExecution(target)
   624→    }
   625→  }

   704→  private hitEnemy(enemyId: number, blade: { bx: number; by: number; tx: number; ty: number }, atk: AttackRuntimeInfo): void {
   705→    const e = this.enemies.find((en) => en.id === enemyId)
   706→    if (!e || e.dead) return
   707→    const sword = this.player.sword
   708→    const dir = Math.sign(blade.tx - blade.bx) || this.player.face
   709→    const hx = (blade.bx + blade.tx) / 2
   710→    const hy = (blade.by + blade.ty) / 2
   711→    const ex = e.x
   712→    // impact FX at the blade's actual contact height, clamped to the visual
   713→    // body (head-top ≈ 88u … feet ≈ 10u) — sparks appear where the edge
   714→    // crossed the silhouette, not at a fixed belly point
   715→    const ey = clamp(hy, e.y - 86 * e.scale, e.y - 10 * e.scale)
   716→
   717→    // elite parry: occasionally deflects light strikes
   718→    if (e.kind === 'elite' && !atk.heavy && Math.random() < 0.16 && e.state !== 'stun' && e.state !== 'launched') {
   719→      this.audio.play('clang')
   720→      this.fx.impact(ex, ey, Math.atan2(ey - hy, ex - hx), true, '#ffffff', '#ff5c7a')
   721→      this.fx.sparks(ex, ey, -Math.PI / 2, 1.4, 12, '#ffffff', 420, 0.4)
   722→      this.addHitstop(0.05)
   723→      this.cam.addTrauma(0.15)
   724→      this.fx.text(ex, ey - 50, 'PARRY!', '#ff5c7a', 12)
   725→      return
   726→    }
   727→
   728→    const res = e.takeHit(atk, dir, this.player.x)
   729→
   730→    // combo & score
   731→    this.combo.onHit()
   732→    this.stats.maxCombo = Math.max(this.stats.maxCombo, this.combo.hits)
   733→    if (this.combo.rankUpT > 0.6) {
   734→      this.audio.play('rankup', { pitch: 1 + this.combo.hits * 0.002 })
   735→      this.emit({ type: 'rank', rank: this.combo.lastRank })
   736→    }
   737→    this.stats.score += Math.round(atk.dmg * this.combo.mult * 0.8)
   738→
   739→    switch (res) {
   740→      case 'killed': {
   741→        this.onEnemyKilled(e)
   742→        break
   743→      }
   744→      case 'blocked': {
   745→        this.audio.play('clang', { pitch: 0.9 })
   746→        this.fx.impact(ex, ey, Math.atan2(ey - hy, ex - hx), false, '#ffffff', '#46ffb4')
   747→        this.fx.text(ex, ey - 46, 'BLOCK', '#46ffb4', 11)
   748→        this.addHitstop(0.03)
   749→        this.cam.addTrauma(0.06)
   750→        break
   751→      }
   752→      case 'armored': {
   753→        this.audio.play('clang', { pitch: 0.6, volume: 0.8 })
   754→        this.fx.sparks(ex, ey, Math.atan2(ey - hy, ex - hx), 0.9, 8, '#ffae2e', 320, 0.35)
   755→        this.fx.text(ex, ey - 46, 'ARMOR', '#ffae2e', 11)
   756→        this.addHitstop(0.02)
   757→        break
   758→      }
   759→      default: {
   760→        // solid hit
   761→        this.audio.play(atk.heavy ? 'hitHeavy' : 'hit', {
   762→          volume: clamp(0.55 + atk.dmg * 0.01, 0.55, 1),
   763→          pitch: rand(0.94, 1.08),
   764→          intensity: clamp01(atk.dmg / 30),
   765→        })
   766→        this.fx.impact(ex, ey, Math.atan2(ey - hy, ex - hx), atk.heavy, sword.colors.glow, sword.colors.accent)
   767→        const dmgShown = Math.round(atk.dmg * (e.vulnerableT > 0 ? 2 : 1) * (e.kind === 'heavy' || e.kind === 'boss' ? 0.72 : 1))
   768→        this.fx.text(ex, ey - 46, `${dmgShown}`, atk.heavy ? sword.colors.accent : '#f4f9fc', atk.heavy ? 17 : 13)
   769→        // crash-zoom punch on heavies (Z-axis push-pull) + impact frames on
   770→        // the big connects — the anime kill-frame language
   771→        this.addHitstop(atk.hitstop, atk.heavy ? 0.09 : 0)
   772→        if (atk.heavy && atk.dmg >= 15) {
   773→          this.fx.impactFrame(ex, ey, sword.colors.glow)
   774→        } else if (atk.heavy) {
   775→          this.fx.chroma(0.45)
   776→        }
   777→        this.cam.addTrauma(atk.shake)
   778→        if (e.vulnerableT > 0) this.fx.text(ex, ey - 70, 'WEAK POINT', '#ffe14d', 11)
   779→        break
   780→      }
   781→    }
   782→  }

   880→  private updateCinematics(dt: number): void {

  1021→  private buildHudState(): HudState {
  1022→    const p = this.player
  1023→    const boss = this.enemies.find((e) => e.kind === 'boss' && !e.dead)
  1024→    let execPrompt: { x: number; y: number } | null = null
  1025→    if (this.phase === 'playing' && !this.exec && !this.ult) {
  1026→      const target = this.findExecuteTarget()
  1027→      if (target) {
  1028→        const zoom = this.cam.zoom
  1029→        execPrompt = {
  1030→          x: (target.x - this.cam.x - this.cam.offX) * zoom + this.cssW / 2,
  1031→          y: (target.y - 120 - this.cam.y - this.cam.offY) * zoom + this.cssH / 2,
  1032→        }
  1033→      }
  1034→    }
  1035→    return {
  1036→      hp: p.hp,
  1037→      maxHp: p.maxHp,
  1038→      energy: p.energy,
  1039→      maxEnergy: p.maxEnergy,
  1040→      meter: p.meter,
  1041→      swordName: p.sword.name,
  1042→      swordColor: p.sword.colors.accent,
  1043→      abilityName: p.sword.ability.name,
  1044→      abilityCost: p.sword.ability.cost,
  1045→      combo: {
  1046→        hits: this.combo.hits,
  1047→        mult: this.combo.mult,
  1048→        rank: this.combo.lastRank,
  1049→        timer: this.combo.timer,
  1050→        timerMax: this.combo.timerMax,
  1051→        rankUpT: this.combo.rankUpT,
  1052→        lastHitT: 0,
  1053→      },
  1054→      wave: Math.max(1, this.waves.wave),
  1055→      enemiesLeft: this.enemies.filter((e) => !e.dead).length + this.waves.enemiesRemaining,
  1056→      kills: this.stats.kills,
  1057→      score: this.stats.score,
  1058→      boss: boss ? { hp: boss.hp, maxHp: boss.maxHp, name: 'GEMINI MK-II' } : null,

  1060→      executePrompt: execPrompt,
