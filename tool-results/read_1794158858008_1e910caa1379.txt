     1→/** STICKMAN RISING — animation inspection lab + QA harness (spec §12).
     2→ *
     3→ * A dedicated diagnostic environment rendered ON TOP of the game:
     4→ *  · SOLO view     — the live character, zoomed, skeleton overlay with
     5→ *                    joint pivots + anatomical labels + foot-contact
     6→ *                    markers + ground line + animation state readout
     7→ *  · SHEET view    — 8-frame contact sheets of every locomotion cycle
     8→ *                    (walk / run / sprint / combat / backpedal / idle)
     9→ *                    side by side for reference comparison, each frame
    10→ *                    solved through the REAL skeleton — inspect the
    11→ *                    actual rendered poses, not code-level assertions
    12→ *  · QA harness    — `window.__srQA.run()`: scripted input sequences
    13→ *                    (walk→run→sprint→combat→backpedal→attacks) with
    14→ *                    anatomy invariants, NaN scan, foot-lock check, loco
    15→ *                    transition log, and combo verification. Result lands
    16→ *                    on `window.__qaResult`.
    17→ */

    37→export type InspectMode = 'off' | 'solo' | 'sheet'

    39→export class Inspect {
    40→  mode: InspectMode = 'off'
    41→  showBones = true
    42→  showLabels = true
    43→  showFeet = true
    44→
    45→  private scratchPose: Pose = makePose()
    46→  private scratchGeom: SkelGeom = makeSkelGeom()
    47→
    48→  setMode(m: InspectMode): void {
    49→    this.mode = m
    50→  }
    51→
    52→  toggleBones(): void {
    53→    this.showBones = !this.showBones
    54→  }
    55→
    56→  /** bottom-left HUD readout (always on while mode != off) */
    57→  draw(ctx: CanvasRenderingContext2D, game: Game, time: number): void {
    58→    if (this.mode === 'off') return
    59→    const p = game.player
    60→    ctx.save()
    61→    ctx.font = '600 12px ui-monospace, monospace'
    62→    ctx.textBaseline = 'top'
    63→
    64→    if (this.mode === 'solo') {
    65→      this.drawSolo(ctx, game, time)
    66→    } else if (this.mode === 'sheet') {
    67→      this.drawSheets(ctx)
    68→    }
    69→
    70→    // state readout (uses public player surface + bracket access for privates)
    71→    const dark = this.mode === 'sheet' // light backdrop → dark text
    72→    const priv = p as unknown as Record<string, unknown>
    73→    const lines = [
    74→      `INSPECT: ${this.mode}${this.showBones ? ' +bones' : ''}`,
    75→      `state=${p.state}  loco=${p.loco}  locoT=${Number(priv.locoT ?? 0).toFixed(2)}s`,
    76→      `dash=${p.dashVariant}${p.dashHandoff ? ' (handoff)' : ''}  cd=${Number(priv.dashCd ?? 0).toFixed(2)}  rec=${Number(priv.dashRecharge ?? 0).toFixed(2)}`,
    77→      `vx=${p.vx.toFixed(0)}  vy=${p.vy.toFixed(0)}  grounded=${p.grounded}  face=${p.face}`,
    78→      `enemyW=${p.enemyNearW.toFixed(2)}  threat=${p.hasThreat ? 'yes' : 'no'}  surrounded=${Boolean(priv.surrounded)}`,
    79→      `postAtk=${Number(priv.postAttackT ?? 0).toFixed(2)}  turn=${Number(priv.turnT ?? 0).toFixed(2)}`,
    80→      `footLock=${Number(priv.footLock ?? 0).toFixed(1)}  hipY=${p.hipY.toFixed(1)}`,
    81→    ]
    82→    let y = 10
    83→    for (const ln of lines) {
    84→      ctx.fillStyle = dark ? 'rgba(240,246,250,0.85)' : 'rgba(2,8,14,0.72)'
    85→      const wdt = ctx.measureText(ln).width
    86→      ctx.fillRect(8, y - 2, wdt + 10, 15)
    87→      ctx.fillStyle = dark ? '#12354e' : '#8ff3ff'
    88→      ctx.fillText(ln, 13, y)
    89→      y += 17
    90→    }
    91→    ctx.restore()
    92→  }

    94→  /** big live character + skeleton overlay, centered-left — drawn on a
    95→   * light diagnostic panel so the black silhouette reads at full contrast */
    96→  private drawSolo(ctx: CanvasRenderingContext2D, game: Game, time: number): void {
    97→    const p = game.player
    98→    const dpr = window.devicePixelRatio || 1
    99→    const Z = PSCALE / 1.32 // zoom factor over the in-game world scale
   100→    const cx = (ctx.canvas.width / dpr) * 0.26
   101→    const groundY = (ctx.canvas.height / dpr) * 0.82
   102→    const hipScreenY = groundY + p.hipY * Z
   103→    // light diagnostic panel behind the figure (contrast for the black body)
   104→    ctx.save()
   105→    ctx.fillStyle = '#dfe7ee'
   106→    ctx.beginPath()
   107→    ctx.roundRect(cx - 250, groundY - 330, 500, 348, 14)
   108→    ctx.fill()
   109→    ctx.strokeStyle = 'rgba(40,70,100,0.35)'
   110→    ctx.lineWidth = 1.5
   111→    ctx.stroke()
   112→    ctx.restore()
   113→    // body: local geom → screen at PSCALE
   114→    ctx.save()
   115→    ctx.translate(cx, hipScreenY)
   116→    ctx.scale(p.face * PSCALE, PSCALE)
   117→    drawStick(ctx, p.geom, PLAYER_STYLE, { time })
   118→    this.drawSkeleton(ctx, p.geom)
   119→    ctx.restore()
   120→    // ground line + foot contact markers at their true screen positions
   121→    ctx.save()
   122→    ctx.strokeStyle = 'rgba(30,90,140,0.65)'
   123→    ctx.lineWidth = 2
   124→    ctx.setLineDash([7, 5])
   125→    ctx.beginPath()
   126→    ctx.moveTo(cx - 240, groundY)
   127→    ctx.lineTo(cx + 240, groundY)
   128→    ctx.stroke()
   129→    ctx.setLineDash([])
   130→    if (this.showFeet) {
   131→      const ptScreen = (j: { x: number; y: number }) => ({
   132→        x: cx + p.face * PSCALE * j.x,
   133→        y: hipScreenY + PSCALE * j.y,
   134→      })
   135→      const rA = ptScreen(p.geom.ankleR)
   136→      const lA = ptScreen(p.geom.ankleL)
   137→      // planted = lowest foot within reach of the ground line
   138→      const rPlanted = p.grounded && p.geom.ankleR.y >= p.geom.ankleL.y
   139→      const lPlanted = p.grounded && p.geom.ankleL.y > p.geom.ankleR.y
   140→      this.drawFootMarker(ctx, rA.x, rA.y, rPlanted)
   141→      this.drawFootMarker(ctx, lA.x, lA.y, lPlanted)
   142→    }
   143→    ctx.restore()
   144→  }

   146→  private drawFootMarker(ctx: CanvasRenderingContext2D, x: number, y: number, planted: boolean): void {
   147→    ctx.fillStyle = planted ? '#51ff8e' : 'rgba(255,170,60,0.8)'
   148→    ctx.beginPath()
   149→    ctx.arc(x, y, 4.5, 0, Math.PI * 2)
   150→    ctx.fill()
   151→  }

   153→  /** bones + joints + labels on any solved geom (local space) */
   154→  private drawSkeleton(ctx: CanvasRenderingContext2D, g: SkelGeom): void {
   155→    if (!this.showBones) return
   156→    ctx.save()
   157→    ctx.strokeStyle = 'rgba(65,246,255,0.55)'
   158→    ctx.lineWidth = 1.1
   159→    const seg = (a: { x: number; y: number }, b: { x: number; y: number }) => {
   160→      ctx.beginPath()
   161→      ctx.moveTo(a.x, a.y)
   162→      ctx.lineTo(b.x, b.y)
   163→      ctx.stroke()
   164→    }
   165→    seg(g.hip, g.waist); seg(g.waist, g.chest); seg(g.chest, g.neckTop); seg(g.neckTop, g.headC)
   166→    seg(g.hipPtR, g.kneeR); seg(g.kneeR, g.ankleR); seg(g.ankleR, g.toeR)
   167→    seg(g.hipPtL, g.kneeL); seg(g.kneeL, g.ankleL); seg(g.ankleL, g.toeL)
   168→    seg(g.shoulderR, g.elbR); seg(g.elbR, g.handR)
   169→    seg(g.shoulderL, g.elbL); seg(g.elbL, g.handL)
   170→    // shoulder girdle line (the anatomy fix, visualized)
   171→    ctx.strokeStyle = 'rgba(255,120,180,0.8)'
   172→    ctx.lineWidth = 1.3
   173→    ctx.beginPath()
   174→    ctx.moveTo(g.shoulderL.x, g.shoulderL.y)
   175→    ctx.lineTo(g.neckTop.x, g.neckTop.y)
   176→    ctx.lineTo(g.shoulderR.x, g.shoulderR.y)
   177→    ctx.stroke()
   178→    // joints
   179→    ctx.fillStyle = '#ffd24a'
   180→    const joints = [
   181→      g.hip, g.hipPtR, g.hipPtL, g.waist, g.chest, g.neckTop, g.headC,
   182→      g.shoulderR, g.shoulderL, g.elbR, g.elbL, g.handR, g.handL,
   183→      g.kneeR, g.kneeL, g.ankleR, g.ankleL,
   184→    ]
   185→    for (const j of joints) {
   186→      ctx.beginPath()
   187→      ctx.arc(j.x, j.y, 1.7, 0, Math.PI * 2)
   188→      ctx.fill()
   189→    }
   190→    // THE proof joints: shoulders in pink
   191→    ctx.fillStyle = '#ff6fb5'
   192→    for (const j of [g.shoulderR, g.shoulderL]) {
   193→      ctx.beginPath()
   194→      ctx.arc(j.x, j.y, 3.0, 0, Math.PI * 2)
   195→      ctx.fill()
   196→    }
   197→    if (this.showLabels) {
   198→      ctx.fillStyle = 'rgba(160,240,255,0.95)'
   199→      ctx.font = '600 4.2px ui-monospace, monospace'
   200→      const lbl = (j: { x: number; y: number }, s: string) => ctx.fillText(s, j.x + 2.4, j.y - 2.4)
   201→      lbl(g.headC, 'skull')
   202→      lbl(g.neckTop, 'neckTop')
   203→      lbl(g.shoulderR, 'SHOULDER-R')
   204→      lbl(g.shoulderL, 'shoulder-l')
   205→      lbl(g.chest, 'chest')
   206→      lbl(g.waist, 'waist')
   207→      lbl(g.hip, 'pelvis')
   208→      lbl(g.kneeR, 'kneeR')
   209→      lbl(g.ankleR, 'ankleR')
   210→      lbl(g.elbR, 'elbR')
   211→      lbl(g.handR, 'handR')
   212→    }
   213→    ctx.restore()
   214→  }

   216→  /** contact sheets: every locomotion cycle, 8 frames each, real solve —
   217→   * light diagnostic background, per-cell ground lines + foot-contact
   218→   * markers (green = support, orange = swing), 2 clip-groups per band */
   219→  private drawSheets(ctx: CanvasRenderingContext2D): void {
   220→    const rows: Array<[string, typeof WALK_CLIP]> = [
   221→      ['WALK', WALK_CLIP],
   222→      ['RUN', RUN_CLIP],
   223→      ['SPRINT', SPRINT_CLIP],
   224→      ['COMBAT STEP', COMBAT_STEP_CLIP],
   225→      ['COMBAT-IDLE (rooted guard)', COMBAT_IDLE_CLIP],
   226→      ['BACKPEDAL', BACKPEDAL_CLIP],
   227→      ['IDLE', IDLE_CLIP],
   228→      ['JUMP', JUMP_CLIP],
   229→      ['DASH LAUNCH (iai)', DASH_LAUNCH_CLIP],
   230→      ['DASH CARRY (shunshin)', DASH_CARRY_CLIP],
   231→      ['DASH HUNT (gap-close)', DASH_HUNT_CLIP],
   232→      ['DASH PIERCE (lunge)', DASH_PIERCE_CLIP],
   233→      ['DASH RETREAT (guard)', DASH_RETREAT_CLIP],
   234→      ['DASH CARVE (breakaway)', DASH_CARVE_CLIP],
   235→    ]
   236→    const dpr = window.devicePixelRatio || 1
   237→    const w = ctx.canvas.width / dpr
   238→    const h = ctx.canvas.height / dpr
   239→    const cols = 8
   240→    const groupsPerRow = 2
   241→    const bands = Math.ceil(rows.length / groupsPerRow) // 3 bands
   242→    const cellW = w / (cols * groupsPerRow)
   243→    const rowH = h / (bands + 0.25)
   244→    // figure scale: fit character height inside a band minus label space
   245→    const S = Math.min(1.0, (rowH - 34) / 138)
   246→    ctx.save()
   247→    // light diagnostic backdrop — black silhouette at full contrast
   248→    ctx.fillStyle = '#e9eef3'
   249→    ctx.fillRect(0, 0, w, h)
   250→    ctx.font = '700 12px ui-monospace, monospace'
   251→    for (let r = 0; r < rows.length; r++) {
   252→      const band = Math.floor(r / groupsPerRow)
   253→      const group = r % groupsPerRow
   254→      const [label, clip] = rows[r]
   255→      const baseY = rowH * (band + 1)
   256→      const labelX = group * (w / groupsPerRow) + 10
   257→      ctx.fillStyle = '#1a3a52'
   258→      ctx.fillText(`${label} · ${clip.dur.toFixed(2)}s cycle`, labelX, baseY - rowH + 16)
   259→      for (let c = 0; c < cols; c++) {
   260→        const t = (clip.dur * c) / cols
   261→        sampleClip(clip, t, this.scratchPose)
   262→        const g = solveSkeleton(this.scratchPose, PLAYER_SPEC, this.scratchGeom)
   263→        // normalize: lowest foot contact sits ON this cell's ground line
   264→        const lowContact = Math.max(g.ankleL.y, g.ankleR.y, g.toeL.y, g.toeR.y)
   265→        const cx = group * (w / groupsPerRow) + cellW * (c + 0.5)
   266→        const groundY = baseY - 14
   267→        ctx.save()
   268→        ctx.translate(cx, groundY - (lowContact + 2.4) * S)
   269→        ctx.scale(S, S)
   270→        drawStick(ctx, g, PLAYER_STYLE, { time: t, simple: true })
   271→        ctx.restore()
   272→        // per-cell ground line
   273→        ctx.strokeStyle = 'rgba(40,80,110,0.45)'
   274→        ctx.lineWidth = 1
   275→        ctx.beginPath()
   276→        ctx.moveTo(cx - cellW * 0.42, groundY)
   277→        ctx.lineTo(cx + cellW * 0.42, groundY)
   278→        ctx.stroke()
   279→        // foot-contact markers: green = support (near ground), orange = swing
   280→        const hipY = groundY - (lowContact + 2.4) * S
   281→        const markFoot = (j: { x: number; y: number }) => {
   282→          const airborne = (lowContact - j.y) > 4 // clearly lifted vs support
   283→          ctx.fillStyle = airborne ? 'rgba(255,140,30,0.9)' : '#18c25a'
   284→          ctx.beginPath()
   285→          ctx.arc(cx + j.x * S, hipY + j.y * S, 3, 0, Math.PI * 2)
   286→          ctx.fill()
   287→        }
   288→        markFoot(g.ankleR)
   289→        markFoot(g.ankleL)
   290→        // frame tick
   291→        ctx.fillStyle = 'rgba(30,70,100,0.55)'
   292→        ctx.font = '600 9px ui-monospace, monospace'
   293→        ctx.fillText(`f${c}`, cx - 5, baseY - 3)
   294→      }
   295→    }
   296→    ctx.restore()
   297→  }
   298→}

   301→/* QA harness                                                          */

   303→
   304→interface QaCheck {
   305→  name: string
   306→  pass: boolean
   307→  detail: string
   308→}
   309→
   310→const sleep = (ms: number) => new Promise<void>((r) => setTimeout(r, ms))
   311→
   312→/** Scripted end-to-end verification. Parks the report on
   313→ * window.__qaResult. Drive from the browser console / agent-browser. */
   314→export async function runQA(game: Game): Promise<void> {
   315→  const checks: QaCheck[] = []
   316→  const log: string[] = []
   317→  const p = game.player
   318→  const input = game.input
   319→
   320→  const samplePose = () => {
   321→    let bad = 0
   322→    for (let i = 0; i < 17; i++) if (!Number.isFinite(p.pose[i])) bad++
   323→    return bad
   324→  }
   325→
   326→  try {
   327→    // ---- setup: controlled empty arena (waves suppressed, player alive) ----
   328→    game.qaMode = true
   329→    game.phase = 'playing'
   330→    game.enemies.length = 0
   331→    game.waves.reset()
   332→    // release every virtual input left held by prior console driving —
   333→    // stale held buttons contaminate the whole scripted sequence
   334→    input.setVirtualAxis(0, 0)
   335→    for (const a of ['left', 'right', 'up', 'down', 'light', 'heavy', 'dash', 'jump', 'parry', 'ability', 'ult', 'execute', 'sheathe'] as const) {
   336→      input.setVirtual(a, false)
   337→    }
   338→    p.reset(400)
   339→    input.setEnabled(true)
   340→    log.push('setup ok')
   341→
   342→    // ---- 1. anatomy invariants (the arms-from-the-neck fix) ----
   343→    {
   344→      const g = p.geom
   345→      const spec = PLAYER_SPEC
   346→      const armLen = Math.hypot(g.elbR.x - g.shoulderR.x, g.elbR.y - g.shoulderR.y)
   347→      // local y grows DOWN: shoulder below neck-top means shoulderR.y GREATER
   348→      const shoulderBelowNeck = g.shoulderR.y > g.neckTop.y + 1.5
   349→      const armFromShoulder = Math.abs(armLen - spec.upperArm) < 0.6
   350→      // neck column rises above chest: neckTop.y LESS (more negative) than chest
   351→      const neckVisible = g.neckTop.y < g.chest.y - 2.5
   352→      checks.push({
   353→        name: 'anatomy: shoulder joints below neck top',
   354→        pass: shoulderBelowNeck,
   355→        detail: `shoulderR.y=${g.shoulderR.y.toFixed(1)} neckTop.y=${g.neckTop.y.toFixed(1)}`,
   356→      })
   357→      checks.push({
   358→        name: 'anatomy: arm chains FROM shoulder (len=upperArm)',
   359→        pass: armFromShoulder,
   360→        detail: `len=${armLen.toFixed(2)} spec=${spec.upperArm}`,
   361→      })
   362→      checks.push({
   363→        name: 'anatomy: neck column visible above chest',
   364→        pass: neckVisible,
   365→        detail: `neckTop.y=${g.neckTop.y.toFixed(1)} chest.y=${g.chest.y.toFixed(1)}`,
   366→      })
   367→    }
   368→
   369→    // ---- 2. walk → run escalation (Condition A, analog speed mapping) ----
   370→    input.setVirtualAxis(0.55, 0)
   371→    await sleep(700)
   372→    const earlyMode = p.loco
   373→    const earlyVx = Math.abs(p.vx)
   374→    input.setVirtualAxis(1, 0)
   375→    await sleep(1500)
   376→    const fullMode = p.loco
   377→    input.setVirtualAxis(0, 0)
   378→    await sleep(500)
   379→    checks.push({
   380→      name: 'locomotion: WALK at partial input',
   381→      pass: earlyMode === 'walk' && earlyVx > 60 && earlyVx < 210,
   382→      detail: `mode@0.7s=${earlyMode} vx=${earlyVx.toFixed(0)}`,
   383→    })
   384→    checks.push({
   385→      name: 'locomotion: RUN at full input',
   386→      pass: fullMode === 'run',
   387→      detail: `mode@2.2s=${fullMode}`,
   388→    })
   389→
   390→    // ---- 3. sprint latch via double-tap (button events → press stamps) ----
   391→    input.setVirtual('right', true)
   392→    await sleep(120)
   393→    input.setVirtual('right', false)
   394→    await sleep(60)
   395→    input.setVirtual('right', true)
   396→    await sleep(900)
   397→    const sprintMode = p.loco
   398→    const sprintVx = Math.abs(p.vx)
   399→    input.setVirtual('right', false)
   400→    await sleep(400)
   401→    checks.push({
   402→      name: 'locomotion: sprint after double-tap',
   403→      pass: sprintMode === 'sprint' && sprintVx > 380,
   404→      detail: `loco=${sprintMode} vx=${sprintVx.toFixed(0)}`,
   405→    })
   406→
   407→    // ---- 4. combat mode when enemy near (Condition B) ----
   408→    const { Enemy } = await import('../entities/enemy')
   409→    const foe = new Enemy('grunt', p.x + 240, 1)
   410→    foe.spawning = false
   411→    foe.state = 'stun' // stun STATE + huge timer = truly frozen (stunT alone
   412→    foe.stunT = 999   // does nothing from 'enter' — the foe chased & hit us)
   413→    game.enemies.push(foe)
   414→    input.setVirtualAxis(0.5, 0)
   415→    await sleep(800)
   416→    const combatMode = p.loco
   417→    const enemyW = p.enemyNearW
   418→    checks.push({
   419→      name: 'context: combat locomotion near enemy',
   420→      pass: combatMode === 'combat' && enemyW > 0.5,
   421→      detail: `loco=${combatMode} enemyW=${enemyW.toFixed(2)}`,
   422→    })
   423→
   424→    // ---- 5. backpedal away from threat (facing HOLDS toward the enemy) ----
   425→    input.setVirtualAxis(-0.5, 0) // partial retreat: stays under the escape speed
   426→    await sleep(600)
   427→    const backMode = p.loco
   428→    const backFace = p.face
   429→    input.setVirtualAxis(0, 0)
   430→    await sleep(300)
   431→    checks.push({
   432→      name: 'context: backpedal retreating vs threat',
   433→      pass: backMode === 'backpedal' && backFace === 1,
   434→      detail: `loco=${backMode} face=${backFace}`,
   435→    })
   436→
   437→    // ---- 5b. THE CONDITIONED DASH SYSTEM ----
   438→    // One movement, nine authored animations: the variant must follow the
   439→    // CONDITIONS at the input instant (momentum, threat geometry, chain).
   440→    // (a) LAUNCH: stationary + no threat → the iai explosion
   441→    {
   442→      game.enemies.length = 0
   443→      p.reset(400)
   444→      input.setVirtualAxis(0, 0)
   445→      await sleep(500)
   446→      const x0 = p.x
   447→      input.setVirtualAxis(1, 0) // HOLD the direction → handoff at exit
   448→      input.setVirtual('dash', true)
   449→      await sleep(50)
   450→      const midVariant = p.dashVariant
   451→      const midState = p.state
   452→      input.setVirtual('dash', false)
   453→      // poll for the exit frame — sample the handed-off velocity exactly at
   454→      // the dash→move transition (before locomotion friction bleeds it)
   455→      let tailVx = 0
   456→      for (let i = 0; i < 30; i++) {
   457→        await sleep(16)
   458→        if (p.state === 'move') { tailVx = Math.abs(p.vx); break }
   459→      }
   460→      await sleep(250) // dash (0.17s) + exit bleed
   461→      const dx = p.x - x0
