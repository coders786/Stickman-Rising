    27→    27→class ComboTracker {
    28→    28→  hits = 0
    29→    29→  timer = 0
    30→    30→  timerMax = 2.6
    31→    31→  mult = 1
    32→    32→  best = 0
    33→    33→  bestRank = ''
    34→    34→  rankUpT = 0
    35→    35→  lastRank = ''
    36→    36→
    37→    37→  onHit(): void {
    38→    38→    this.hits++
    39→    39→    this.timer = this.timerMax
    40→    40→    this.mult = Math.min(1 + Math.floor(this.hits / 5) * 0.5, 5)
    41→    41→    if (this.hits > this.best) this.best = this.hits
    42→    42→    const { rank } = rankForHits(this.hits)
    43→    43→    if (rank && rank !== this.lastRank) {
    44→    44→      this.lastRank = rank
    45→    45→      if (rank.length > this.bestRank.length) this.bestRank = rank
    46→    46→      this.rankUpT = 0.7
    47→    47→    }
    48→    48→  }
    49→    49→
    50→    50→  update(dt: number): void {
    51→    51→    this.rankUpT = Math.max(0, this.rankUpT - dt)
    52→    52→    if (this.hits > 0) {
    53→    53→      this.timer -= dt
    54→    54→      if (this.timer <= 0) {
    55→    55→        this.hits = 0
    56→    56→        this.mult = 1
    57→    57→        this.lastRank = ''
    58→    58→      }
    59→    59→    }
    60→    60→  }
    61→    61→
    62→    62→  reset(): void {
    63→    63→    this.hits = 0
    64→    64→    this.timer = 0
    65→    65→    this.mult = 1
    66→    66→    this.best = 0
    67→    67→    this.bestRank = ''
    68→    68→    this.rankUpT = 0
    69→    69→    this.lastRank = ''
    70→    70→  }
    71→    71→}

   336→   336→
   337→   337→  private loop = (now: number): void => {
   338→   338→    if (this.destroyed) return
   339→   339→    this.raf = requestAnimationFrame(this.loop)
   340→   340→    let dtRaw = (now - this.lastFrame) / 1000
   341→   341→    this.lastFrame = now
   342→   342→    if (dtRaw > 0.05) dtRaw = 0.05
   343→   343→    if (dtRaw <= 0) dtRaw = 0.0001
   344→   344→    this.fps = damp(this.fps, 1 / dtRaw, 3, dtRaw)
   345→   345→
   346→   346→    const ctx = this.ctx
   347→   347→    if (!ctx) return
   348→   348→
   349→   349→    if (this.phase === 'paused') {
   350→   350→      // frozen frame, HUD still drawn
   351→   351→    } else {
   352→   352→      // --- timing: hitstop & slow-mo ---
   353→   353→      let simDt = dtRaw
   354→   354→      if (this.hitstopT > 0) {
   355→   355→        this.hitstopT -= dtRaw
   356→   356→        simDt = dtRaw * 0.05
   357→   357→        this.timeScale = 0.05 // world consumers (blade impact deflection) must SEE the freeze
   358→   358→      } else if (this.slowmoT > 0) {
   359→   359→        this.slowmoT -= dtRaw
   360→   360→        simDt = dtRaw * this.slowmoScale
   361→   361→        this.timeScale = this.slowmoScale
   362→   362→        if (this.slowmoT <= 0) this.timeScale = 1
   363→   363→      } else {
   364→   364→        this.timeScale = 1
   365→   365→      }
   366→   366→
   367→   367→      // --- cinematics run on real time ---
   368→   368→      this.updateCinematics(dtRaw)
   369→   369→
   370→   370→      // --- fixed-ish substeps for stable physics ---
   371→   371→      let remaining = simDt
   372→   372→      while (remaining > 1e-6) {
   373→   373→        const stepDt = Math.min(remaining, 1 / 90)
   374→   374→        this.step(stepDt)
   375→   375→        remaining -= stepDt
   376→   376→      }
   377→   377→    }
   378→   378→
   379→   379→    // --- render ---
   380→   380→    ctx.setTransform(this.dpr, 0, 0, this.dpr, 0, 0)
   381→   381→    const world: RenderWorld = {
   382→   382→      cam: this.cam,
   383→   383→      fx: this.fx,
   384→   384→      bg: this.bg,
   385→   385→      enemies: this.enemies,
   386→   386→      player: this.player,
   387→   387→      projectiles: this.projectiles,
   388→   388→      hud: this.hud,
   389→   389→      hudState: this.buildHudState(),
   390→   390→      time: this.time,
   391→   391→    }
   392→   392→    this.renderer.draw(ctx, world, this.cssW, this.cssH, dtRaw)
   393→   393→    // inspection overlay (solo/sheet views + state readout)
   394→   394→    this.inspect.draw(ctx, this, this.time)
   395→   395→
   396→   396→    this.input.endFrame(dtRaw)
   397→   397→  }
   398→   398→
   399→   399→  private step(dt: number): void {
   400→   400→    this.time += dt
   401→   401→    this.worldRef.time = this.time
   402→   402→    this.worldRef.timeScale = this.timeScale
   403→   403→    this.worldRef.quality = this.settings.quality
   404→   404→    this.worldRef.gore = this.settings.gore
   405→   405→    PLAYER_X_REF.x = this.player.x
   406→   406→    PLAYER_X_REF.y = this.player.y
   407→   407→    PLAYER_POS.x = this.player.x
   408→   408→    PLAYER_POS.y = this.player.y
   409→   409→
   410→   410→    if (this.phase === 'playing' || this.phase === 'gameover') {
   411→   411→      if (this.phase === 'playing') {
   412→   412→        this.handleMetaInput()
   413→   413→      }
   414→   414→      this.player.update(dt, this.worldRef)
   415→   415→      for (const e of this.enemies) {
   416→   416→        e.update(dt, this.worldRef)
   417→   417→      }
   418→   418→      // purge dead
   419→   419→      for (let i = this.enemies.length - 1; i >= 0; i--) {
   420→   420→        if (this.enemies[i].dead) this.enemies.splice(i, 1)
   421→   421→      }
   422→   422→      this.projectiles.update(dt, this.worldRef)
   423→   423→
   424→   424→      if (this.phase === 'playing') {
   425→   425→        if (!this.qaMode) this.waves.update(dt, this.enemies.length)
   426→   426→        this.combo.update(dt)
   427→   427→        this.runTime += dt
   428→   428→        this.stats.timeAlive = this.runTime
   429→   429→        // death → game over
   430→   430→        if (!this.qaMode && this.player.isDead && this.player.deadTime > 1.4) {
   431→   431→          this.gameOver()
   432→   432→        }
   433→   433→      }
   434→   434→    } else if (this.phase === 'attract') {
   435→   435→      // idle hero breathing on the rooftop (framed below the title logo)
   436→   436→      this.player.update(dt, this.worldRef)
   437→   437→      this.cam.focus(this.player.x + Math.sin(this.time * 0.14) * 140, this.player.y - 110, 0, 0)
   438→   438→    }
   439→   439→
   440→   440→    this.fx.update(dt)
   441→   441→    this.bg.update(dt)               // living storm: rain, gusts, fog, lightning
   442→   442→    this.updateCamera(dt)
   443→   443→  }
   444→   444→
   445→   445→  private updateCamera(dt: number): void {
   446→   446→    if (this.phase === 'playing' || this.phase === 'gameover') {
   447→   447→      const p = this.player
   448→   448→      let tx = p.x
   449→   449→      let ty = p.y
   450→   450→      let vx = p.vx
   451→   451→      let vy = p.vy
   452→   452→      // during execution / ult, focus the action
   453→   453→      if (this.exec) {
   454→   454→        tx = (p.x + this.exec.enemy.x) / 2
   455→   455→        ty = (p.y + this.exec.enemy.y) / 2 - 30
   456→   456→        vx = 0
   457→   457→        vy = 0
   458→   458→      }
   459→   459→      this.cam.focus(tx, ty, vx, vy)
   460→   460→      // dynamic zoom: pull back when many enemies, punch in during cinematics
   461→   461→      let zoom = 1.04
   462→   462→      const n = this.enemies.length
   463→   463→      if (n > 10) zoom = 0.94
   464→   464→      if (n > 18) zoom = 0.88
   465→   465→      if (this.exec || this.ult) zoom = 1.3
   466→   466→      this.cam.setZoom(zoom)
   467→   467→    }
   468→   468→    this.cam.update(dt)
   469→   469→  }
   470→   470→
   471→   471→  private handleMetaInput(): void {
   472→   472→    // pause
   473→   473→    if (this.input.pressed('pause')) {
   474→   474→      this.pause()
   475→   475→      return
   476→   476→    }
   477→   477→    // sword switching
   478→   478→    const swordIds: SwordId[] = ['katana', 'inferno', 'thunder', 'void', 'gravity', 'phantom']

   482→   482→        this.trySwitchSword(swordIds[i])
   483→   483→      }
   484→   484→    }
   485→   485→    if (this.input.pressed('cycleSword')) {
   486→   486→      const idx = SWORD_ORDER.indexOf(this.player.sword.id)
   487→   487→      for (let i = 1; i <= 6; i++) {
   488→   488→        const next = SWORD_ORDER[(idx + i) % 6]
   489→   489→        if (this.unlocked.includes(next)) {
   490→   490→          this.trySwitchSword(next)
   491→   491→          break
   492→   492→        }
   493→   493→      }
   494→   494→    }
   495→   495→    // ultimate
   496→   496→    if (this.input.pressed('ult') && this.player.meter >= 100 && !this.ult && !this.exec) {
   497→   497→      this.startUlt()
   498→   498→    }
   499→   499→    // execution
   500→   500→    if (this.input.pressed('execute') && !this.exec && !this.ult) {
   501→   501→      const target = this.findExecuteTarget()
   502→   502→      if (target) this.startExecution(target)
   503→   503→    }
   504→   504→  }

   583→   583→  private hitEnemy(enemyId: number, blade: { bx: number; by: number; tx: number; ty: number }, atk: AttackRuntimeInfo): void {
   584→   584→    const e = this.enemies.find((en) => en.id === enemyId)
   585→   585→    if (!e || e.dead) return
   586→   586→    const sword = this.player.sword
   587→   587→    const dir = Math.sign(blade.tx - blade.bx) || this.player.face
   588→   588→    const hx = (blade.bx + blade.tx) / 2
   589→   589→    const hy = (blade.by + blade.ty) / 2
   590→   590→    const ex = e.x
   591→   591→    // impact FX at the blade's actual contact height, clamped to the visual
   592→   592→    // body (head-top ≈ 88u … feet ≈ 10u) — sparks appear where the edge
   593→   593→    // crossed the silhouette, not at a fixed belly point
   594→   594→    const ey = clamp(hy, e.y - 86 * e.scale, e.y - 10 * e.scale)
   595→   595→
   596→   596→    // elite parry: occasionally deflects light strikes
   597→   597→    if (e.kind === 'elite' && !atk.heavy && Math.random() < 0.16 && e.state !== 'stun' && e.state !== 'launched') {
   598→   598→      this.audio.play('clang')
   599→   599→      this.fx.impact(ex, ey, Math.atan2(ey - hy, ex - hx), true, '#ffffff', '#ff5c7a')
   600→   600→      this.fx.sparks(ex, ey, -Math.PI / 2, 1.4, 12, '#ffffff', 420, 0.4)
   601→   601→      this.addHitstop(0.05)
   602→   602→      this.cam.addTrauma(0.15)
   603→   603→      this.fx.text(ex, ey - 50, 'PARRY!', '#ff5c7a', 12)
   604→   604→      return
   605→   605→    }
   606→   606→
   607→   607→    const res = e.takeHit(atk, dir, this.player.x)
   608→   608→
   609→   609→    // combo & score
   610→   610→    this.combo.onHit()
   611→   611→    this.stats.maxCombo = Math.max(this.stats.maxCombo, this.combo.hits)
   612→   612→    if (this.combo.rankUpT > 0.6) {
   613→   613→      this.audio.play('rankup', { pitch: 1 + this.combo.hits * 0.002 })
   614→   614→      this.emit({ type: 'rank', rank: this.combo.lastRank })
   615→   615→    }
   616→   616→    this.stats.score += Math.round(atk.dmg * this.combo.mult * 0.8)
   617→   617→
   618→   618→    switch (res) {
   619→   619→      case 'killed': {
   620→   620→        this.onEnemyKilled(e)
   621→   621→        break
   622→   622→      }
   623→   623→      case 'blocked': {
   624→   624→        this.audio.play('clang', { pitch: 0.9 })
   625→   625→        this.fx.impact(ex, ey, Math.atan2(ey - hy, ex - hx), false, '#ffffff', '#46ffb4')
   626→   626→        this.fx.text(ex, ey - 46, 'BLOCK', '#46ffb4', 11)
   627→   627→        this.addHitstop(0.03)
   628→   628→        this.cam.addTrauma(0.06)
   629→   629→        break
   630→   630→      }
   631→   631→      case 'armored': {
   632→   632→        this.audio.play('clang', { pitch: 0.6, volume: 0.8 })
   633→   633→        this.fx.sparks(ex, ey, Math.atan2(ey - hy, ex - hx), 0.9, 8, '#ffae2e', 320, 0.35)
   634→   634→        this.fx.text(ex, ey - 46, 'ARMOR', '#ffae2e', 11)
   635→   635→        this.addHitstop(0.02)
   636→   636→        break
   637→   637→      }
   638→   638→      default: {
   639→   639→        // solid hit
   640→   640→        this.audio.play(atk.heavy ? 'hitHeavy' : 'hit', {
   641→   641→          volume: clamp(0.55 + atk.dmg * 0.01, 0.55, 1),
   642→   642→          pitch: rand(0.94, 1.08),
   643→   643→          intensity: clamp01(atk.dmg / 30),
   644→   644→        })
   645→   645→        this.fx.impact(ex, ey, Math.atan2(ey - hy, ex - hx), atk.heavy, sword.colors.glow, sword.colors.accent)
   646→   646→        const dmgShown = Math.round(atk.dmg * (e.vulnerableT > 0 ? 2 : 1) * (e.kind === 'heavy' || e.kind === 'boss' ? 0.72 : 1))
   647→   647→        this.fx.text(ex, ey - 46, `${dmgShown}`, atk.heavy ? sword.colors.accent : '#f4f9fc', atk.heavy ? 17 : 13)
   648→   648→        // crash-zoom punch on heavies (Z-axis push-pull) + impact frames on
   649→   649→        // the big connects — the anime kill-frame language
   650→   650→        this.addHitstop(atk.hitstop, atk.heavy ? 0.09 : 0)
   651→   651→        if (atk.heavy && atk.dmg >= 15) {
   652→   652→          this.fx.impactFrame(ex, ey, sword.colors.glow)
   653→   653→        } else if (atk.heavy) {
   654→   654→          this.fx.chroma(0.45)
   655→   655→        }
   656→   656→        this.cam.addTrauma(atk.shake)
   657→   657→        if (e.vulnerableT > 0) this.fx.text(ex, ey - 70, 'WEAK POINT', '#ffe14d', 11)
   658→   658→        break
   659→   659→      }
   660→   660→    }
   661→   661→  }

   900→   900→  private buildHudState(): HudState {
   901→   901→    const p = this.player
   902→   902→    const boss = this.enemies.find((e) => e.kind === 'boss' && !e.dead)
   903→   903→    let execPrompt: { x: number; y: number } | null = null
   904→   904→    if (this.phase === 'playing' && !this.exec && !this.ult) {
   905→   905→      const target = this.findExecuteTarget()
   906→   906→      if (target) {
   907→   907→        const zoom = this.cam.zoom
   908→   908→        execPrompt = {
   909→   909→          x: (target.x - this.cam.x - this.cam.offX) * zoom + this.cssW / 2,
   910→   910→          y: (target.y - 120 - this.cam.y - this.cam.offY) * zoom + this.cssH / 2,
   911→   911→        }
   912→   912→      }
   913→   913→    }
   914→   914→    return {
