    16→/** circle vs swept blade quad (prev segment + current segment) — prevents tunneling */
    17→function circleHitsSweptQuad(
    18→  cx: number, cy: number, r: number,
    19→  pbx: number, pby: number, ptx: number, pty: number,
    20→  bx: number, by: number, tx: number, ty: number,
    21→): boolean {
    22→  // current + previous blade segments
    23→  if (segHitsCircle(bx, by, tx, ty, cx, cy, r)) return true
    24→  if (segHitsCircle(pbx, pby, ptx, pty, cx, cy, r)) return true
    25→  // swept edges (tip path + base path)
    26→  if (segHitsCircle(ptx, pty, tx, ty, cx, cy, r)) return true
    27→  if (segHitsCircle(pbx, pby, bx, by, cx, cy, r)) return true
    28→  // circle center inside the swept quad (spins / wide arcs)
    29→  if (pointInTri(cx, cy, pbx, pby, ptx, pty, tx, ty)) return true
    30→  if (pointInTri(cx, cy, pbx, pby, tx, ty, bx, by)) return true
    31→  return false
    32→}

    34→function pointInTri(px: number, py: number, x1: number, y1: number, x2: number, y2: number, x3: number, y3: number): boolean {
    35→  const d1 = (px - x2) * (y1 - y2) - (x1 - x2) * (py - y2)
    36→  const d2 = (px - x3) * (y2 - y3) - (x2 - x3) * (py - y3)
    37→  const d3 = (px - x1) * (y3 - y1) - (x3 - x1) * (py - y1)
    38→  const hasNeg = d1 < 0 || d2 < 0 || d3 < 0
    39→  const hasPos = d1 > 0 || d2 > 0 || d3 > 0
    40→  return !(hasNeg && hasPos)
    41→}

    47→const SCALE = 1.32
    48→/** nominal (airborne) hip height = thigh(25)+shin(24) = 49 local × SCALE.
    49→ * Grounded height is NOT fixed — it derives from the planted foot via the
    50→ * foot-ground lock (updateFootLock): idle bent knees, lunges and landing
    51→ * squashes all sit exactly ON the ground line. */
    52→const HIP_H = (25 + 24) * SCALE
    53→
    54→/**
    55→ * Enemy body spheres: [centerY offset above ground, radius] in enemy-local
    56→ * units (× e.scale). Traced from the ENEMY_SPEC skeleton — hip at 47u, torso
    57→ * 32u, head top ≈ 94u. The old single circle (y-34, r 26) only covered the
    58→ * lower 2/3 of the body, so chest/head-height slashes passed straight
    59→ * through the sprite without connecting. Three spheres hug the visual body.
    60→ */
    61→const BODY_SPHERES: ReadonlyArray<{ oy: number; r: number }> = [
    62→  { oy: 14, r: 15 }, // legs / hip mass
    63→  { oy: 48, r: 21 }, // torso (the main target)
    64→  { oy: 78, r: 17 }, // chest / head zone
    65→]
    66→
    67→/** blade segment vs an enemy's full visual body (any sphere, swept-aware).
    68→ * `pad` fattens the radius per-attack — a katana edge is a plane; the hit
    69→ * volume must match the visual footprint of the blade edge corridor. */
    70→function bladeHitsBody(
    71→  e: { x: number; y: number; scale: number },
    72→  hasPrev: boolean,
    73→  prev: { bx: number; by: number; tx: number; ty: number },
    74→  cur: { bx: number; by: number; tx: number; ty: number },
    75→  pad = 3,
    76→): boolean {
    77→  const s = e.scale
    78→  for (const sp of BODY_SPHERES) {
    79→    const cy = e.y - sp.oy * s
    80→    const r = sp.r * s + pad
    81→    if (hasPrev
    82→      ? circleHitsSweptQuad(e.x, cy, r, prev.bx, prev.by, prev.tx, prev.ty, cur.bx, cur.by, cur.tx, cur.ty)
    83→      : segHitsCircle(cur.bx, cur.by, cur.tx, cur.ty, e.x, cy, r)
    84→    ) return true
    85→  }
    86→  return false
    87→}

   320→  update(dt: number, w: WorldRef): void {
   321→    this.iframes = Math.max(0, this.iframes - dt)
   322→    this.armorT = Math.max(0, this.armorT - dt)
   323→    this.flashT = Math.max(0, this.flashT - dt * 5)
   324→    this.swingGlow = Math.max(0, this.swingGlow - dt * 5)
   325→    this.riposteT = Math.max(0, this.riposteT - dt)
   326→    this.energy = Math.min(this.maxEnergy, this.energy + dt * 3)
   327→
   328→    switch (this.state) {
   329→      case 'move': this.updateMove(dt, w); break
   330→      case 'attack': this.updateAttack(dt, w); break
   331→      case 'dash': this.updateDash(dt, w); break
   332→      case 'parry': this.updateParry(dt, w); break
   333→      case 'charge': this.updateCharge(dt, w); break
   334→      case 'hurt': this.updateHurt(dt, w); break
   335→      case 'dead': this.updateDead(dt, w); break
   336→      case 'scripted': this.updateScripted(dt); break
   337→    }
   338→
   339→    // physics integration (shared)
   340→    if (this.state !== 'dash') {
   341→      let g: number
   342→      if (this.state === 'attack' && this.action && this.action.def.float) {
   343→        // authored air-combo float overrides the shaping (juggle windows)
   344→        g = GRAVITY * this.action.def.float
   345→      } else if (this.vy < 0) {
   346→        // ANTI-FLOAT: apex decay — no moon hovering
   347→        g = GRAVITY * (this.vy < -260 ? GRAV_RISE : GRAV_APEX)
   348→      } else {
   349→        // heavy fall: the body plummets
   350→        g = GRAVITY * GRAV_FALL
   351→      }
   352→      this.vy += g * dt
   353→      if (this.vy > TERMINAL_VY) this.vy = TERMINAL_VY
   354→    }
   355→    this.x += this.vx * dt
   356→    this.y += this.vy * dt
   357→
   358→    // ground
   359→    const wasAir = !this.grounded
   360→    if (this.y >= 0) {
   361→      this.y = 0
   362→      if (this.vy > 140) {
   363→        this.landSpring = clamp(this.vy / 700, 0, 1)
   364→        if (wasAir) this.onLand(w)
   365→      } else if (wasAir) {
   366→        this.landSpring = 0.4
   367→        this.onLand(w)
   368→      }
   369→      this.vy = 0
   370→      this.grounded = true
   371→      this.jumpsUsed = 0
   372→      this.airDashUsed = false
   373→    } else {
   374→      this.grounded = false
   375→    }
   376→    // arena bounds
   377→    if (this.x < w.arenaMinX + 20) {
   378→      this.x = w.arenaMinX + 20
   379→      if (this.vx < -200) w.fx.dust(this.x - 12, 0, 5)
   380→      this.vx = Math.max(0, this.vx)
   381→    }
   382→    if (this.x > w.arenaMaxX - 20) {
   383→      this.x = w.arenaMaxX - 20
   384→      if (this.vx > 200) w.fx.dust(this.x + 12, 0, 5)
   385→      this.vx = Math.min(0, this.vx)
   386→    }
   387→
   388→    this.dashCd = Math.max(0, this.dashCd - dt)
   389→    this.dashRecharge = Math.max(0, this.dashRecharge - dt)
   390→    if (this.dashRecharge <= 0) this.dashConsec = 0
   391→
   392→    // ability script
   393→    if (this.ability) this.updateAbility(dt, w)
   394→
   395→    // animation + solve
   396→    this.animator.update(dt)
   397→    // Base layer: the clip FULLY owns the pose each frame (write, never
   398→    // accumulate). applyProcedural() is a bounded additive layer on top.
   399→    // Without this copy, procedural += deltas integrate unbounded → spine
   400→    // spin + root sink. (Same contract as Spine's AnimationState.apply /
   401→    // Godot AnimationTree → Skeleton2D.)
   402→    this.pose.set(this.animator.pose)
   403→    this.applyProcedural(dt)
   404→    // procedural layers (bounded, frame-local — never accumulate):
   405→    // SWORD-LED ARC first: it rewrites the arm channels, so the blade
   406→    // dynamics spring below sees the ARC's angular velocity — the whip
   407→    // lags/overshoots the real swing, not the raw keyframes.
   408→    this.swordArc.update(dt, {
   409→      pose: this.pose,
   410→      chestX: this.geom.chest.x, chestY: this.geom.chest.y,
   411→      handX: this.geom.handR.x, handY: this.geom.handR.y,
   412→      def: this.state === 'attack' ? this.action?.def ?? null : null,
   413→      animT: this.action ? this.action.animT : 0,
   414→      enemies: w.enemies,
   415→      face: this.face,
   416→      x: this.x,
   417→      hipY: this.hipY,
   418→      reach: PLAYER_SPEC.blade * this.sword.stats.range * SCALE * 1.1 + 130,
   419→    })
   420→    this.updateBladeDynamics(dt, w) // wrist lag spring + kinetic chain — feeds hitboxes
   421→    this.applyTwoHandGrip(dt)       // off-hand IK onto the tsuka
   422→    solveSkeleton(this.pose, PLAYER_SPEC, this.geom)
   423→    this.updateFootLock(dt)         // weight: hips sit on the planted foot
   424→    this.updateBladeWorld()
   425→    this.updateSwingScale(dt)
   426→    this.updateBladeZ(dt)
   427→  }

   429→  private onLand(w: WorldRef): void {
   430→    w.audio.play('land', { volume: 0.5 })
   431→    w.fx.dust(this.x, 0, 7)
   432→    // slam impact while attacking
   433→    const a = this.action
   434→    if (a && a.def.slamImpact && !a.impactDone) {
   435→      a.impactDone = true
   436→      this.slamAoe(w, a.def.slamImpact.r, a.def.slamImpact.dmg, a.def.slamImpact.launch)
   437→    }
   438→    if (this.state === 'move' && this.lastGrounded === false) {
   439→      this.animator.play(LAND_CLIP, { blend: 0.06 })
   440→    }
   441→    this.airIndex = 0
   442→  }

   444→  private slamAoe(w: WorldRef, r: number, dmg: number, launch: number): void {
   445→    const info: AttackRuntimeInfo = {
   446→      id: 'slam', dmg: dmg * this.sword.stats.dmg, heavy: true,
   447→      stagger: 0.5, kb: 240, kbY: 0, launch, juggle: 0,
   448→      hitstop: 0.11, shake: 0.45, aoe: r, breakGuard: true, kind: 'heavy',
   449→    }
   450→    const colors = this.sword.colors
   451→    w.fx.shockwave(this.x, 0, r, colors.glow, 0.5, 8)
   452→    w.fx.groundCrack(this.x, 0, r * 0.6)
   453→    w.fx.burst(this.x, -6, 14, colors.glow, 340, 0.5, 'spark')
   454→    w.fx.screenFlash(colors.trail, 0.16, 5)
   455→    w.addHitstop(0.09, 0.06)
   456→    w.cam.addTrauma(0.4)
   457→    w.audio.play('slam', { intensity: 1 })
   458→    const bx = this.x
   459→    for (const e of w.enemies) {
   460→      if (e.dead || e.spawning) continue
   461→      const dx = e.x - bx
   462→      if (Math.abs(dx) < r && e.y > -140) {
   463→        w.hitEnemy(e.id, { bx: bx - 20, by: -20, tx: bx + 20, ty: -20 }, info)
   464→      }
   465→    }
   466→  }

   472→  private updateMove(dt: number, w: WorldRef): void {
   473→    const input = w.input
   474→    const ax = input.axisX()
   475→    const ay = input.axisY()
   476→
   477→    // facing follows input
   478→    if (ax !== 0) this.face = ax > 0 ? 1 : -1
   479→
   480→    // horizontal movement
   481→    if (ax !== 0) {
   482→      const accel = this.grounded ? ACCEL : AIR_ACCEL
   483→      this.vx += ax * accel * dt
   484→      this.vx = clamp(this.vx, -MOVE_MAX, MOVE_MAX)
   485→    } else if (this.grounded) {
   486→      const f = FRICTION * dt
   487→      this.vx = Math.abs(this.vx) <= f ? 0 : this.vx - Math.sign(this.vx) * f
   488→    } else {
   489→      this.vx = damp(this.vx, 0, 0.4, dt)
   490→    }
   491→
   492→    // locomotion clip selection
   493→    const speed = Math.abs(this.vx)
   494→    if (this.grounded) {
   495→      if (speed > 40) {
   496→        this.animator.play(RUN_CLIP, { blend: 0.09, speed: clamp(speed / MOVE_MAX, 0.65, 2.1) })
   497→      } else if (this.vy === 0) {
   498→        this.animator.play(IDLE_CLIP, { blend: 0.16 })
   499→      }
   500→    } else if (this.vy < -60) {
   501→      this.animator.play(JUMP_CLIP, { blend: 0.06 })
   502→    } else {
   503→      this.animator.play(FALL_CLIP, { blend: 0.12 })
   504→    }
   505→
   506→    // ---- actions ----
   507→    if (this.tryCombatInputs(w, ax, ay)) return
   508→  }

   510→  /** returns true if a new state was entered */
   511→  private tryCombatInputs(w: WorldRef, ax: number, ay: number): boolean {
   512→    const input = w.input
   513→
   514→    // dash (anytime, from move)
   515→    if ((input.pressed('dash') || input.buffered('dash')) && this.canDash()) {
   516→      this.startDash(w, ax, ay)
   517→      return true
   518→    }
   519→    // jump (grounded = jump 1, plus one air jump)
   520→    if (input.pressed('jump') || input.buffered('jump', 0.18)) {
   521→      if (this.jumpsUsed < 2) {
   522→        this.doJump(w)
   523→        return false
   524→      }
   525→    }
   526→    // parry
   527→    if (input.pressed('parry')) {
   528→      this.enterParry(w)
   529→      return true
   530→    }
   531→    // ability
   532→    if (input.pressed('ability')) {
   533→      if (this.energy >= this.sword.ability.cost) {
   534→        this.startAbility(w)
   535→        return true
   536→      }
   537→    }
   538→    // ult is handled by Game (cinematic)
   539→
   540→    // heavy: charge vs tap
   541→    if (input.down('heavy') && this.grounded) {
   542→      this.enterCharge(w)
   543→      return true
   544→    }
   545→    if (input.pressed('heavy') || input.buffered('heavy')) {
   546→      input.takeBuffered('heavy')
   547→      this.startAttack(this.pickHeavy(w), w)
   548→      return true
   549→    }
   550→    // light
   551→    if (input.pressed('light') || input.buffered('light')) {
   552→      input.takeBuffered('light')
   553→      // DASH MOMENTUM STRIKE: attacking while the last dash's momentum
   554→      // still carries the body converts into the through-cut — the
   555→      // dash-strike feel (Hades). Without this, the 150ms in-dash input
   556→      // window was brutally tight and dash attacks "randomly" whiffed.
   557→      if (this.grounded
   558→        && w.time - this.dashEndedAt < 0.18
   559→        && Math.abs(this.vx) > 110
   560→        && this.riposteT <= 0) {
   561→        this.startAttack(ATTACKS.dashSlash, w)
   562→        return true
   563→      }
   564→      const id = this.riposteT > 0 ? 'riposte' : this.pickLight(w, ax, ay)
   565→      this.startAttack(ATTACKS[id], w)
   566→      return true
   567→    }
   568→    return false
   569→  }

   571→  private doJump(w: WorldRef): void {
   572→    w.input.takeBuffered('jump')
   573→    this.jumpsUsed++
   574→    this.vy = this.jumpsUsed === 1 ? JUMP_V : AIR_JUMP_V
   575→    this.grounded = false
   576→    w.audio.play('jump', { volume: 0.4 })
   577→    w.fx.dust(this.x, 0, 4)
   578→    if (this.jumpsUsed === 2) {
   579→      w.fx.burst(this.x, this.y - 20, 6, '#9fb4c4', 160, 0.4, 'dot')
   580→    }
   581→  }

   587→  private canDash(): boolean {
   588→    if (this.dashCd > 0 || this.dashRecharge > 0) return false
   589→    if (!this.grounded && this.airDashUsed) return false
   590→    return true
   591→  }

   593→  startDash(w: WorldRef, ax: number, ay: number): void {
   594→    w.input.takeBuffered('dash')
   595→    let dx = ax
   596→    let dy = ay
   597→    if (dx === 0 && dy === 0) dx = this.face
   598→    const len = Math.hypot(dx, dy) || 1
   599→    dx /= len
   600→    dy /= len
   601→    if (dx !== 0) this.face = dx > 0 ? 1 : -1
   602→    this.dashDx = dx
   603→    this.dashDy = dy
   604→    this.state = 'dash'
   605→    this.dashT = DASH_TIME
   606→    this.dashCd = DASH_CD
   607→    this.dashConsec++
   608→    if (this.dashConsec >= 2) this.dashRecharge = DASH_RECHARGE
   609→    if (!this.grounded) this.airDashUsed = true
   610→    this.iframes = Math.max(this.iframes, 0.12)
   611→    this.action = null
   612→    this.charging = false
   613→    this.vx = dx * DASH_SPEED
   614→    this.vy = dy * DASH_SPEED * (this.grounded && dy < -0.4 ? 0.55 : 1)
   615→    this.animator.play(this.grounded ? DASH_CLIP : DASH_AIR_CLIP, { blend: 0.05 })
   616→    w.audio.play('dash', { pitch: rand(0.95, 1.1) })
   617→    const colors = this.sword.colors
   618→    w.fx.speedStreaks(this.x, this.y - 40, dx * DASH_SPEED * 0.9, 6, rgba(colors.trail, 1))
   619→    w.fx.chroma(0.4) // lens strain on the launch
   620→    w.fx.dust(this.x, this.grounded ? 0 : this.y, 5)
   621→    w.cam.addTrauma(0.06)
   622→  }

   624→  private updateDash(dt: number, w: WorldRef): void {
   625→    this.dashT -= dt
   626→    // afterimages
   627→    this.afterimgTimer -= dt
   628→    if (this.afterimgTimer <= 0) {
   629→      this.afterimgTimer = 0.026
   630→      w.fx.afterimage(this.pose, this.x, this.hipY, this.face, this.sword.colors.trail, 0.3, SCALE)
   631→    }
   632→    // maintain velocity with slight decay
   633→    const k = 1 - this.dashT / DASH_TIME
   634→    const sp = DASH_SPEED * (1 - k * 0.25)
   635→    this.vx = this.dashDx * sp
   636→    this.vy = this.dashDy * sp * (this.grounded ? (this.dashDy < 0 ? 0.4 : 1) : 1)
   637→
   638→    // dash attack
   639→    const input = w.input
   640→    if (input.pressed('light') || input.buffered('light')) {
   641→      input.takeBuffered('light')
   642→      this.startAttack(ATTACKS.dashSlash, w)
   643→      return
   644→    }
   645→    if (input.pressed('heavy') || input.buffered('heavy')) {
   646→      if (!this.grounded) {
   647→        input.takeBuffered('heavy')
   648→        this.startAttack(ATTACKS.slam, w)
   649→        return
   650→      }
   651→      input.takeBuffered('heavy')
   652→      this.startAttack(ATTACKS.dashHeavy, w)
   653→      return
   654→    }
   655→    if (this.dashT <= 0) {
   656→      this.state = 'move'
   657→      this.dashEndedAt = w.time
   658→      this.vx *= 0.55
   659→      if (!this.grounded) {
   660→        this.vy *= 0.5
   661→      } else if (Math.abs(this.vx) > 260) {
   662→        // THE BRAKING SKID: momentum must shed through a visible physical
   663→        // pose — heel-dig crouch + a dust scrape, never a dead stop.
   664→        w.fx.dust(this.x - this.face * 12, 0, 5)
   665→        this.landSpring = 0.35
   666→        this.animator.play(LAND_CLIP, { blend: 0.05 })
   667→      }
   668→    }
   669→  }

   675→  private pickLight(w: WorldRef, ax: number, ay: number): string {
   676→    // contextual branches
   677→    if (ay < -0.4 && this.grounded) return 'launcher'
   678→    if (ay > 0.4 && this.grounded) return 'spin'
   679→    if (this.grounded && ax !== 0 && Math.sign(ax) !== this.face) return 'retreat'
   680→    if (!this.grounded) {
   681→      if (ay > 0.4) return 'dive'
   682→      this.airIndex = (this.airIndex % 3) + 1
   683→      return `a${this.airIndex}`
   684→    }
   685→    // ground light chain (with time decay)
   686→    if (w.time - this.lastLightAt > 1.1) this.lightIndex = 0
   687→    this.lightIndex = Math.min(this.lightIndex + 1, 5)
   688→    return `l${Math.min(this.lightIndex, 5)}`
   689→  }

   691→  /** Heavy CHAIN: Kiri Oroshi → Tsuki → Grand Slam. Sequential like the
   692→   * light chain, with time decay — heavy attacks finally have combo
   693→   * pairing instead of one animation forever. */
   694→  private pickHeavy(w: WorldRef): AttackDef {
   695→    if (w.time - this.lastHeavyAt > 1.15) this.heavyIndex = 0
   696→    this.heavyIndex = Math.min(this.heavyIndex + 1, 3)
   697→    return ATTACKS[`h${this.heavyIndex}`] ?? ATTACKS.h1
   698→  }

   700→  startAttack(def: AttackDef, w: WorldRef, fromParry = false): void {
   701→    if (!def) return
   702→    // IN-BETWEEN LAW: capture the source state BEFORE switching — chained
   703→    // attacks crossfade slower (0.11s) than fresh entries (0.07s) so every
   704→    // transition between animation instances gets real in-betweens.
   705→    const wasAttack = this.state === 'attack'
   706→    // requirement checks
   707→    if (def.requires === 'ground' && !this.grounded) def = ATTACKS.a1
   708→    if (def.requires === 'air' && this.grounded) def = ATTACKS.l1
   709→
   710→    // face toward held direction at attack start
   711→    const ax = w.input.axisX()
   712→    if (ax !== 0 && !fromParry) {
   713→      this.face = ax > 0 ? 1 : -1
   714→    } else if (!fromParry) {
   715→      // SOFT AIM: no direction held → face the nearest threat. Every swing
   716→      // commits toward a real target — the root of "aimed at different
   717→      // angles": cuts should converge on the enemy, not on empty planes.
   718→      const tgt = this.nearestEnemy(w, 600)
   719→      if (tgt && Math.abs(tgt.x - this.x) > 24) {
   720→        this.face = tgt.x > this.x ? 1 : -1
   721→      }
   722→    }
   723→
   724→    // heavy chain bookkeeping (h1 → h2 → h3 sequencing)
   725→    const hm = /^h([123])$/.exec(def.id)
   726→    if (hm) {
   727→      this.heavyIndex = Number(hm[1])
   728→      this.lastHeavyAt = w.time
   729→    }
   730→    if (/^l[1-5]$/.test(def.id)) this.lastLightAt = w.time
   731→
   732→    // HYPER-ARMOR: the charged release cannot be interrupted (the root
   733→    // cause of "charged never hits" — enemies struck mid-swing cancelled it)
   734→    if (def.id === 'charged') this.armorT = 0.5
   735→
   736→    const trail = w.fx.startTrail(this.sword.colors.trail, this.sword.colors.trail2, this.sword.trailWidth)
   737→    this.hasPrevBlade = false
   738→    this.hasPrevAng = false
   739→    this.action = {
   740→      def,
   741→      animT: 0,
   742→      hitMap: new Map(),
   743→      trail,
   744→      sfxDone: false,
   745→      moveDone: false,
   746→      moveYDone: false,
   747→      impactDone: false,
   748→      overshootDone: false,
   749→      flashDone: !def.drawFlash,
   750→      fromParry,
   751→    }
   752→    if (fromParry) this.riposteT = 0
   753→    this.swordArc.begin() // sword-led arc: reset the engage capture
   754→    this.state = 'attack'
   755→    this.charging = false
   756→    // ANIME TIMING: attack bodies play "on twos" (24Hz stepped keys —
   757→    // crisp, keyframed combat beats à la GG Strive) while blade physics,
   758→    // smears, and trails run full-rate on the procedural layers.
   759→    this.animator.play(def.clip, { blend: wasAttack ? 0.11 : 0.07, speed: this.atkSpeed, stepRate: 24 })
   760→    // slight ground friction during attacks
   761→  }

   763→  private updateAttack(dt: number, w: WorldRef): void {
   764→    const a = this.action
   765→    if (!a) {
   766→      this.state = 'move'
   767→      return
   768→    }
   769→    const def = a.def
   770→    a.animT = this.animator.getTime()
   771→    const animT = a.animT
   772→    const spd = this.atkSpeed
   773→
   774→    // ground friction / air drag during attack
   775→    if (this.grounded) {
   776→      const f = 900 * dt
   777→      this.vx = Math.abs(this.vx) <= f ? 0 : this.vx - Math.sign(this.vx) * f
   778→    }
   779→
   780→    // sfx event
   781→    if (!a.sfxDone && animT >= def.swingSfx.t) {
   782→      a.sfxDone = true
   783→      w.audio.play(def.swingSfx.name, { pitch: rand(0.94, 1.08) })
   784→    }
   785→    // iai SNAP-DRAW flash release: the blade clears the saya in one frame —
   786→    // a localized light event + a 2-frame stretch on the steel sells it.
   787→    if (!a.flashDone && animT >= def.trail.start) {
   788→      a.flashDone = true
   789→      const c = this.sword.colors
   790→      w.fx.burst(this.blade.tx, this.blade.ty, 7, c.core, 300, 0.3, 'spark')
   791→      w.fx.shockwave(this.blade.tx, this.blade.ty, 46, c.glow, 0.18, 3)
   792→      w.cam.addTrauma(0.05)
   793→      this.swingScale = Math.max(this.swingScale, 1.3) // velocity stretch
   794→    }
   795→    // forward impulse
   796→    if (!a.moveDone && def.moveX !== undefined && animT >= (def.moveXAt ?? def.hit.start)) {
   797→      a.moveDone = true
   798→      this.vx = this.face * def.moveX * (def.kind === 'dash' ? 1 : 1)
   799→    }
   800→    // vertical impulse (launcher hop / slam plunge)
   801→    if (!a.moveYDone && def.moveY !== undefined && animT >= (def.moveYAt ?? 0)) {
   802→      a.moveYDone = true
   803→      this.vy = def.moveY
   804→      if (def.moveY < 0) {
   805→        this.grounded = false
   806→        w.fx.dust(this.x, 0, 6)
   807→      }
   808→    }
   809→
   810→    // hit window: sample blade segment & test enemies (swept, anti-tunneling)
   811→    const h = def.hit
   812→    if (animT >= h.start && animT <= h.end) {
   813→      this.swingGlow = 1
   814→      const info = makeRuntimeInfo(def, this.sword.stats.dmg)
   815→      // DASH THROUGH-CUT: the iai pass cuts everything the dash swept
   816→      // through — the player crosses the target at 980px/s BEFORE the
   817→      // attack anim even starts, so the blade-only segment points past
   818→      // them. The corridor is the DASH PATH itself: a fixed chest-height
   819→      // band from 124px behind the player to 66px ahead — anchored to the
   820→      // path (not the wandering kissaki, which wraps behind on the
   821→      // follow-through and would collapse the span). This is the swept
   822→      // volume of the through-cut.
   823→      let curSeg = this.blade
   824→      let usePrev = this.hasPrevBlade
   825→      if (def.kind === 'dash') {
   826→        curSeg = {
   827→          bx: this.x - this.face * 124,
   828→          by: this.y - 55,
   829→          tx: this.x + this.face * 66,
   830→          ty: this.y - 55,
   831→        }
   832→        usePrev = false
   833→      }
   834→      for (const e of w.enemies) {
   835→        // spawning enemies within blade reach (landing pods, y > -90) are
   836→        // fair game — the swing connects the moment the body crosses the arc
   837→        if (e.dead || (e.spawning && e.y < -90)) continue
   838→        const last = a.hitMap.get(e.id)
   839→        const reArm = h.multiHit ? (last === undefined || animT - last >= h.multiHit) : last === undefined
   840→        if (!reArm) continue
   841→        if (bladeHitsBody(e, usePrev, this.prevBlade, curSeg, h.pad ?? 3)) {
   842→          a.hitMap.set(e.id, animT)
   843→          w.hitEnemy(e.id, this.blade, info)
   844→          this.onLandedHit(w, info, e.x, e.y - 48 * e.scale)
   845→        }
   846→      }
   847→      // remember this frame's blade for the next sweep test
   848→      this.prevBlade.bx = this.blade.bx
   849→      this.prevBlade.by = this.blade.by
   850→      this.prevBlade.tx = this.blade.tx
   851→      this.prevBlade.ty = this.blade.ty
   852→      this.hasPrevBlade = true
   853→      // aoe shockwave at strike moment for heavy hits — the wave is REAL:
   854→      // enemies inside the radius eat the strike (the charged/h3 promise:
   855→      // the shockwave visual must connect, not just decorate)
   856→      if (h.aoe && h.aoe > 0 && !a.impactDone && this.grounded && animT >= h.start + 0.02) {
   857→        a.impactDone = true
   858→        w.fx.shockwave(this.x + this.face * 30, 0, h.aoe, this.sword.colors.glow, 0.4, 6)
   859→        if (this.grounded) w.fx.groundCrack(this.x + this.face * 34, 0, h.aoe * 0.45)
   860→        for (const e of w.enemies) {
   861→          if (e.dead || e.spawning) continue
   862→          if (a.hitMap.has(e.id) && !h.multiHit) continue
   863→          if (Math.abs(e.x - this.x) < h.aoe && e.y > -140) {
   864→            a.hitMap.set(e.id, animT)
   865→            w.hitEnemy(e.id, this.blade, info)
   866→            this.onLandedHit(w, info, e.x, e.y - 48 * e.scale)
   867→          }
   868→        }
   869→      }
   870→    }
   871→
   872→    // OVERSHOOT (follow-through law): as the active window closes, inject a
   873→    // 2-frame rotational impulse — the tip pushes PAST the stop point and
   874→    // the wrist spring elastic-snaps it back. Stopping dead reads weightless.
   875→    if (!a.overshootDone && animT > h.end && this.armVel !== 0) {
   876→      a.overshootDone = true
   877→      this.bladeLagVel += Math.sign(this.armVel) * 2.4
   878→    }
   879→
   880→    // trail emission
   881→    if (animT >= def.trail.start && animT <= def.trail.end) {
   882→      w.fx.pushTrail(a.trail, this.blade.bx, this.blade.by, this.blade.tx, this.blade.ty)
   883→    } else if (animT > def.trail.end && a.trail.emitting) {
   884→      w.fx.endTrail(a.trail)
   885→    }
   886→
   887→    // cancel window → allow chaining / dash-cancel / jump-cancel
   888→    const dashCancel = animT >= h.end * 0.6
   889→    if (animT >= def.cancel || dashCancel) {
   890→      const input = w.input
   891→      if ((input.pressed('dash') || input.buffered('dash')) && this.canDash()) {
   892→        w.fx.endTrail(a.trail)
   893→        this.startDash(w, input.axisX(), input.axisY())
   894→        return
   895→      }
   896→      if (input.pressed('parry')) {
   897→        w.fx.endTrail(a.trail)
   898→        this.enterParry(w)
   899→        return
   900→      }
   901→      if (animT >= def.cancel) {
   902→        // chain next attack
   903→        const ax = input.axisX()
   904→        const ay = input.axisY()
   905→        let next: string | null = null
   906→        if (input.buffered('light')) {
   907→          input.takeBuffered('light')
   908→          next = this.riposteT > 0 && this.grounded ? 'riposte' : this.pickLight(w, ax, ay)
   909→        } else if (input.buffered('heavy')) {
   910→          input.takeBuffered('heavy')
   911→          if (!this.grounded) {
   912→            next = 'slam'
   913→          } else if (input.down('heavy')) {
   914→            // still holding → convert to charge
   915→            this.enterCharge(w)
   916→            w.fx.endTrail(a.trail)
   917→            return
   918→          } else {
   919→            // advance the heavy CHAIN (h1 → h2 → h3) like the light chain
   920→            const heavyNext = def.chain.find((c) => /^h[123]$/.test(c))
   921→            next = heavyNext ?? (def.kind === 'heavy' ? 'l1' : 'h1')
   922→          }
   923→        }
   924→        if (next) {
   925→          // validate chain (or allow from-parry riposte)
   926→          if (fromChain(def, next, this)) {
   927→            w.fx.endTrail(a.trail)
   928→            this.startAttack(ATTACKS[next], w)
   929→            return
   930→          }
   931→        }
   932→        // jump cancel on ground
   933→        if (this.grounded && (input.pressed('jump') || input.buffered('jump', 0.15))) {
   934→          w.fx.endTrail(a.trail)
   935→          this.doJump(w)
   936→          this.state = 'move'
   937→          return
   938→        }
   939→      }
   940→    }
   941→
   942→    // finish
   943→    if (this.animator.isDone()) {
   944→      w.fx.endTrail(a.trail, 0.18)
   945→      this.action = null
   946→      this.state = 'move'
   947→    }
   948→  }

   950→  private onLandedHit(w: WorldRef, info: AttackRuntimeInfo, ex: number, ey: number): void {
   951→    this.energy = Math.min(this.maxEnergy, this.energy + 5)
   952→    this.meter = Math.min(100, this.meter + info.dmg * 0.32)
   953→    // ANIME IMPACT FRAME (research: 1-2 frame white beat at contact): heavy
   954→    // connects flash the whole silhouette white through the hit-stop — the
   955→    // classic anime cut-in. Lighter hits keep the spark burst only.
   956→    if (info.hitstop >= 0.07 && this.impactFlashT <= 0) {
   957→      this.impactFlashT = 0.085
   958→    }
   959→    // elemental on-hit flourishes
   960→    const c = this.sword.colors
   961→    const dir = Math.atan2(ey - this.y + 40, ex - this.x)
   962→    switch (this.sword.id) {
   963→      case 'inferno':
   964→        w.fx.embers(ex, ey, 5, c.accent, 18)
   965→        break
   966→      case 'thunder':
   967→        if (Math.random() < 0.3) w.fx.lightningArc(ex, ey, ex + rand(-40, 40), ey + rand(-30, 30), c.trail)
   968→        break
   969→      case 'void':
   970→        w.fx.burst(ex, ey, 4, c.core, 120, 0.4, 'shard')
   971→        break
   972→      case 'gravity':
   973→        w.fx.burst(ex, ey, 3, c.accent, 60, 0.5, 'ember')
   974→        break
   975→      case 'phantom':
   976→        w.fx.slashArc(ex, ey, dir + rand(-0.5, 0.5), 1.4, 30, c.trail, c.trail2)
   977→        break
   978→      default:
   979→        break
   980→    }
   981→  }

  1000→  private updateParry(dt: number, w: WorldRef): void {

  1431→  /** FOOT-GROUND LOCK — the hip sits at the planted foot's extension. This
  1432→   * is the weight system: idle bent knees no longer float (measured 9.5px
  1433→   * gap), lunges/crouches stay exactly grounded, landing squashes through
  1434→   * the leg bend instead of sinking into the floor. Airborne → nominal
  1435→   * hip height (physics owns y). */
  1436→  private updateFootLock(dt: number): void {
  1437→    const g = this.geom
  1438→    // lowest contact point: ankles AND toes (a pointed trailing toe grazes
  1439→    // the ground at toe-off — real gait), plus the sole half-width so the
  1440→    // visual foot bottom kisses the line, not the ankle joint center
  1441→    const lowContact = Math.max(g.ankleL.y, g.ankleR.y, g.toeL.y, g.toeR.y) + 2.4
  1442→    const support = lowContact * SCALE
  1443→    const groundedPose = this.grounded && this.state !== 'dead'
  1444→    let target: number
  1445→    if (groundedPose && support >= HIP_H * 0.86) {
  1446→      // SUPPORT phase — the lowest foot is load-bearing (within 14% of full
  1447→      // leg extension): pin the hip so the sole EXACTLY contacts. This is
  1448→      // the weight: no float, no dig, crouches stay crouched.
  1449→      target = support
  1450→    } else {
  1451→      // FLIGHT (both feet swinging/tucked mid-stride) or airborne: physics
  1452→      // owns height — honor the clip's authored bob instead of dragging the
  1453→      // hip down to a tucked toe (which inverted the run's natural bounce)
  1454→      target = HIP_H + this.pose[1] * SCALE
  1455→    }
  1456→    // k=24: tracks a 0.3s stride with ~2px residual — crisp, never mushy
  1457→    this.footLock = damp(this.footLock, target, 24, dt)

  1460→  /** BLADE DYNAMICS — rotational inertia of the katana as a spring-damper
  1461→   * on the wrist channel. When the arm accelerates (windup → cut), the tip
  1462→   * lags behind the drive angle; when the arm decelerates, the stored
  1463→   * energy whips the tip THROUGH the target line with a small overshoot —
  1464→   * the physical furi-otoshi snap real swordsmen feel. Under-damped
  1465→   * (ζ≈0.3): lively, never mushy. Applied BEFORE solve → hitboxes inherit
  1466→   * the physical blade, not just the keyframed one.
  1467→   *
  1468→   * During hit-stop (world timeScale ≈ 0.05) the blade takes a ±5° impact
  1469→   * resistance deflection AGAINST its swing direction — the cut bites into
  1470→   * something with mass (spec: kinetic impact modulation). */
  1471→  private updateBladeDynamics(dt: number, w: WorldRef): void {
  1472→    if (dt <= 1e-5) return
  1473→    const p = this.pose
  1474→    const armSum = p[CH.shR] + p[CH.elR] + p[CH.wrR]
  1475→    const armVel = (armSum - this.prevArmSum) / dt
  1476→    this.armVel = armVel
  1477→    this.prevArmSum = armSum
  1478→    // lag target: proportional to angular velocity (inertia resisting)
  1479→    const target = w.timeScale < 0.5
  1480→      ? clamp(-Math.sign(armVel || this.bladeLag || 1) * 0.09, -0.15, 0.15) // impact resistance ~5°
  1481→      : clamp(-armVel * 0.045, -0.15, 0.15)
  1482→    // spring toward target, under-damped for the whip-through
  1483→    const K = 260
  1484→    const C = 9
  1485→    this.bladeLagVel += (K * (target - this.bladeLag) - C * this.bladeLagVel) * dt
  1486→    this.bladeLag += this.bladeLagVel * dt
  1487→    this.bladeLag = clamp(this.bladeLag, -0.17, 0.17)
  1488→    this.bladeLagVel = clamp(this.bladeLagVel, -6, 6)
  1489→    p[CH.wrR] += this.bladeLag
  1490→
  1491→    // KINETIC CHAIN (hips 30% · torso 50% · shoulders 20%): a slower spring
  1492→    // driven by the same arm whip. Accelerating into a cut → the torso loads
  1493→    // INTO the swing; the deceleration at impact → the body recoils back.
  1494→    // The upper body can never again rotate independently of the hips.
  1495→    const k2 = 90
  1496→    const c2 = 7.5
  1497→    const chainTarget = clamp(-armVel * 0.02, -0.14, 0.14)
  1498→    this.chainLagVel += (k2 * (chainTarget - this.chainLag) - c2 * this.chainLagVel) * dt
  1499→    this.chainLag = clamp(this.chainLag + this.chainLagVel * dt, -0.16, 0.16)
  1500→    p[CH.rootRot] += -this.chainLag * 0.3
  1501→    p[CH.spine] += -this.chainLag * 0.5
  1502→    p[CH.shR] += this.chainLag * 0.2
  1503→  }

  1505→  private nearestEnemy(w: WorldRef, range: number): { x: number; y: number } | null {

  1551→  /** TWO-HANDED GRIP — the off hand solves 2-bone IK onto the tsuka butt

  1553→   * katana is actually swung). Blended over the authored off-arm channels
  1554→   * with a state-driven weight so running/dashing pumps stay authored. */
  1555→  private applyTwoHandGrip(dt: number): void {
  1556→    // desired grip weight by state
  1557→    let target = 0
  1558→    switch (this.state) {
  1559→      case 'move':
  1560→        target = this.grounded && Math.abs(this.vx) < 60 ? 1 : 0.1
  1561→        break
  1562→      case 'attack':
  1563→        target = this.action ? (this.action.def.grip ?? 0.45) : 0.45
  1564→        break
  1565→      case 'charge':
  1566→      case 'parry':
  1567→        target = 1
  1568→        break
  1569→      case 'scripted':
  1570→        target = 0.5
  1571→        break
  1572→      default:
  1573→        target = 0
  1574→    }
  1575→    this.twoHandW = damp(this.twoHandW, target, 11, dt)
  1576→    const w = this.twoHandW
  1577→    if (w < 0.03) return
  1578→
  1579→    const g = this.geom // last frame's solve — continuous, 1-frame stale
  1580→    const tsuka = PLAYER_SPEC.blade * this.sword.stats.range * 0.3
  1581→    // tsuka butt (pommel end) in local space
  1582→    const bax = Math.cos(g.bladeAngle)
  1583→    const bay = Math.sin(g.bladeAngle)
  1584→    const tx = g.handR.x - bax * tsuka
  1585→    const ty = g.handR.y - bay * tsuka

  1587→    const sx = g.chest.x
  1588→    const sy = g.chest.y
  1589→    const U = PLAYER_SPEC.upperArm
  1590→    const F = PLAYER_SPEC.foreArm
  1591→    let dx = tx - sx
  1592→    let dy = ty - sy
  1593→    let d = Math.hypot(dx, dy)
  1594→    const maxD = (U + F) * 0.98
  1595→    if (d > maxD) {
  1596→      const k = maxD / d
  1597→      dx *= k
  1598→      dy *= k
  1599→      d = maxD
  1600→    }
  1601→    if (d < 2) return
  1602→    // law of cosines: interior angles of the shoulder-elbow-target triangle
  1603→    const cosB = clamp((U * U + d * d - F * F) / (2 * U * d), -1, 1)
  1604→    const B = Math.acos(cosB)
  1605→    const base = Math.atan2(dy, dx)
  1606→    // elbow bends DOWN (anatomical) → rotate the upper arm below the line
  1607→    const upperAng = base + B
  1608→    const ex = sx + Math.cos(upperAng) * U
  1609→    const ey = sy + Math.sin(upperAng) * U
  1610→    // convert elbow/hand to the channel convention (from straight-down, +fwd)
  1611→    const shIk = Math.atan2(ex - sx, ey - sy)
  1612→    const elIk = Math.atan2(tx - ex, ty - ey) - shIk
  1613→    const p = this.pose
  1614→    p[CH.shL] += (shIk - p[CH.shL]) * w
  1615→    p[CH.elL] += (elIk - p[CH.elL]) * w
  1616→  }

  1618→  private applyProcedural(dt: number): void {

  1620→    this.animClock += dt

  1624→    const speedLean = clamp((this.vx * this.face) / MOVE_MAX, -1, 1)

  1628→      p[2] += clamp(this.vy / 1400, -0.22, 0.3)

  1631→    if (this.landSpring > 0) {
  1632→      this.landSpring = Math.max(0, this.landSpring - dt * 5)
  1633→      p[1] += this.landSpring * 6

  1648→    if (this.state === 'move' && this.grounded && Math.abs(this.vx) < 40 && this.hurtT <= 0) {
