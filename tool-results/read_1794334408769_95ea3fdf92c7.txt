     1→/** STICKMAN RISING — the modular combo moveset, re-authored around REAL
     2→ * swordsmanship (kenjutsu/iai techniques — see the per-attack notes).
     3→ *
     4→ * Design law for this revision:
     5→ *  1. KAMAE CONTINUITY — every clip starts AND ends in chūdan-no-kamae (the
     6→ *     middle guard). Chained attacks blend guard→chamber→cut→guard; the
     7→ *     shared pose base means no snap between animation instances.
     8→ *  2. HASUJI (blade alignment) — blade dir = π/2 − (shR+elR+wrR) in canvas
     9→ *     space (+Y down), so: sum 1.57 → level forward · sum ≈ 3.0 → blade UP
    10→ *     (jōdan) · sum ≈ 0.3 → blade DOWN (follow-through) · sum < 0 → blade
    11→ *     back (waki / sheathed). Cuts sweep through the plane they name:
    12→ *     kesa descends diagonally, gyaku-kesa rises, yoko stays level, kiri
    13→ *     oroshi falls vertically, tsuki spears flat. NO 360° windmills.
    14→ *  3. TEMPO + IN-BETWEENS — every fast span (>1.4 rad) carries an explicit
    15→ *     intermediate "ignition" keyframe, so each frame is a real authored
    16→ *     in-between pose: chamber → ignition → strike → follow → zanshin.
    17→ *  4. The blade is drawn from the SAYA (scabbard at the hip) for iai cuts
    18→ *     (l1, riposte): hand at the hip, blade lying back along the scabbard.
    19→ */

    48→export interface AttackDef {
    49→  /** Empty-hand contact uses the visible forearm/fist, never a hidden blade. */
    50→  strikeHand?: 'left' | 'right'
    51→  strikeJoint?:'hand'|'elbow'
    52→  id: string
    53→  label: string
    54→  kind: AttackKind
    55→  clip: Clip
    56→  hit: AttackHit
    57→  trail: { start: number; end: number }
    58→  swingSfx: { t: number; name: SfxName }
    59→  cancel: number
    60→  chain: string[]
    61→  moveX?: number
    62→  moveXAt?: number
    63→  /** Opt-in finite ground drive. Legacy moves remain unchanged until qualified. */
    64→  groundDrive?: {duration:number;acceleration:number}
    65→  moveY?: number
    66→  moveYAt?: number
    67→  requires: 'ground' | 'air' | 'any'
    68→  /** gravity scale while the attack is active (air-combo float) */
    69→  float?: number
    70→  /** AoE + fx fire when the player touches ground mid-attack (slams / dive) */
    71→  slamImpact?: { r: number; dmg: number; launch: number }
    72→  /** two-handed grip weight (0..1): how strongly the off-hand IK takes the
    73→   * tsuka during this attack — real kenjutsu cuts are two-handed */
    74→  grip?: number
    75→  /** 2.8D Z-EXCURSION (0..1): legacy visual-only depth pulse — superseded
    76→   * by `traj` (the real 3D yaw path); still drives the pierce flare for
    77→   * attacks without a declared trajectory */
    78→  zArc?: number
    79→  /** iai snap-draw: spawn the flash-release burst at ignition (blade
    80→   * clearing the saya reads as a 1-frame light event) */
    81→  drawFlash?: boolean
    82→  /** THE 3D TRAJECTORY: the blade's path through real 3D space during the
    83→   * strike. Every attack family gets a genuinely different combination —
    84→   * horizontal cuts carve wide depth arcs, diagonals tilt through depth,
    85→   * thrusts spear the lens, risers reverse the plane. The body follows:
    86→   * hips rotate with the yaw, the chest counter-rotates further. */
    87→  traj?: Traj3D
    88→}
    89→
    90→/** Blade path spec (see armarc.ts — the 3D trajectory solver):
    91→ *  a0→a1  pitch: screen-plane blade angle, chamber → contact
    92→ *         (0 = level forward, + = tip down, − = tip up)
    93→ *  y0→y1  yaw: depth-plane rotation, chamber → contact
    94→ *         (− = kissaki behind the body plane, + = toward the camera)
    95→ *  e0→e1  arm extension fraction (coil → drive through)
    96→ *  band   strike height (world px above feet: 14 legs / 48 torso / 78 chest) */
    97→export interface Traj3D {
    98→  a0: number
    99→  a1: number
   100→  y0: number
   101→  y1: number
   102→  e0: number
   103→  e1: number
   104→  band?: number
   105→}
