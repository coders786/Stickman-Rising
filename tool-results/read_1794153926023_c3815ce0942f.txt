    26→    26→/** swaps left/right LEG channels only — the arms keep their authored
    27→    27→ * channels (the sword stays in the right hand across both strides). */
    28→    28→function mirrorLegs(src: PoseInput): PoseInput {
    29→    29→  const out: PoseInput = { ...src }
    30→    30→  const pairs: [keyof PoseInput, keyof PoseInput][] = [
    31→    31→    ['hipL', 'hipR'], ['kneeL', 'kneeR'], ['footL', 'footR'],
    32→    32→  ]
    33→    33→  for (const [l, r] of pairs) {
    34→    34→    const lv = src[l]
    35→    35→    const rv = src[r]
    36→    36→    if (lv !== undefined) out[r] = lv
    37→    37→    if (rv !== undefined) out[l] = rv
    38→    38→  }
    39→    39→  return out
    40→    40→}
    41→    41→
    42→    42→/** STRIDE MIRROR for bipedal cycles: legs swap, the SWORD ARM (right)
    43→    43→ * keeps the exact same carry across both strides (a held weapon does not
    44→    44→ * teleport between arms / change carry mid-cycle), and the off-arm
    45→    45→ * negates its swing so the contralateral counter-phase survives.
    46→    46→ * DEPTH channels mirror with the stride: pelvis/chest yaw negate (the
    47→    47→ * opposite hip now drives), the off-arm's depth swing negates — while the
    48→    48→ * sword-side channels (armRz, bladeYaw) carry unchanged, like the arm. */
    49→    49→function mirrorStride(src: PoseInput): PoseInput {
    50→    50→  const out = mirrorLegs(src)
    51→    51→  if (src.shL !== undefined) out.shL = -src.shL
    52→    52→  if (src.pelvisYaw !== undefined) out.pelvisYaw = -src.pelvisYaw
    53→    53→  if (src.chestYaw !== undefined) out.chestYaw = -src.chestYaw
    54→    54→  if (src.armLz !== undefined) out.armLz = -src.armLz
    55→    55→  if (src.armRz !== undefined) out.armRz = src.armRz
    56→    56→  if (src.bladeYaw !== undefined) out.bladeYaw = src.bladeYaw
    57→    57→  return out
    58→    58→}
