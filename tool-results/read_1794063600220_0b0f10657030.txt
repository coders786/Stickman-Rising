    15→/** circle vs swept blade quad (prev segment + current segment) — prevents tunneling */
    16→function circleHitsSweptQuad(
    17→  cx: number, cy: number, r: number,
    18→  pbx: number, pby: number, ptx: number, pty: number,
    19→  bx: number, by: number, tx: number, ty: number,
    20→): boolean {
    21→  // current + previous blade segments
    22→  if (segHitsCircle(bx, by, tx, ty, cx, cy, r)) return true
    23→  if (segHitsCircle(pbx, pby, ptx, pty, cx, cy, r)) return true
    24→  // swept edges (tip path + base path)
    25→  if (segHitsCircle(ptx, pty, tx, ty, cx, cy, r)) return true
    26→  if (segHitsCircle(pbx, pby, bx, by, cx, cy, r)) return true
    27→  // circle center inside the swept quad (spins / wide arcs)
    28→  if (pointInTri(cx, cy, pbx, pby, ptx, pty, tx, ty)) return true
    29→  if (pointInTri(cx, cy, pbx, pby, tx, ty, bx, by)) return true
    30→  return false
    31→}

    33→function pointInTri(px: number, py: number, x1: number, y1: number, x2: number, y2: number, x3: number, y3: number): boolean {
    34→  const d1 = (px - x2) * (y1 - y2) - (x1 - x2) * (py - y2)
    35→  const d2 = (px - x3) * (y2 - y3) - (x2 - x3) * (py - y3)
    36→  const d3 = (px - x1) * (y3 - y1) - (x3 - x1) * (py - y1)
    37→  const hasNeg = d1 < 0 || d2 < 0 || d3 < 0
    38→  const hasPos = d1 > 0 || d2 > 0 || d3 > 0
    39→  return !(hasNeg && hasPos)
    40→}

   206→  update(dt: number, w: WorldRef): void {
   207→    this.iframes = Math.max(0, this.iframes - dt)
   208→    this.flashT = Math.max(0, this.flashT - dt * 5)
   209→    this.swingGlow = Math.max(0, this.swingGlow - dt * 5)
   210→    this.riposteT = Math.max(0, this.riposteT - dt)
   211→    this.energy = Math.min(this.maxEnergy, this.energy + dt * 3)
   212→
   213→    switch (this.state) {
   214→      case 'move': this.updateMove(dt, w); break
   215→      case 'attack': this.updateAttack(dt, w); break
   216→      case 'dash': this.updateDash(dt, w); break
   217→      case 'parry': this.updateParry(dt, w); break
   218→      case 'charge': this.updateCharge(dt, w); break
   219→      case 'hurt': this.updateHurt(dt, w); break
   220→      case 'dead': this.updateDead(dt, w); break
   221→      case 'scripted': this.updateScripted(dt); break
   222→    }
   223→
   224→    // physics integration (shared)
   225→    if (this.state !== 'dash') {
   226→      const g = this.state === 'attack' && this.action && this.action.def.float
   227→        ? GRAVITY * this.action.def.float
   228→        : GRAVITY
   229→      this.vy += g * dt
   230→    }
   231→    this.x += this.vx * dt
   232→    this.y += this.vy * dt
   233→
   234→    // ground
   235→    const wasAir = !this.grounded
   236→    if (this.y >= 0) {
   237→      this.y = 0
   238→      if (this.vy > 140) {
   239→        this.landSpring = clamp(this.vy / 700, 0, 1)
   240→        if (wasAir) this.onLand(w)
   241→      } else if (wasAir) {
   242→        this.landSpring = 0.4
   243→        this.onLand(w)
   244→      }
   245→      this.vy = 0
   246→      this.grounded = true
   247→      this.jumpsUsed = 0
   248→      this.airDashUsed = false
   249→    } else {
   250→      this.grounded = false
   251→    }
   252→    // arena bounds
   253→    if (this.x < w.arenaMinX + 20) {
   254→      this.x = w.arenaMinX + 20
   255→      if (this.vx < -200) w.fx.dust(this.x - 12, 0, 5)
   256→      this.vx = Math.max(0, this.vx)
   257→    }
   258→    if (this.x > w.arenaMaxX - 20) {
   259→      this.x = w.arenaMaxX - 20
   260→      if (this.vx > 200) w.fx.dust(this.x + 12, 0, 5)
   261→      this.vx = Math.min(0, this.vx)
   262→    }
   263→
   264→    this.dashCd = Math.max(0, this.dashCd - dt)
   265→    this.dashRecharge = Math.max(0, this.dashRecharge - dt)
   266→    if (this.dashRecharge <= 0) this.dashConsec = 0
   267→
   268→    // ability script
   269→    if (this.ability) this.updateAbility(dt, w)
   270→
   271→    // animation + solve
   272→    this.animator.update(dt)
   273→    // Base layer: the clip FULLY owns the pose each frame (write, never
   274→    // accumulate). applyProcedural() is a bounded additive layer on top.
   275→    // Without this copy, procedural += deltas integrate unbounded → spine
   276→    // spin + root sink. (Same contract as Spine's AnimationState.apply /
   277→    // Godot AnimationTree → Skeleton2D.)
   278→    this.pose.set(this.animator.pose)
   279→    this.applyProcedural(dt)
   280→    solveSkeleton(this.pose, PLAYER_SPEC, this.geom)
   281→    this.updateBladeWorld()
   282→  }

   284→  private onLand(w: WorldRef): void {
   285→    w.audio.play('land', { volume: 0.5 })
   286→    w.fx.dust(this.x, 0, 7)
   287→    // slam impact while attacking
   288→    const a = this.action
   289→    if (a && a.def.slamImpact && !a.impactDone) {
   290→      a.impactDone = true
   291→      this.slamAoe(w, a.def.slamImpact.r, a.def.slamImpact.dmg, a.def.slamImpact.launch)
   292→    }
   293→    if (this.state === 'move' && this.lastGrounded === false) {
   294→      this.animator.play(LAND_CLIP, { blend: 0.03 })
   295→    }
   296→    this.airIndex = 0
   297→  }

   299→  private slamAoe(w: WorldRef, r: number, dmg: number, launch: number): void {
   300→    const info: AttackRuntimeInfo = {
   301→      id: 'slam', dmg: dmg * this.sword.stats.dmg, heavy: true,
   302→      stagger: 0.5, kb: 240, kbY: 0, launch, juggle: 0,
   303→      hitstop: 0.11, shake: 0.45, aoe: r, breakGuard: true, kind: 'heavy',
   304→    }
   305→    const colors = this.sword.colors
   306→    w.fx.shockwave(this.x, 0, r, colors.glow, 0.5, 8)
   307→    w.fx.groundCrack(this.x, 0, r * 0.6)
   308→    w.fx.burst(this.x, -6, 14, colors.glow, 340, 0.5, 'spark')
   309→    w.fx.screenFlash(colors.trail, 0.16, 5)
   310→    w.addHitstop(0.09, 0.06)
   311→    w.cam.addTrauma(0.4)
   312→    w.audio.play('slam', { intensity: 1 })
   313→    const bx = this.x
   314→    for (const e of w.enemies) {
   315→      if (e.dead || e.spawning) continue
   316→      const dx = e.x - bx
   317→      if (Math.abs(dx) < r && e.y > -140) {
   318→        w.hitEnemy(e.id, { bx: bx - 20, by: -20, tx: bx + 20, ty: -20 }, info)
   319→      }
   320→    }
   321→  }

   327→  private updateMove(dt: number, w: WorldRef): void {
   328→    const input = w.input
   329→    const ax = input.axisX()
   330→    const ay = input.axisY()
   331→
   332→    // facing follows input
   333→    if (ax !== 0) this.face = ax > 0 ? 1 : -1
   334→
   335→    // horizontal movement
   336→    if (ax !== 0) {
   337→      const accel = this.grounded ? ACCEL : AIR_ACCEL
   338→      this.vx += ax * accel * dt
   339→      this.vx = clamp(this.vx, -MOVE_MAX, MOVE_MAX)
   340→    } else if (this.grounded) {
   341→      const f = FRICTION * dt
   342→      this.vx = Math.abs(this.vx) <= f ? 0 : this.vx - Math.sign(this.vx) * f
   343→    } else {
   344→      this.vx = damp(this.vx, 0, 0.4, dt)
   345→    }
   346→
   347→    // locomotion clip selection
   348→    const speed = Math.abs(this.vx)
   349→    if (this.grounded) {
   350→      if (speed > 40) {
   351→        this.animator.play(RUN_CLIP, { blend: 0.09, speed: clamp(speed / MOVE_MAX, 0.65, 2.1) })
   352→      } else if (this.vy === 0) {
   353→        this.animator.play(IDLE_CLIP, { blend: 0.16 })
   354→      }
   355→    } else if (this.vy < -60) {
   356→      this.animator.play(JUMP_CLIP, { blend: 0.06 })
   357→    } else {
   358→      this.animator.play(FALL_CLIP, { blend: 0.12 })
   359→    }
   360→
   361→    // ---- actions ----
   362→    if (this.tryCombatInputs(w, ax, ay)) return
   363→  }

   365→  /** returns true if a new state was entered */
   366→  private tryCombatInputs(w: WorldRef, ax: number, ay: number): boolean {
   367→    const input = w.input
   368→
   369→    // dash (anytime, from move)
   370→    if ((input.pressed('dash') || input.buffered('dash')) && this.canDash()) {
   371→      this.startDash(w, ax, ay)
   372→      return true
   373→    }
   374→    // jump (grounded = jump 1, plus one air jump)
   375→    if (input.pressed('jump') || input.buffered('jump', 0.18)) {
   376→      if (this.jumpsUsed < 2) {
   377→        this.doJump(w)
   378→        return false
   379→      }
   380→    }
   381→    // parry
   382→    if (input.pressed('parry')) {
   383→      this.enterParry(w)
   384→      return true
   385→    }
   386→    // ability
   387→    if (input.pressed('ability')) {
   388→      if (this.energy >= this.sword.ability.cost) {
   389→        this.startAbility(w)
   390→        return true
   391→      }
   392→    }
   393→    // ult is handled by Game (cinematic)
   394→
   395→    // heavy: charge vs tap
   396→    if (input.down('heavy') && this.grounded) {
   397→      this.enterCharge(w)
   398→      return true
   399→    }
   400→    if (input.pressed('heavy') || input.buffered('heavy')) {
   401→      input.takeBuffered('heavy')
   402→      this.startAttack(this.pickHeavy(w), w)
   403→      return true
   404→    }
   405→    // light
   406→    if (input.pressed('light') || input.buffered('light')) {
   407→      input.takeBuffered('light')
   408→      const id = this.riposteT > 0 ? 'riposte' : this.pickLight(w, ax, ay)
   409→      this.startAttack(ATTACKS[id], w)
   410→      return true
   411→    }
   412→    return false
   413→  }

   415→  private doJump(w: WorldRef): void {
   416→    w.input.takeBuffered('jump')
   417→    this.jumpsUsed++
   418→    this.vy = this.jumpsUsed === 1 ? JUMP_V : AIR_JUMP_V
   419→    this.grounded = false
   420→    w.audio.play('jump', { volume: 0.4 })
   421→    w.fx.dust(this.x, 0, 4)
   422→    if (this.jumpsUsed === 2) {
   423→      w.fx.burst(this.x, this.y - 20, 6, '#9fb4c4', 160, 0.4, 'dot')
   424→    }
   425→  }

   431→  private canDash(): boolean {
   432→    if (this.dashCd > 0 || this.dashRecharge > 0) return false
   433→    if (!this.grounded && this.airDashUsed) return false
   434→    return true
   435→  }
   436→
   437→  startDash(w: WorldRef, ax: number, ay: number): void {
   438→    w.input.takeBuffered('dash')
   439→    let dx = ax
   440→    let dy = ay
   441→    if (dx === 0 && dy === 0) dx = this.face
   442→    const len = Math.hypot(dx, dy) || 1
   443→    dx /= len
   444→    dy /= len
   445→    if (dx !== 0) this.face = dx > 0 ? 1 : -1
   446→    this.dashDx = dx
   447→    this.dashDy = dy
   448→    this.state = 'dash'
   449→    this.dashT = DASH_TIME
   450→    this.dashCd = DASH_CD
   451→    this.dashConsec++
   452→    if (this.dashConsec >= 2) this.dashRecharge = DASH_RECHARGE
   453→    if (!this.grounded) this.airDashUsed = true
   454→    this.iframes = Math.max(this.iframes, 0.12)
   455→    this.action = null
   456→    this.charging = false
   457→    this.vx = dx * DASH_SPEED
   458→    this.vy = dy * DASH_SPEED * (this.grounded && dy < -0.4 ? 0.55 : 1)
   459→    this.animator.play(this.grounded ? DASH_CLIP : DASH_AIR_CLIP, { blend: 0.03 })
   460→    w.audio.play('dash', { pitch: rand(0.95, 1.1) })
   461→    const colors = this.sword.colors
   462→    w.fx.speedStreaks(this.x, this.y - 40, dx * DASH_SPEED * 0.9, 6, rgba(colors.trail, 1))
   463→    w.fx.dust(this.x, this.grounded ? 0 : this.y, 5)
   464→    w.cam.addTrauma(0.06)
   465→  }
   466→
   467→  private updateDash(dt: number, w: WorldRef): void {
   468→    this.dashT -= dt
   469→    // afterimages
   470→    this.afterimgTimer -= dt
   471→    if (this.afterimgTimer <= 0) {
   472→      this.afterimgTimer = 0.026
   473→      w.fx.afterimage(this.pose, this.x, this.hipY, this.face, this.sword.colors.trail, 0.3, SCALE)
   474→    }
   475→    // maintain velocity with slight decay
   476→    const k = 1 - this.dashT / DASH_TIME
   477→    const sp = DASH_SPEED * (1 - k * 0.25)
   478→    this.vx = this.dashDx * sp
   479→    this.vy = this.dashDy * sp * (this.grounded ? (this.dashDy < 0 ? 0.4 : 1) : 1)
   480→
   481→    // dash attack
   482→    const input = w.input
   483→    if (input.pressed('light') || input.buffered('light')) {
   484→      input.takeBuffered('light')
   485→      this.startAttack(ATTACKS.dashSlash, w)
   486→      return
   487→    }
   488→    if (input.pressed('heavy') || input.buffered('heavy')) {
   489→      if (!this.grounded) {
   490→        input.takeBuffered('heavy')
   491→        this.startAttack(ATTACKS.slam, w)
   492→        return
   493→      }
   494→      input.takeBuffered('heavy')
   495→      this.startAttack(ATTACKS.dashHeavy, w)
   496→      return
   497→    }
   498→    if (this.dashT <= 0) {
   499→      this.state = 'move'
   500→      this.vx *= 0.55
   501→      if (!this.grounded) this.vy *= 0.5
   502→    }
   503→  }

   509→  private pickLight(w: WorldRef, ax: number, ay: number): string {
   510→    // contextual branches
   511→    if (ay < -0.4 && this.grounded) return 'launcher'
   512→    if (ay > 0.4 && this.grounded) return 'spin'
   513→    if (this.grounded && ax !== 0 && Math.sign(ax) !== this.face) return 'retreat'
   514→    if (!this.grounded) {
   515→      if (ay > 0.4) return 'dive'
   516→      this.airIndex = (this.airIndex % 3) + 1
   517→      return `a${this.airIndex}`
   518→    }
   519→    // ground light chain (with time decay)
   520→    if (w.time - this.lastLightAt > 1.1) this.lightIndex = 0
   521→    this.lightIndex = Math.min(this.lightIndex + 1, 5)
   522→    return `l${Math.min(this.lightIndex, 5)}`
   523→  }

   529→  startAttack(def: AttackDef, w: WorldRef, fromParry = false): void {
   530→    if (!def) return
   531→    // requirement checks
   532→    if (def.requires === 'ground' && !this.grounded) def = ATTACKS.a1
   533→    if (def.requires === 'air' && this.grounded) def = ATTACKS.l1
   534→
   535→    // face toward held direction at attack start
   536→    const ax = w.input.axisX()
   537→    if (ax !== 0 && !fromParry) this.face = ax > 0 ? 1 : -1
   538→
   539→    const trail = w.fx.startTrail(this.sword.colors.trail, this.sword.colors.trail2, this.sword.trailWidth)
   540→    this.hasPrevBlade = false
   541→    this.action = {
   542→      def,
   543→      animT: 0,
   544→      hitMap: new Map(),
   545→      trail,
   546→      sfxDone: false,
   547→      moveDone: false,
   548→      moveYDone: false,
   549→      impactDone: false,
   550→      fromParry,
   551→    }
   552→    if (fromParry) this.riposteT = 0
   553→    this.state = 'attack'
   554→    this.charging = false
   555→    if (def === ATTACKS.l1 || def === ATTACKS.l2 || def === ATTACKS.l3 || def === ATTACKS.l4 || def === ATTACKS.l5) {
   556→      this.lastLightAt = w.time
   557→    }
   558→    this.animator.play(def.clip, { blend: 0.045, speed: this.atkSpeed })
   559→    // slight ground friction during attacks
   560→  }
   561→
   562→  private updateAttack(dt: number, w: WorldRef): void {
   563→    const a = this.action
   564→    if (!a) {
   565→      this.state = 'move'
   566→      return
   567→    }
   568→    const def = a.def
   569→    a.animT = this.animator.getTime()
   570→    const animT = a.animT
   571→    const spd = this.atkSpeed
   572→
   573→    // ground friction / air drag during attack
   574→    if (this.grounded) {
   575→      const f = 900 * dt
   576→      this.vx = Math.abs(this.vx) <= f ? 0 : this.vx - Math.sign(this.vx) * f
   577→    }
   578→
   579→    // sfx event
   580→    if (!a.sfxDone && animT >= def.swingSfx.t) {
   581→      a.sfxDone = true
   582→      w.audio.play(def.swingSfx.name, { pitch: rand(0.94, 1.08) })
   583→    }
   584→    // forward impulse
   585→    if (!a.moveDone && def.moveX !== undefined && animT >= (def.moveXAt ?? def.hit.start)) {
   586→      a.moveDone = true
   587→      this.vx = this.face * def.moveX * (def.kind === 'dash' ? 1 : 1)
   588→    }
   589→    // vertical impulse (launcher hop / slam plunge)
   590→    if (!a.moveYDone && def.moveY !== undefined && animT >= (def.moveYAt ?? 0)) {
   591→      a.moveYDone = true
   592→      this.vy = def.moveY
   593→      if (def.moveY < 0) {
   594→        this.grounded = false
   595→        w.fx.dust(this.x, 0, 6)
   596→      }
   597→    }
   598→
   599→    // hit window: sample blade segment & test enemies (swept, anti-tunneling)
   600→    const h = def.hit
   601→    if (animT >= h.start && animT <= h.end) {
   602→      this.swingGlow = 1
   603→      const info = makeRuntimeInfo(def, this.sword.stats.dmg)
   604→      for (const e of w.enemies) {
   605→        if (e.dead || e.spawning) continue
   606→        const last = a.hitMap.get(e.id)
   607→        const reArm = h.multiHit ? (last === undefined || animT - last >= h.multiHit) : last === undefined
   608→        if (!reArm) continue
   609→        const cy = e.y - 34 * e.scale
   610→        const r = 26 * e.scale + 6
   611→        const hit = this.hasPrevBlade
   612→          ? circleHitsSweptQuad(e.x, cy, r, this.prevBlade.bx, this.prevBlade.by, this.prevBlade.tx, this.prevBlade.ty, this.blade.bx, this.blade.by, this.blade.tx, this.blade.ty)
   613→          : segHitsCircle(this.blade.bx, this.blade.by, this.blade.tx, this.blade.ty, e.x, cy, r)
   614→        if (hit) {
   615→          a.hitMap.set(e.id, animT)
   616→          w.hitEnemy(e.id, this.blade, info)
   617→          this.onLandedHit(w, info, e.x, cy)
   618→        }
   619→      }
   620→      // remember this frame's blade for the next sweep test
   621→      this.prevBlade.bx = this.blade.bx
   622→      this.prevBlade.by = this.blade.by
   623→      this.prevBlade.tx = this.blade.tx
   624→      this.prevBlade.ty = this.blade.ty
   625→      this.hasPrevBlade = true
   626→      // aoe shockwave at strike moment for heavy hits
   627→      if (h.aoe && h.aoe > 0 && !a.impactDone && this.grounded && animT >= h.start + 0.02) {
   628→        a.impactDone = true
   629→        w.fx.shockwave(this.x + this.face * 30, 0, h.aoe, this.sword.colors.glow, 0.4, 6)
   630→        if (this.grounded) w.fx.groundCrack(this.x + this.face * 34, 0, h.aoe * 0.45)
   631→      }
   632→    }
   633→
   634→    // trail emission
   635→    if (animT >= def.trail.start && animT <= def.trail.end) {
   636→      w.fx.pushTrail(a.trail, this.blade.bx, this.blade.by, this.blade.tx, this.blade.ty)
   637→    } else if (animT > def.trail.end && a.trail.emitting) {
   638→      w.fx.endTrail(a.trail)
   639→    }
   640→
   641→    // cancel window → allow chaining / dash-cancel / jump-cancel
   642→    const dashCancel = animT >= h.end * 0.6
   643→    if (animT >= def.cancel || dashCancel) {
   644→      const input = w.input
   645→      if ((input.pressed('dash') || input.buffered('dash')) && this.canDash()) {
   646→        w.fx.endTrail(a.trail)
   647→        this.startDash(w, input.axisX(), input.axisY())
   648→        return
   649→      }
   650→      if (input.pressed('parry')) {
   651→        w.fx.endTrail(a.trail)
   652→        this.enterParry(w)
   653→        return
   654→      }
   655→      if (animT >= def.cancel) {
   656→        // chain next attack
   657→        const ax = input.axisX()
   658→        const ay = input.axisY()
   659→        let next: string | null = null
   660→        if (input.buffered('light')) {
   661→          input.takeBuffered('light')
   662→          next = this.riposteT > 0 && this.grounded ? 'riposte' : this.pickLight(w, ax, ay)
   663→        } else if (input.buffered('heavy')) {
   664→          input.takeBuffered('heavy')
   665→          if (!this.grounded) {
   666→            next = 'slam'
   667→          } else if (input.down('heavy')) {
   668→            // still holding → convert to charge
   669→            this.enterCharge(w)
   670→            w.fx.endTrail(a.trail)
   671→            return
   672→          } else {
   673→            next = def.chain.find((c) => c.startsWith('h')) ?? 'h1'
   674→          }
   675→        }
   676→        if (next) {
   677→          // validate chain (or allow from-parry riposte)
   678→          if (fromChain(def, next, this)) {
   679→            w.fx.endTrail(a.trail)
   680→            this.startAttack(ATTACKS[next], w)
   681→            return
   682→          }
   683→        }
   684→        // jump cancel on ground
   685→        if (this.grounded && (input.pressed('jump') || input.buffered('jump', 0.15))) {
   686→          w.fx.endTrail(a.trail)
   687→          this.doJump(w)
   688→          this.state = 'move'
   689→          return
   690→        }
   691→      }
   692→    }
   693→
   694→    // finish
   695→    if (this.animator.isDone()) {
   696→      w.fx.endTrail(a.trail, 0.18)
   697→      this.action = null
   698→      this.state = 'move'
   699→    }
   700→  }

  1010→  /** set > 0 to auto-exit scripted state after t seconds */

  1013→  private updateAbility(dt: number, w: WorldRef): void {
  1014→    const ab = this.ability
  1015→    if (!ab) return
  1016→    ab.t += dt
  1017→    if (this.exitScriptedSoon > 0) {
  1018→      this.exitScriptedSoon -= dt
  1019→      if (this.exitScriptedSoon <= 0) {
  1020→        this.exitScripted()
  1021→        if (ab.kind !== 'gravity') this.ability = null
  1022→      }
  1023→    }
  1024→
  1025→    const c = this.sword.colors
  1026→
  1027→    switch (ab.kind) {
  1028→      case 'katana': {
  1029→        // storm: dash-slash through a new enemy every 0.13s, 6 steps
  1030→        const interval = 0.13
  1031→        if (ab.t >= ab.step * interval && ab.step < 6) {
  1032→          ab.step++
  1033→          const targets = this.collectNearest(w, 6, 520).filter((e) => !ab.targets.includes(e.id))

  1035→          if (tgt) {
  1036→            ab.targets.push(tgt.id)
  1037→            w.fx.afterimage(this.pose, this.x, this.hipY, this.face, c.trail, 0.35, SCALE)

  1047→            w.cam.addTrauma(0.12)
  1048→            w.addHitstop(0.03)
  1049→          }
  1050→        }
  1051→        if (ab.step >= 6 && ab.t > 6 * interval + 0.2) {
  1052→          this.ability = null
  1053→          this.exitScripted()
  1054→        }
  1055→        break
  1056→      }
  1057→      case 'gravity': {
  1058→        // suspend hold, then crush at t=0.55
  1059→        if (ab.step === 0 && ab.t >= 0.55) {

  1076→          this.ability = null
  1077→          this.exitScripted()
  1078→        }
  1079→        break
  1080→      }
  1081→      case 'phantom': {
  1082→        // echoes every 0.2s
  1083→        const interval = 0.2
  1084→        if (ab.t >= ab.step * interval && ab.step < ab.targets.length) {
  1085→          const eid = ab.targets[ab.step]
  1086→          ab.step++

  1098→          this.ability = null

  1100→        }
  1101→        break
  1102→      }
  1103→      default:
  1104→        if (ab.t > 1.2) {
  1105→          this.ability = null
  1106→        }
  1107→    }
  1108→  }

  1110→  private collectNearest(w: WorldRef, n: number, range: number): EnemyLike[] {
  1111→    const list: EnemyLike[] = []
  1112→    for (const e of w.enemies) {
  1113→      if (e.dead || e.spawning) continue
  1114→      const d = Math.hypot(e.x - this.x, e.y - this.y)
  1115→      if (d < range) list.push({ id: e.id, x: e.x, y: e.y, d })
  1116→    }
  1117→    list.sort((a, b) => (a.d ?? 0) - (b.d ?? 0))
  1118→    return list.slice(0, n)
  1119→  }

  1121→  private findDensest(w: WorldRef, range: number): EnemyLike | null {
  1122→    let best: EnemyLike | null = null
  1123→    let bestScore = -1
  1124→    for (const e of w.enemies) {
  1125→      if (e.dead || e.spawning) continue
  1126→      let score = 0
  1127→      for (const o of w.enemies) {
  1128→        if (o.dead || o.spawning) continue
  1129→        if (Math.abs(o.x - e.x) < range) score++
  1130→      }
  1131→      if (score > bestScore) {
  1132→        bestScore = score
  1133→        best = { id: e.id, x: e.x, y: e.y, d: 0 }
  1134→      }
  1135→    }
  1136→    return best
  1137→  }

  1154→  private applyProcedural(dt: number): void {
  1155→    const p = this.pose
  1156→    // velocity lean (proportional, eased)
  1157→    const speedLean = clamp((this.vx * this.face) / MOVE_MAX, -1, 1)
  1158→    p[2] += speedLean * 0.14
  1159→    // vertical pitch while airborne
  1160→    if (!this.grounded && this.state !== 'attack') {
  1161→      p[2] += clamp(this.vy / 1400, -0.22, 0.3)
  1162→    }
  1163→    // landing spring (squash)
  1164→    if (this.landSpring > 0) {
  1165→      this.landSpring = Math.max(0, this.landSpring - dt * 5)
  1166→      p[1] += this.landSpring * 6
  1167→    }
  1168→    // charge tremble
  1169→    if (this.charging) {
  1170→      p[0] += rand(-0.8, 0.8) * this.chargeRatio
  1171→      p[1] += rand(-0.5, 0.5) * this.chargeRatio
  1172→    }
  1173→    // hurt shake
  1174→    if (this.hurtT > 0) {
  1175→      p[0] += rand(-1.4, 1.4) * this.hurtT
  1176→    }
  1177→  }

  1257→function fromChain(def: AttackDef, next: string, player: Player): boolean {
  1258→  if (next === 'riposte') return true
  1259→  // launcher chains naturally into air attacks
  1260→  if (def.id === 'launcher' && (next.startsWith('a') || next === 'dive' || next === 'slam')) return true
  1261→  if (def.chain.includes(next)) return true
  1262→  // any light may restart the chain if the window allows
  1263→  if (player.grounded && next.startsWith('l') && def.chain.some((c) => c.startsWith('l'))) return true
  1264→  if (!player.grounded && (next.startsWith('a') || next === 'dive' || next === 'slam') && def.chain.some((c) => c.startsWith('a') || c === 'dive' || c === 'slam')) return true
  1265→  return false
  1266→}
