    27→class ComboTracker {
    28→  hits = 0
    29→  timer = 0
    30→  timerMax = 2.6
    31→  mult = 1
    32→  best = 0
    33→  bestRank = ''
    34→  rankUpT = 0
    35→  lastRank = ''
    36→
    37→  onHit(): void {
    38→    this.hits++
    39→    this.timer = this.timerMax
    40→    this.mult = Math.min(1 + Math.floor(this.hits / 5) * 0.5, 5)
    41→    if (this.hits > this.best) this.best = this.hits
    42→    const { rank } = rankForHits(this.hits)
    43→    if (rank && rank !== this.lastRank) {
    44→      this.lastRank = rank
    45→      if (rank.length > this.bestRank.length) this.bestRank = rank
    46→      this.rankUpT = 0.7
    47→    }
    48→  }
    49→
    50→  update(dt: number): void {
    51→    this.rankUpT = Math.max(0, this.rankUpT - dt)
    52→    if (this.hits > 0) {
    53→      this.timer -= dt
    54→      if (this.timer <= 0) {
    55→        this.hits = 0
    56→        this.mult = 1
    57→        this.lastRank = ''
    58→      }
    59→    }
    60→  }
    61→
    62→  reset(): void {
    63→    this.hits = 0
    64→    this.timer = 0
    65→    this.mult = 1
    66→    this.best = 0
    67→    this.bestRank = ''
    68→    this.rankUpT = 0
    69→    this.lastRank = ''
    70→  }
    71→}

   246→  /** resolution governor: every 0.8s of PLAYING, compare the damped frame
   247→   * rate against thresholds and step the render DPR down (or back up) with
   248→   * hysteresis. If DPR 1 still can't hold ~40fps, fall a quality tier
   249→   * (rain/mote/fog/mist/reflection density all scale with tier). */
   250→  private govern(dt: number): void {
   251→    if (!this.autoQuality || this.phase !== 'playing') return
   252→    this.govT += dt
   253→    this.govCool = Math.max(0, this.govCool - dt)
   254→    if (this.govT < 0.8) return
   255→    this.govT = 0
   256→    if (this.govCool > 0) return
   257→    const cap = this.settings.quality === 'high' ? 1.5 : this.settings.quality === 'medium' ? 1.25 : 1
   258→    if (this.fpsAvg < 44 && this.dprTarget > 1) {
   259→      this.dprTarget = Math.max(1, this.dprTarget - 0.25)
   260→      this.govCool = 1.6
   261→      this.resize()
   262→    } else if (this.fpsAvg < 36 && this.dprTarget <= 1) {
   263→      // resolution floor reached and still drowning → drop a quality tier
   264→      const order: GameSettings['quality'][] = ['high', 'medium', 'low']
   265→      const i = order.indexOf(this.settings.quality)
   266→      if (i < order.length - 1) {
   267→        this.applySettings({ ...this.settings, quality: order[i + 1] })
   268→        this.govCool = 4
   269→      }
   270→    } else if (this.fpsAvg > 57 && this.dprTarget < cap) {
   271→      this.dprTarget = Math.min(cap, this.dprTarget + 0.25)
   272→      this.govCool = 3
   273→      this.resize()
   274→    }
   275→  }

   383→  private loop = (now: number): void => {
   384→    if (this.destroyed) return
   385→    this.raf = requestAnimationFrame(this.loop)
   386→    let dtRaw = (now - this.lastFrame) / 1000
   387→    this.lastFrame = now
   388→    if (dtRaw > 0.05) dtRaw = 0.05
   389→    if (dtRaw <= 0) dtRaw = 0.0001
   390→    this.fps = damp(this.fps, 1 / dtRaw, 3, dtRaw)
   391→    this.fpsAvg = damp(this.fpsAvg, 1 / dtRaw, 1.2, dtRaw)
   392→    this.govern(dtRaw)
   393→
   394→    const ctx = this.ctx
   395→    if (!ctx) return
   396→
   397→    if (this.phase === 'paused') {
   398→      // frozen frame, HUD still drawn
   399→    } else {
   400→      // --- timing: hitstop & slow-mo ---
   401→      let simDt = dtRaw
   402→      if (this.hitstopT > 0) {
   403→        this.hitstopT -= dtRaw
   404→        simDt = dtRaw * 0.05
   405→        this.timeScale = 0.05 // world consumers (blade impact deflection) must SEE the freeze
   406→      } else if (this.slowmoT > 0) {
   407→        this.slowmoT -= dtRaw
   408→        simDt = dtRaw * this.slowmoScale
   409→        this.timeScale = this.slowmoScale
   410→        if (this.slowmoT <= 0) this.timeScale = 1
   411→      } else {
   412→        this.timeScale = 1
   413→      }
   414→
   415→      // --- cinematics run on real time ---
   416→      this.updateCinematics(dtRaw)
   417→
   418→      // --- capped substeps for stable physics (NO death spiral: cost of a
   419→      // frame is bounded no matter how slow the machine runs) ---
   420→      let remaining = simDt
   421→      let substeps = 0
   422→      while (remaining > 1e-6 && substeps < 3) {
   423→        const stepDt = Math.min(remaining, 1 / 60)
   424→        this.step(stepDt)
   425→        remaining -= stepDt
   426→        substeps++
   427→      }
   428→      if (remaining > 1e-6) {
   429→        // very slow frame: consume the backlog as one clamped catch-up tick
   430→        // so the world keeps roughly real-time without multiplying cost
   431→        this.step(Math.min(remaining, 0.05))
   432→      }
   433→      // --- visual systems tick ONCE per frame (never per substep) ---
   434→      this.fx.update(simDt)
   435→      this.bg.update(simDt)
   436→      this.updateCamera(simDt)
   437→    }
   438→
   439→    // --- render ---
   440→    ctx.setTransform(this.dpr, 0, 0, this.dpr, 0, 0)
   441→    const world: RenderWorld = {
   442→      cam: this.cam,
   443→      fx: this.fx,
   444→      bg: this.bg,
   445→      enemies: this.enemies,
   446→      player: this.player,
   447→      projectiles: this.projectiles,
   448→      hud: this.hud,
   449→      hudState: this.buildHudState(),
   450→      time: this.time,
   451→    }
   452→    this.renderer.draw(ctx, world, this.cssW, this.cssH, dtRaw)
   453→    // inspection overlay (solo/sheet views + state readout)
   454→    this.inspect.draw(ctx, this, this.time)
   455→
   456→    this.input.endFrame(dtRaw)
   457→  }
   458→
   459→  private step(dt: number): void {
   460→    this.time += dt
   461→    this.worldRef.time = this.time
   462→    this.worldRef.timeScale = this.timeScale
   463→    this.worldRef.quality = this.settings.quality
   464→    this.worldRef.gore = this.settings.gore
   465→    PLAYER_X_REF.x = this.player.x
   466→    PLAYER_X_REF.y = this.player.y
   467→    PLAYER_POS.x = this.player.x
   468→    PLAYER_POS.y = this.player.y
   469→
   470→    if (this.phase === 'playing' || this.phase === 'gameover') {
   471→      if (this.phase === 'playing') {
   472→        this.handleMetaInput()
   473→      }
   474→      this.player.update(dt, this.worldRef)
   475→      for (const e of this.enemies) {
   476→        e.update(dt, this.worldRef)
   477→      }
   478→      // purge dead
   479→      for (let i = this.enemies.length - 1; i >= 0; i--) {
   480→        if (this.enemies[i].dead) this.enemies.splice(i, 1)
   481→      }
   482→      this.projectiles.update(dt, this.worldRef)
   483→
   484→      if (this.phase === 'playing') {
   485→        if (!this.qaMode) this.waves.update(dt, this.enemies.length)
   486→        this.combo.update(dt)
   487→        this.runTime += dt
   488→        this.stats.timeAlive = this.runTime
   489→        // death → game over
   490→        if (!this.qaMode && this.player.isDead && this.player.deadTime > 1.4) {
   491→          this.gameOver()
   492→        }
   493→      }
   494→    } else if (this.phase === 'attract') {
   495→      // idle hero breathing on the rooftop (framed below the title logo)
   496→      this.player.update(dt, this.worldRef)
   497→      this.cam.focus(this.player.x + Math.sin(this.time * 0.14) * 140, this.player.y - 110, 0, 0)
   498→    }
   499→  }
   500→
   501→  private updateCamera(dt: number): void {
   502→    if (this.phase === 'playing' || this.phase === 'gameover') {
   503→      const p = this.player
   504→      let tx = p.x
   505→      let ty = p.y
   506→      let vx = p.vx
   507→      let vy = p.vy
   508→      // during execution / ult, focus the action
   509→      if (this.exec) {
   510→        tx = (p.x + this.exec.enemy.x) / 2
   511→        ty = (p.y + this.exec.enemy.y) / 2 - 30
   512→        vx = 0
   513→        vy = 0
   514→      }
   515→      this.cam.focus(tx, ty, vx, vy)
   516→      // dynamic zoom: pull back when many enemies, punch in during cinematics
   517→      let zoom = 1.04
   518→      const n = this.enemies.length
   519→      if (n > 10) zoom = 0.94
   520→      if (n > 18) zoom = 0.88
   521→      if (this.exec || this.ult) zoom = 1.3
   522→      this.cam.setZoom(zoom)
   523→    }
   524→    this.cam.update(dt)
   525→  }
   526→
   527→  private handleMetaInput(): void {
   528→    // pause
   529→    if (this.input.pressed('pause')) {
   530→      this.pause()
   531→      return
   532→    }
   533→    // sword switching
   534→    const swordIds: SwordId[] = ['katana', 'inferno', 'thunder', 'void', 'gravity', 'phantom']
   535→    const swordActions = ['sword1', 'sword2', 'sword3', 'sword4', 'sword5', 'sword6'] as const
   536→    for (let i = 0; i < 6; i++) {
   537→      if (this.input.pressed(swordActions[i])) {
   538→        this.trySwitchSword(swordIds[i])
   539→      }
   540→    }
   541→    if (this.input.pressed('cycleSword')) {
   542→      const idx = SWORD_ORDER.indexOf(this.player.sword.id)
   543→      for (let i = 1; i <= 6; i++) {
   544→        const next = SWORD_ORDER[(idx + i) % 6]
   545→        if (this.unlocked.includes(next)) {
   546→          this.trySwitchSword(next)
   547→          break
   548→        }
   549→      }
   550→    }
   551→    // ultimate
   552→    if (this.input.pressed('ult') && this.player.meter >= 100 && !this.ult && !this.exec) {
   553→      this.startUlt()
   554→    }
   555→    // execution
   556→    if (this.input.pressed('execute') && !this.exec && !this.ult) {
   557→      const target = this.findExecuteTarget()
   558→      if (target) this.startExecution(target)

   639→  private hitEnemy(enemyId: number, blade: { bx: number; by: number; tx: number; ty: number }, atk: AttackRuntimeInfo): void {
   640→    const e = this.enemies.find((en) => en.id === enemyId)
   641→    if (!e || e.dead) return
   642→    const sword = this.player.sword
   643→    const dir = Math.sign(blade.tx - blade.bx) || this.player.face
   644→    const hx = (blade.bx + blade.tx) / 2
   645→    const hy = (blade.by + blade.ty) / 2
   646→    const ex = e.x
   647→    // impact FX at the blade's actual contact height, clamped to the visual
   648→    // body (head-top ≈ 88u … feet ≈ 10u) — sparks appear where the edge
   649→    // crossed the silhouette, not at a fixed belly point
   650→    const ey = clamp(hy, e.y - 86 * e.scale, e.y - 10 * e.scale)
   651→
   652→    // elite parry: occasionally deflects light strikes
   653→    if (e.kind === 'elite' && !atk.heavy && Math.random() < 0.16 && e.state !== 'stun' && e.state !== 'launched') {
   654→      this.audio.play('clang')
   655→      this.fx.impact(ex, ey, Math.atan2(ey - hy, ex - hx), true, '#ffffff', '#ff5c7a')
   656→      this.fx.sparks(ex, ey, -Math.PI / 2, 1.4, 12, '#ffffff', 420, 0.4)
   657→      this.addHitstop(0.05)
   658→      this.cam.addTrauma(0.15)
   659→      this.fx.text(ex, ey - 50, 'PARRY!', '#ff5c7a', 12)
   660→      return
   661→    }
   662→
   663→    const res = e.takeHit(atk, dir, this.player.x)
   664→
   665→    // combo & score
   666→    this.combo.onHit()
   667→    this.stats.maxCombo = Math.max(this.stats.maxCombo, this.combo.hits)
   668→    if (this.combo.rankUpT > 0.6) {
   669→      this.audio.play('rankup', { pitch: 1 + this.combo.hits * 0.002 })
   670→      this.emit({ type: 'rank', rank: this.combo.lastRank })
   671→    }
   672→    this.stats.score += Math.round(atk.dmg * this.combo.mult * 0.8)
   673→
   674→    switch (res) {
   675→      case 'killed': {
   676→        this.onEnemyKilled(e)
   677→        break
   678→      }
   679→      case 'blocked': {
   680→        this.audio.play('clang', { pitch: 0.9 })
   681→        this.fx.impact(ex, ey, Math.atan2(ey - hy, ex - hx), false, '#ffffff', '#46ffb4')
   682→        this.fx.text(ex, ey - 46, 'BLOCK', '#46ffb4', 11)
   683→        this.addHitstop(0.03)
   684→        this.cam.addTrauma(0.06)
   685→        break
   686→      }
   687→      case 'armored': {
   688→        this.audio.play('clang', { pitch: 0.6, volume: 0.8 })
   689→        this.fx.sparks(ex, ey, Math.atan2(ey - hy, ex - hx), 0.9, 8, '#ffae2e', 320, 0.35)
   690→        this.fx.text(ex, ey - 46, 'ARMOR', '#ffae2e', 11)
   691→        this.addHitstop(0.02)
   692→        break
   693→      }
   694→      default: {
   695→        // solid hit
   696→        this.audio.play(atk.heavy ? 'hitHeavy' : 'hit', {
   697→          volume: clamp(0.55 + atk.dmg * 0.01, 0.55, 1),
   698→          pitch: rand(0.94, 1.08),
   699→          intensity: clamp01(atk.dmg / 30),
   700→        })
   701→        this.fx.impact(ex, ey, Math.atan2(ey - hy, ex - hx), atk.heavy, sword.colors.glow, sword.colors.accent)
   702→        const dmgShown = Math.round(atk.dmg * (e.vulnerableT > 0 ? 2 : 1) * (e.kind === 'heavy' || e.kind === 'boss' ? 0.72 : 1))
   703→        this.fx.text(ex, ey - 46, `${dmgShown}`, atk.heavy ? sword.colors.accent : '#f4f9fc', atk.heavy ? 17 : 13)
   704→        // crash-zoom punch on heavies (Z-axis push-pull) + impact frames on
   705→        // the big connects — the anime kill-frame language
   706→        this.addHitstop(atk.hitstop, atk.heavy ? 0.09 : 0)
   707→        if (atk.heavy && atk.dmg >= 15) {
   708→          this.fx.impactFrame(ex, ey, sword.colors.glow)
   709→        } else if (atk.heavy) {
   710→          this.fx.chroma(0.45)
   711→        }
   712→        this.cam.addTrauma(atk.shake)
   713→        if (e.vulnerableT > 0) this.fx.text(ex, ey - 70, 'WEAK POINT', '#ffe14d', 11)
   714→        break
   715→      }
   716→    }
   717→  }

   719→  private onEnemyKilled(e: Enemy): void {

   815→  private updateCinematics(dt: number): void {
   816→    if (this.banner) {
   817→      this.banner.t += dt
   818→      if (this.banner.t >= this.banner.dur) this.banner = null
   819→    }

   821→      const ex = this.exec
   822→      ex.t += dt

   824→      const sword = this.player.sword
   825→      const midX = (this.player.x + e.x) / 2
   826→      const midY = (this.player.y + e.y) / 2 - 40

   828→        ex.step = 1
   829→        this.execSlash(midX, midY, sword.colors.trail, 0)
   830→        this.player.playClip(ATTACKS.l1.clip, 1.7)

   832→        ex.step = 2
   833→        this.execSlash(midX, midY, sword.colors.trail, 1)
   834→        this.player.playClip(ATTACKS.l2.clip, 1.7)

   840→        ex.step = 4
   841→        // final blow

   843→        this.audio.play('deathHeavy', { volume: 1 })

   916→            id: 'ult', dmg: 30 * sword.stats.dmg, heavy: true, stagger: 0.6, kb: 300, kbY: 0,

   956→  private buildHudState(): HudState {

   958→    const boss = this.enemies.find((e) => e.kind === 'boss' && !e.dead)
   959→    let execPrompt: { x: number; y: number } | null = null
   960→    if (this.phase === 'playing' && !this.exec && !this.ult) {
   961→      const target = this.findExecuteTarget()
   962→      if (target) {
   963→        const zoom = this.cam.zoom

   965→          x: (target.x - this.cam.x - this.cam.offX) * zoom + this.cssW / 2,
   966→          y: (target.y - 120 - this.cam.y - this.cam.offY) * zoom + this.cssH / 2,
   967→        }
   968→      }
   969→    }
   970→    return {
   971→      hp: p.hp,
   972→      maxHp: p.maxHp,
   973→      energy: p.energy,
