    15→/** circle vs swept blade quad (prev segment + current segment) — prevents tunneling */
    16→function circleHitsSweptQuad(
    17→  cx: number, cy: number, r: number,
    18→  pbx: number, pby: number, ptx: number, pty: number,
    19→  bx: number, by: number, tx: number, ty: number,
    20→): boolean {
    21→  // current + previous blade segments
    22→  if (segHitsCircle(bx, by, tx, ty, cx, cy, r)) return true
    23→  if (segHitsCircle(pbx, pby, ptx, pty, cx, cy, r)) return true
    24→  // swept edges (tip path + base path)
    25→  if (segHitsCircle(ptx, pty, tx, ty, cx, cy, r)) return true
    26→  if (segHitsCircle(pbx, pby, bx, by, cx, cy, r)) return true
    27→  // circle center inside the swept quad (spins / wide arcs)
    28→  if (pointInTri(cx, cy, pbx, pby, ptx, pty, tx, ty)) return true
    29→  if (pointInTri(cx, cy, pbx, pby, tx, ty, bx, by)) return true
    30→  return false
    31→}

    33→function pointInTri(px: number, py: number, x1: number, y1: number, x2: number, y2: number, x3: number, y3: number): boolean {
    34→  const d1 = (px - x2) * (y1 - y2) - (x1 - x2) * (py - y2)
    35→  const d2 = (px - x3) * (y2 - y3) - (x2 - x3) * (py - y3)
    36→  const d3 = (px - x1) * (y3 - y1) - (x3 - x1) * (py - y1)
    37→  const hasNeg = d1 < 0 || d2 < 0 || d3 < 0
    38→  const hasPos = d1 > 0 || d2 > 0 || d3 > 0
    39→  return !(hasNeg && hasPos)
    40→}

    46→const SCALE = 1.32
    47→/** standing hip height = thigh(25)+shin(24)=49 local × SCALE ≈ 64.7 — keeps
    48→ *  straight legs exactly on the ground line instead of sinking 4px into it */
    49→const HIP_H = 64
    50→
    51→/**
    52→ * Enemy body spheres: [centerY offset above ground, radius] in enemy-local
    53→ * units (× e.scale). Traced from the ENEMY_SPEC skeleton — hip at 47u, torso
    54→ * 32u, head top ≈ 94u. The old single circle (y-34, r 26) only covered the
    55→ * lower 2/3 of the body, so chest/head-height slashes passed straight
    56→ * through the sprite without connecting. Three spheres hug the visual body.
    57→ */
    58→const BODY_SPHERES: ReadonlyArray<{ oy: number; r: number }> = [
    59→  { oy: 14, r: 15 }, // legs / hip mass
    60→  { oy: 48, r: 21 }, // torso (the main target)
    61→  { oy: 78, r: 17 }, // chest / head zone
    62→]
    63→
    64→/** blade segment vs an enemy's full visual body (any sphere, swept-aware).
    65→ * Radius carries +3px edge forgiveness — a katana edge is a plane, not a
    66→ * line; the visual blade is ~3px wide so the hit volume matches the art. */
    67→function bladeHitsBody(
    68→  e: { x: number; y: number; scale: number },
    69→  hasPrev: boolean,
    70→  prev: { bx: number; by: number; tx: number; ty: number },
    71→  cur: { bx: number; by: number; tx: number; ty: number },
    72→): boolean {
    73→  const s = e.scale
    74→  for (const sp of BODY_SPHERES) {
    75→    const cy = e.y - sp.oy * s
    76→    const r = sp.r * s + 3
    77→    if (hasPrev
    78→      ? circleHitsSweptQuad(e.x, cy, r, prev.bx, prev.by, prev.tx, prev.ty, cur.bx, cur.by, cur.tx, cur.ty)
    79→      : segHitsCircle(cur.bx, cur.by, cur.tx, cur.ty, e.x, cy, r)
    80→    ) return true
    81→  }
    82→  return false
    83→}

   238→  /* ------------------------------------------------------------------ */
   239→  /* main update                                                          */

   241→
   242→  update(dt: number, w: WorldRef): void {
   243→    this.iframes = Math.max(0, this.iframes - dt)
   244→    this.flashT = Math.max(0, this.flashT - dt * 5)
   245→    this.swingGlow = Math.max(0, this.swingGlow - dt * 5)
   246→    this.riposteT = Math.max(0, this.riposteT - dt)
   247→    this.energy = Math.min(this.maxEnergy, this.energy + dt * 3)
   248→
   249→    switch (this.state) {
   250→      case 'move': this.updateMove(dt, w); break
   251→      case 'attack': this.updateAttack(dt, w); break
   252→      case 'dash': this.updateDash(dt, w); break
   253→      case 'parry': this.updateParry(dt, w); break
   254→      case 'charge': this.updateCharge(dt, w); break
   255→      case 'hurt': this.updateHurt(dt, w); break
   256→      case 'dead': this.updateDead(dt, w); break
   257→      case 'scripted': this.updateScripted(dt); break
   258→    }
   259→
   260→    // physics integration (shared)
   261→    if (this.state !== 'dash') {
   262→      const g = this.state === 'attack' && this.action && this.action.def.float
   263→        ? GRAVITY * this.action.def.float
   264→        : GRAVITY
   265→      this.vy += g * dt
   266→    }
   267→    this.x += this.vx * dt
   268→    this.y += this.vy * dt
   269→
   270→    // ground
   271→    const wasAir = !this.grounded
   272→    if (this.y >= 0) {
   273→      this.y = 0
   274→      if (this.vy > 140) {
   275→        this.landSpring = clamp(this.vy / 700, 0, 1)
   276→        if (wasAir) this.onLand(w)
   277→      } else if (wasAir) {
   278→        this.landSpring = 0.4
   279→        this.onLand(w)
   280→      }
   281→      this.vy = 0
   282→      this.grounded = true
   283→      this.jumpsUsed = 0
   284→      this.airDashUsed = false
   285→    } else {
   286→      this.grounded = false
   287→    }
   288→    // arena bounds
   289→    if (this.x < w.arenaMinX + 20) {
   290→      this.x = w.arenaMinX + 20
   291→      if (this.vx < -200) w.fx.dust(this.x - 12, 0, 5)
   292→      this.vx = Math.max(0, this.vx)
   293→    }
   294→    if (this.x > w.arenaMaxX - 20) {
   295→      this.x = w.arenaMaxX - 20
   296→      if (this.vx > 200) w.fx.dust(this.x + 12, 0, 5)
   297→      this.vx = Math.min(0, this.vx)
   298→    }
   299→
   300→    this.dashCd = Math.max(0, this.dashCd - dt)
   301→    this.dashRecharge = Math.max(0, this.dashRecharge - dt)
   302→    if (this.dashRecharge <= 0) this.dashConsec = 0
   303→
   304→    // ability script
   305→    if (this.ability) this.updateAbility(dt, w)
   306→
   307→    // animation + solve
   308→    this.animator.update(dt)
   309→    // Base layer: the clip FULLY owns the pose each frame (write, never
   310→    // accumulate). applyProcedural() is a bounded additive layer on top.
   311→    // Without this copy, procedural += deltas integrate unbounded → spine
   312→    // spin + root sink. (Same contract as Spine's AnimationState.apply /
   313→    // Godot AnimationTree → Skeleton2D.)
   314→    this.pose.set(this.animator.pose)
   315→    this.applyProcedural(dt)
   316→    solveSkeleton(this.pose, PLAYER_SPEC, this.geom)
   317→    this.updateBladeWorld()
   318→  }

   335→  private slamAoe(w: WorldRef, r: number, dmg: number, launch: number): void {
   336→    const info: AttackRuntimeInfo = {
   337→      id: 'slam', dmg: dmg * this.sword.stats.dmg, heavy: true,
   338→      stagger: 0.5, kb: 240, kbY: 0, launch, juggle: 0,
   339→      hitstop: 0.11, shake: 0.45, aoe: r, breakGuard: true, kind: 'heavy',
   340→    }
   341→    const colors = this.sword.colors
   342→    w.fx.shockwave(this.x, 0, r, colors.glow, 0.5, 8)
   343→    w.fx.groundCrack(this.x, 0, r * 0.6)
   344→    w.fx.burst(this.x, -6, 14, colors.glow, 340, 0.5, 'spark')
   345→    w.fx.screenFlash(colors.trail, 0.16, 5)
   346→    w.addHitstop(0.09, 0.06)
   347→    w.cam.addTrauma(0.4)
   348→    w.audio.play('slam', { intensity: 1 })
   349→    const bx = this.x
   350→    for (const e of w.enemies) {
   351→      if (e.dead || e.spawning) continue
   352→      const dx = e.x - bx
   353→      if (Math.abs(dx) < r && e.y > -140) {
   354→        w.hitEnemy(e.id, { bx: bx - 20, by: -20, tx: bx + 20, ty: -20 }, info)
   355→      }
   356→    }
   357→  }

   363→  private updateMove(dt: number, w: WorldRef): void {
   364→    const input = w.input
   365→    const ax = input.axisX()
   366→    const ay = input.axisY()
   367→
   368→    // facing follows input
   369→    if (ax !== 0) this.face = ax > 0 ? 1 : -1
   370→
   371→    // horizontal movement
   372→    if (ax !== 0) {
   373→      const accel = this.grounded ? ACCEL : AIR_ACCEL
   374→      this.vx += ax * accel * dt
   375→      this.vx = clamp(this.vx, -MOVE_MAX, MOVE_MAX)
   376→    } else if (this.grounded) {
   377→      const f = FRICTION * dt
   378→      this.vx = Math.abs(this.vx) <= f ? 0 : this.vx - Math.sign(this.vx) * f
   379→    } else {
   380→      this.vx = damp(this.vx, 0, 0.4, dt)
   381→    }
   382→
   383→    // locomotion clip selection
   384→    const speed = Math.abs(this.vx)
   385→    if (this.grounded) {
   386→      if (speed > 40) {
   387→        this.animator.play(RUN_CLIP, { blend: 0.09, speed: clamp(speed / MOVE_MAX, 0.65, 2.1) })
   388→      } else if (this.vy === 0) {
   389→        this.animator.play(IDLE_CLIP, { blend: 0.16 })
   390→      }
   391→    } else if (this.vy < -60) {
   392→      this.animator.play(JUMP_CLIP, { blend: 0.06 })
   393→    } else {
   394→      this.animator.play(FALL_CLIP, { blend: 0.12 })
   395→    }
   396→
   397→    // ---- actions ----
   398→    if (this.tryCombatInputs(w, ax, ay)) return
   399→  }
   400→
   401→  /** returns true if a new state was entered */
   402→  private tryCombatInputs(w: WorldRef, ax: number, ay: number): boolean {
   403→    const input = w.input
   404→
   405→    // dash (anytime, from move)
   406→    if ((input.pressed('dash') || input.buffered('dash')) && this.canDash()) {
   407→      this.startDash(w, ax, ay)
   408→      return true
   409→    }
   410→    // jump (grounded = jump 1, plus one air jump)
   411→    if (input.pressed('jump') || input.buffered('jump', 0.18)) {
   412→      if (this.jumpsUsed < 2) {
   413→        this.doJump(w)
   414→        return false
   415→      }
   416→    }
   417→    // parry
   418→    if (input.pressed('parry')) {
   419→      this.enterParry(w)
   420→      return true
   421→    }
   422→    // ability
   423→    if (input.pressed('ability')) {
   424→      if (this.energy >= this.sword.ability.cost) {
   425→        this.startAbility(w)
   426→        return true
   427→      }
   428→    }
   429→    // ult is handled by Game (cinematic)
   430→
   431→    // heavy: charge vs tap
   432→    if (input.down('heavy') && this.grounded) {
   433→      this.enterCharge(w)
   434→      return true
   435→    }
   436→    if (input.pressed('heavy') || input.buffered('heavy')) {
   437→      input.takeBuffered('heavy')
   438→      this.startAttack(this.pickHeavy(w), w)
   439→      return true
   440→    }
   441→    // light
   442→    if (input.pressed('light') || input.buffered('light')) {
   443→      input.takeBuffered('light')
   444→      const id = this.riposteT > 0 ? 'riposte' : this.pickLight(w, ax, ay)
   445→      this.startAttack(ATTACKS[id], w)
   446→      return true
   447→    }
   448→    return false
   449→  }

   451→  private doJump(w: WorldRef): void {
   452→    w.input.takeBuffered('jump')
   453→    this.jumpsUsed++
   454→    this.vy = this.jumpsUsed === 1 ? JUMP_V : AIR_JUMP_V

   456→    w.audio.play('jump', { volume: 0.4 })

   458→    if (this.jumpsUsed === 2) {
   459→      w.fx.burst(this.x, this.y - 20, 6, '#9fb4c4', 160, 0.4, 'dot')
   460→    }
   461→  }

   467→  private canDash(): boolean {
   468→    if (this.dashCd > 0 || this.dashRecharge > 0) return false
   469→    if (!this.grounded && this.airDashUsed) return false
   470→    return true
   471→  }
   472→
   473→  startDash(w: WorldRef, ax: number, ay: number): void {
   474→    w.input.takeBuffered('dash')
   475→    let dx = ax
   476→    let dy = ay
   477→    if (dx === 0 && dy === 0) dx = this.face
   478→    const len = Math.hypot(dx, dy) || 1
   479→    dx /= len
   480→    dy /= len
   481→    if (dx !== 0) this.face = dx > 0 ? 1 : -1
   482→    this.dashDx = dx
   483→    this.dashDy = dy
   484→    this.state = 'dash'
   485→    this.dashT = DASH_TIME
   486→    this.dashCd = DASH_CD
   487→    this.dashConsec++
   488→    if (this.dashConsec >= 2) this.dashRecharge = DASH_RECHARGE
   489→    if (!this.grounded) this.airDashUsed = true
   490→    this.iframes = Math.max(this.iframes, 0.12)
   491→    this.action = null
   492→    this.charging = false
   493→    this.vx = dx * DASH_SPEED
   494→    this.vy = dy * DASH_SPEED * (this.grounded && dy < -0.4 ? 0.55 : 1)
   495→    this.animator.play(this.grounded ? DASH_CLIP : DASH_AIR_CLIP, { blend: 0.05 })
   496→    w.audio.play('dash', { pitch: rand(0.95, 1.1) })
   497→    const colors = this.sword.colors
   498→    w.fx.speedStreaks(this.x, this.y - 40, dx * DASH_SPEED * 0.9, 6, rgba(colors.trail, 1))
   499→    w.fx.dust(this.x, this.grounded ? 0 : this.y, 5)
   500→    w.cam.addTrauma(0.06)
   501→  }
   502→
   503→  private updateDash(dt: number, w: WorldRef): void {
   504→    this.dashT -= dt
   505→    // afterimages
   506→    this.afterimgTimer -= dt
   507→    if (this.afterimgTimer <= 0) {
   508→      this.afterimgTimer = 0.026
   509→      w.fx.afterimage(this.pose, this.x, this.hipY, this.face, this.sword.colors.trail, 0.3, SCALE)
   510→    }
   511→    // maintain velocity with slight decay
   512→    const k = 1 - this.dashT / DASH_TIME
   513→    const sp = DASH_SPEED * (1 - k * 0.25)
   514→    this.vx = this.dashDx * sp
   515→    this.vy = this.dashDy * sp * (this.grounded ? (this.dashDy < 0 ? 0.4 : 1) : 1)
   516→
   517→    // dash attack
   518→    const input = w.input
   519→    if (input.pressed('light') || input.buffered('light')) {
   520→      input.takeBuffered('light')
   521→      this.startAttack(ATTACKS.dashSlash, w)
   522→      return
   523→    }
   524→    if (input.pressed('heavy') || input.buffered('heavy')) {
   525→      if (!this.grounded) {
   526→        input.takeBuffered('heavy')
   527→        this.startAttack(ATTACKS.slam, w)
   528→        return
   529→      }
   530→      input.takeBuffered('heavy')
   531→      this.startAttack(ATTACKS.dashHeavy, w)
   532→      return
   533→    }
   534→    if (this.dashT <= 0) {
   535→      this.state = 'move'
   536→      this.vx *= 0.55
   537→      if (!this.grounded) this.vy *= 0.5
   538→    }
   539→  }

   545→  private pickLight(w: WorldRef, ax: number, ay: number): string {
   546→    // contextual branches
   547→    if (ay < -0.4 && this.grounded) return 'launcher'
   548→    if (ay > 0.4 && this.grounded) return 'spin'
   549→    if (this.grounded && ax !== 0 && Math.sign(ax) !== this.face) return 'retreat'
   550→    if (!this.grounded) {
   551→      if (ay > 0.4) return 'dive'
   552→      this.airIndex = (this.airIndex % 3) + 1
   553→      return `a${this.airIndex}`
   554→    }
   555→    // ground light chain (with time decay)
   556→    if (w.time - this.lastLightAt > 1.1) this.lightIndex = 0
   557→    this.lightIndex = Math.min(this.lightIndex + 1, 5)
   558→    return `l${Math.min(this.lightIndex, 5)}`
   559→  }

   565→  startAttack(def: AttackDef, w: WorldRef, fromParry = false): void {
   566→    if (!def) return
   567→    // IN-BETWEEN LAW: capture the source state BEFORE switching — chained
   568→    // attacks crossfade slower (0.11s) than fresh entries (0.07s) so every
   569→    // transition between animation instances gets real in-betweens.
   570→    const wasAttack = this.state === 'attack'
   571→    // requirement checks
   572→    if (def.requires === 'ground' && !this.grounded) def = ATTACKS.a1
   573→    if (def.requires === 'air' && this.grounded) def = ATTACKS.l1
   574→
   575→    // face toward held direction at attack start
   576→    const ax = w.input.axisX()
   577→    if (ax !== 0 && !fromParry) this.face = ax > 0 ? 1 : -1
   578→
   579→    const trail = w.fx.startTrail(this.sword.colors.trail, this.sword.colors.trail2, this.sword.trailWidth)
   580→    this.hasPrevBlade = false
   581→    this.action = {
   582→      def,
   583→      animT: 0,
   584→      hitMap: new Map(),
   585→      trail,
   586→      sfxDone: false,
   587→      moveDone: false,
   588→      moveYDone: false,
   589→      impactDone: false,
   590→      fromParry,
   591→    }
   592→    if (fromParry) this.riposteT = 0
   593→    this.state = 'attack'
   594→    this.charging = false
   595→    if (def === ATTACKS.l1 || def === ATTACKS.l2 || def === ATTACKS.l3 || def === ATTACKS.l4 || def === ATTACKS.l5) {
   596→      this.lastLightAt = w.time
   597→    }
   598→    this.animator.play(def.clip, { blend: wasAttack ? 0.11 : 0.07, speed: this.atkSpeed })
   599→    // slight ground friction during attacks
   600→  }

   602→  private updateAttack(dt: number, w: WorldRef): void {
   603→    const a = this.action
   604→    if (!a) {
   605→      this.state = 'move'
   606→      return
   607→    }
   608→    const def = a.def
   609→    a.animT = this.animator.getTime()
   610→    const animT = a.animT
   611→    const spd = this.atkSpeed
   612→
   613→    // ground friction / air drag during attack
   614→    if (this.grounded) {
   615→      const f = 900 * dt
   616→      this.vx = Math.abs(this.vx) <= f ? 0 : this.vx - Math.sign(this.vx) * f
   617→    }
   618→
   619→    // sfx event
   620→    if (!a.sfxDone && animT >= def.swingSfx.t) {
   621→      a.sfxDone = true
   622→      w.audio.play(def.swingSfx.name, { pitch: rand(0.94, 1.08) })
   623→    }
   624→    // forward impulse
   625→    if (!a.moveDone && def.moveX !== undefined && animT >= (def.moveXAt ?? def.hit.start)) {
   626→      a.moveDone = true
   627→      this.vx = this.face * def.moveX * (def.kind === 'dash' ? 1 : 1)
   628→    }
   629→    // vertical impulse (launcher hop / slam plunge)
   630→    if (!a.moveYDone && def.moveY !== undefined && animT >= (def.moveYAt ?? 0)) {
   631→      a.moveYDone = true
   632→      this.vy = def.moveY
   633→      if (def.moveY < 0) {
   634→        this.grounded = false
   635→        w.fx.dust(this.x, 0, 6)
   636→      }
   637→    }
   638→
   639→    // hit window: sample blade segment & test enemies (swept, anti-tunneling)
   640→    const h = def.hit
   641→    if (animT >= h.start && animT <= h.end) {
   642→      this.swingGlow = 1
   643→      const info = makeRuntimeInfo(def, this.sword.stats.dmg)
   644→      for (const e of w.enemies) {
   645→        if (e.dead || e.spawning) continue
   646→        const last = a.hitMap.get(e.id)
   647→        const reArm = h.multiHit ? (last === undefined || animT - last >= h.multiHit) : last === undefined
   648→        if (!reArm) continue
   649→        if (bladeHitsBody(e, this.hasPrevBlade, this.prevBlade, this.blade)) {
   650→          a.hitMap.set(e.id, animT)
   651→          w.hitEnemy(e.id, this.blade, info)
   652→          this.onLandedHit(w, info, e.x, e.y - 48 * e.scale)
   653→        }
   654→      }
   655→      // remember this frame's blade for the next sweep test
   656→      this.prevBlade.bx = this.blade.bx
   657→      this.prevBlade.by = this.blade.by
   658→      this.prevBlade.tx = this.blade.tx
   659→      this.prevBlade.ty = this.blade.ty
   660→      this.hasPrevBlade = true
   661→      // aoe shockwave at strike moment for heavy hits
   662→      if (h.aoe && h.aoe > 0 && !a.impactDone && this.grounded && animT >= h.start + 0.02) {
   663→        a.impactDone = true
   664→        w.fx.shockwave(this.x + this.face * 30, 0, h.aoe, this.sword.colors.glow, 0.4, 6)
   665→        if (this.grounded) w.fx.groundCrack(this.x + this.face * 34, 0, h.aoe * 0.45)
   666→      }
   667→    }
   668→
   669→    // trail emission
   670→    if (animT >= def.trail.start && animT <= def.trail.end) {
   671→      w.fx.pushTrail(a.trail, this.blade.bx, this.blade.by, this.blade.tx, this.blade.ty)
   672→    } else if (animT > def.trail.end && a.trail.emitting) {
   673→      w.fx.endTrail(a.trail)
   674→    }
   675→
   676→    // cancel window → allow chaining / dash-cancel / jump-cancel
   677→    const dashCancel = animT >= h.end * 0.6
   678→    if (animT >= def.cancel || dashCancel) {
   679→      const input = w.input
   680→      if ((input.pressed('dash') || input.buffered('dash')) && this.canDash()) {
   681→        w.fx.endTrail(a.trail)
   682→        this.startDash(w, input.axisX(), input.axisY())
   683→        return
   684→      }
   685→      if (input.pressed('parry')) {
   686→        w.fx.endTrail(a.trail)
   687→        this.enterParry(w)
   688→        return
   689→      }
   690→      if (animT >= def.cancel) {
   691→        // chain next attack
   692→        const ax = input.axisX()
   693→        const ay = input.axisY()
   694→        let next: string | null = null
   695→        if (input.buffered('light')) {
   696→          input.takeBuffered('light')
   697→          next = this.riposteT > 0 && this.grounded ? 'riposte' : this.pickLight(w, ax, ay)
   698→        } else if (input.buffered('heavy')) {
   699→          input.takeBuffered('heavy')
   700→          if (!this.grounded) {
   701→            next = 'slam'
   702→          } else if (input.down('heavy')) {
   703→            // still holding → convert to charge
   704→            this.enterCharge(w)
   705→            w.fx.endTrail(a.trail)
   706→            return
   707→          } else {
   708→            next = def.chain.find((c) => c.startsWith('h')) ?? 'h1'
   709→          }
   710→        }
   711→        if (next) {
   712→          // validate chain (or allow from-parry riposte)
   713→          if (fromChain(def, next, this)) {
   714→            w.fx.endTrail(a.trail)
   715→            this.startAttack(ATTACKS[next], w)
   716→            return
   717→          }
   718→        }
   719→        // jump cancel on ground
   720→        if (this.grounded && (input.pressed('jump') || input.buffered('jump', 0.15))) {
   721→          w.fx.endTrail(a.trail)
   722→          this.doJump(w)
   723→          this.state = 'move'
   724→          return
   725→        }
   726→      }
   727→    }
   728→
   729→    // finish
   730→    if (this.animator.isDone()) {
   731→      w.fx.endTrail(a.trail, 0.18)
   732→      this.action = null
   733→      this.state = 'move'
   734→    }
   735→  }

   737→  private onLandedHit(w: WorldRef, info: AttackRuntimeInfo, ex: number, ey: number): void {
   738→    this.energy = Math.min(this.maxEnergy, this.energy + 5)

   740→    // elemental on-hit flourishes

   743→    switch (this.sword.id) {
   744→      case 'inferno':
   745→        w.fx.embers(ex, ey, 5, c.accent, 18)
   746→        break
   747→      case 'thunder':
   748→        if (Math.random() < 0.3) w.fx.lightningArc(ex, ey, ex + rand(-40, 40), ey + rand(-30, 30), c.trail)
   749→        break
   750→      case 'void':
   751→        w.fx.burst(ex, ey, 4, c.core, 120, 0.4, 'shard')
   752→        break
   753→      case 'gravity':

   755→        break
   756→      case 'phantom':
   757→        w.fx.slashArc(ex, ey, dir + rand(-0.5, 0.5), 1.4, 30, c.trail, c.trail2)
   758→        break
   759→      default:
   760→        break
   761→    }
   762→  }

  1048→  private updateAbility(dt: number, w: WorldRef): void {
  1049→    const ab = this.ability
  1050→    if (!ab) return
  1051→    ab.t += dt
  1052→    if (this.exitScriptedSoon > 0) {
  1053→      this.exitScriptedSoon -= dt
  1054→      if (this.exitScriptedSoon <= 0) {
  1055→        this.exitScripted()

  1057→      }
  1058→    }
  1059→
  1060→    const c = this.sword.colors
  1061→
  1062→    switch (ab.kind) {
  1063→      case 'katana': {
  1064→        // storm: dash-slash through a new enemy every 0.13s, 6 steps
  1065→        const interval = 0.13
  1066→        if (ab.t >= ab.step * interval && ab.step < 6) {
  1067→          ab.step++
  1068→          const targets = this.collectNearest(w, 6, 520).filter((e) => !ab.targets.includes(e.id))
  1069→          const tgt = targets[0]
  1070→          if (tgt) {
  1071→            ab.targets.push(tgt.id)
  1072→            w.fx.afterimage(this.pose, this.x, this.hipY, this.face, c.trail, 0.35, SCALE)
  1073→            w.fx.speedStreaks(this.x, this.y - 40, tgt.x - this.x, 6, rgba(c.trail, 1))
  1074→            this.x = clamp(tgt.x + (tgt.x > this.x ? -54 : 54), w.arenaMinX + 30, w.arenaMaxX - 30)
  1075→            this.face = tgt.x > this.x ? 1 : -1
  1076→            const ang = Math.atan2(tgt.y - 40 - (this.y - 55), tgt.x - this.x)
  1077→            w.fx.slashArc(tgt.x, tgt.y - 40, ang, 2.6, 66, c.trail, c.trail2)
  1078→            w.fx.impact(tgt.x, tgt.y - 40, ang, false, c.glow, c.accent)
  1079→            this.dealAbilityDamage(w, tgt, 16, 0.3, 140, -220)
  1080→            w.audio.play('swing', { pitch: rand(1.1, 1.3), volume: 0.7 })
  1081→            this.animator.play(ATTACKS.dashSlash.clip, { blend: 0.02, speed: 1.4 })
  1082→            w.cam.addTrauma(0.12)
  1083→            w.addHitstop(0.03)
  1084→          }
  1085→        }
  1086→        if (ab.step >= 6 && ab.t > 6 * interval + 0.2) {
  1087→          this.ability = null
  1088→          this.exitScripted()
  1089→        }
  1090→        break
  1091→      }
  1092→      case 'gravity': {
  1093→        // suspend hold, then crush at t=0.55
  1094→        if (ab.step === 0 && ab.t >= 0.55) {
  1095→          ab.step = 1
  1096→          w.audio.play('slam', { intensity: 1 })
  1097→          w.fx.shockwave(this.x, 0, 220, c.glow, 0.5, 10)
  1098→          w.fx.groundCrack(this.x, 0, 130)
  1099→          w.addHitstop(0.1)
  1100→          w.cam.addTrauma(0.5)
  1101→          for (const e of w.enemies) {
  1102→            if (e.dead || !ab.targets.includes(e.id)) continue
  1103→            if (Math.abs(e.x - this.x) < 240) {
  1104→              e.vy = 520
  1105→              this.dealAbilityDamage(w, e, 30, 0.5, 0, 0)
  1106→            }
  1107→          }
  1108→          this.animator.play(ATTACKS.h2.clip, { blend: 0.03, speed: 1.25 })
  1109→        }
  1110→        if (ab.t >= 1.0) {
  1111→          this.ability = null
  1112→          this.exitScripted()
  1113→        }
  1114→        break
  1115→      }
  1116→      case 'phantom': {
  1117→        // echoes every 0.2s
  1118→        const interval = 0.2
  1119→        if (ab.t >= ab.step * interval && ab.step < ab.targets.length) {
  1120→          const eid = ab.targets[ab.step]
  1121→          ab.step++
  1122→          const e = w.enemies.find((en) => en.id === eid && !en.dead)
  1123→          if (e) {
  1124→            const ang = rand(TAU_LIKE)
  1125→            w.fx.slashArc(e.x, e.y - 40, ang, 2.4, 60, c.trail, c.trail2, 1.2)
  1126→            w.fx.afterimage(this.pose, e.x - this.face * 60, this.hipY, this.face, c.trail, 0.4, SCALE)
  1127→            w.fx.impact(e.x, e.y - 40, ang, false, c.glow, c.accent)
  1128→            this.dealAbilityDamage(w, e, 18, 0.34, 120, -160)
  1129→            w.audio.play('swing', { pitch: rand(0.9, 1.1), volume: 0.6 })
  1130→          }
  1131→        }
  1132→        if (ab.step >= Math.max(ab.targets.length, 1) && ab.t > ab.targets.length * interval + 0.25) {
  1133→          this.ability = null
  1134→          this.exitScripted()
  1135→        }
  1136→        break
  1137→      }
  1138→      default:
  1139→        if (ab.t > 1.2) {
  1140→          this.ability = null
  1141→        }
  1142→    }
  1143→  }

  1145→  private collectNearest(w: WorldRef, n: number, range: number): EnemyLike[] {
  1146→    const list: EnemyLike[] = []
  1147→    for (const e of w.enemies) {
  1148→      if (e.dead || e.spawning) continue
  1149→      const d = Math.hypot(e.x - this.x, e.y - this.y)
  1150→      if (d < range) list.push({ id: e.id, x: e.x, y: e.y, d })
  1151→    }
  1152→    list.sort((a, b) => (a.d ?? 0) - (b.d ?? 0))
  1153→    return list.slice(0, n)
  1154→  }

  1156→  private findDensest(w: WorldRef, range: number): EnemyLike | null {
  1157→    let best: EnemyLike | null = null
  1158→    let bestScore = -1
  1159→    for (const e of w.enemies) {
  1160→      if (e.dead || e.spawning) continue
  1161→      let score = 0
  1162→      for (const o of w.enemies) {
  1163→        if (o.dead || o.spawning) continue
  1164→        if (Math.abs(o.x - e.x) < range) score++
  1165→      }
  1166→      if (score > bestScore) {
  1167→        bestScore = score
  1168→        best = { id: e.id, x: e.x, y: e.y, d: 0 }
  1169→      }
  1170→    }
  1171→    return best
  1172→  }

  1189→  private applyProcedural(dt: number): void {
  1190→    const p = this.pose
  1191→    // velocity lean (proportional, eased)
  1192→    const speedLean = clamp((this.vx * this.face) / MOVE_MAX, -1, 1)
  1193→    p[2] += speedLean * 0.14
  1194→    // vertical pitch while airborne
  1195→    if (!this.grounded && this.state !== 'attack') {
  1196→      p[2] += clamp(this.vy / 1400, -0.22, 0.3)
  1197→    }
  1198→    // landing spring (squash)
  1199→    if (this.landSpring > 0) {
  1200→      this.landSpring = Math.max(0, this.landSpring - dt * 5)
  1201→      p[1] += this.landSpring * 6
  1202→    }
  1203→    // charge tremble
  1204→    if (this.charging) {
  1205→      p[0] += rand(-0.8, 0.8) * this.chargeRatio
  1206→      p[1] += rand(-0.5, 0.5) * this.chargeRatio
  1207→    }
  1208→    // hurt shake
  1209→    if (this.hurtT > 0) {
  1210→      p[0] += rand(-1.4, 1.4) * this.hurtT
  1211→    }
  1212→  }

  1330→function fromChain(def: AttackDef, next: string, player: Player): boolean {
  1331→  if (next === 'riposte') return true
  1332→  // launcher chains naturally into air attacks
  1333→  if (def.id === 'launcher' && (next.startsWith('a') || next === 'dive' || next === 'slam')) return true
  1334→  if (def.chain.includes(next)) return true
  1335→  // any light may restart the chain if the window allows
  1336→  if (player.grounded && next.startsWith('l') && def.chain.some((c) => c.startsWith('l'))) return true
  1337→  if (!player.grounded && (next.startsWith('a') || next === 'dive' || next === 'slam') && def.chain.some((c) => c.startsWith('a') || c === 'dive' || c === 'slam')) return true
  1338→  return false
  1339→}
