    26→/** swaps left/right LEG channels only — the arms keep their authored
    27→ * channels (the sword stays in the right hand across both strides). */
    28→function mirrorLegs(src: PoseInput): PoseInput {
    29→  const out: PoseInput = { ...src }
    30→  const pairs: [keyof PoseInput, keyof PoseInput][] = [
    31→    ['hipL', 'hipR'], ['kneeL', 'kneeR'], ['footL', 'footR'],
    32→  ]
    33→  for (const [l, r] of pairs) {
    34→    const lv = src[l]
    35→    const rv = src[r]
    36→    if (lv !== undefined) out[r] = lv
    37→    if (rv !== undefined) out[l] = rv
    38→  }
    39→  return out
    40→}
    41→
    42→/** STRIDE MIRROR for bipedal cycles: legs swap, the SWORD ARM (right)
    43→ * keeps the exact same carry across both strides (a held weapon does not
    44→ * teleport between arms / change carry mid-cycle), and the off-arm
    45→ * negates its swing so the contralateral counter-phase survives.
    46→ * DEPTH channels mirror with the stride: pelvis/chest yaw negate (the
    47→ * opposite hip now drives), the off-arm's depth swing negates — while the
    48→ * sword-side channels (armRz, bladeYaw) carry unchanged, like the arm. */
    49→function mirrorStride(src: PoseInput): PoseInput {
    50→  const out = mirrorLegs(src)
    51→  if (src.shL !== undefined) out.shL = -src.shL
    52→  if (src.pelvisYaw !== undefined) out.pelvisYaw = -src.pelvisYaw
    53→  if (src.chestYaw !== undefined) out.chestYaw = -src.chestYaw
    54→  if (src.armLz !== undefined) out.armLz = -src.armLz
    55→  // the off FOREARM's own depth yaw negates WITH its upper arm — leaving
    56→  // it un-mirrored desyncs the elbow's depth plane on stride B (the
    57→  // "forearm flips against the upper arm" depth incoherence)
    58→  if (src.foreLz !== undefined) out.foreLz = -src.foreLz
    59→  if (src.armRz !== undefined) out.armRz = src.armRz
    60→  if (src.bladeYaw !== undefined) out.bladeYaw = src.bladeYaw
    61→  return out
    62→}
