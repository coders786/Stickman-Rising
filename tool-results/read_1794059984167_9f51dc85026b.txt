     1→/* ============================================================================
     2→ * STICKMAN RISING — src/game/world/background.ts
     3→ * Self-contained parallax cyberpunk-city backdrop (rooftop-at-night stage).
     4→ * ----------------------------------------------------------------------------
     5→ * Contract with the caller (everything in CSS pixels — the caller owns any
     6→ * devicePixelRatio scaling of the context):
     7→ *   drawSky(ctx, cam)      screen-space pass, full viewport, drawn first.
     8→ *   drawArena(ctx)         world-space pass — caller ALREADY applied
     9→ *                          translate(w/2,h/2) · scale(zoom) · translate(-cam).
    10→ *   drawWeather(ctx, cam)  screen-space pass drawn after world entities.
    11→ * World conventions: ground plane is y = 0 (entities live at y <= 0), the
    12→ * arena floor spans world x in [-150, 2750], the facade fades to black by
    13→ * y ≈ +260, the camera hovers around y = -170 with zoom 0.85..1.3.
    14→ *
    15→ * Engineering notes:
    16→ *  · All layout comes from fixed-seed mulberry32 RNGs → identical art on
    17→ *    every run and every resize.
    18→ *  · Every heavy layer is pre-rendered once into an offscreen canvas whose
    19→ *    content is horizontally periodic (period 2048 layer-px); per-frame cost
    20→ *    is a handful of drawImage blits plus tiny animated overlays. Tiles live
    21→ *    in a resolution-independent layer space, so resize() only records the
    22→ *    new viewport size — a rebuild would only ever be needed at resize time
    23→ *    and would be pixel-identical, so none is performed.
    24→ *  · Zero heap allocations in update()/draw*(): particle state lives in
    25→ *    preallocated typed arrays, style strings/gradients are precomputed and
    26→ *    animated overlays only blit pre-rendered sprites.
    27→ *  · update()/draw*() are safe before resize() (defaults 1280x720).
    28→ * ==========================================================================*/
    29→

    41→/* ============================ seeded RNG ================================= */
    42→
    43→/** mulberry32 — small, fast, deterministic 32-bit PRNG. */
    44→function mulberry32(seed: number): () => number {
    45→  let a = seed >>> 0;
    46→  return (): number => {
    47→    a = (a + 0x6d2b79f5) >>> 0;
    48→    let t = a;
    49→    t = Math.imul(t ^ (t >>> 15), t | 1);
    50→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    51→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    52→  };
    53→}

    77→function makeCanvas(w: number, h: number): { c: HTMLCanvasElement; g: CanvasRenderingContext2D } {
    78→  const c = document.createElement('canvas');
    79→  c.width = w;
    80→  c.height = h;
    81→  const g = c.getContext('2d');
    82→  if (!g) throw new Error('2D canvas context unavailable');
    83→  return { c, g };
    84→}

    86→/** Rounded-rect subpath (no dependency on ctx.roundRect). */
    87→function rrPath(g: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number): void {
    88→  const k = Math.min(r, w * 0.5, h * 0.5);
    89→  g.beginPath();
    90→  g.moveTo(x + k, y);
    91→  g.arcTo(x + w, y, x + w, y + h, k);
    92→  g.arcTo(x + w, y + h, x, y + h, k);
    93→  g.arcTo(x, y + h, x, y, k);
    94→  g.arcTo(x, y, x + w, y, k);
    95→  g.closePath();
    96→}

   195→/** soft wispy fog band, horizontally seamless (blobs wrap around) */
   196→function buildFogSprite(): HTMLCanvasElement {
   197→  const { c, g } = makeCanvas(FOG_W, FOG_H);
   198→  const rng = mulberry32(0xf06b17);
   199→  for (let i = 0; i < 9; i++) {
   200→    const cx = rng() * FOG_W;
   201→    const cy = 100 + (rng() - 0.5) * 46;
   202→    const r = 70 + rng() * 100;
   203→    const a = 0.08 + rng() * 0.09;
   204→    for (let k = -1; k <= 1; k++) {
   205→      const x = cx + k * FOG_W;
   206→      const gr = g.createRadialGradient(x, cy, 0, x, cy, r);
   207→      gr.addColorStop(0, `rgba(128,158,168,${a.toFixed(3)})`);
   208→      gr.addColorStop(1, 'rgba(128,158,168,0)');
   209→      g.fillStyle = gr;
   210→      g.beginPath(); g.arc(x, cy, r, 0, TAU); g.fill();
   211→    }
   212→  }
   213→  return c;
   214→}

   254→/** two puddle decals (magenta / cyan neon reflections), stretched per decal */
   255→function buildPuddleSprites(): HTMLCanvasElement[] {
   256→  const cols: ReadonlyArray<string> = ['255,45,149', '41,230,210'];
   257→  const out: HTMLCanvasElement[] = [];
   258→  for (let ci = 0; ci < 2; ci++) {
   259→    const { c, g } = makeCanvas(96, 16);
   260→    g.save();
   261→    g.translate(48, 8);
   262→    g.scale(1, 0.136);                                     // squash to ellipse
   263→    let gr = g.createRadialGradient(0, 0, 0, 0, 0, 46);
   264→    gr.addColorStop(0, `rgba(${cols[ci]},0.3)`);
   265→    gr.addColorStop(0.55, `rgba(${cols[ci]},0.16)`);
   266→    gr.addColorStop(1, `rgba(${cols[ci]},0)`);
   267→    g.fillStyle = gr;
   268→    g.beginPath(); g.arc(0, 0, 46, 0, TAU); g.fill();
   269→    gr = g.createLinearGradient(-30, 0, 30, 0);            // bright streak
   270→    gr.addColorStop(0, `rgba(${cols[ci]},0)`);
   271→    gr.addColorStop(0.5, `rgba(${cols[ci]},0.55)`);
   272→    gr.addColorStop(1, `rgba(${cols[ci]},0)`);
   273→    g.fillStyle = gr;
   274→    g.beginPath(); g.arc(0, 0, 30, 0, TAU); g.fill();
   275→    g.restore();
   276→    out.push(c);
   277→  }
   278→  return out;
   279→}

   281→/** abstract sci-fi glyph panel for one neon sign (no real letters) */
   282→function buildGlyph(ci: number): HTMLCanvasElement {
   283→  const { c, g } = makeCanvas(128, 48);
   284→  const rng = mulberry32(0x9a7c00 + ci * 7919);
   285→  const ink = SIGN_INK[ci];
   286→  const st: GlyphStroke[] = [];
   287→  for (let cell = 0; cell < 5; cell++) {
   288→    const bx = 8 + cell * 24;
   289→    const n = 3 + ((rng() * 3) | 0);
   290→    for (let s = 0; s < n; s++) {
   291→      const t = rng();
   292→      if (t < 0.34) {                                       // vertical bar
   293→        const x = bx + 3 + rng() * 15, y = 10 + rng() * 8;
   294→        st.push({ x0: x, y0: y, x1: x, y1: y + 8 + rng() * 18, dot: false });
   295→      } else if (t < 0.58) {                                // horizontal bar
   296→        const y = 10 + rng() * 24, x = bx + 2 + rng() * 7;
   297→        st.push({ x0: x, y0: y, x1: x + 8 + rng() * 11, y1: y, dot: false });
   298→      } else if (t < 0.82) {                                // diagonal slash
   299→        const x = bx + 2 + rng() * 9, y = 10 + rng() * 18;
   300→        const d = (rng() < 0.5 ? -1 : 1) * (5 + rng() * 9);
   301→        st.push({ x0: x, y0: y, x1: x + d, y1: y + 6 + rng() * 11, dot: false });
   302→      } else {                                              // dot
   303→        st.push({ x0: bx + 4 + rng() * 15, y0: 11 + rng() * 24, x1: 0, y1: 0, dot: true });
   304→      }
   305→    }
   306→  }
   307→  st.push({ x0: 7, y0: 43, x1: 114, y1: 43, dot: false });  // underline rail
   308→  st.push({ x0: 120, y0: 43, x1: 0, y1: 0, dot: true });    // end pip
   309→  const pass = (lw: number, stroke: string, blur: number, r: number): void => {
   310→    g.lineWidth = lw;
   311→    g.lineCap = 'round';
   312→    g.lineJoin = 'round';
   313→    g.strokeStyle = stroke;
   314→    g.fillStyle = stroke;
   315→    g.shadowColor = ink;
   316→    g.shadowBlur = blur;
   317→    for (let i = 0; i < st.length; i++) {
   318→      const s = st[i];
   319→      if (s.dot) { g.beginPath(); g.arc(s.x0, s.y0, r, 0, TAU); g.fill(); }
   320→      else { g.beginPath(); g.moveTo(s.x0, s.y0); g.lineTo(s.x1, s.y1); g.stroke(); }
   321→    }
   322→  };
   323→  pass(3.4, ink, 7, 1.8);                                   // soft neon glow
   324→  pass(1.3, SIGN_CORE[ci], 0, 1.1);                         // bright tube core
   325→  g.shadowBlur = 0;
   326→  return c;
   327→}

   384→  constructor(quality: Quality) {
   385→    this.rainCap = quality === 'high' ? 130 : quality === 'medium' ? 70 : 0;
   386→    this.moteCap = quality === 'high' ? 20 : quality === 'medium' ? 10 : 0;
   387→    this.fogCount = quality === 'low' ? 2 : 3;
   388→    const nRain = this.rainCap, nMote = this.moteCap;
   389→    this.rainX = new Float32Array(nRain);
   390→    this.rainY = new Float32Array(nRain);
   391→    this.rainL = new Float32Array(nRain);
   392→    this.rainS = new Float32Array(nRain);
   393→    this.rainB = new Uint8Array(nRain);
   394→    this.moteX = new Float32Array(nMote);
   395→    this.moteY = new Float32Array(nMote);
   396→    this.moteP = new Float32Array(nMote);
   397→    this.moteB = new Uint8Array(nMote);
   398→    // deterministic first scatter for rain / motes / ripples
   399→    const rng = mulberry32(0x5170394);
   400→    for (let i = 0; i < nRain; i++) {
   401→      this.rainX[i] = rng() * 1360 - 40;
   402→      this.rainY[i] = rng() * 800 - 40;
   403→      this.rainL[i] = 10 + rng() * 16;
   404→      this.rainS[i] = 480 + rng() * 300;
   405→      this.rainB[i] = (rng() * 8) | 0;
   406→    }
   407→    for (let i = 0; i < nMote; i++) {
   408→      this.moteX[i] = rng() * 1320 - 20;
   409→      this.moteY[i] = rng() * 760 - 20;
   410→      this.moteP[i] = rng() * TAU;
   411→      this.moteB[i] = (rng() * 4) | 0;
   412→    }

   414→      this.ripples.push({ wx: 140 + rng() * 2470, oy: 3 + rng() * 9, ph: rng(), sp: 0.45 + rng() * 0.4, r: 6 + rng() * 8 });

   416→    this.ltnNext = 4 + rng() * 6;

   419→  /** CSS-pixel size of the game canvas viewport. */
   420→  resize(w: number, h: number): void {
   421→    this.vw = w > 0 ? w : 1;
   422→    this.vh = h > 0 ? h : 1;
   423→    // All art is pre-rendered in a fixed layer space from a fixed seed, so a
   424→    // resize only changes the viewport dims — the layout stays identical and
   425→    // no tile ever needs re-rendering.
   426→  }
   427→
   428→  /** advance ambient animation; dt in seconds (clamped internally). */
   429→  update(dt: number): void {
   430→    const d = dt > 0 ? (dt < 0.1 ? dt : 0.1) : 0;
   431→    this.t += d;
   432→    const t = this.t;
   433→    this.slant = 0.72 + 0.16 * Math.sin(t * 0.09) + 0.08 * Math.sin(t * 0.037 + 2.1);
   434→
   435→    /* neon-sign buzz / brown-out / re-ignite state machines */
   436→    for (let i = 0; i < this.signs.length; i++) {
   437→      const s = this.signs[i];
   438→      if (s.state === 0) {
   439→        s.next -= d;
   440→        if (s.next <= 0) { s.state = 1; s.timer = 0.45 + Math.random() * 1.05; }
   441→      } else if (s.state === 1) {
   442→        s.timer -= d;
   443→        if (s.timer <= 0) { s.state = 2; s.timer = 0.4; }
   444→      } else {
   445→        s.timer -= d;
   446→        if (s.timer <= 0) { s.state = 0; s.next = 5 + Math.random() * 9; }
   447→      }
   448→    }
   449→
   450→    /* puddle shimmer envelopes (consumed by drawArena) */
   451→    for (let i = 0; i < this.puddles.length; i++) {
   452→      const p = this.puddles[i];
   453→      this.puddleA[i] = 0.5 + 0.28 * Math.sin(t * p.sp + p.ph) + 0.1 * Math.sin(t * p.sp * 2.7 + p.ph * 2);
   454→    }
   455→
   456→    /* rain + motes (in-place wrap, no allocations) */
   457→    if (this.rainCap > 0) {
   458→      const w = this.vw, h = this.vh, sl = this.slant;
   459→      for (let i = 0; i < this.rainCap; i++) {
   460→        let y = this.rainY[i] + this.rainS[i] * d;
   461→        let x = this.rainX[i] + this.rainS[i] * sl * d;
   462→        if (y > h + 40) { y = -30 - Math.random() * 90; x = Math.random() * (w + 80) - 40; }
   463→        if (x > w + 40) x -= w + 80;
   464→        else if (x < -40) x += w + 80;
   465→        this.rainX[i] = x;
   466→        this.rainY[i] = y;
   467→      }
   468→    }
   469→    if (this.moteCap > 0) {
   470→      for (let i = 0; i < this.moteCap; i++) {
   471→        const ph = this.moteP[i];
   472→        let x = this.moteX[i] + (5 + 3 * Math.sin(t * 0.05 + ph)) * d;
   473→        let y = this.moteY[i] + Math.sin(t * 0.21 + ph) * 5 * d;
   474→        if (x > this.vw + 20) x -= this.vw + 40;
   475→        if (y > this.vh + 20) y -= this.vh + 40;
   476→        else if (y < -20) y += this.vh + 40;
   477→        this.moteX[i] = x;
   478→        this.moteY[i] = y;
   479→      }
   480→    }
   481→
   482→    /* distant lightning scheduler (8..15 s cadence) */
   483→    if (this.ltnT >= 0) {
   484→      this.ltnT += d;
   485→      if (this.ltnT > 1) this.ltnT = -1;
   486→    } else {
   487→      this.ltnNext -= d;
   488→      if (this.ltnNext <= 0) { this.ltnT = 0; this.ltnNext = 8 + Math.random() * 7; }
   489→    }
   490→  }

   492→  /* ------------------------ screen-space sky pass ------------------------ */
   493→
   494→  drawSky(ctx: CanvasRenderingContext2D, cam: BgCam): void {
   495→    this.ensureBuilt();
   496→    const cx = Number.isFinite(cam.x) ? cam.x : 0;
   497→    const cy = Number.isFinite(cam.y) ? cam.y : NOMINAL_Y;
   498→    const cz = Number.isFinite(cam.zoom) ? cam.zoom : 1;
   499→    const w = this.vw, h = this.vh, t = this.t;
   500→
   501→    ctx.save();
   502→
   503→    /* 1 — sky gradient anchored to the world ground line */
   504→    const hy = h * 0.5 + (0 - cy) * cz;                     // screen y of world y=0
   505→    const skyH = hy > 1 ? (hy < h ? hy : h) : 1;
   506→    ctx.drawImage(this.skyStrip, 0, 0, 8, 600, 0, 0, w, skyH);
   507→    ctx.fillStyle = SKY_BELOW;
   508→    this.fillBelow(ctx, hy);
   509→
   510→    /* 2 — stars + moon (celestial parallax) */
   511→    const starBase = hy - 70 > h * 0.25 ? hy - 70 : h * 0.25;
   512→    const ms = mod(cx * 0.02 * cz, TILE);
   513→    this.blitRow(ctx, this.starTile, ms, starBase - STAR_H);
   514→    ctx.fillStyle = TWINK_COL;
   515→    for (let i = 0; i < this.twinks.length; i++) {
   516→      const tr = this.twinks[i];
   517→      ctx.globalAlpha = 0.2 + 0.55 * (0.5 + 0.5 * Math.sin(t * tr.sp + tr.ph));
   518→      const sy = starBase - STAR_H + tr.y;
   519→      for (let x = mod(tr.x - ms, TILE) - TILE; x < w + 3; x += TILE) {
   520→        if (x > -3) ctx.fillRect(x, sy, 2, 2);
   521→      }
   522→    }
   523→    ctx.globalAlpha = 1;
   524→    ctx.drawImage(this.moonSpr, w * 0.62 - cx * 0.03 * cz - 80, h * 0.16 - (cy - NOMINAL_Y) * 0.08 * cz - 80);
   525→
   526→    /* 3 — far mega-city (parallax 0.05) + blinking antenna beacons */
   527→    const byF = h * 0.5 + (LAY_FAR.lift - cy * LAY_FAR.vf) * cz;
   528→    const mF = mod(cx * LAY_FAR.f * cz, TILE);
   529→    ctx.fillStyle = LAY_FAR.below;
   530→    this.fillBelow(ctx, byF);
   531→    this.blitRow(ctx, this.farTile, mF, byF - FAR_H);
   532→    ctx.fillStyle = BEACON_CORE;
   533→    for (let i = 0; i < this.beacons.length; i++) {
   534→      const b = this.beacons[i];
   535→      const s = Math.max(0, Math.sin(t * b.sp + b.ph));
   536→      const a = 0.1 + 0.9 * s * s * s * s;                 // slow soft blink
   537→      const by = byF - b.yA;
   538→      for (let x = mod(b.x - mF, TILE) - TILE; x < w + 8; x += TILE) {
   539→        if (x < -8 || x > w + 8) continue;
   540→        ctx.globalAlpha = a * 0.55;
   541→        ctx.drawImage(this.beaconSpr, x - 8, by - 8);
   542→        ctx.globalAlpha = a + 0.2 > 1 ? 1 : a + 0.2;
   543→        ctx.fillRect(x - 1, by - 1, 2, 2);
   544→      }
   545→    }
   546→    ctx.globalAlpha = 1;
   547→
   548→    /* 4 — mid city (parallax 0.14) + flickering neon signs */
   549→    const byM = h * 0.5 + (LAY_MID.lift - cy * LAY_MID.vf) * cz;
   550→    const mM = mod(cx * LAY_MID.f * cz, TILE);
   551→    ctx.fillStyle = LAY_MID.below;
   552→    this.fillBelow(ctx, byM);
   553→    this.blitRow(ctx, this.midTile, mM, byM - MID_H);
   554→    ctx.lineWidth = 1;
   555→    for (let i = 0; i < this.signs.length; i++) {
   556→      const s = this.signs[i];
   557→      let mul = 1;
   558→      if (s.state === 1) mul = 0.07 + 0.05 * Math.sin(t * 41 + s.ph);            // brown-out
   559→      else if (s.state === 2) {                                                  // re-ignite
   560→        const k = 1 - s.timer / 0.4;
   561→        mul = k * (0.35 + 0.65 * Math.abs(Math.sin(t * 66 + s.ph)));
   562→      }
   563→      let a = mul * (0.72 + 0.19 * Math.sin(t * 31 + s.ph) + 0.09 * Math.sin(t * 97 + s.ph * 3));
   564→      if (a < 0.03) a = 0.03;
   565→      else if (a > 1) a = 1;
   566→      const y0 = byM - s.yA;
   567→      const iw = s.w - 10, ih = s.h - 10;
   568→      for (let x = mod(s.x - s.w * 0.5 - mM, TILE) - TILE; x < w + 4; x += TILE) {
   569→        if (x + s.w < -4 || x > w + 4) continue;
   570→        ctx.globalAlpha = 0.92;
   571→        ctx.fillStyle = SIGN_PANEL;
   572→        rrPath(ctx, x, y0, s.w, s.h, 4);
   573→        ctx.fill();
   574→        ctx.globalAlpha = a * 0.55;
   575→        ctx.strokeStyle = SIGN_BORDER[s.ci];
   576→        ctx.stroke();
   577→        ctx.globalCompositeOperation = 'lighter';          // fake bloom
   578→        ctx.globalAlpha = a * 0.22;
   579→        ctx.drawImage(this.glyphSpr[s.ci], x + 5 - iw * 0.38, y0 + 5 - ih * 0.45, iw * 1.76, ih * 1.9);
   580→        ctx.globalCompositeOperation = 'source-over';
   581→        ctx.globalAlpha = a;
   582→        ctx.drawImage(this.glyphSpr[s.ci], x + 5, y0 + 5, iw, ih);
   583→      }
   584→    }
   585→    ctx.globalAlpha = 1;
   586→
   587→    /* 5 — near-mid towers (parallax 0.30): elevator dots + searchlights */
   588→    const byN = h * 0.5 + (LAY_NEAR.lift - cy * LAY_NEAR.vf) * cz;
   589→    const mN = mod(cx * LAY_NEAR.f * cz, TILE);
   590→    ctx.fillStyle = LAY_NEAR.below;
   591→    this.fillBelow(ctx, byN);
   592→    this.blitRow(ctx, this.nearTile, mN, byN - NEAR_H);
   593→    for (let i = 0; i < this.lifts.length; i++) {
   594→      const L = this.lifts[i];
   595→      const v = t * L.sp + L.ph;
   596→      const u = v - Math.floor(v);
   597→      const tri = 1 - Math.abs(u * 2 - 1);                // ping-pong travel
   598→      const sy = byN - (L.yA1 + (L.yA0 - L.yA1) * tri);
   599→      ctx.fillStyle = LIFT_COL[L.ci];
   600→      for (let x = mod(L.x - mN, TILE) - TILE; x < w + 4; x += TILE) {
   601→        if (x < -4) continue;
   602→        ctx.globalAlpha = 0.75;
   603→        ctx.fillRect(x - 1, sy - 2, 2, 3);
   604→        ctx.globalAlpha = 0.12;
   605→        ctx.fillRect(x - 0.5, sy - 9, 1, 10);
   606→      }
   607→    }
   608→    ctx.globalCompositeOperation = 'lighter';             // sweeping searchlights
   609→    for (let i = 0; i < this.beams.length; i++) {
   610→      const B = this.beams[i];
   611→      const ang = -Math.PI * 0.5 + B.amp * Math.sin(t * B.sp + B.ph);
   612→      const pulse = 0.55 + 0.45 * Math.sin(t * 0.43 + B.ph * 2.3);
   613→      const by = byN - B.yA;
   614→      for (let x = mod(B.x - mN, TILE) - TILE; x < w + 8; x += TILE) {
   615→        if (x < -BEAM_LEN - 8 || x > w + BEAM_LEN + 8) continue;
   616→        ctx.save();
   617→        ctx.translate(x, by);
   618→        ctx.rotate(ang);
   619→        ctx.globalAlpha = pulse;
   620→        ctx.drawImage(this.beamSpr, 0, -80);
   621→        ctx.restore();
   622→      }
   623→    }
   624→    ctx.globalCompositeOperation = 'source-over';
   625→    ctx.globalAlpha = 1;
   626→
   627→    /* 6 — drifting fog bands */
   628→    for (let i = 0; i < this.fogCount; i++) {
   629→      const F = FOG_BANDS[i];
   630→      const fy = hy + F.oy - (cy - NOMINAL_Y) * 0.15 * cz;
   631→      ctx.globalAlpha = F.a * (0.85 + 0.15 * Math.sin(t * 0.05 + F.ph));
   632→      const off = mod(t * F.sp - cx * F.f * cz, FOG_W);
   633→      for (let x = off - FOG_W; x < w; x += FOG_W) {
   634→        ctx.drawImage(this.fogSpr, x, fy - FOG_H * 0.5);
   635→      }
   636→    }
   637→    ctx.globalAlpha = 1;
   638→    ctx.restore();
   639→  }

   670→  drawWeather(ctx: CanvasRenderingContext2D, cam: BgCam): void {
   671→    this.ensureBuilt();
   672→    const cx = Number.isFinite(cam.x) ? cam.x : 0;
   673→    const cy = Number.isFinite(cam.y) ? cam.y : NOMINAL_Y;
   674→    const cz = Number.isFinite(cam.zoom) ? cam.zoom : 1;
   675→    const w = this.vw, h = this.vh, t = this.t;
   676→
   677→    ctx.save();
   678→    ctx.lineCap = 'round';
   679→
   680→    /* rain streaks (quality-scaled) — batched per alpha bucket */
   681→    if (this.rainCap > 0) {
   682→      const ox = -cx * cz * 0.5, oy = -(cy - NOMINAL_Y) * cz * 0.3;
   683→      const inv = 1 / Math.sqrt(1 + this.slant * this.slant);
   684→      const ux = this.slant * inv, uy = inv;
   685→      const wrapW = w + 80, wrapH = h + 80;
   686→      for (let b = 0; b < 8; b++) {
   687→        ctx.strokeStyle = RAIN_STYLES[b];
   688→        ctx.beginPath();
   689→        for (let i = 0; i < this.rainCap; i++) {
   690→          if (this.rainB[i] !== b) continue;
   691→          const x = mod(this.rainX[i] + ox, wrapW) - 40;
   692→          const y = mod(this.rainY[i] + oy, wrapH) - 40;
   693→          ctx.moveTo(x, y);
   694→          ctx.lineTo(x - ux * this.rainL[i], y - uy * this.rainL[i]);
   695→        }
   696→        ctx.stroke();
   697→      }
   698→      ctx.strokeStyle = RAIN_TIP;                          // subtle bright tips
   699→      ctx.beginPath();
   700→      for (let i = 0; i < this.rainCap; i++) {
   701→        if (this.rainB[i] < 4) continue;
   702→        const x = mod(this.rainX[i] + ox, wrapW) - 40;
   703→        const y = mod(this.rainY[i] + oy, wrapH) - 40;
   704→        ctx.moveTo(x, y);
   705→        ctx.lineTo(x - ux * 4, y - uy * 4);
   706→      }
   707→      ctx.stroke();
   708→
   709→      /* tiny ripple ellipses pinned to the arena floor */
   710→      const hy = h * 0.5 + (0 - cy) * cz;
   711→      if (hy > -30 && hy < h + 40) {
   712→        ctx.strokeStyle = RIPPLE_COL;
   713→        ctx.lineWidth = 1;
   714→        for (let i = 0; i < this.ripples.length; i++) {
   715→          const r = this.ripples[i];
   716→          let u = (t * r.sp + r.ph) % 1;
   717→          if (u < 0) u += 1;
   718→          const sx = w * 0.5 + (r.wx - cx) * cz;
   719→          if (sx < -24 || sx > w + 24) continue;
   720→          const e = Math.sin(u * Math.PI);
   721→          ctx.globalAlpha = e * e * 0.12;
   722→          ctx.beginPath();
   723→          ctx.ellipse(sx, hy + r.oy * cz, r.r * (0.3 + 0.7 * u) * cz, r.r * (0.3 + 0.7 * u) * cz * 0.24, 0, 0, TAU);
   724→          ctx.stroke();
   725→        }
   726→        ctx.globalAlpha = 1;
   727→      }
   728→    }
   729→
   730→    /* drifting motes */
   731→    if (this.moteCap > 0) {
   732→      const ox = -cx * cz * 0.22, oy = -(cy - NOMINAL_Y) * cz * 0.15;
   733→      const wrapW = w + 40, wrapH = h + 40;
   734→      for (let b = 0; b < 4; b++) {
   735→        ctx.fillStyle = MOTE_STYLES[b];
   736→        for (let i = 0; i < this.moteCap; i++) {
   737→          if (this.moteB[i] !== b) continue;
   738→          ctx.fillRect(mod(this.moteX[i] + ox, wrapW) - 20, mod(this.moteY[i] + oy, wrapH) - 20, 2, 2);
   739→        }
   740→      }
   741→    }
   742→
   743→    /* distant lightning — 2-frame silhouette flash, then a slow fade */
   744→    if (this.ltnT >= 0) {
   745→      const u = this.ltnT;
   746→      let I = 0;
   747→      if (u < 0.05) I = 1;
   748→      else if (u < 0.09) I = 0.22;
   749→      else if (u < 0.14) I = 0.78;
   750→      else I = 0.78 * Math.exp(-(u - 0.14) * 7);
   751→      if (I > 0.012) {
   752→        ctx.globalAlpha = 0.085 * I;
   753→        ctx.fillStyle = FLASH_COL;
   754→        ctx.fillRect(0, 0, w, h);
   755→        ctx.globalAlpha = 1;
   756→      }
   757→    }
   758→    ctx.restore();
   759→  }

   792→  private buildStarTile(): HTMLCanvasElement {
   793→    const { c, g } = makeCanvas(TILE, STAR_H);
   794→    const rng = mulberry32(0x51a2c9);
   795→    for (let i = 0; i < 180; i++) {
   796→      const x = rng() * TILE;
   797→      const y = rng() * STAR_H;
   798→      const f = y / STAR_H;
   799→      const fade = f < 0.5 ? 1 : Math.max(0.06, 1 - (f - 0.5) * 1.85);   // city glow wash
   800→      const a = (0.16 + rng() * 0.6) * fade;
   801→      const big = rng() < 0.1;
   802→      g.fillStyle = rng() < 0.12
   803→        ? `rgba(232,220,200,${a.toFixed(3)})`
   804→        : `rgba(201,216,221,${a.toFixed(3)})`;
   805→      g.fillRect(x, y, big ? 2 : 1, big ? 2 : 1);
   806→    }
   807→    for (let i = 0; i < 16; i++) {
   808→      this.twinks.push({ x: rng() * TILE, y: 170 + rng() * 260, ph: rng() * TAU, sp: 1.4 + rng() * 2.4 });
   809→    }
   810→    return c;
   811→  }

   813→  private buildFar(): HTMLCanvasElement {
   814→    const { c, g } = makeCanvas(TILE, FAR_H);
   815→    const rng = mulberry32(0xc1f3a0);
   816→    const shades = ['#0b1219', '#0d151d', '#0e1720', '#101a24'];
   817→    let x = 0;
   818→    while (x < TILE - 24) {
   819→      let bw = 46 + rng() * 130;
   820→      if (x + bw > TILE) bw = TILE - x;
   821→      let bh = 110 + rng() * 330;
   822→      if (rng() < 0.18) bh += 60 + rng() * 150;
   823→      if (bh > FAR_H - 8) bh = FAR_H - 8;
   824→      const shade = shades[(rng() * shades.length) | 0];
   825→      g.fillStyle = shade;
   826→      g.fillRect(x, FAR_H - bh, bw, bh);
   827→      if (rng() < 0.25) {                                  // crowning setback
   828→        const sw = bw * (0.35 + rng() * 0.3);
   829→        let sh = 22 + rng() * 70;
   830→        if (bh + sh > FAR_H - 6) sh = FAR_H - 6 - bh;
   831→        if (sh > 6) {
   832→          g.fillRect(x + (bw - sw) * rng(), FAR_H - bh - sh, sw, sh);
   833→          g.fillStyle = 'rgba(60,80,92,0.12)';
   834→          g.fillRect(x + (bw - sw) * rng(), FAR_H - bh - sh, sw, 1);
   835→        }
   836→      }
   837→      if (bh > 330 && rng() < 0.6 && this.beacons.length < 9) {  // antenna + beacon
   838→        const ax = x + bw * (0.2 + rng() * 0.6);
   839→        let aH = 16 + rng() * 42;
   840→        if (bh + aH > FAR_H - 4) aH = FAR_H - 4 - bh;
   841→        if (aH > 8) {
   842→          g.fillStyle = '#0d141b';
   843→          g.fillRect(ax, FAR_H - bh - aH, 1, aH);
   844→          this.beacons.push({ x: ax + 0.5, yA: bh + aH - 2, ph: rng() * TAU, sp: 0.8 + rng() * 1.5 });
   845→        }
