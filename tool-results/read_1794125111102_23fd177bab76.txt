    21→/** circle vs swept blade quad (prev segment + current segment) — prevents tunneling */
    22→function circleHitsSweptQuad(
    23→  cx: number, cy: number, r: number,
    24→  pbx: number, pby: number, ptx: number, pty: number,
    25→  bx: number, by: number, tx: number, ty: number,
    26→): boolean {
    27→  // current + previous blade segments
    28→  if (segHitsCircle(bx, by, tx, ty, cx, cy, r)) return true
    29→  if (segHitsCircle(pbx, pby, ptx, pty, cx, cy, r)) return true
    30→  // swept edges (tip path + base path)
    31→  if (segHitsCircle(ptx, pty, tx, ty, cx, cy, r)) return true
    32→  if (segHitsCircle(pbx, pby, bx, by, cx, cy, r)) return true
    33→  // circle center inside the swept quad (spins / wide arcs)
    34→  if (pointInTri(cx, cy, pbx, pby, ptx, pty, tx, ty)) return true
    35→  if (pointInTri(cx, cy, pbx, pby, tx, ty, bx, by)) return true
    36→  return false
    37→}

    39→function pointInTri(px: number, py: number, x1: number, y1: number, x2: number, y2: number, x3: number, y3: number): boolean {
    40→  const d1 = (px - x2) * (y1 - y2) - (x1 - x2) * (py - y2)
    41→  const d2 = (px - x3) * (y2 - y3) - (x2 - x3) * (py - y3)
    42→  const d3 = (px - x1) * (y3 - y1) - (x3 - x1) * (py - y1)
    43→  const hasNeg = d1 < 0 || d2 < 0 || d3 < 0
    44→  const hasPos = d1 > 0 || d2 > 0 || d3 > 0
    45→  return !(hasNeg && hasPos)
    46→}

    65→const SCALE = 1.32
    66→/** nominal (airborne) hip height = thigh(25)+shin(24) = 49 local × SCALE.
    67→ * Grounded height is NOT fixed — it derives from the planted foot via the
    68→ * foot-ground lock (updateFootLock): idle bent knees, lunges and landing
    69→ * squashes all sit exactly ON the ground line. */
    70→const HIP_H = (25 + 24) * SCALE
    71→
    72→/**
    73→ * Enemy body spheres: [centerY offset above ground, radius] in enemy-local
    74→ * units (× e.scale). Traced from the ENEMY_SPEC skeleton — hip at 47u, torso
    75→ * 32u, head top ≈ 94u. The old single circle (y-34, r 26) only covered the
    76→ * lower 2/3 of the body, so chest/head-height slashes passed straight
    77→ * through the sprite without connecting. Three spheres hug the visual body.
    78→ */
    79→const BODY_SPHERES: ReadonlyArray<{ oy: number; r: number }> = [
    80→  { oy: 14, r: 15 }, // legs / hip mass
    81→  { oy: 48, r: 21 }, // torso (the main target)
    82→  { oy: 78, r: 17 }, // chest / head zone
    83→]
    84→
    85→/** blade segment vs an enemy's full visual body (any sphere, swept-aware).
    86→ * `pad` fattens the radius per-attack — a katana edge is a plane; the hit
    87→ * volume must match the visual footprint of the blade edge corridor. */
    88→function bladeHitsBody(
    89→  e: { x: number; y: number; scale: number },
    90→  hasPrev: boolean,
    91→  prev: { bx: number; by: number; tx: number; ty: number },
    92→  cur: { bx: number; by: number; tx: number; ty: number },
    93→  pad = 3,
    94→): boolean {
    95→  const s = e.scale
    96→  for (const sp of BODY_SPHERES) {
    97→    const cy = e.y - sp.oy * s
    98→    const r = sp.r * s + pad
    99→    if (hasPrev
   100→      ? circleHitsSweptQuad(e.x, cy, r, prev.bx, prev.by, prev.tx, prev.ty, cur.bx, cur.by, cur.tx, cur.ty)
   101→      : segHitsCircle(cur.bx, cur.by, cur.tx, cur.ty, e.x, cy, r)
   102→    ) return true
   103→  }
   104→  return false
   105→}

   416→  update(dt: number, w: WorldRef): void {
   417→    this.iframes = Math.max(0, this.iframes - dt)
   418→    this.armorT = Math.max(0, this.armorT - dt)
   419→    this.flashT = Math.max(0, this.flashT - dt * 5)
   420→    this.swingGlow = Math.max(0, this.swingGlow - dt * 5)
   421→    this.riposteT = Math.max(0, this.riposteT - dt)
   422→    this.energy = Math.min(this.maxEnergy, this.energy + dt * 3)
   423→
   424→    // context-system clocks
   425→    this.now += dt
   426→    this.locoT += dt
   427→    if (this.turnT > 0) this.turnT = Math.max(0, this.turnT - dt * 6) // ~0.17s pivot
   428→    if (this.postAttackT > 0) this.postAttackT = Math.max(0, this.postAttackT - dt)
   429→
   430→    switch (this.state) {
   431→      case 'move': this.updateMove(dt, w); break
   432→      case 'attack': this.updateAttack(dt, w); break
   433→      case 'dash': this.updateDash(dt, w); break
   434→      case 'parry': this.updateParry(dt, w); break
   435→      case 'charge': this.updateCharge(dt, w); break
   436→      case 'hurt': this.updateHurt(dt, w); break
   437→      case 'dead': this.updateDead(dt, w); break
   438→      case 'scripted': this.updateScripted(dt); break
   439→    }
   440→
   441→    // physics integration (shared)
   442→    if (this.state !== 'dash') {
   443→      let g: number
   444→      if (this.state === 'attack' && this.action && this.action.def.float) {
   445→        // authored air-combo float overrides the shaping (juggle windows)
   446→        g = GRAVITY * this.action.def.float
   447→      } else if (this.vy < 0) {
   448→        // ANTI-FLOAT: apex decay — no moon hovering
   449→        g = GRAVITY * (this.vy < -260 ? GRAV_RISE : GRAV_APEX)
   450→      } else {
   451→        // heavy fall: the body plummets
   452→        g = GRAVITY * GRAV_FALL
   453→      }
   454→      this.vy += g * dt
   455→      if (this.vy > TERMINAL_VY) this.vy = TERMINAL_VY
   456→    }
   457→    this.x += this.vx * dt
   458→    this.y += this.vy * dt
   459→
   460→    // ground
   461→    const wasAir = !this.grounded
   462→    if (this.y >= 0) {
   463→      this.y = 0
   464→      if (this.vy > 140) {
   465→        this.landSpring = clamp(this.vy / 700, 0, 1)
   466→        if (wasAir) this.onLand(w)
   467→      } else if (wasAir) {
   468→        this.landSpring = 0.4
   469→        this.onLand(w)
   470→      }
   471→      this.vy = 0
   472→      this.grounded = true
   473→      this.jumpsUsed = 0
   474→      this.airDashUsed = false
   475→    } else {
   476→      this.grounded = false
   477→    }
   478→    // arena bounds
   479→    if (this.x < w.arenaMinX + 20) {
   480→      this.x = w.arenaMinX + 20
   481→      if (this.vx < -200) w.fx.dust(this.x - 12, 0, 5)
   482→      this.vx = Math.max(0, this.vx)
   483→    }
   484→    if (this.x > w.arenaMaxX - 20) {
   485→      this.x = w.arenaMaxX - 20
   486→      if (this.vx > 200) w.fx.dust(this.x + 12, 0, 5)
   487→      this.vx = Math.min(0, this.vx)
   488→    }
   489→
   490→    this.dashCd = Math.max(0, this.dashCd - dt)
   491→    this.dashRecharge = Math.max(0, this.dashRecharge - dt)
   492→    if (this.dashRecharge <= 0) this.dashConsec = 0
   493→
   494→    // ability script
   495→    if (this.ability) this.updateAbility(dt, w)
   496→
   497→    // animation + solve
   498→    this.animator.update(dt)
   499→    // Base layer: the clip FULLY owns the pose each frame (write, never
   500→    // accumulate). applyProcedural() is a bounded additive layer on top.
   501→    // Without this copy, procedural += deltas integrate unbounded → spine
   502→    // spin + root sink. (Same contract as Spine's AnimationState.apply /
   503→    // Godot AnimationTree → Skeleton2D.)
   504→    this.pose.set(this.animator.pose)
   505→    this.applyProcedural(dt)
   506→    this.updateBladeZ(dt) // zArc → bladeYaw channel (BEFORE the 3D solve)
   507→    // procedural layers (bounded, frame-local — never accumulate):
   508→    // SWORD-LED ARC first: it rewrites the arm channels, so the blade
   509→    // dynamics spring below sees the ARC's angular velocity — the whip
   510→    // lags/overshoots the real swing, not the raw keyframes.
   511→    this.swordArc.update(dt, {
   512→      pose: this.pose,
   513→      shoulderX: this.geom.shoulderR.x, shoulderY: this.geom.shoulderR.y,
   514→      shoulderZ: this.geom.shoulderR.z,
   515→      handX: this.geom.handR.x, handY: this.geom.handR.y,
   516→      handZ: this.geom.handR.z,
   517→      def: this.state === 'attack' ? this.action?.def ?? null : null,
   518→      animT: this.action ? this.action.animT : 0,
   519→      enemies: w.enemies,
   520→      face: this.face,
   521→      x: this.x,
   522→      hipY: this.hipY,
   523→      reach: PLAYER_SPEC.blade * this.sword.stats.range * SCALE * 1.1 + 130,
   524→    })
   525→    this.updateBladeDynamics(dt, w) // wrist lag spring + kinetic chain — feeds hitboxes
   526→    this.applyTwoHandGrip(dt)       // off-hand IK onto the tsuka
   527→    // 3D SOLVE: sagittal channels + depth channels → projected joints + blade
   528→    solveFighter3D(
   529→      this.pose, PLAYER_SPEC, this.geom,
   530→      PLAYER_SPEC.blade * this.sword.stats.range * 1.06,
   531→      this.swingScale,
   532→    )
   533→    this.updateFootLock(dt)         // weight: hips sit on the planted foot
   534→    this.updateBladeWorld()
   535→    this.updateSwingScale(dt)
   536→  }

   553→  private slamAoe(w: WorldRef, r: number, dmg: number, launch: number): void {

   581→  /** Per-frame enemy-formation scan (spec §5): nearest threat + proximity
   582→   * weight (with hysteresis) + surrounded flag. Feeds the locomotion
   583→   * selector AND the procedural posture layers. NEVER takes control away
   584→   * from the player — posture only. */
   585→  private scanEnemies(dt: number, w: WorldRef): void {
   586→    const RADIUS = 430
   587→    let bestD = Infinity
   588→    let ahead = false
   589→    let behind = false
   590→    for (const e of w.enemies) {
   591→      if (e.dead || e.spawning) continue
   592→      const dx = e.x - this.x
   593→      const ad = Math.abs(dx)
   594→      if (ad < RADIUS) {
   595→        if (dx * this.face > 0) ahead = true
   596→        else behind = true
   597→      }
   598→      if (ad < bestD) {
   599→        bestD = ad
   600→        this.threatX = e.x
   601→        this.threatY = e.y
   602→        this.hasThreat = true
   603→      }
   604→    }
   605→    if (bestD === Infinity) this.hasThreat = false
   606→    this.surrounded = ahead && behind && bestD < 300
   607→    // asymmetric damp = hysteresis: fast to raise the guard, slow to relax
   608→    const target = bestD < RADIUS ? 1 : 0
   609→    const k = target > this.enemyNearW ? 7 : 2.2
   610→    this.enemyNearW = damp(this.enemyNearW, target, k, dt)
   611→    // head-tracking weight: quick to look, slower to look away
   612→    const ht = this.hasThreat && bestD < 520 ? 1 : 0
   613→    const hk = ht > this.headTrackW ? 8 : 3
   614→    this.headTrackW = damp(this.headTrackW, ht, hk, dt)
   615→  }
   616→
   617→  /** Select the locomotion mode with dwell hysteresis (no flicker): a mode
   618→   * must be held ≥0.1s before another can take over (except brake/turn
   619→   * interrupts). Priority: brake > sprint latch > combat (enemy near) >
   620→   * backpedal (retreating vs facing) > run > walk > idle. */
   621→  private pickLoco(speed: number, ax: number): LocoMode {
   622→    const fwd = this.vx * this.face
   623→    if (this.braking) return 'brake'
   624→    if (!this.grounded) return this.loco // airborne handled by clips, not modes
   625→    if (speed < 36) {
   626→      return this.enemyNearW > 0.5 || this.surrounded ? 'combat' : 'idle'
   627→    }
   628→    // retreating while a threat is engaged → guarded backpedal
   629→    if (fwd < -60 && this.enemyNearW > 0.4) return 'backpedal'
   630→    // explicit sprint outranks proximity awareness (player intent wins)
   631→    if (this.sprintLatch && speed > 300) return 'sprint'
   632→    // Condition B: combat shuffle while enemies are near (but full-speed
   633→    // committed running still uses the run mechanics)
   634→    if (this.enemyNearW > 0.55 && speed < 310 && ax !== 0) return 'combat'
   635→    if (speed > 210) return 'run'
   636→    return 'walk'
   637→  }
   638→
   639→  private setLoco(mode: LocoMode): void {
   640→    if (mode === this.loco) return
   641→    // dwell hysteresis: hold each mode ≥0.1s (brake interrupts instantly)
   642→    if (this.locoT < 0.1 && mode !== 'brake' && this.loco !== 'brake') return
   643→    this.loco = mode
   644→    this.locoT = 0
   645→  }

   647→  private updateMove(dt: number, w: WorldRef): void {
   648→    const input = w.input
   649→    const ax = input.axisX()
   650→    const ay = input.axisY()
   651→
   652→    // ---- enemy-formation awareness (posture influence ONLY) ----
   653→    this.scanEnemies(dt, w)
   654→
   655→    // ---- facing: hold toward an engaged threat at maneuvering speed ----
   656→    // (Condition B: circle/retreat keeps the guard trained on the enemy —
   657→    // backpedal instead of a civilian turn-away. Full-speed escape or
   658→    // open-space movement flips freely.)
   659→    if (ax !== 0 && this.face !== (ax > 0 ? 1 : -1)) {
   660→      const engaged = this.enemyNearW > 0.55 && this.hasThreat
   661→      const slow = Math.abs(this.vx) < 260 && !this.sprintLatch
   662→      if (!(engaged && slow)) {
   663→        if (Math.abs(this.vx) > 120) {
   664→          this.turnT = 1
   665→          this.turnDir = -Math.sign(ax)
   666→          w.fx.dust(this.x, 0, 4)
   667→        }
   668→        this.face = ax > 0 ? 1 : -1
   669→      }
   670→    }
   671→
   672→    // ---- double-tap direction → sprint latch (DMC-style, event-driven) ----
   673→    // Press stamps come from the keydown EVENTS (input clock), so a rapid
   674→    // tap-release-tap registers even if the release window contained zero
   675→    // game frames (low fps / throttled rAF).
   676→    {
   677→      const sr = w.input.pressStamp('right')
   678→      const sl = w.input.pressStamp('left')
   679→      const newR = sr !== this.stampR && sr >= 0
   680→      const newL = sl !== this.stampL && sl >= 0
   681→      if (newR || newL) {
   682→        const stamp = newR ? sr : sl
   683→        const dir = newR ? 1 : -1
   684→        if (stamp - this.lastTapT < 0.3 && dir === this.lastTapDir) {
   685→          this.sprintLatch = true
   686→        }
   687→        this.lastTapDir = dir
   688→        this.lastTapT = stamp
   689→      }
   690→      this.stampR = sr
   691→      this.stampL = sl
   692→    }
   693→    if (ax === 0) this.sprintLatch = false
   694→
   695→    // ---- horizontal movement (sprint has its own accel curve + ceiling) ----
   696→    const sprinting = this.sprintLatch && this.grounded && ax !== 0 && Math.sign(ax) === Math.sign(this.vx || ax)
   697→    const vMax = sprinting ? SPRINT_MAX : MOVE_MAX
   698→    if (ax !== 0) {
   699→      // input reversal at speed → BRAKE (skid decel + one-shot clip + dust)
   700→      const reversing = Math.sign(ax) !== Math.sign(this.vx) && Math.abs(this.vx) > 240
   701→      if (reversing && this.grounded && !this.braking) {
   702→        this.braking = true
   703→        this.animator.play(BRAKE_CLIP, { blend: 0.04 })
   704→        w.fx.dust(this.x - this.face * 10, 0, 7)
   705→        w.audio.play('land', { volume: 0.35 })
   706→      }
   707→      if (reversing || this.braking) {
   708→        // hard skid deceleration toward zero before re-accelerating
   709→        const d = BRAKE_DECEL * dt
   710→        this.vx = Math.abs(this.vx) <= d ? 0 : this.vx - Math.sign(this.vx) * d
   711→      } else {
   712→        // analog target speed: axis magnitude maps to a target velocity —
   713→        // partial input = partial speed (walk reachable; touch-friendly),
   714→        // full keys still hit the ceiling. Accelerate toward it, never past.
   715→        const targetV = ax * vMax
   716→        const movingToward = Math.sign(this.vx) === Math.sign(targetV) || this.vx === 0
   717→        if (!movingToward || Math.abs(this.vx) < Math.abs(targetV)) {
   718→          const accel = sprinting ? SPRINT_ACCEL : this.grounded ? ACCEL : AIR_ACCEL
   719→          this.vx += ax * accel * dt
   720→          if (Math.sign(this.vx) === Math.sign(targetV) && Math.abs(this.vx) > Math.abs(targetV)) {
   721→            this.vx = targetV
   722→          }
   723→        } else if (Math.sign(this.vx) === Math.sign(targetV) && Math.abs(this.vx) > Math.abs(targetV)) {
   724→          // ease down toward the lower analog target (stick pushed softer)
   725→          const f = FRICTION * dt
   726→          const dv = Math.min(f, Math.abs(this.vx) - Math.abs(targetV))
   727→          this.vx -= Math.sign(this.vx) * dv
   728→        }
   729→      }
   730→      this.vx = clamp(this.vx, -vMax, vMax)
   731→    } else if (this.grounded) {
   732→      const f = FRICTION * dt
   733→      this.vx = Math.abs(this.vx) <= f ? 0 : this.vx - Math.sign(this.vx) * f
   734→    } else {
   735→      this.vx = damp(this.vx, 0, 0.4, dt)
   736→    }
   737→
   738→    // ---- locomotion clip selection (context-aware, hysteresis) ----
   739→    const speed = Math.abs(this.vx)
   740→    if (this.grounded) {
   741→      if (this.braking) {
   742→        if (this.animator.isDone() || speed < 90) this.braking = false
   743→      } else {
   744→        this.setLoco(this.pickLoco(speed, ax))
   745→        switch (this.loco) {
   746→          case 'walk':
   747→            this.animator.play(WALK_CLIP, { blend: 0.18, speed: clamp(speed / 210, 0.7, 1.35) })
   748→            break
   749→          case 'run':
   750→            this.animator.play(RUN_CLIP, { blend: 0.12, speed: clamp(speed / MOVE_MAX, 0.8, 1.5) })
   751→            break
   752→          case 'sprint':
   753→            this.animator.play(SPRINT_CLIP, { blend: 0.1, speed: clamp(speed / SPRINT_MAX, 0.85, 1.25) })
   754→            break
   755→          case 'combat':
   756→            this.animator.play(COMBAT_STEP_CLIP, { blend: 0.16, speed: clamp(speed / 260, 0.7, 1.7) })
   757→            break
   758→          case 'backpedal':
   759→            this.animator.play(BACKPEDAL_CLIP, { blend: 0.14, speed: clamp(speed / 240, 0.7, 1.6) })
   760→            break
   761→          default:
   762→            this.animator.play(IDLE_CLIP, { blend: 0.16 })
   763→        }
   764→      }
   765→    } else if (this.vy < -60) {
   766→      this.animator.play(JUMP_CLIP, { blend: 0.06 })
   767→    } else {
   768→      this.animator.play(FALL_CLIP, { blend: 0.12 })
   769→    }
   770→
   771→    // ---- actions ----
   772→    if (this.tryCombatInputs(w, ax, ay)) return
   773→  }
   774→
   775→  /** returns true if a new state was entered */
   776→  private tryCombatInputs(w: WorldRef, ax: number, ay: number): boolean {
   777→    const input = w.input
   778→
   779→    // dash (anytime, from move)
   780→    if ((input.pressed('dash') || input.buffered('dash')) && this.canDash()) {
   781→      this.startDash(w, ax, ay)
   782→      return true
   783→    }
   784→    // jump (grounded = jump 1, plus one air jump)
   785→    if (input.pressed('jump') || input.buffered('jump', 0.18)) {
   786→      if (this.jumpsUsed < 2) {
   787→        this.doJump(w)
   788→        return false
   789→      }
   790→    }
   791→    // parry
   792→    if (input.pressed('parry')) {
   793→      this.enterParry(w)
   794→      return true
   795→    }
   796→    // ability
   797→    if (input.pressed('ability')) {
   798→      if (this.energy >= this.sword.ability.cost) {
   799→        this.startAbility(w)
   800→        return true
   801→      }
   802→    }
   803→    // ult is handled by Game (cinematic)
   804→
   805→    // heavy: charge vs tap
   806→    if (input.down('heavy') && this.grounded) {
   807→      this.enterCharge(w)
   808→      return true
   809→    }
   810→    if (input.pressed('heavy') || input.buffered('heavy')) {
   811→      input.takeBuffered('heavy')
   812→      this.startAttack(this.pickHeavy(w), w)
   813→      return true
   814→    }
   815→    // light
   816→    if (input.pressed('light') || input.buffered('light')) {
   817→      input.takeBuffered('light')
   818→      // DASH MOMENTUM STRIKE: attacking while the last dash's momentum
   819→      // still carries the body converts into the through-cut — the
   820→      // dash-strike feel (Hades). Without this, the 150ms in-dash input
   821→      // window was brutally tight and dash attacks "randomly" whiffed.
   822→      if (this.grounded
   823→        && w.time - this.dashEndedAt < 0.18
   824→        && Math.abs(this.vx) > 110
   825→        && this.riposteT <= 0) {
   826→        this.startAttack(ATTACKS.dashSlash, w)
   827→        return true
   828→      }
   829→      const id = this.riposteT > 0 ? 'riposte' : this.pickLight(w, ax, ay)
   830→      this.startAttack(ATTACKS[id], w)
   831→      return true
   832→    }
   833→    return false
   834→  }

   836→  private doJump(w: WorldRef): void {
   837→    w.input.takeBuffered('jump')
   838→    this.jumpsUsed++
   839→    this.vy = this.jumpsUsed === 1 ? JUMP_V : AIR_JUMP_V
   840→    this.grounded = false
   841→    w.audio.play('jump', { volume: 0.4 })
   842→    w.fx.dust(this.x, 0, 4)
   843→    if (this.jumpsUsed === 2) {
   844→      w.fx.burst(this.x, this.y - 20, 6, '#9fb4c4', 160, 0.4, 'dot')
   845→    }
   846→  }

   852→  private canDash(): boolean {
   853→    if (this.dashCd > 0 || this.dashRecharge > 0) return false
   854→    if (!this.grounded && this.airDashUsed) return false
   855→    return true
   856→  }
   857→
   858→  startDash(w: WorldRef, ax: number, ay: number): void {
   859→    w.input.takeBuffered('dash')
   860→    let dx = ax
   861→    let dy = ay
   862→    if (dx === 0 && dy === 0) dx = this.face
   863→    const len = Math.hypot(dx, dy) || 1
   864→    dx /= len
   865→    dy /= len
   866→    if (dx !== 0) this.face = dx > 0 ? 1 : -1
   867→    this.dashDx = dx
   868→    this.dashDy = dy
   869→    this.state = 'dash'
   870→    this.dashT = DASH_TIME
   871→    this.dashCd = DASH_CD
   872→    this.dashConsec++
   873→    if (this.dashConsec >= 2) this.dashRecharge = DASH_RECHARGE
   874→    if (!this.grounded) this.airDashUsed = true
   875→    this.iframes = Math.max(this.iframes, 0.12)
   876→    this.action = null
   877→    this.charging = false
   878→    this.vx = dx * DASH_SPEED
   879→    this.vy = dy * DASH_SPEED * (this.grounded && dy < -0.4 ? 0.55 : 1)
   880→    this.animator.play(this.grounded ? DASH_CLIP : DASH_AIR_CLIP, { blend: 0.05 })
   881→    w.audio.play('dash', { pitch: rand(0.95, 1.1) })
   882→    const colors = this.sword.colors
   883→    w.fx.speedStreaks(this.x, this.y - 40, dx * DASH_SPEED * 0.9, 6, rgba(colors.trail, 1))
   884→    w.fx.chroma(0.4) // lens strain on the launch
   885→    w.fx.dust(this.x, this.grounded ? 0 : this.y, 5)
   886→    w.cam.addTrauma(0.06)
   887→  }

   889→  private updateDash(dt: number, w: WorldRef): void {
   890→    this.dashT -= dt
   891→    // afterimages
   892→    this.afterimgTimer -= dt
   893→    if (this.afterimgTimer <= 0) {
   894→      this.afterimgTimer = 0.026
   895→      w.fx.afterimage(this.pose, this.x, this.hipY, this.face, this.sword.colors.trail, 0.3, SCALE)
   896→    }
   897→    // maintain velocity with slight decay
   898→    const k = 1 - this.dashT / DASH_TIME
   899→    const sp = DASH_SPEED * (1 - k * 0.25)
   900→    this.vx = this.dashDx * sp
   901→    this.vy = this.dashDy * sp * (this.grounded ? (this.dashDy < 0 ? 0.4 : 1) : 1)
   902→
   903→    // dash attack
   904→    const input = w.input
   905→    if (input.pressed('light') || input.buffered('light')) {
   906→      input.takeBuffered('light')
   907→      this.startAttack(ATTACKS.dashSlash, w)
   908→      return
   909→    }
   910→    if (input.pressed('heavy') || input.buffered('heavy')) {
   911→      if (!this.grounded) {
   912→        input.takeBuffered('heavy')
   913→        this.startAttack(ATTACKS.slam, w)
   914→        return
   915→      }
   916→      input.takeBuffered('heavy')
   917→      this.startAttack(ATTACKS.dashHeavy, w)
   918→      return
   919→    }
   920→    if (this.dashT <= 0) {
   921→      this.state = 'move'
   922→      this.dashEndedAt = w.time
   923→      this.vx *= 0.55
   924→      if (!this.grounded) {
   925→        this.vy *= 0.5
   926→      } else if (Math.abs(this.vx) > 260) {
   927→        // THE BRAKING SKID: momentum must shed through a visible physical
   928→        // pose — heel-dig crouch + a dust scrape, never a dead stop.
   929→        w.fx.dust(this.x - this.face * 12, 0, 5)
   930→        this.landSpring = 0.35
   931→        this.animator.play(LAND_CLIP, { blend: 0.05 })
   932→      }
   933→    }
   934→  }

   940→  private pickLight(w: WorldRef, ax: number, ay: number): string {
   941→    // contextual branches
   942→    if (ay < -0.4 && this.grounded) return 'launcher'
   943→    if (ay > 0.4 && this.grounded) return 'spin'
   944→    if (this.grounded && ax !== 0 && Math.sign(ax) !== this.face) return 'retreat'
   945→    if (!this.grounded) {
   946→      if (ay > 0.4) return 'dive'
   947→      this.airIndex = (this.airIndex % 3) + 1
   948→      return `a${this.airIndex}`
   949→    }
   950→    // ground light chain (with time decay)
   951→    if (w.time - this.lastLightAt > 1.1) this.lightIndex = 0
   952→    this.lightIndex = Math.min(this.lightIndex + 1, 5)
   953→    return `l${Math.min(this.lightIndex, 5)}`
   954→  }

   956→  /** Heavy CHAIN: Kiri Oroshi → Tsuki → Grand Slam. Sequential like the
   957→   * light chain, with time decay — heavy attacks finally have combo
   958→   * pairing instead of one animation forever. */
   959→  private pickHeavy(w: WorldRef): AttackDef {
   960→    if (w.time - this.lastHeavyAt > 1.15) this.heavyIndex = 0
   961→    this.heavyIndex = Math.min(this.heavyIndex + 1, 3)
   962→    return ATTACKS[`h${this.heavyIndex}`] ?? ATTACKS.h1
   963→  }
   964→
   965→  startAttack(def: AttackDef, w: WorldRef, fromParry = false): void {
   966→    if (!def) return
   967→    // IN-BETWEEN LAW: capture the source state BEFORE switching — chained
   968→    // attacks crossfade slower (0.11s) than fresh entries (0.07s) so every
   969→    // transition between animation instances gets real in-betweens.
   970→    const wasAttack = this.state === 'attack'
   971→    // requirement checks
   972→    if (def.requires === 'ground' && !this.grounded) def = ATTACKS.a1
   973→    if (def.requires === 'air' && this.grounded) def = ATTACKS.l1
   974→
   975→    // face toward held direction at attack start
   976→    const ax = w.input.axisX()
   977→    if (ax !== 0 && !fromParry) {
   978→      this.face = ax > 0 ? 1 : -1
   979→    } else if (!fromParry) {
   980→      // SOFT AIM: no direction held → face the nearest threat. Every swing
   981→      // commits toward a real target — the root of "aimed at different
   982→      // angles": cuts should converge on the enemy, not on empty planes.
   983→      const tgt = this.nearestEnemy(w, 600)
   984→      if (tgt && Math.abs(tgt.x - this.x) > 24) {
   985→        this.face = tgt.x > this.x ? 1 : -1
   986→      }
   987→    }
   988→

  1006→      animT: 0,

  1023→    // smears, and trails run full-rate on the procedural layers.

  1026→  }

  1028→  private updateAttack(dt: number, w: WorldRef): void {
  1029→    const a = this.action
  1030→    if (!a) {
  1031→      this.state = 'move'
  1032→      return
  1033→    }
  1034→    const def = a.def
  1035→    a.animT = this.animator.getTime()
  1036→    const animT = a.animT
  1037→    const spd = this.atkSpeed
  1038→
  1039→    // ground friction / air drag during attack
  1040→    if (this.grounded) {
  1041→      const f = 900 * dt
  1042→      this.vx = Math.abs(this.vx) <= f ? 0 : this.vx - Math.sign(this.vx) * f
  1043→    }
  1044→

  1047→      a.sfxDone = true

  1049→    }

  1051→    // a localized light event + a 2-frame stretch on the steel sells it.
  1052→    if (!a.flashDone && animT >= def.trail.start) {
  1053→      a.flashDone = true
  1054→      const c = this.sword.colors
  1055→      w.fx.burst(this.blade.tx, this.blade.ty, 7, c.core, 300, 0.3, 'spark')
  1056→      w.fx.shockwave(this.blade.tx, this.blade.ty, 46, c.glow, 0.18, 3)
  1057→      w.cam.addTrauma(0.05)

  1059→    }
  1060→    // forward impulse
  1061→    if (!a.moveDone && def.moveX !== undefined && animT >= (def.moveXAt ?? def.hit.start)) {
  1062→      a.moveDone = true
  1063→      this.vx = this.face * def.moveX * (def.kind === 'dash' ? 1 : 1)
  1064→    }
  1065→    // vertical impulse (launcher hop / slam plunge)
  1066→    if (!a.moveYDone && def.moveY !== undefined && animT >= (def.moveYAt ?? 0)) {
  1067→      a.moveYDone = true
  1068→      this.vy = def.moveY
  1069→      if (def.moveY < 0) {
  1070→        this.grounded = false
  1071→        w.fx.dust(this.x, 0, 6)
  1072→      }
  1073→    }

  1076→    const h = def.hit
  1077→    if (animT >= h.start && animT <= h.end) {
  1078→      this.swingGlow = 1
  1079→      const info = makeRuntimeInfo(def, this.sword.stats.dmg)
  1080→      // DASH THROUGH-CUT: the iai pass cuts everything the dash swept
  1081→      // through — the player crosses the target at 980px/s BEFORE the
  1082→      // attack anim even starts, so the blade-only segment points past
  1083→      // them. The corridor is the DASH PATH itself: a fixed chest-height
  1084→      // band from 124px behind the player to 66px ahead — anchored to the
  1085→      // path (not the wandering kissaki, which wraps behind on the
  1086→      // follow-through and would collapse the span). This is the swept
  1087→      // volume of the through-cut.
  1088→      let curSeg = this.blade
  1089→      let usePrev = this.hasPrevBlade
  1090→      if (def.kind === 'dash') {
  1091→        curSeg = {
  1092→          bx: this.x - this.face * 124,
  1093→          by: this.y - 55,
  1094→          tx: this.x + this.face * 66,
  1095→          ty: this.y - 55,
  1096→        }
  1097→        usePrev = false
  1098→      }
  1099→      for (const e of w.enemies) {
  1100→        // spawning enemies within blade reach (landing pods, y > -90) are
  1101→        // fair game — the swing connects the moment the body crosses the arc
  1102→        if (e.dead || (e.spawning && e.y < -90)) continue
  1103→        const last = a.hitMap.get(e.id)
  1104→        const reArm = h.multiHit ? (last === undefined || animT - last >= h.multiHit) : last === undefined
  1105→        if (!reArm) continue

  1109→          this.onLandedHit(w, info, e.x, e.y - 48 * e.scale)
  1110→        }
  1111→      }
  1112→      // remember this frame's blade for the next sweep test
  1113→      this.prevBlade.bx = this.blade.bx
  1114→      this.prevBlade.by = this.blade.by
  1115→      this.prevBlade.tx = this.blade.tx
  1116→      this.prevBlade.ty = this.blade.ty
  1117→      this.hasPrevBlade = true
  1118→      // aoe shockwave at strike moment for heavy hits — the wave is REAL:
  1119→      // enemies inside the radius eat the strike (the charged/h3 promise:
  1120→      // the shockwave visual must connect, not just decorate)
  1121→      if (h.aoe && h.aoe > 0 && !a.impactDone && this.grounded && animT >= h.start + 0.02) {
  1122→        a.impactDone = true
  1123→        w.fx.shockwave(this.x + this.face * 30, 0, h.aoe, this.sword.colors.glow, 0.4, 6)
  1124→        if (this.grounded) w.fx.groundCrack(this.x + this.face * 34, 0, h.aoe * 0.45)
  1125→        for (const e of w.enemies) {
  1126→          if (e.dead || e.spawning) continue
  1127→          if (a.hitMap.has(e.id) && !h.multiHit) continue
  1128→          if (Math.abs(e.x - this.x) < h.aoe && e.y > -140) {
  1129→            a.hitMap.set(e.id, animT)
  1130→            w.hitEnemy(e.id, this.blade, info)
  1131→            this.onLandedHit(w, info, e.x, e.y - 48 * e.scale)
  1132→          }
  1133→        }
  1134→      }
  1135→    }
  1136→
  1137→    // OVERSHOOT (follow-through law): as the active window closes, inject a
  1138→    // 2-frame rotational impulse — the tip pushes PAST the stop point and
  1139→    // the wrist spring elastic-snaps it back. Stopping dead reads weightless.
  1140→    if (!a.overshootDone && animT > h.end && this.armVel !== 0) {
  1141→      a.overshootDone = true
  1142→      this.bladeLagVel += Math.sign(this.armVel) * 2.4
  1143→    }
  1144→
  1145→    // trail emission
  1146→    if (animT >= def.trail.start && animT <= def.trail.end) {
  1147→      w.fx.pushTrail(a.trail, this.blade.bx, this.blade.by, this.blade.tx, this.blade.ty)
  1148→    } else if (animT > def.trail.end && a.trail.emitting) {
  1149→      w.fx.endTrail(a.trail)
  1150→    }
  1151→
  1152→    // cancel window → allow chaining / dash-cancel / jump-cancel
  1153→    const dashCancel = animT >= h.end * 0.6
  1154→    if (animT >= def.cancel || dashCancel) {
  1155→      const input = w.input
  1156→      if ((input.pressed('dash') || input.buffered('dash')) && this.canDash()) {
  1157→        w.fx.endTrail(a.trail)
  1158→        this.startDash(w, input.axisX(), input.axisY())
  1159→        return
  1160→      }
  1161→      if (input.pressed('parry')) {
  1162→        w.fx.endTrail(a.trail)
  1163→        this.enterParry(w)
  1164→        return
  1165→      }
  1166→      if (animT >= def.cancel) {
  1167→        // chain next attack
  1168→        const ax = input.axisX()
  1169→        const ay = input.axisY()
  1170→        let next: string | null = null
  1171→        if (input.buffered('light')) {
  1172→          input.takeBuffered('light')
  1173→          next = this.riposteT > 0 && this.grounded ? 'riposte' : this.pickLight(w, ax, ay)
  1174→        } else if (input.buffered('heavy')) {
  1175→          input.takeBuffered('heavy')
  1176→          if (!this.grounded) {
  1177→            next = 'slam'
  1178→          } else if (input.down('heavy')) {
  1179→            // still holding → convert to charge
  1180→            this.enterCharge(w)
  1181→            w.fx.endTrail(a.trail)
  1182→            return
  1183→          } else {
  1184→            // advance the heavy CHAIN (h1 → h2 → h3) like the light chain
  1185→            const heavyNext = def.chain.find((c) => /^h[123]$/.test(c))
  1186→            next = heavyNext ?? (def.kind === 'heavy' ? 'l1' : 'h1')
  1187→          }
  1188→        }
  1189→        if (next) {
  1190→          // validate chain (or allow from-parry riposte)
  1191→          if (fromChain(def, next, this)) {
  1192→            w.fx.endTrail(a.trail)
  1193→            this.startAttack(ATTACKS[next], w)
  1194→            return
  1195→          }
  1196→        }
  1197→        // jump cancel on ground
  1198→        if (this.grounded && (input.pressed('jump') || input.buffered('jump', 0.15))) {
  1199→          w.fx.endTrail(a.trail)
  1200→          this.doJump(w)
  1201→          this.state = 'move'
  1202→          return
  1203→        }
  1204→      }
  1205→    }
  1206→
  1207→    // finish — capture the swing's final rotational momentum: the body
  1208→    // carries it into locomotion (post-attack recovery, spec §7)
  1209→    if (this.animator.isDone()) {
  1210→      w.fx.endTrail(a.trail, 0.18)
  1211→      this.postAttackT = 0.34
  1212→      this.postArmVel = this.armVel
  1213→      this.action = null
  1214→      this.state = 'move'
  1215→    }
  1216→  }

  1371→      }
  1372→      return true

  1375→    this.flashT = 1
  1376→    this.hurtT = 0.34
  1377→    this.iframes = 0.75
  1378→    this.action = null
  1379→    this.charging = false

  1381→    this.state = 'hurt'
  1382→    this.vx = dir * (heavy ? 380 : 240)

  1386→    this.animator.play(HURT_CLIP, { blend: 0.05 })
  1387→    w.audio.play('hurt', { volume: 0.8, intensity: heavy ? 1 : 0.5 })

  1390→    w.cam.addTrauma(heavy ? 0.45 : 0.3)
  1391→    w.addHitstop(0.06)
  1392→    if (this.hp <= 0) {
  1393→      this.state = 'dead'
  1394→      this.deadT = 0
  1395→      this.vx = dir * 300
  1396→      this.vy = -420
  1397→    }

  1699→  /** FOOT-GROUND LOCK — the hip sits at the planted foot's extension. This
  1700→   * is the weight system: idle bent knees no longer float (measured 9.5px
  1701→   * gap), lunges/crouches stay exactly grounded, landing squashes through
  1702→   * the leg bend instead of sinking into the floor. Airborne → nominal
  1703→   * hip height (physics owns y). */
  1704→  private updateFootLock(dt: number): void {
  1705→    const g = this.geom
  1706→    // lowest contact point: ankles AND toes (a pointed trailing toe grazes
  1707→    // the ground at toe-off — real gait), plus the sole half-width so the
  1708→    // visual foot bottom kisses the line, not the ankle joint center
  1709→    const lowContact = Math.max(g.ankleL.y, g.ankleR.y, g.toeL.y, g.toeR.y) + 2.4
  1710→    const support = lowContact * SCALE
  1711→    const groundedPose = this.grounded && this.state !== 'dead'
  1712→    let target: number
  1713→    if (groundedPose && support >= HIP_H * 0.86) {
  1714→      // SUPPORT phase — the lowest foot is load-bearing (within 14% of full
  1715→      // leg extension): pin the hip so the sole EXACTLY contacts. This is
  1716→      // the weight: no float, no dig, crouches stay crouched.
  1717→      target = support
  1718→    } else {
  1719→      // FLIGHT (both feet swinging/tucked mid-stride) or airborne: physics
  1720→      // owns height — honor the clip's authored bob instead of dragging the
  1721→      // hip down to a tucked toe (which inverted the run's natural bounce)
  1722→      target = HIP_H + this.pose[1] * SCALE
  1723→    }
  1724→    // k=24: tracks a 0.3s stride with ~2px residual — crisp, never mushy
  1725→    this.footLock = damp(this.footLock, target, 24, dt)
  1726→  }

  1728→  /** BLADE DYNAMICS — rotational inertia of the katana as a spring-damper
  1729→   * on the wrist channel. When the arm accelerates (windup → cut), the tip
  1730→   * lags behind the drive angle; when the arm decelerates, the stored
  1731→   * energy whips the tip THROUGH the target line with a small overshoot —
  1732→   * the physical furi-otoshi snap real swordsmen feel. Under-damped
  1733→   * (ζ≈0.3): lively, never mushy. Applied BEFORE solve → hitboxes inherit
  1734→   * the physical blade, not just the keyframed one.
  1735→   *
  1736→   * During hit-stop (world timeScale ≈ 0.05) the blade takes a ±5° impact
  1737→   * resistance deflection AGAINST its swing direction — the cut bites into
  1738→   * something with mass (spec: kinetic impact modulation). */
  1739→  private updateBladeDynamics(dt: number, w: WorldRef): void {
  1740→    if (dt <= 1e-5) return
  1741→    const p = this.pose
  1742→    const armSum = p[CH.shR] + p[CH.elR] + p[CH.wrR]
  1743→    const armVel = (armSum - this.prevArmSum) / dt
  1744→    this.armVel = armVel
  1745→    this.prevArmSum = armSum
  1746→    // lag target: proportional to angular velocity (inertia resisting)
  1747→    const target = w.timeScale < 0.5
  1748→      ? clamp(-Math.sign(armVel || this.bladeLag || 1) * 0.09, -0.15, 0.15) // impact resistance ~5°
  1749→      : clamp(-armVel * 0.045, -0.15, 0.15)
  1750→    // spring toward target, under-damped for the whip-through
  1751→    const K = 260
  1752→    const C = 9
  1753→    this.bladeLagVel += (K * (target - this.bladeLag) - C * this.bladeLagVel) * dt
  1754→    this.bladeLag += this.bladeLagVel * dt
  1755→    this.bladeLag = clamp(this.bladeLag, -0.17, 0.17)
  1756→    this.bladeLagVel = clamp(this.bladeLagVel, -6, 6)
  1757→    p[CH.wrR] += this.bladeLag
  1758→
  1759→    // KINETIC CHAIN (hips 30% · torso 50% · shoulders 20%): a slower spring
  1760→    // driven by the same arm whip. Accelerating into a cut → the torso loads
  1761→    // INTO the swing; the deceleration at impact → the body recoils back.
  1762→    // The upper body can never again rotate independently of the hips.
  1763→    const k2 = 90
  1764→    const c2 = 7.5
  1765→    const chainTarget = clamp(-armVel * 0.02, -0.14, 0.14)
  1766→    this.chainLagVel += (k2 * (chainTarget - this.chainLag) - c2 * this.chainLagVel) * dt
  1767→    this.chainLag = clamp(this.chainLag + this.chainLagVel * dt, -0.16, 0.16)
  1768→    p[CH.rootRot] += -this.chainLag * 0.3
  1769→    p[CH.spine] += -this.chainLag * 0.5
  1770→    p[CH.shR] += this.chainLag * 0.2
  1771→
  1772→    // GROUND CLEARANCE — the floor is SOLID: the kissaki can never pass
  1773→    // below the ground plane. Clip-blends between carries (guard ↔ walk
  1774→    // side-carry ↔ run trail) linearly sweep the wrist through
  1775→    // straight-down angles; without this, the tip pierces the floor for
  1776→    // ~10 frames on every transition. A positional wrist constraint rotates
  1777→    // the tip toward the NEARER horizon (forward-level or back-level,
  1778→    // preserving the carry direction), rate-limited so the lowering reads
  1779→    // as the kissaki TAPPING the floor and sliding along it.
  1780→    // (Attacks exempt: their tips target enemy bodies above the floor.)
  1781→    if (this.state !== 'attack') {
  1782→      const sum = p[CH.shR] + p[CH.elR] + p[CH.wrR]
  1783→      const bladeA = Math.PI / 2 - sum
  1784→      const sinA = Math.sin(bladeA)
  1785→      const cosA = Math.cos(bladeA)
  1786→      if (sinA > 0.05) {
  1787→        const len = PLAYER_SPEC.blade * this.sword.stats.range * 1.06 * this.swingScale
  1788→        const tipLY = this.geom.handR.y + sinA * len
  1789→        // local floor = the stance foot's contact point (crouch-aware),
  1790→        // small margin for the drawn edge width
  1791→        const floorLocal = Math.max(
  1792→          this.geom.ankleL.y, this.geom.ankleR.y, this.geom.toeL.y, this.geom.toeR.y,
  1793→        ) + 2.5
  1794→        if (tipLY > floorLocal) {
  1795→          const dY = tipLY - floorLocal
  1796→          // solve for the angle that puts the tip exactly on the floor
  1797→          const sinT = Math.max(0.06, (sinA * len - dY) / len)
  1798→          const aT = Math.asin(clamp01(sinT))
  1799→          // quadrant-safe (raw channels are up-wound): cos<0 → back-down
  1800→          // carry rotates toward straight-back; cos≥0 → forward-down
  1801→          // rotates toward level-forward — the carry direction is preserved
  1802→          if (cosA < 0) {
  1803→            const aNorm = Math.PI - Math.asin(clamp01(sinA))
  1804→            const delta = Math.PI - aT - aNorm
  1805→            p[CH.wrR] -= clamp(delta, 0, 20 * dt)
  1806→          } else {
  1807→            const aNorm = Math.asin(clamp01(sinA))
  1808→            const delta = aNorm - aT
  1809→            p[CH.wrR] += clamp(delta, 0, 20 * dt)
  1810→          }
  1811→        }
  1812→      }
  1813→    }
  1814→  }

  1864→  /** TWO-HANDED GRIP — the off hand solves 2-bone IK onto the tsuka butt
  1865→   * (te-no-uchi: left hand at the pommel, right near the guard — how a
  1866→   * katana is actually swung). Blended over the authored off-arm channels
  1867→   * with a state-driven weight so running/dashing pumps stay authored. */
  1868→  private applyTwoHandGrip(dt: number): void {
  1869→    // desired grip weight by state
  1870→    let target = 0
  1871→    switch (this.state) {
  1872→      case 'move':
  1873→        target = this.grounded && Math.abs(this.vx) < 60 ? 1 : 0.1
  1874→        break
  1875→      case 'attack':
  1876→        target = this.action ? (this.action.def.grip ?? 0.45) : 0.45
  1877→        break
  1878→      case 'charge':
