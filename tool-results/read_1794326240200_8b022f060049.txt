    27→/** swaps left/right LEG channels only — the arms keep their authored
    28→ * channels (the sword stays in the right hand across both strides). */
    29→function mirrorLegs(src: PoseInput): PoseInput {
    30→  const out: PoseInput = { ...src }
    31→  const pairs: [keyof PoseInput, keyof PoseInput][] = [
    32→    ['hipL', 'hipR'], ['kneeL', 'kneeR'], ['footL', 'footR'],
    33→  ]
    34→  for (const [l, r] of pairs) {
    35→    const lv = src[l]
    36→    const rv = src[r]
    37→    if (lv !== undefined) out[r] = lv
    38→    if (rv !== undefined) out[l] = rv
    39→  }
    40→  return out
    41→}
    42→
    43→/** STRIDE MIRROR for bipedal cycles: legs swap, the SWORD ARM (right)
    44→ * keeps the exact same carry across both strides (a held weapon does not
    45→ * teleport between arms / change carry mid-cycle), and the off-arm
    46→ * negates its swing so the contralateral counter-phase survives.
    47→ * DEPTH channels mirror with the stride: pelvis/chest yaw negate (the
    48→ * opposite hip now drives), the off-arm's depth swing negates — while the
    49→ * sword-side channels (armRz, bladeYaw) carry unchanged, like the arm. */
    50→function mirrorStride(src: PoseInput): PoseInput {
    51→  const out = mirrorLegs(src)
    52→  if (src.shL !== undefined) out.shL = -src.shL
    53→  if (src.pelvisYaw !== undefined) out.pelvisYaw = -src.pelvisYaw
    54→  if (src.chestYaw !== undefined) out.chestYaw = -src.chestYaw
    55→  if (src.armLz !== undefined) out.armLz = -src.armLz
    56→  // the off FOREARM's own depth yaw negates WITH its upper arm — leaving
    57→  // it un-mirrored desyncs the elbow's depth plane on stride B (the
    58→  // "forearm flips against the upper arm" depth incoherence)
    59→  if (src.foreLz !== undefined) out.foreLz = -src.foreLz
    60→  if (src.armRz !== undefined) out.armRz = src.armRz
    61→  if (src.bladeYaw !== undefined) out.bladeYaw = src.bladeYaw
    62→  return out
    63→}

    65→/** Hand-space authoring for human gait. This fixes the old shoulder-only
    66→ * mirroring, which moved both hands behind the torso instead of producing
    67→ * a genuine contralateral arm swing. The elbow pole is solved by the rig.
    68→ */
    69→function humanFreeArm(x: number, y: number): PoseInput {
    70→  const upper = PLAYER_SPEC.upperArm, fore = PLAYER_SPEC.foreArm
    71→  const r = Math.max(12, Math.min(upper + fore - 2, Math.hypot(x, y)))
    72→  const angle = Math.acos(Math.max(-1, Math.min(1, (upper*upper + r*r - fore*fore) / (2*upper*r))))
    73→  const flex = -(Math.PI - Math.acos(Math.max(-1, Math.min(1, (upper*upper + fore*fore - r*r) / (2*upper*fore)))))
    74→  return { shL: Math.atan2(x, y) + angle, elL: flex }
    75→}
    76→function mirrorHumanStride(src: PoseInput): PoseInput {
    77→  const sh = src.shL ?? 0, el = src.elL ?? -.45
    78→  const x = PLAYER_SPEC.upperArm*Math.sin(sh) + PLAYER_SPEC.foreArm*Math.sin(sh + el)
    79→  const y = PLAYER_SPEC.upperArm*Math.cos(sh) + PLAYER_SPEC.foreArm*Math.cos(sh + el)
    80→  // Sprint backswing drops the hand toward the hip; reflecting only X
    81→  // would put the rear fist unnaturally up behind the shoulder.
    82→  const mirroredY = (src.rootRot ?? 0) > .13
    83→    ? y + (-x < 0 ? Math.min(12, Math.abs(x) * .8) : -Math.min(8, Math.abs(x) * .4)) : y
    84→  return { ...mirrorStride(src), ...humanFreeArm(-x, mirroredY) }
    85→}
