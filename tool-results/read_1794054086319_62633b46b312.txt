     1→/** STICKMAN RISING — the modular combo moveset.
     2→ *
     3→ * Each attack = a keyframed clip (anticipation → strike → follow-through →
     4→ * recovery) + hitbox window (the physical blade sweep is the hitbox) + trail
     5→ * window + cancel window + legal follow-up chain. Directional inputs,
     6→ * dash-cancels and buffers are resolved by the Player combo machine.
     7→ *
     8→ * Angle conventions (see pose.ts): arm angles measured from straight-down,
     9→ * positive swings toward facing. Blade direction = π/2 − (shR+elR+wrR).
    10→ */
    11→
    12→import { Clip, keyframe, makeClip } from '../anim/animator'
    13→import { GUARD, GUARD_AIR } from '../anim/clips'
    14→import type { PoseInput } from '../anim/pose'
    15→import type { SfxName } from '../audio/audio'
    16→
    17→export type AttackKind =
    18→  | 'light' | 'heavy' | 'launcher' | 'air' | 'slam' | 'dash'
    19→  | 'spin' | 'charged' | 'riposte' | 'finisher' | 'ability'
    20→
    21→export interface AttackHit {
    22→  start: number
    23→  end: number
    24→  dmg: number
    25→  stagger: number
    26→  kb: number
    27→  kbY?: number
    28→  launch?: number
    29→  juggle?: number
    30→  hitstop: number
    31→  shake: number
    32→  heavy: boolean
    33→  aoe?: number
    34→  multiHit?: number
    35→  breakGuard?: boolean
    36→}
    37→
    38→export interface AttackDef {
    39→  id: string
    40→  label: string
    41→  kind: AttackKind
    42→  clip: Clip
    43→  hit: AttackHit
    44→  trail: { start: number; end: number }
    45→  swingSfx: { t: number; name: SfxName }
    46→  cancel: number
    47→  chain: string[]
    48→  moveX?: number
    49→  moveXAt?: number
    50→  moveY?: number
    51→  moveYAt?: number
    52→  requires: 'ground' | 'air' | 'any'
    53→  /** gravity scale while the attack is active (air-combo float) */
    54→  float?: number
    55→  /** AoE + fx fire when the player touches ground mid-attack (slams / dive) */
    56→  slamImpact?: { r: number; dmg: number; launch: number }
    57→}
