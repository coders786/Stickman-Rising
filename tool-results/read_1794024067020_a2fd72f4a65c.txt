   145→  /* ---------------- particles ---------------- */

   147→  private nextParticle(): Particle | null {
   148→    let p = this.pFree.pop()
   149→    if (!p) {
   150→      if (this.particles.length >= MAX_PARTICLES) return null
   151→      p = {
   152→        active: false, x: 0, y: 0, vx: 0, vy: 0, age: 0, life: 1,
   153→        size: 2, color: '#fff', type: 'dot', grav: 0, drag: 0,
   154→        rot: 0, vrot: 0, pts: null,
   155→      }
   156→      this.particles.push(p)
   157→    }
   158→    p.active = true
   159→    return p
   160→  }

   162→  private scaleCount(n: number): number {
   163→    return Math.max(1, Math.round(n * this.qualityScale))

   166→  /** directional spark cone */
   167→  sparks(x: number, y: number, dir: number, spread: number, count: number, color: string, speed = 320, life = 0.4): void {
   168→    const n = this.scaleCount(count)
   169→    for (let i = 0; i < n; i++) {
   170→      const p = this.nextParticle()
   171→      if (!p) return
   172→      const a = dir + rand(-spread, spread)
   173→      const sp = speed * rand(0.45, 1.25)
   174→      p.x = x; p.y = y
   175→      p.vx = Math.cos(a) * sp
   176→      p.vy = Math.sin(a) * sp
   177→      p.age = 0; p.life = life * rand(0.6, 1.3)
   178→      p.size = rand(1.4, 3.2)
   179→      p.color = color
   180→      p.type = 'spark'
   181→      p.grav = 640
   182→      p.drag = 2.4
   183→      p.rot = 0; p.vrot = 0; p.pts = null
   184→    }
   185→  }

   191→      const p = this.nextParticle()

   269→  /** jagged lightning bolt between two points */
   270→  lightningArc(x0: number, y0: number, x1: number, y1: number, color: string, segments = 6): void {
   271→    const p = this.nextParticle()
   272→    if (!p) return
   273→    const pts: number[] = [x0, y0]
   274→    for (let i = 1; i < segments; i++) {
   275→      const t = i / segments
   276→      const jitter = 14 * (1 - Math.abs(t - 0.5) * 2 + 0.2)
   277→      pts.push(
   278→        x0 + (x1 - x0) * t + rand(-jitter, jitter),
   279→        y0 + (y1 - y0) * t + rand(-jitter, jitter),
   280→      )
   281→    }
   282→    pts.push(x1, y1)
   283→    p.x = x0; p.y = y0
   284→    p.vx = 0; p.vy = 0
   285→    p.age = 0; p.life = 0.22
   286→    p.size = 2.4
   287→    p.color = color
   288→    p.type = 'arc'
   289→    p.grav = 0; p.drag = 0
   290→    p.rot = 0; p.vrot = 0
   291→    p.pts = pts
   292→    // spark burst at both ends
   293→    this.sparks(x1, y1, Math.atan2(y1 - y0, x1 - x0), 0.7, 5, color, 240, 0.3)
   294→  }

   302→      r.x = x; r.y = y

   304→      r.r1 = heavy ? 78 : 40
   305→      r.age = 0; r.life = heavy ? 0.3 : 0.2
   306→      r.color = heavy ? color : accent
   307→      r.width = heavy ? 5 : 3
   308→      r.squash = 1
   309→    }
   310→    if (heavy) {
   311→      this.burst(x, y, 8, accent, 200, 0.4, 'shard')
   312→    }
   313→  }

   316→  dismember(x: number, y: number, dir: number, bodyColor: string, accent: string): void {
   317→    const gore = this.gore === 'blood'
   318→    const n = this.scaleCount(6)
   319→    for (let i = 0; i < n; i++) {
   320→      const d = this.dFree.pop() ?? {
   321→        active: false, x: 0, y: 0, vx: 0, vy: 0, rot: 0, vrot: 0, len: 10, color: '#fff', age: 0, life: 1,
   322→      }
   323→      d.active = true
   324→      d.x = x; d.y = y - rand(0, 40)
   325→      const a = dir + rand(-0.9, 0.9)
   326→      const sp = rand(220, 560)
   327→      d.vx = Math.cos(a) * sp
   328→      d.vy = -rand(120, 420)
   329→      d.rot = rand(TAU)
   330→      d.vrot = rand(-16, 16)
   331→      d.len = rand(9, 22)
   332→      d.color = i % 3 === 0 ? accent : bodyColor
   333→      d.age = 0; d.life = rand(1.2, 2.2)
   334→      this.debris.push(d)
   335→    }
   336→    this.burst(x, y - 20, gore ? 18 : 14, gore ? '#c31432' : accent, 380, 0.7, gore ? 'dot' : 'spark')
   337→    if (!gore) this.burst(x, y - 20, 10, '#ffffff', 300, 0.4, 'spark')
   338→  }

   381→  groundCrack(x: number, y: number, size: number, color = '#0b0f14'): void {
   382→    const c = this.cFree.pop() ?? {
   383→      active: false, x: 0, y: 0, pts: [] as number[], age: 0, life: 1, color: '#0b0f14',
   384→    }
   385→    c.active = true
   386→    c.x = x; c.y = y
   387→    c.age = 0; c.life = 5
   388→    c.color = color
   389→    const pts: number[] = [0, 0]
   390→    const branches = 3
   391→    for (let b = 0; b < branches; b++) {
   392→      let a = rand(TAU)
   393→      if (b === 0) a = 0
   394→      if (b === 1) a = Math.PI
   395→      let px = 0, py = 0
   396→      const segs = 3
   397→      for (let s = 0; s < segs; s++) {
   398→        a += rand(-0.5, 0.5)
   399→        px += Math.cos(a) * size * rand(0.24, 0.4)
   400→        py += Math.sin(a) * size * rand(0.08, 0.16)
   401→        pts.push(px, py)
   402→      }
   403→    }
   404→    c.pts = pts

   474→  pushTrail(t: Trail, bx: number, by: number, tx: number, ty: number): void {

   477→    t.pts[i + 1] = by

   479→    t.pts[i + 3] = ty

   491→  /** a standalone slash arc (ability echoes, cinematic slashes) */
   492→  slashArc(x: number, y: number, angle: number, spread: number, radius: number, color: string, color2: string, width = 1): void {
   493→    const t = this.startTrail(color, color2, width)
   494→    t.life = 0.3
   495→    t.emitting = false
   496→    t.age = 0
   497→    const steps = 9
   498→    for (let i = 0; i <= steps; i++) {
   499→      const a = angle - spread / 2 + (spread * i) / steps
   500→      const r0 = radius * 0.25
   501→      pushPolar(t, x, y, a, r0, radius)
   502→    }
   503→  }

   505→  /* ---------------- update ---------------- */
   506→
   507→  update(dt: number): void {
   508→    // particles
   509→    for (let i = this.particles.length - 1; i >= 0; i--) {
   510→      const p = this.particles[i]
   511→      if (!p.active) continue
   512→      p.age += dt
   513→      if (p.age >= p.life) {
   514→        p.active = false
   515→        this.pFree.push(p)
   516→        continue
   517→      }
   518→      p.vy += p.grav * dt
   519→      const dr = 1 - p.drag * dt
   520→      p.vx *= dr > 0 ? dr : 0
   521→      p.vy *= dr > 0 ? dr : 0
   522→      p.x += p.vx * dt
   523→      p.y += p.vy * dt
   524→    }
   525→    // trails
   526→    for (let i = this.trails.length - 1; i >= 0; i--) {
   527→      const t = this.trails[i]
   528→      if (!t.active) continue
   529→      if (!t.emitting) {
   530→        t.age += dt
   531→        if (t.age >= t.life) {
   532→          t.active = false
   533→          this.tFree.push(t)
   534→        }
   535→      }
   536→    }
   537→    // afterimages
   538→    for (let i = this.afterimages.length - 1; i >= 0; i--) {
   539→      const a = this.afterimages[i]
   540→      a.age += dt
   541→      if (a.age >= a.life) {
   542→        a.active = false
   543→        this.aFree.push(a)
   544→      }
   545→    }
   546→    // rings
   547→    for (let i = this.rings.length - 1; i >= 0; i--) {
   548→      const r = this.rings[i]
   549→      r.age += dt
   550→      if (r.age >= r.life) {
   551→        r.active = false
   552→        this.rFree.push(r)
   553→      }
   554→    }
   555→    // cracks
   556→    for (let i = this.cracks.length - 1; i >= 0; i--) {
   557→      const c = this.cracks[i]
   558→      c.age += dt
   559→      if (c.age >= c.life) {
   560→        c.active = false
   561→        this.cFree.push(c)
   562→      }
   563→    }
   564→    // texts
   565→    for (let i = this.texts.length - 1; i >= 0; i--) {
   566→      const t = this.texts[i]
   567→      t.age += dt
   568→      if (t.age >= t.life) {
   569→        t.active = false
   570→        this.txtFree.push(t)
   571→      }
   572→      t.vy += 320 * dt
   573→      t.x += t.vx * dt
   574→      t.y += t.vy * dt
   575→    }
   576→    // debris
   577→    for (let i = this.debris.length - 1; i >= 0; i--) {
   578→      const d = this.debris[i]
   579→      d.age += dt
   580→      if (d.age >= d.life) {
   581→        d.active = false
   582→        this.dFree.push(d)
   583→        continue
   584→      }
   585→      d.vy += 900 * dt
   586→      d.x += d.vx * dt
   587→      d.y += d.vy * dt
   588→      d.rot += d.vrot * dt
   589→      if (d.y > 0) {
   590→        d.y = 0
   591→        d.vy *= -0.42
   592→        d.vx *= 0.6
   593→        d.vrot *= 0.5
   594→        if (Math.abs(d.vy) < 40) d.vy = 0
   595→      }
   596→    }
   597→    // flash
   598→    if (this.flashAlpha > 0) {
   599→      this.flashAlpha -= this.flashDecay * dt
   600→      if (this.flashAlpha < 0) this.flashAlpha = 0
   601→    }
   602→  }

   606→  /** ground layer: cracks + debris + afterimages (under entities) */
   607→  drawUnder(ctx: CanvasRenderingContext2D): void {
   608→    ctx.save()
   609→    // cracks
   610→    for (const c of this.cracks) {
   611→      if (!c.active) continue
   612→      const a = 1 - c.age / c.life
   613→      ctx.strokeStyle = rgba('#000000', a * 0.5)
   614→      ctx.lineWidth = 3
   615→      drawCrack(ctx, c)
   616→      ctx.strokeStyle = rgba('#39505f', a * 0.4)
   617→      ctx.lineWidth = 1.2
   618→      drawCrack(ctx, c)
   619→    }
   620→    // debris
   621→    ctx.lineCap = 'round'
   622→    for (const d of this.debris) {
   623→      if (!d.active) continue
   624→      const a = 1 - d.age / d.life
   625→      ctx.strokeStyle = rgba(d.color, a)
   626→      ctx.lineWidth = 3.4
   627→      const c = Math.cos(d.rot) * d.len / 2
   628→      const s = Math.sin(d.rot) * d.len / 2
   629→      ctx.beginPath()
   630→      ctx.moveTo(d.x - c, d.y - s)
   631→      ctx.lineTo(d.x + c, d.y + s)
   632→      ctx.stroke()
   633→    }
   634→    ctx.restore()
   635→  }

   637→  /** additive layer: afterimages, trails, particles, rings */
   638→  drawAdditive(ctx: CanvasRenderingContext2D, time: number): void {
   639→    ctx.save()
   640→    ctx.globalCompositeOperation = 'lighter'
   641→
   642→    // afterimages (ghost stickmen)
   643→    for (const a of this.afterimages) {
   644→      const k = 1 - a.age / a.life
   645→      ctx.save()
   646→      ctx.translate(a.x, a.y)
   647→      ctx.scale(a.face * a.scale, a.scale)
   648→      aiPose.set(a.pose)
   649→      solveSkeleton(aiPose, PLAYER_SPEC, scrashGeom)
   650→      drawStick(ctx, scrashGeom, PLAYER_STYLE, {
   651→        ghost: a.color,
   652→        ghostAlpha: k * 0.5,
   653→        time,
   654→      })
   655→      ctx.restore()
   656→    }
   657→
   658→    // slash trail ribbons
   659→    for (const t of this.trails) {
   660→      if (!t.active || t.count < 2) continue
   661→      const fade = t.emitting ? 1 : 1 - t.age / t.life
   662→      drawTrail(ctx, t, fade)
   663→    }
   664→
   665→    // particles
   666→    for (const p of this.particles) {
   667→      if (!p.active) continue
   668→      const k = 1 - p.age / p.life
   669→      switch (p.type) {
   670→        case 'spark': {
   671→          ctx.strokeStyle = rgba(p.color, k)
   672→          ctx.lineWidth = p.size * 0.8
   673→          ctx.beginPath()
   674→          ctx.moveTo(p.x, p.y)
   675→          ctx.lineTo(p.x - p.vx * 0.03, p.y - p.vy * 0.03)
   676→          ctx.stroke()
   677→          break
   678→        }
   679→        case 'streak': {
   680→          ctx.strokeStyle = rgba(p.color, k * 0.5)
   681→          ctx.lineWidth = 1.6
   682→          ctx.beginPath()
   683→          ctx.moveTo(p.x, p.y)
   684→          ctx.lineTo(p.x - p.vx * 0.08, p.y)
   685→          ctx.stroke()
   686→          break
   687→        }
   688→        case 'ember': {
   689→          const fl = 0.6 + 0.4 * Math.sin(p.age * 30 + p.x)
   690→          ctx.fillStyle = rgba(p.color, k * fl)
   691→          ctx.beginPath()
   692→          ctx.arc(p.x, p.y, p.size * (0.7 + k * 0.5), 0, TAU)
   693→          ctx.fill()
   694→          break
   695→        }
   696→        case 'smoke': {
   697→          ctx.fillStyle = rgba(p.color, k * 0.24)
   698→          ctx.beginPath()
   699→          ctx.arc(p.x, p.y, p.size * (1.6 - k * 0.6), 0, TAU)
   700→          ctx.fill()
   701→          break
   702→        }
   703→        case 'shard': {
   704→          ctx.strokeStyle = rgba(p.color, k)
   705→          ctx.lineWidth = 2
   706→          ctx.save()
   707→          ctx.translate(p.x, p.y)
   708→          ctx.rotate(p.age * 9)
   709→          ctx.beginPath()
   710→          ctx.moveTo(-p.size, 0)
   711→          ctx.lineTo(p.size, 0)
   712→          ctx.stroke()
   713→          ctx.restore()
   714→          break
   715→        }
   716→        case 'arc': {
   717→          if (p.pts) {
   718→            const fl = k * (0.7 + 0.3 * Math.sin(time * 60 + p.x))
   719→            ctx.strokeStyle = rgba(p.color, fl)
   720→            ctx.lineWidth = p.size
   721→            ctx.beginPath()
   722→            ctx.moveTo(p.pts[0], p.pts[1])
   723→            for (let i = 2; i < p.pts.length; i += 2) {
   724→              ctx.lineTo(p.pts[i], p.pts[i + 1])
   725→            }
   726→            ctx.stroke()
   727→            ctx.strokeStyle = rgba('#ffffff', fl * 0.8)
   728→            ctx.lineWidth = 1
   729→            ctx.stroke()
   730→          }
   731→          break
   732→        }
   733→        default: {
   734→          ctx.fillStyle = rgba(p.color, k)
   735→          ctx.beginPath()
   736→          ctx.arc(p.x, p.y, p.size, 0, TAU)
   737→          ctx.fill()
   738→        }
   739→      }
   740→    }
   741→
   742→    // rings
   743→    for (const r of this.rings) {
   744→      if (!r.active) continue
   745→      const k = 1 - r.age / r.life
   746→      const rr = r.r0 + (r.r1 - r.r0) * (1 - k * k)
   747→      ctx.strokeStyle = rgba(r.color, k * 0.9)
   748→      ctx.lineWidth = r.width * k + 0.5
   749→      ctx.beginPath()
   750→      ctx.ellipse(r.x, r.y, rr, rr * r.squash, 0, 0, TAU)
   751→      ctx.stroke()
   752→      ctx.strokeStyle = rgba('#ffffff', k * 0.5)
   753→      ctx.lineWidth = 1.2
   754→      ctx.beginPath()
   755→      ctx.ellipse(r.x, r.y, rr * 0.92, rr * 0.92 * r.squash, 0, 0, TAU)
   756→      ctx.stroke()
   757→    }
   758→
   759→    // damage numbers (crisp, non-additive, on top of the glow layers)
   760→    ctx.globalCompositeOperation = 'source-over'
   761→    ctx.textAlign = 'center'
   762→    for (const t of this.texts) {
   763→      if (!t.active) continue
   764→      const k = 1 - t.age / t.life
   765→      const pop = t.age < 0.09 ? 1 + (0.09 - t.age) * 4 : 1
   766→      ctx.save()
   767→      ctx.translate(t.x, t.y)
   768→      ctx.scale(pop, pop)
   769→      ctx.font = `900 ${t.size}px "Arial Black", Arial, sans-serif`
   770→      ctx.lineWidth = 3
   771→      ctx.strokeStyle = rgba('#04060a', Math.min(1, k * 1.4))
   772→      ctx.strokeText(t.str, 0, 0)
   773→      ctx.fillStyle = rgba(t.color, Math.min(1, k * 1.6))
   774→      ctx.fillText(t.str, 0, 0)
   775→      ctx.restore()
   776→    }
   777→
   778→    ctx.restore()
   779→  }

   795→function pushPolar(t: Trail, x: number, y: number, a: number, r0: number, r1: number): void {
   796→  const i = t.head * 4
   797→  t.pts[i] = x + Math.cos(a) * r0
   798→  t.pts[i + 1] = y + Math.sin(a) * r0
   799→  t.pts[i + 2] = x + Math.cos(a) * r1
   800→  t.pts[i + 3] = y + Math.sin(a) * r1
   801→  t.head = (t.head + 1) % t.N
   802→  if (t.count < t.N) t.count++
   803→}
   804→
   805→function drawTrail(ctx: CanvasRenderingContext2D, t: Trail, fade: number): void {
   806→  const n = t.count
   807→  const pts = t.pts
   808→  // render oldest → newest
   809→  const start = (t.head - n + t.N + t.N) % t.N
   810→
   811→  const seg = (idx: number, out: number[]): void => {
   812→    const i = ((start + idx) % t.N) * 4
   813→    out[0] = pts[i]; out[1] = pts[i + 1]; out[2] = pts[i + 2]; out[3] = pts[i + 3]
   814→  }
   815→
   816→  const a1: number[] = [0, 0, 0, 0]
   817→  const a2: number[] = [0, 0, 0, 0]
   818→
   819→  // wide soft layer + bright core layer
   820→  for (let layer = 0; layer < 2; layer++) {
   821→    const wide = layer === 0
   822→    const color = wide ? t.color2 : t.color
   823→    const baseAlpha = wide ? 0.3 : 0.55
   824→    for (let i = 0; i < n - 1; i++) {
   825→      seg(i, a1)
   826→      seg(i + 1, a2)
   827→      const headBias = i / Math.max(1, n - 1)
   828→      const alpha = baseAlpha * fade * (0.35 + 0.65 * headBias)
   829→      ctx.fillStyle = rgba(color, alpha)
   830→      // quad between consecutive blade segments
   831→      ctx.beginPath()
   832→      ctx.moveTo(a1[0], a1[1])
   833→      ctx.lineTo(a1[2], a1[3])
   834→      ctx.lineTo(a2[2], a2[3])
   835→      ctx.lineTo(a2[0], a2[1])
   836→      ctx.closePath()
   837→      ctx.fill()
   838→    }
   839→  }
   840→
   841→  // bright cutting edge along the tip path
   842→  if (n >= 2) {
   843→    ctx.strokeStyle = rgba('#ffffff', 0.5 * fade)
   844→    ctx.lineWidth = 1.8 * t.width
   845→    ctx.beginPath()
   846→    seg(0, a1)
   847→    ctx.moveTo(a1[2], a1[3])
   848→    for (let i = 1; i < n; i++) {
   849→      seg(i, a1)
   850→      ctx.lineTo(a1[2], a1[3])
   851→    }
   852→    ctx.stroke()
   853→  }
   854→}

   856→function drawCrack(ctx: CanvasRenderingContext2D, c: Crack): void {
   857→  if (c.pts.length < 4) return
   858→  ctx.beginPath()
   859→  ctx.moveTo(c.x + c.pts[0], c.y + c.pts[1])
   860→  for (let i = 2; i < c.pts.length; i += 2) {
   861→    ctx.lineTo(c.x + c.pts[i], c.y + c.pts[i + 1])
   862→  }
   863→  ctx.stroke()
   864→}
