   164→  /* ---------------- particles ---------------- */

   166→  private nextParticle(): Particle | null {
   167→    let p = this.pFree.pop()
   168→    if (!p) {
   169→      if (this.particles.length >= MAX_PARTICLES) return null
   170→      p = {
   171→        active: false, x: 0, y: 0, vx: 0, vy: 0, age: 0, life: 1,
   172→        size: 2, color: '#fff', type: 'dot', grav: 0, drag: 0,
   173→        rot: 0, vrot: 0, pts: null,
   174→      }
   175→      this.particles.push(p)
   176→    }
   177→    p.active = true
   178→    return p
   179→  }

   181→  private scaleCount(n: number): number {
   182→    return Math.max(1, Math.round(n * this.qualityScale))

   185→  /** directional spark cone */
   186→  sparks(x: number, y: number, dir: number, spread: number, count: number, color: string, speed = 320, life = 0.4): void {
   187→    const n = this.scaleCount(count)
   188→    for (let i = 0; i < n; i++) {
   189→      const p = this.nextParticle()
   190→      if (!p) return
   191→      const a = dir + rand(-spread, spread)
   192→      const sp = speed * rand(0.45, 1.25)
   193→      p.x = x; p.y = y
   194→      p.vx = Math.cos(a) * sp
   195→      p.vy = Math.sin(a) * sp
   196→      p.age = 0; p.life = life * rand(0.6, 1.3)
   197→      p.size = rand(1.4, 3.2)
   198→      p.color = color
   199→      p.type = 'spark'
   200→      p.grav = 640
   201→      p.drag = 2.4
   202→      p.rot = 0; p.vrot = 0; p.pts = null
   203→    }
   204→  }

   218→      p.size = rand(1.6, 3.6)

   288→  /** jagged lightning bolt between two points */
   289→  lightningArc(x0: number, y0: number, x1: number, y1: number, color: string, segments = 6): void {
   290→    const p = this.nextParticle()
   291→    if (!p) return
   292→    const pts: number[] = [x0, y0]
   293→    for (let i = 1; i < segments; i++) {
   294→      const t = i / segments
   295→      const jitter = 14 * (1 - Math.abs(t - 0.5) * 2 + 0.2)
   296→      pts.push(
   297→        x0 + (x1 - x0) * t + rand(-jitter, jitter),
   298→        y0 + (y1 - y0) * t + rand(-jitter, jitter),
   299→      )
   300→    }
   301→    pts.push(x1, y1)
   302→    p.x = x0; p.y = y0
   303→    p.vx = 0; p.vy = 0
   304→    p.age = 0; p.life = 0.22
   305→    p.size = 2.4
   306→    p.color = color
   307→    p.type = 'arc'
   308→    p.grav = 0; p.drag = 0
   309→    p.rot = 0; p.vrot = 0
   310→    p.pts = pts
   311→    // spark burst at both ends
   312→    this.sparks(x1, y1, Math.atan2(y1 - y0, x1 - x0), 0.7, 5, color, 240, 0.3)
   313→  }

   334→  /** dismemberment: spinning limb debris + gore burst */
   335→  dismember(x: number, y: number, dir: number, bodyColor: string, accent: string): void {
   336→    const gore = this.gore === 'blood'
   337→    const n = this.scaleCount(6)
   338→    for (let i = 0; i < n; i++) {
   339→      const d = this.dFree.pop() ?? {
   340→        active: false, x: 0, y: 0, vx: 0, vy: 0, rot: 0, vrot: 0, len: 10, color: '#fff', age: 0, life: 1,
   341→      }
   342→      d.active = true
   343→      d.x = x; d.y = y - rand(0, 40)
   344→      const a = dir + rand(-0.9, 0.9)
   345→      const sp = rand(220, 560)
   346→      d.vx = Math.cos(a) * sp
   347→      d.vy = -rand(120, 420)
   348→      d.rot = rand(TAU)
   349→      d.vrot = rand(-16, 16)
   350→      d.len = rand(9, 22)
   351→      d.color = i % 3 === 0 ? accent : bodyColor
   352→      d.age = 0; d.life = rand(1.2, 2.2)
   353→      this.debris.push(d)
   354→    }
   355→    this.burst(x, y - 20, gore ? 18 : 14, gore ? '#c31432' : accent, 380, 0.7, gore ? 'dot' : 'spark')
   356→    if (!gore) this.burst(x, y - 20, 10, '#ffffff', 300, 0.4, 'spark')
   357→  }

   400→  groundCrack(x: number, y: number, size: number, color = '#0b0f14'): void {
   401→    const c = this.cFree.pop() ?? {
   402→      active: false, x: 0, y: 0, pts: [] as number[], age: 0, life: 1, color: '#0b0f14',
   403→    }
   404→    c.active = true
   405→    c.x = x; c.y = y
   406→    c.age = 0; c.life = 5
   407→    c.color = color
   408→    const pts: number[] = [0, 0]
   409→    const branches = 3
   410→    for (let b = 0; b < branches; b++) {
   411→      let a = rand(TAU)
   412→      if (b === 0) a = 0
   413→      if (b === 1) a = Math.PI
   414→      let px = 0, py = 0
   415→      const segs = 3
   416→      for (let s = 0; s < segs; s++) {
   417→        a += rand(-0.5, 0.5)
   418→        px += Math.cos(a) * size * rand(0.24, 0.4)
   419→        py += Math.sin(a) * size * rand(0.08, 0.16)
   420→        pts.push(px, py)
   421→      }
   422→    }
   423→    c.pts = pts
   424→    this.cracks.push(c)

   508→  pushTrail(t: Trail, bx: number, by: number, tx: number, ty: number): void {
   509→    const i = t.head * 4
   510→    t.pts[i] = bx
   511→    t.pts[i + 1] = by
   512→    t.pts[i + 2] = tx
   513→    t.pts[i + 3] = ty
   514→    t.head = (t.head + 1) % t.N
   515→    if (t.count < t.N) t.count++
   516→  }

   525→  /** a standalone slash arc (ability echoes, cinematic slashes) */
   526→  slashArc(x: number, y: number, angle: number, spread: number, radius: number, color: string, color2: string, width = 1): void {
   527→    const t = this.startTrail(color, color2, width)
   528→    t.life = 0.3
   529→    t.emitting = false
   530→    t.age = 0
   531→    const steps = 9
   532→    for (let i = 0; i <= steps; i++) {
   533→      const a = angle - spread / 2 + (spread * i) / steps
   534→      const r0 = radius * 0.25
   535→      pushPolar(t, x, y, a, r0, radius)
   536→    }
   537→  }

   539→  /* ---------------- update ---------------- */
   540→
   541→  update(dt: number): void {
   542→    // particles
   543→    for (let i = this.particles.length - 1; i >= 0; i--) {
   544→      const p = this.particles[i]
   545→      if (!p.active) continue
   546→      p.age += dt
   547→      if (p.age >= p.life) {
   548→        p.active = false
   549→        this.pFree.push(p)
   550→        continue
   551→      }
   552→      p.vy += p.grav * dt
   553→      const dr = 1 - p.drag * dt
   554→      p.vx *= dr > 0 ? dr : 0
   555→      p.vy *= dr > 0 ? dr : 0
   556→      p.x += p.vx * dt
   557→      p.y += p.vy * dt
   558→    }
   559→    // trails
   560→    for (let i = this.trails.length - 1; i >= 0; i--) {
   561→      const t = this.trails[i]
   562→      if (!t.active) continue
   563→      if (!t.emitting) {
   564→        t.age += dt
   565→        if (t.age >= t.life) {
   566→          t.active = false
   567→          this.tFree.push(t)
   568→        }
   569→      }
   570→    }
   571→    // afterimages
   572→    for (let i = this.afterimages.length - 1; i >= 0; i--) {
   573→      const a = this.afterimages[i]
   574→      a.age += dt
   575→      if (a.age >= a.life) {
   576→        a.active = false
   577→        this.aFree.push(a)
   578→      }
   579→    }
   580→    // rings
   581→    for (let i = this.rings.length - 1; i >= 0; i--) {
   582→      const r = this.rings[i]
   583→      r.age += dt
   584→      if (r.age >= r.life) {
   585→        r.active = false
   586→        this.rFree.push(r)
   587→      }
   588→    }
   589→    // cracks
   590→    for (let i = this.cracks.length - 1; i >= 0; i--) {
   591→      const c = this.cracks[i]
   592→      c.age += dt
   593→      if (c.age >= c.life) {
   594→        c.active = false
   595→        this.cFree.push(c)
   596→      }
   597→    }
   598→    // texts
   599→    for (let i = this.texts.length - 1; i >= 0; i--) {
   600→      const t = this.texts[i]
   601→      t.age += dt
   602→      if (t.age >= t.life) {
   603→        t.active = false
   604→        this.txtFree.push(t)
   605→      }
   606→      t.vy += 320 * dt
   607→      t.x += t.vx * dt
   608→      t.y += t.vy * dt
   609→    }
   610→    // debris
   611→    for (let i = this.debris.length - 1; i >= 0; i--) {
   612→      const d = this.debris[i]
   613→      d.age += dt
   614→      if (d.age >= d.life) {
   615→        d.active = false
   616→        this.dFree.push(d)
   617→        continue
   618→      }
   619→      d.vy += 900 * dt
   620→      d.x += d.vx * dt
   621→      d.y += d.vy * dt
   622→      d.rot += d.vrot * dt
   623→      if (d.y > 0) {
   624→        d.y = 0
   625→        d.vy *= -0.42
   626→        d.vx *= 0.6
   627→        d.vrot *= 0.5
   628→        if (Math.abs(d.vy) < 40) d.vy = 0
   629→      }
   630→    }
   631→    // flash
   632→    if (this.flashAlpha > 0) {
   633→      this.flashAlpha -= this.flashDecay * dt
   634→      if (this.flashAlpha < 0) this.flashAlpha = 0
   635→    }
   636→    // chromatic strain decay
   637→    if (this.caIntensity > 0) {
   638→      this.caIntensity = Math.max(0, this.caIntensity - dt * 3.4)
   639→    }
   640→    // impact frames
   641→    for (let i = this.impactFrames.length - 1; i >= 0; i--) {
   642→      const f = this.impactFrames[i]
   643→      f.age += dt
   644→      if (f.age >= f.life) {
   645→        f.active = false
   646→        this.impactFrames.splice(i, 1)
   647→      }
   648→    }
   649→  }

   684→  /** additive layer: afterimages, trails, particles, rings */
   685→  drawAdditive(ctx: CanvasRenderingContext2D, time: number): void {
   686→    ctx.save()
   687→    ctx.globalCompositeOperation = 'lighter'
   688→
   689→    // afterimages (ghost stickmen)
   690→    for (const a of this.afterimages) {
   691→      const k = 1 - a.age / a.life
   692→      ctx.save()
   693→      ctx.translate(a.x, a.y)
   694→      ctx.scale(a.face * a.scale, a.scale)
   695→      aiPose.set(a.pose)
   696→      solveSkeleton(aiPose, PLAYER_SPEC, scrashGeom)
   697→      drawStick(ctx, scrashGeom, PLAYER_STYLE, {
   698→        ghost: a.color,
   699→        ghostAlpha: k * 0.5,
   700→        time,
   701→      })
   702→      ctx.restore()
   703→    }
   704→
   705→    // slash trail ribbons
   706→    for (const t of this.trails) {
   707→      if (!t.active || t.count < 2) continue
   708→      const fade = t.emitting ? 1 : 1 - t.age / t.life
   709→      drawTrail(ctx, t, fade)
   710→    }
   711→
   712→    // particles
   713→    for (const p of this.particles) {
   714→      if (!p.active) continue
   715→      const k = 1 - p.age / p.life
   716→      switch (p.type) {
   717→        case 'spark': {
   718→          ctx.strokeStyle = rgba(p.color, k)
   719→          ctx.lineWidth = p.size * 0.8
   720→          ctx.beginPath()
   721→          ctx.moveTo(p.x, p.y)
   722→          ctx.lineTo(p.x - p.vx * 0.03, p.y - p.vy * 0.03)
   723→          ctx.stroke()
   724→          break
   725→        }
   726→        case 'streak': {
   727→          ctx.strokeStyle = rgba(p.color, k * 0.5)
   728→          ctx.lineWidth = 1.6
   729→          ctx.beginPath()
   730→          ctx.moveTo(p.x, p.y)
   731→          ctx.lineTo(p.x - p.vx * 0.08, p.y)
   732→          ctx.stroke()
   733→          break
   734→        }
   735→        case 'ember': {
   736→          const fl = 0.6 + 0.4 * Math.sin(p.age * 30 + p.x)
   737→          ctx.fillStyle = rgba(p.color, k * fl)
   738→          ctx.beginPath()
   739→          ctx.arc(p.x, p.y, p.size * (0.7 + k * 0.5), 0, TAU)
   740→          ctx.fill()
   741→          break
   742→        }
   743→        case 'smoke': {
   744→          ctx.fillStyle = rgba(p.color, k * 0.24)
   745→          ctx.beginPath()
   746→          ctx.arc(p.x, p.y, p.size * (1.6 - k * 0.6), 0, TAU)
   747→          ctx.fill()
   748→          break
   749→        }
   750→        case 'shard': {
   751→          ctx.strokeStyle = rgba(p.color, k)
   752→          ctx.lineWidth = 2
   753→          ctx.save()
   754→          ctx.translate(p.x, p.y)
   755→          ctx.rotate(p.age * 9)
   756→          ctx.beginPath()
   757→          ctx.moveTo(-p.size, 0)
   758→          ctx.lineTo(p.size, 0)
   759→          ctx.stroke()
   760→          ctx.restore()
   761→          break
   762→        }
   763→        case 'arc': {
   764→          if (p.pts) {
   765→            const fl = k * (0.7 + 0.3 * Math.sin(time * 60 + p.x))
   766→            ctx.strokeStyle = rgba(p.color, fl)
   767→            ctx.lineWidth = p.size
   768→            ctx.beginPath()
   769→            ctx.moveTo(p.pts[0], p.pts[1])
   770→            for (let i = 2; i < p.pts.length; i += 2) {
   771→              ctx.lineTo(p.pts[i], p.pts[i + 1])
   772→            }
   773→            ctx.stroke()
   774→            ctx.strokeStyle = rgba('#ffffff', fl * 0.8)
   775→            ctx.lineWidth = 1
   776→            ctx.stroke()
   777→          }
   778→          break
   779→        }
   780→        default: {
   781→          ctx.fillStyle = rgba(p.color, k)
   782→          ctx.beginPath()
   783→          ctx.arc(p.x, p.y, p.size, 0, TAU)
   784→          ctx.fill()
   785→        }
   786→      }
   787→    }
   788→
   789→    // rings
   790→    for (const r of this.rings) {
   791→      if (!r.active) continue
   792→      const k = 1 - r.age / r.life
   793→      const rr = r.r0 + (r.r1 - r.r0) * (1 - k * k)
   794→      ctx.strokeStyle = rgba(r.color, k * 0.9)
   795→      ctx.lineWidth = r.width * k + 0.5
   796→      ctx.beginPath()
   797→      ctx.ellipse(r.x, r.y, rr, rr * r.squash, 0, 0, TAU)
   798→      ctx.stroke()
   799→      ctx.strokeStyle = rgba('#ffffff', k * 0.5)
   800→      ctx.lineWidth = 1.2
   801→      ctx.beginPath()
   802→      ctx.ellipse(r.x, r.y, rr * 0.92, rr * 0.92 * r.squash, 0, 0, TAU)
   803→      ctx.stroke()
   804→    }
   805→
   806→    // damage numbers (crisp, non-additive, on top of the glow layers)
   807→    ctx.globalCompositeOperation = 'source-over'
   808→    ctx.textAlign = 'center'
   809→    for (const t of this.texts) {
   810→      if (!t.active) continue
   811→      const k = 1 - t.age / t.life
   812→      const pop = t.age < 0.09 ? 1 + (0.09 - t.age) * 4 : 1
   813→      ctx.save()
   814→      ctx.translate(t.x, t.y)
   815→      ctx.scale(pop, pop)
   816→      ctx.font = `900 ${t.size}px "Arial Black", Arial, sans-serif`
   817→      ctx.lineWidth = 3
   818→      ctx.strokeStyle = rgba('#04060a', Math.min(1, k * 1.4))
   819→      ctx.strokeText(t.str, 0, 0)
   820→      ctx.fillStyle = rgba(t.color, Math.min(1, k * 1.6))
   821→      ctx.fillText(t.str, 0, 0)
   822→      ctx.restore()
   823→    }
   824→
   825→    ctx.restore()
   826→  }

   837→  /** MANGA IMPACT FRAME pass — draw AFTER the additive layer so the black
   838→   * spikes punch through the glow: white core → black wedges → accent ring.
   839→   * Called by the renderer in world space. */
   840→  drawImpactFrames(ctx: CanvasRenderingContext2D): void {
   841→    for (const f of this.impactFrames) {
   842→      if (!f.active) continue
   843→      const k = 1 - f.age / f.life
   844→      // white inversion core
   845→      ctx.save()
   846→      ctx.globalCompositeOperation = 'lighter'
   847→      ctx.fillStyle = rgba('#ffffff', 0.9 * k)
   848→      ctx.beginPath()
   849→      ctx.arc(f.x, f.y, 34 + (1 - k) * 26, 0, TAU)
   850→      ctx.fill()
   851→      ctx.restore()
   852→      // black radial spikes — alternating long/short, jittered by seed
   853→      ctx.save()
   854→      ctx.fillStyle = rgba('#04060a', 0.96 * k)
   855→      const spikes = 13
   856→      for (let i = 0; i < spikes; i++) {
   857→        const base = f.seed + (i / spikes) * TAU
   858→        const long = i % 2 === 0
   859→        const r0 = 26 * k + 8
   860→        const r1 = (long ? 250 : 140) * (0.55 + 0.45 * k)
   861→        const halfW = (long ? 0.055 : 0.03) * (0.6 + 0.4 * k)
   862→        ctx.beginPath()
   863→        ctx.moveTo(f.x + Math.cos(base - halfW) * r0, f.y + Math.sin(base - halfW) * r0)
   864→        ctx.lineTo(f.x + Math.cos(base) * r1, f.y + Math.sin(base) * r1)
   865→        ctx.lineTo(f.x + Math.cos(base + halfW) * r0, f.y + Math.sin(base + halfW) * r0)
   866→        ctx.closePath()
   867→        ctx.fill()
   868→      }
   869→      // thin accent ring at the shock front
   870→      ctx.strokeStyle = rgba(f.color, 0.85 * k)
   871→      ctx.lineWidth = 2.6
   872→      ctx.beginPath()
   873→      ctx.arc(f.x, f.y, 46 + (1 - k) * 90, 0, TAU)
   874→      ctx.stroke()
   875→      ctx.restore()
   876→    }
   877→  }

   884→function pushPolar(t: Trail, x: number, y: number, a: number, r0: number, r1: number): void {
   885→  const i = t.head * 4
   886→  t.pts[i] = x + Math.cos(a) * r0
   887→  t.pts[i + 1] = y + Math.sin(a) * r0
   888→  t.pts[i + 2] = x + Math.cos(a) * r1
   889→  t.pts[i + 3] = y + Math.sin(a) * r1
   890→  t.head = (t.head + 1) % t.N
   891→  if (t.count < t.N) t.count++
   892→}

   894→const segA: number[] = [0, 0, 0, 0]
   895→const segB: number[] = [0, 0, 0, 0]
   896→
   897→function drawTrail(ctx: CanvasRenderingContext2D, t: Trail, fade: number): void {
   898→  const n = t.count
   899→  const pts = t.pts
   900→  // render oldest → newest
   901→  const start = (t.head - n + t.N + t.N) % t.N
   902→
   903→  const seg = (idx: number, out: number[]): void => {
   904→    const i = ((start + idx) % t.N) * 4
   905→    out[0] = pts[i]; out[1] = pts[i + 1]; out[2] = pts[i + 2]; out[3] = pts[i + 3]
   906→  }
   907→
   908→  // ---- wide soft halo + mid core ribbon ----
   909→  for (let layer = 0; layer < 2; layer++) {
   910→    const wide = layer === 0
   911→    const color = wide ? t.color2 : t.color
   912→    const baseAlpha = wide ? 0.3 : 0.55
   913→    for (let i = 0; i < n - 1; i++) {
   914→      seg(i, segA)
   915→      seg(i + 1, segB)
   916→      const headBias = i / Math.max(1, n - 1)
   917→      const alpha = baseAlpha * fade * (0.35 + 0.65 * headBias)
   918→      ctx.fillStyle = rgba(color, alpha)
   919→      // quad between consecutive blade segments
   920→      ctx.beginPath()
   921→      ctx.moveTo(segA[0], segA[1])
   922→      ctx.lineTo(segA[2], segA[3])
   923→      ctx.lineTo(segB[2], segB[3])
   924→      ctx.lineTo(segB[0], segB[1])
   925→      ctx.closePath()
   926→      ctx.fill()
   927→    }
   928→  }
   929→
   930→  // ---- hot inner core: thin near-white strip hugging the blade path ----
   931→  // (strided every 2nd segment — plasma-line look at half the fill cost)
   932→  for (let i = 0; i < n - 1; i += 2) {
   933→    seg(i, segA)
   934→    seg(Math.min(i + 1, n - 1), segB)
   935→    const headBias = i / Math.max(1, n - 1)
   936→    const alpha = 0.5 * fade * (0.3 + 0.7 * headBias)
   937→    // inset both blade endpoints 45% toward the tip → thin strip near the edge
   938→    const ax = segA[0] + (segA[2] - segA[0]) * 0.55
   939→    const ay = segA[1] + (segA[3] - segA[1]) * 0.55
   940→    const bx = segB[0] + (segB[2] - segB[0]) * 0.55
   941→    const by = segB[1] + (segB[3] - segB[1]) * 0.55
   942→    ctx.fillStyle = rgba('#ffffff', alpha)
   943→    ctx.beginPath()
   944→    ctx.moveTo(ax, ay)
   945→    ctx.lineTo(segA[2], segA[3])
   946→    ctx.lineTo(segB[2], segB[3])
   947→    ctx.lineTo(bx, by)
   948→    ctx.closePath()
   949→    ctx.fill()
   950→  }
   951→
   952→  // ---- layered cutting edges along the tip path ----
   953→  if (n >= 2) {
   954→    ctx.beginPath()
   955→    seg(0, segA)
   956→    ctx.moveTo(segA[2], segA[3])
   957→    for (let i = 1; i < n; i++) {
   958→      seg(i, segA)
   959→      ctx.lineTo(segA[2], segA[3])
   960→    }
   961→    // soft colored edge under a hot white edge → layered "energy blade" look
   962→    ctx.strokeStyle = rgba(t.color2, 0.34 * fade)
   963→    ctx.lineWidth = 3.2 * t.width
   964→    ctx.stroke()
   965→    ctx.strokeStyle = rgba('#ffffff', 0.55 * fade)
   966→    ctx.lineWidth = 1.5 * t.width
   967→    ctx.stroke()
   968→  }
   969→
   970→  // ---- tip glint + motion-blur streaks (only while the swing is live) ----
   971→  if (t.emitting && n >= 3) {
   972→    seg(n - 1, segA)
   973→    seg(n - 2, segB)
   974→    const dx = segA[2] - segB[2]
   975→    const dy = segA[3] - segB[3]
   976→    const spd = Math.sqrt(dx * dx + dy * dy)
   977→    if (spd > 3) {
   978→      const ix = 1 / spd
   979→      const nx = -dy * ix // perpendicular
   980→      const ny = dx * ix
   981→      const glint = Math.min(1, spd / 26) * fade
   982→      // 4-point star flare at the newest tip
   983→      const gx = segA[2]
   984→      const gy = segA[3]
   985→      const g1 = 4 + t.width * 2.4
   986→      const g2 = g1 * 0.42
   987→      ctx.strokeStyle = rgba('#ffffff', 0.75 * glint)
   988→      ctx.lineWidth = 1.1
   989→      ctx.beginPath()
   990→      ctx.moveTo(gx - g1, gy)
   991→      ctx.lineTo(gx + g1, gy)
   992→      ctx.moveTo(gx, gy - g1)
   993→      ctx.lineTo(gx, gy + g1)
   994→      ctx.moveTo(gx - g2 * nx, gy - g2 * ny)
   995→      ctx.lineTo(gx + g2 * nx, gy + g2 * ny)
   996→      ctx.stroke()
   997→      ctx.fillStyle = rgba(t.color, 0.9 * glint)
   998→      ctx.beginPath()
   999→      ctx.arc(gx, gy, 1.4 + t.width * 0.7, 0, TAU)
  1000→      ctx.fill()
  1001→      // two faint speed streaks trailing behind the tip
  1002→      ctx.strokeStyle = rgba(t.color2, 0.3 * glint)
  1003→      ctx.lineWidth = 1
  1004→      const off = 2.6 + t.width
  1005→      ctx.beginPath()
  1006→      ctx.moveTo(gx - nx * off, gy - ny * off)
  1007→      ctx.lineTo(gx - dx * 2.4 - nx * off, gy - dy * 2.4 - ny * off)
  1008→      ctx.moveTo(gx + nx * off, gy + ny * off)
  1009→      ctx.lineTo(gx - dx * 2.4 + nx * off, gy - dy * 2.4 + ny * off)
  1010→      ctx.stroke()
  1011→    }
  1012→  }
  1013→}

  1015→function drawCrack(ctx: CanvasRenderingContext2D, c: Crack): void {
  1016→  if (c.pts.length < 4) return
  1017→  ctx.beginPath()
  1018→  ctx.moveTo(c.x + c.pts[0], c.y + c.pts[1])
  1019→  for (let i = 2; i < c.pts.length; i += 2) {
  1020→    ctx.lineTo(c.x + c.pts[i], c.y + c.pts[i + 1])
  1021→  }
  1022→  ctx.stroke()
  1023→}
  1024→
