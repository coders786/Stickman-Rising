    29→    29→interface ScoreRow {
    30→    30→  id: string
    31→    31→  name: string

    39→    39→const SAVE_KEY = 'sr_save_v1'
    40→    40→const SETTINGS_KEY = 'sr_settings_v1'
    41→    41→const NAME_KEY = 'sr_name_v1'
    42→    42→
    43→    43→function loadSave(): SaveData {
    44→    44→  if (typeof window === 'undefined') return { totalKills: 0, best: 0, unlocked: ['katana'] }
    45→    45→  try {
    46→    46→    const raw = window.localStorage.getItem(SAVE_KEY)
    47→    47→    if (raw) {
    48→    48→      const parsed = JSON.parse(raw) as SaveData
    49→    49→      if (!parsed.unlocked?.includes('katana')) parsed.unlocked.unshift('katana')
    50→    50→      return parsed
    51→    51→    }
    52→    52→  } catch { /* ignore */ }
    53→    53→  return { totalKills: 0, best: 0, unlocked: ['katana'] }
    54→    54→}
    55→    55→
    56→    56→function loadSettings(): GameSettings {
    57→    57→  if (typeof window === 'undefined') return { ...DEFAULT_SETTINGS }
    58→    58→  try {
    59→    59→    const raw = window.localStorage.getItem(SETTINGS_KEY)
    60→    60→    if (raw) return { ...DEFAULT_SETTINGS, ...(JSON.parse(raw) as Partial<GameSettings>) }
    61→    61→  } catch { /* ignore */ }
    62→    62→  const coarse = typeof window !== 'undefined' && window.matchMedia?.('(pointer: coarse)').matches
    63→    63→  return { ...DEFAULT_SETTINGS, quality: coarse ? 'medium' : 'high' }
    64→    64→}

   650→   650→function Joystick({ input }: { input: GameInput }) {
   651→   651→  const ref = useRef<HTMLDivElement>(null)
   652→   652→  const [knob, setKnob] = useState({ x: 0, y: 0 })
   653→   653→  const activeId = useRef<number | null>(null)
   654→   654→
   655→   655→  const handle = (e: React.PointerEvent) => {
   656→   656→    const el = ref.current
   657→   657→    if (!el) return
   658→   658→    const rect = el.getBoundingClientRect()
   659→   659→    const cx = rect.left + rect.width / 2
   660→   660→    const cy = rect.top + rect.height / 2
   661→   661→    let dx = (e.clientX - cx) / (rect.width / 2)
   662→   662→    let dy = (e.clientY - cy) / (rect.height / 2)
   663→   663→    const len = Math.hypot(dx, dy)
   664→   664→    if (len > 1) {
   665→   665→      dx /= len
   666→   666→      dy /= len
   667→   667→    }
   668→   668→    input.setVirtualAxis(dx, dy)
   669→   669→    setKnob({ x: dx * (rect.width / 2 - 22), y: dy * (rect.height / 2 - 22) })
   670→   670→  }
   671→   671→
   672→   672→  const release = () => {
   673→   673→    activeId.current = null
   674→   674→    input.setVirtualAxis(0, 0)
   675→   675→    setKnob({ x: 0, y: 0 })
   676→   676→  }
   677→   677→
   678→   678→  return (
   679→   679→    <div
   680→   680→      ref={ref}
   681→   681→      className="absolute bottom-6 left-5 z-20 h-32 w-32 touch-none rounded-full border border-zinc-700/70 bg-black/40 backdrop-blur-sm"
   682→   682→      onPointerDown={(e) => {
   683→   683→        e.preventDefault()
   684→   684→        activeId.current = e.pointerId
   685→   685→        ;(e.target as HTMLElement).setPointerCapture?.(e.pointerId)
   686→   686→        handle(e)
   687→   687→      }}
   688→   688→      onPointerMove={(e) => {
   689→   689→        if (activeId.current === e.pointerId) handle(e)
   690→   690→      }}
   691→   691→      onPointerUp={release}
   692→   692→      onPointerCancel={release}
   693→   693→      onContextMenu={(e) => e.preventDefault()}
   694→   694→    >
   695→   695→      <div
   696→   696→        className="pointer-events-none absolute left-1/2 top-1/2 h-11 w-11 -translate-x-1/2 -translate-y-1/2 rounded-full border border-zinc-500 bg-zinc-800/90"
   697→   697→        style={{ transform: `translate(calc(-50% + ${knob.x}px), calc(-50% + ${knob.y}px))` }}
   698→   698→      />
   699→   699→    </div>
   700→   700→  )
   701→   701→}
   702→   702→
