    33→    33→function pointInTri(px: number, py: number, x1: number, y1: number, x2: number, y2: number, x3: number, y3: number): boolean {
    34→    34→  const d1 = (px - x2) * (y1 - y2) - (x1 - x2) * (py - y2)
    35→    35→  const d2 = (px - x3) * (y2 - y3) - (x2 - x3) * (py - y3)
    36→    36→  const d3 = (px - x1) * (y3 - y1) - (x3 - x1) * (py - y1)
    37→    37→  const hasNeg = d1 < 0 || d2 < 0 || d3 < 0
    38→    38→  const hasPos = d1 > 0 || d2 > 0 || d3 > 0
    39→    39→  return !(hasNeg && hasPos)
    40→    40→}

   215→   215→  update(dt: number, w: WorldRef): void {
   216→   216→    this.iframes = Math.max(0, this.iframes - dt)
   217→   217→    this.flashT = Math.max(0, this.flashT - dt * 5)
   218→   218→    this.swingGlow = Math.max(0, this.swingGlow - dt * 5)
   219→   219→    this.riposteT = Math.max(0, this.riposteT - dt)
   220→   220→    this.energy = Math.min(this.maxEnergy, this.energy + dt * 3)
   221→   221→    this.turnT = Math.max(0, this.turnT - dt)
   222→   222→
   223→   223→    switch (this.state) {
   224→   224→      case 'move': this.updateMove(dt, w); break
   225→   225→      case 'attack': this.updateAttack(dt, w); break
   226→   226→      case 'dash': this.updateDash(dt, w); break
   227→   227→      case 'parry': this.updateParry(dt, w); break
   228→   228→      case 'charge': this.updateCharge(dt, w); break
   229→   229→      case 'hurt': this.updateHurt(dt, w); break
   230→   230→      case 'dead': this.updateDead(dt, w); break
   231→   231→      case 'scripted': this.updateScripted(dt); break
   232→   232→    }
   233→   233→
   234→   234→    // physics integration (shared)
   235→   235→    if (this.state !== 'dash') {
   236→   236→      const g = this.state === 'attack' && this.action && this.action.def.float
   237→   237→        ? GRAVITY * this.action.def.float
   238→   238→        : GRAVITY
   239→   239→      this.vy += g * dt
   240→   240→    }
   241→   241→    this.x += this.vx * dt
   242→   242→    this.y += this.vy * dt
   243→   243→
   244→   244→    // ground
   245→   245→    const wasAir = !this.grounded
   246→   246→    if (this.y >= 0) {
   247→   247→      this.y = 0
   248→   248→      if (this.vy > 140) {
   249→   249→        this.landSpring = clamp(this.vy / 700, 0, 1)
   250→   250→        if (wasAir) this.onLand(w)
   251→   251→      } else if (wasAir) {
   252→   252→        this.landSpring = 0.4
   253→   253→        this.onLand(w)
   254→   254→      }
   255→   255→      this.vy = 0
   256→   256→      this.grounded = true
   257→   257→      this.jumpsUsed = 0
   258→   258→      this.airDashUsed = false
   259→   259→    } else {
   260→   260→      this.grounded = false
   261→   261→    }
   262→   262→    // arena bounds
   263→   263→    if (this.x < w.arenaMinX + 20) {
   264→   264→      this.x = w.arenaMinX + 20
   265→   265→      if (this.vx < -200) w.fx.dust(this.x - 12, 0, 5)
   266→   266→      this.vx = Math.max(0, this.vx)
   267→   267→    }
   268→   268→    if (this.x > w.arenaMaxX - 20) {
   269→   269→      this.x = w.arenaMaxX - 20
   270→   270→      if (this.vx > 200) w.fx.dust(this.x + 12, 0, 5)
   271→   271→      this.vx = Math.min(0, this.vx)
   272→   272→    }
   273→   273→
   274→   274→    this.dashCd = Math.max(0, this.dashCd - dt)
   275→   275→    this.dashRecharge = Math.max(0, this.dashRecharge - dt)
   276→   276→    if (this.dashRecharge <= 0) this.dashConsec = 0
   277→   277→
   278→   278→    // ability script
   279→   279→    if (this.ability) this.updateAbility(dt, w)
   280→   280→
   281→   281→    // animation + solve — ROOT-MOTION POLICY: the animator owns the pose
   282→   282→    // baseline; procedural layers re-apply on the fresh sample each frame
   283→   283→    // (without this copy, lean/spring adds accumulate into a never-reset
   284→   284→    // array and the whole torso visibly spins while running)
   285→   285→    this.animator.update(dt)
   286→   286→    copyPose(this.pose, this.animator.pose)
   287→   287→    this.applyProcedural(dt)
   288→   288→    solveSkeleton(this.pose, PLAYER_SPEC, this.geom)
   289→   289→    // soft ground contact — plants the feet via a persistent offset without
   290→   290→    // touching the clip's authored root motion
   291→   291→    if (this.grounded) {
   292→   292→      applyGroundConstraint(this.pose, this.geom, PLAYER_SPEC, 47, dt, this.ground)
   293→   293→    } else {
   294→   294→      this.ground.off += (0 - this.ground.off) * (1 - Math.exp(-15 * dt))
   295→   295→    }
   296→   296→    this.updateBladeWorld()
   297→   297→  }

   342→   342→  private updateMove(dt: number, w: WorldRef): void {
   343→   343→    const input = w.input
   344→   344→    const ax = input.axisX()
   345→   345→    const ay = input.axisY()
   346→   346→
   347→   347→    // horizontal movement
   348→   348→    if (ax !== 0) {
   349→   349→      const accel = this.grounded ? ACCEL : AIR_ACCEL
   350→   350→      this.vx += ax * accel * dt
   351→   351→      this.vx = clamp(this.vx, -MOVE_MAX, MOVE_MAX)
   352→   352→      if (ax > 0 !== (this.face > 0)) this.turnT = 0.09 // anime turn accent
   353→   353→      this.face = ax > 0 ? 1 : -1
   354→   354→    } else if (this.grounded) {
   355→   355→      const f = FRICTION * dt
   356→   356→      this.vx = Math.abs(this.vx) <= f ? 0 : this.vx - Math.sign(this.vx) * f
   357→   357→    } else {
   358→   358→      this.vx = damp(this.vx, 0, 0.4, dt)
   359→   359→    }
   360→   360→
   361→   361→    // locomotion clip selection — gait tempo matched to ground speed so the
   362→   362→    // planted foot tracks (stride ≈ 156px/cycle at SCALE 1.32 → vx/311)
   363→   363→    const speed = Math.abs(this.vx)
   364→   364→    if (this.grounded) {
   365→   365→      if (speed > 40) {
   366→   366→        this.animator.play(RUN_CLIP, { blend: 0.09, speed: clamp(speed / 311, 0.5, 1.45) })
   367→   367→      } else if (this.vy === 0) {
   368→   368→        this.animator.play(IDLE_CLIP, { blend: 0.16 })
   369→   369→      }
   370→   370→    } else if (this.vy < -60) {
   371→   371→      this.animator.play(JUMP_CLIP, { blend: 0.06 })
   372→   372→    } else {
   373→   373→      this.animator.play(FALL_CLIP, { blend: 0.12 })
   374→   374→    }
   375→   375→
   376→   376→    // ---- actions ----
   377→   377→    if (this.tryCombatInputs(w, ax, ay)) return
   378→   378→  }

   380→   380→  /** returns true if a new state was entered */
   381→   381→  private tryCombatInputs(w: WorldRef, ax: number, ay: number): boolean {
   382→   382→    const input = w.input
   383→   383→
   384→   384→    // dash (anytime, from move)
   385→   385→    if ((input.pressed('dash') || input.buffered('dash')) && this.canDash()) {
   386→   386→      this.startDash(w, ax, ay)
   387→   387→      return true
   388→   388→    }
   389→   389→    // jump (grounded = jump 1, plus one air jump)
   390→   390→    if (input.pressed('jump') || input.buffered('jump', 0.18)) {
   391→   391→      if (this.jumpsUsed < 2) {
   392→   392→        this.doJump(w)
   393→   393→        return false
   394→   394→      }
   395→   395→    }
   396→   396→    // parry
   397→   397→    if (input.pressed('parry')) {
   398→   398→      this.enterParry(w)
   399→   399→      return true
   400→   400→    }
   401→   401→    // ability
   402→   402→    if (input.pressed('ability')) {
   403→   403→      if (this.energy >= this.sword.ability.cost) {
   404→   404→        this.startAbility(w)
   405→   405→        return true
   406→   406→      }
   407→   407→    }
   408→   408→    // ult is handled by Game (cinematic)
   409→   409→
   410→   410→    // heavy: charge vs tap
   411→   411→    if (input.down('heavy') && this.grounded) {
   412→   412→      this.enterCharge(w)
   413→   413→      return true
   414→   414→    }
   415→   415→    if (input.pressed('heavy') || input.buffered('heavy')) {
   416→   416→      input.takeBuffered('heavy')
   417→   417→      this.startAttack(this.pickHeavy(w), w)
   418→   418→      return true
   419→   419→    }
   420→   420→    // light
   421→   421→    if (input.pressed('light') || input.buffered('light')) {
   422→   422→      input.takeBuffered('light')
   423→   423→      const id = this.riposteT > 0 ? 'riposte' : this.pickLight(w, ax, ay)
   424→   424→      this.startAttack(ATTACKS[id], w)
   425→   425→      return true
   426→   426→    }
   427→   427→    return false
   428→   428→  }

   437→   437→    if (this.jumpsUsed === 2) {

   439→   439→    }

   446→   446→  private canDash(): boolean {
   447→   447→    if (this.dashCd > 0 || this.dashRecharge > 0) return false
   448→   448→    if (!this.grounded && this.airDashUsed) return false
   449→   449→    return true
   450→   450→  }
   451→   451→
   452→   452→  startDash(w: WorldRef, ax: number, ay: number): void {
   453→   453→    w.input.takeBuffered('dash')
   454→   454→    let dx = ax
   455→   455→    let dy = ay
   456→   456→    if (dx === 0 && dy === 0) dx = this.face
   457→   457→    const len = Math.hypot(dx, dy) || 1
   458→   458→    dx /= len
   459→   459→    dy /= len
   460→   460→    if (dx !== 0) this.face = dx > 0 ? 1 : -1
   461→   461→    this.dashDx = dx
   462→   462→    this.dashDy = dy
   463→   463→    this.state = 'dash'
   464→   464→    this.dashT = DASH_TIME
   465→   465→    this.dashCd = DASH_CD
   466→   466→    this.dashConsec++
   467→   467→    if (this.dashConsec >= 2) this.dashRecharge = DASH_RECHARGE
   468→   468→    if (!this.grounded) this.airDashUsed = true
   469→   469→    this.iframes = Math.max(this.iframes, 0.12)
   470→   470→    this.action = null
   471→   471→    this.charging = false
   472→   472→    this.vx = dx * DASH_SPEED
   473→   473→    this.vy = dy * DASH_SPEED * (this.grounded && dy < -0.4 ? 0.55 : 1)
   474→   474→    this.animator.play(this.grounded ? DASH_CLIP : DASH_AIR_CLIP, { blend: 0.03 })
   475→   475→    w.audio.play('dash', { pitch: rand(0.95, 1.1) })
   476→   476→    const colors = this.sword.colors
   477→   477→    w.fx.speedStreaks(this.x, this.y - 40, dx * DASH_SPEED * 0.9, 6, rgba(colors.trail, 1))
   478→   478→    w.fx.dust(this.x, this.grounded ? 0 : this.y, 5)
   479→   479→    w.cam.addTrauma(0.06)
   480→   480→  }

   482→   482→  private updateDash(dt: number, w: WorldRef): void {
   483→   483→    this.dashT -= dt
   484→   484→    // afterimages
   485→   485→    this.afterimgTimer -= dt
   486→   486→    if (this.afterimgTimer <= 0) {
   487→   487→      this.afterimgTimer = 0.026
   488→   488→      w.fx.afterimage(this.pose, this.x, this.hipY, this.face, this.sword.colors.trail, 0.3, SCALE)
   489→   489→    }
