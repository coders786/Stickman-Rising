    61→    61→    ctx.save()
    62→    62→    ctx.font = '600 12px ui-monospace, monospace'

    64→    64→
    65→    65→    if (this.mode === 'solo') {
    66→    66→      this.drawSolo(ctx, game, time)
    67→    67→    } else if (this.mode === 'sheet') {
    68→    68→      this.drawSheets(ctx)
    69→    69→    }
    70→    70→
    71→    71→    // state readout (uses public player surface + bracket access for privates)

    73→    73→    const priv = p as unknown as Record<string, unknown>

    80→    80→      `postAtk=${Number(priv.postAttackT ?? 0).toFixed(2)}  turn=${Number(priv.turnT ?? 0).toFixed(2)}`,

    82→    82→    ]

    89→    89→      ctx.fillText(ln, 13, y)
    90→    90→      y += 17
    91→    91→    }
    92→    92→    ctx.restore()
    93→    93→  }

    95→    95→  /** big live character + skeleton overlay, centered-left — drawn on a
    96→    96→   * light diagnostic panel so the black silhouette reads at full contrast */
    97→    97→  private drawSolo(ctx: CanvasRenderingContext2D, game: Game, time: number): void {
    98→    98→    const p = game.player
    99→    99→    const dpr = window.devicePixelRatio || 1
   100→   100→    const Z = PSCALE / 1.32 // zoom factor over the in-game world scale
   101→   101→    const cx = (ctx.canvas.width / dpr) * 0.26
   102→   102→    const groundY = (ctx.canvas.height / dpr) * 0.82
   103→   103→    const hipScreenY = groundY + p.hipY * Z
   104→   104→    // light diagnostic panel behind the figure (contrast for the black body)
   105→   105→    ctx.save()
   106→   106→    ctx.fillStyle = '#dfe7ee'
   107→   107→    ctx.beginPath()
   108→   108→    ctx.roundRect(cx - 250, groundY - 330, 500, 348, 14)
   109→   109→    ctx.fill()
   110→   110→    ctx.strokeStyle = 'rgba(40,70,100,0.35)'
   111→   111→    ctx.lineWidth = 1.5
   112→   112→    ctx.stroke()
   113→   113→    ctx.restore()
   114→   114→    // body: local geom → screen at PSCALE
   115→   115→    ctx.save()
   116→   116→    ctx.translate(cx, hipScreenY)
   117→   117→    ctx.scale(p.face * PSCALE, PSCALE)
   118→   118→    drawFighter3D(ctx, p.geom, PLAYER_STYLE, { time })
   119→   119→    this.drawSkeleton(ctx, p.geom)
   120→   120→    ctx.restore()
   121→   121→    // ground line + foot contact markers at their true screen positions
   122→   122→    ctx.save()
   123→   123→    ctx.strokeStyle = 'rgba(30,90,140,0.65)'
   124→   124→    ctx.lineWidth = 2
   125→   125→    ctx.setLineDash([7, 5])
   126→   126→    ctx.beginPath()
   127→   127→    ctx.moveTo(cx - 240, groundY)
   128→   128→    ctx.lineTo(cx + 240, groundY)
   129→   129→    ctx.stroke()
   130→   130→    ctx.setLineDash([])
   131→   131→    if (this.showFeet) {
   132→   132→      const ptScreen = (j: { x: number; y: number }) => ({
   133→   133→        x: cx + p.face * PSCALE * j.x,
   134→   134→        y: hipScreenY + PSCALE * j.y,
   135→   135→      })
   136→   136→      const rA = ptScreen(p.geom.ankleR)
   137→   137→      const lA = ptScreen(p.geom.ankleL)
   138→   138→      // planted = lowest foot within reach of the ground line
   139→   139→      const rPlanted = p.grounded && p.geom.ankleR.y >= p.geom.ankleL.y
   140→   140→      const lPlanted = p.grounded && p.geom.ankleL.y > p.geom.ankleR.y
   141→   141→      this.drawFootMarker(ctx, rA.x, rA.y, rPlanted)
   142→   142→      this.drawFootMarker(ctx, lA.x, lA.y, lPlanted)
   143→   143→    }
   144→   144→    ctx.restore()
   145→   145→  }

   154→   154→  /** bones + joints + labels on any solved geom (local space) */
   155→   155→  private drawSkeleton(ctx: CanvasRenderingContext2D, g: SkelGeom): void {
   156→   156→    if (!this.showBones) return
   157→   157→    ctx.save()
   158→   158→    ctx.strokeStyle = 'rgba(65,246,255,0.55)'
   159→   159→    ctx.lineWidth = 1.1
   160→   160→    const seg = (a: { x: number; y: number }, b: { x: number; y: number }) => {
   161→   161→      ctx.beginPath()
   162→   162→      ctx.moveTo(a.x, a.y)
   163→   163→      ctx.lineTo(b.x, b.y)
   164→   164→      ctx.stroke()
   165→   165→    }
   166→   166→    seg(g.hip, g.waist); seg(g.waist, g.chest); seg(g.chest, g.neckTop); seg(g.neckTop, g.headC)
   167→   167→    seg(g.hipPtR, g.kneeR); seg(g.kneeR, g.ankleR); seg(g.ankleR, g.toeR)
   168→   168→    seg(g.hipPtL, g.kneeL); seg(g.kneeL, g.ankleL); seg(g.ankleL, g.toeL)
   169→   169→    seg(g.shoulderR, g.elbR); seg(g.elbR, g.handR)
   170→   170→    seg(g.shoulderL, g.elbL); seg(g.elbL, g.handL)
   171→   171→    // shoulder girdle line (the anatomy fix, visualized)
   172→   172→    ctx.strokeStyle = 'rgba(255,120,180,0.8)'
   173→   173→    ctx.lineWidth = 1.3
   174→   174→    ctx.beginPath()
   175→   175→    ctx.moveTo(g.shoulderL.x, g.shoulderL.y)
   176→   176→    ctx.lineTo(g.neckTop.x, g.neckTop.y)
   177→   177→    ctx.lineTo(g.shoulderR.x, g.shoulderR.y)
   178→   178→    ctx.stroke()
   179→   179→    // joints
   180→   180→    ctx.fillStyle = '#ffd24a'
   181→   181→    const joints = [
   182→   182→      g.hip, g.hipPtR, g.hipPtL, g.waist, g.chest, g.neckTop, g.headC,
   183→   183→      g.shoulderR, g.shoulderL, g.elbR, g.elbL, g.handR, g.handL,
   184→   184→      g.kneeR, g.kneeL, g.ankleR, g.ankleL,
   185→   185→    ]
   186→   186→    for (const j of joints) {
   187→   187→      ctx.beginPath()
   188→   188→      ctx.arc(j.x, j.y, 1.7, 0, Math.PI * 2)
   189→   189→      ctx.fill()
   190→   190→    }
   191→   191→    // THE proof joints: shoulders in pink
   192→   192→    ctx.fillStyle = '#ff6fb5'
   193→   193→    for (const j of [g.shoulderR, g.shoulderL]) {
   194→   194→      ctx.beginPath()
   195→   195→      ctx.arc(j.x, j.y, 3.0, 0, Math.PI * 2)
   196→   196→      ctx.fill()
   197→   197→    }
   198→   198→    if (this.showLabels) {
   199→   199→      ctx.fillStyle = 'rgba(160,240,255,0.95)'
   200→   200→      ctx.font = '600 4.2px ui-monospace, monospace'
   201→   201→      const lbl = (j: { x: number; y: number }, s: string) => ctx.fillText(s, j.x + 2.4, j.y - 2.4)
   202→   202→      lbl(g.headC, 'skull')
   203→   203→      lbl(g.neckTop, 'neckTop')
   204→   204→      lbl(g.shoulderR, 'SHOULDER-R')
   205→   205→      lbl(g.shoulderL, 'shoulder-l')
   206→   206→      lbl(g.chest, 'chest')
   207→   207→      lbl(g.waist, 'waist')
   208→   208→      lbl(g.hip, 'pelvis')
   209→   209→      lbl(g.kneeR, 'kneeR')
   210→   210→      lbl(g.ankleR, 'ankleR')
   211→   211→      lbl(g.elbR, 'elbR')
   212→   212→      lbl(g.handR, 'handR')
   213→   213→    }
   214→   214→    ctx.restore()
   215→   215→  }

   217→   217→  /** contact sheets: every locomotion cycle, 8 frames each, real solve —
   218→   218→   * light diagnostic background, per-cell ground lines + foot-contact
   219→   219→   * markers (green = support, orange = swing), 2 clip-groups per band */
   220→   220→  private drawSheets(ctx: CanvasRenderingContext2D): void {
   221→   221→    const rows: Array<[string, typeof WALK_CLIP]> = [
   222→   222→      ['WALK', WALK_CLIP],
   223→   223→      ['RUN', RUN_CLIP],
   224→   224→      ['SPRINT', SPRINT_CLIP],
   225→   225→      ['COMBAT STEP', COMBAT_STEP_CLIP],
   226→   226→      ['COMBAT-IDLE (rooted guard)', COMBAT_IDLE_CLIP],
   227→   227→      ['BACKPEDAL', BACKPEDAL_CLIP],
   228→   228→      ['IDLE', IDLE_CLIP],
   229→   229→      ['JUMP', JUMP_CLIP],
   230→   230→      ['DASH LAUNCH (iai)', DASH_LAUNCH_CLIP],
   231→   231→      ['DASH CARRY (momentum)', DASH_CARRY_CLIP],
   232→   232→      ['DASH HUNT (gap-close)', DASH_HUNT_CLIP],
   233→   233→      ['DASH PIERCE (lunge)', DASH_PIERCE_CLIP],
   234→   234→      ['DASH RETREAT (guard)', DASH_RETREAT_CLIP],
   235→   235→      ['DASH CARVE (breakaway)', DASH_CARVE_CLIP],
   236→   236→    ]
   237→   237→    const dpr = window.devicePixelRatio || 1
   238→   238→    const w = ctx.canvas.width / dpr
   239→   239→    const h = ctx.canvas.height / dpr
   240→   240→    const cols = 8
   241→   241→    const groupsPerRow = 2
   242→   242→    const bands = Math.ceil(rows.length / groupsPerRow) // 3 bands
   243→   243→    const cellW = w / (cols * groupsPerRow)
   244→   244→    const rowH = h / (bands + 0.25)
   245→   245→    // figure scale: fit character height inside a band minus label space
   246→   246→    const S = Math.min(1.0, (rowH - 34) / 138)
   247→   247→    ctx.save()
   248→   248→    // light diagnostic backdrop — black silhouette at full contrast
   249→   249→    ctx.fillStyle = '#e9eef3'
   250→   250→    ctx.fillRect(0, 0, w, h)
   251→   251→    ctx.font = '700 12px ui-monospace, monospace'
   252→   252→    for (let r = 0; r < rows.length; r++) {
   253→   253→      const band = Math.floor(r / groupsPerRow)
   254→   254→      const group = r % groupsPerRow
   255→   255→      const [label, clip] = rows[r]
   256→   256→      const baseY = rowH * (band + 1)
   257→   257→      const labelX = group * (w / groupsPerRow) + 10
   258→   258→      ctx.fillStyle = '#1a3a52'
   259→   259→      ctx.fillText(`${label} · ${clip.dur.toFixed(2)}s cycle`, labelX, baseY - rowH + 16)
   260→   260→      for (let c = 0; c < cols; c++) {
   261→   261→        const t = (clip.dur * c) / cols
   262→   262→        sampleClip(clip, t, this.scratchPose)
   263→   263→        const g = solveFighter3D(this.scratchPose, PLAYER_SPEC, this.scratchGeom, PLAYER_SPEC.blade)
   264→   264→        // normalize: lowest foot contact sits ON this cell's ground line
   265→   265→        const lowContact = Math.max(g.ankleL.y, g.ankleR.y, g.toeL.y, g.toeR.y)
   266→   266→        const cx = group * (w / groupsPerRow) + cellW * (c + 0.5)
   267→   267→        const groundY = baseY - 14
   268→   268→        ctx.save()
   269→   269→        ctx.translate(cx, groundY - (lowContact + 2.4) * S)
   270→   270→        ctx.scale(S, S)
   271→   271→        drawFighter3D(ctx, g, PLAYER_STYLE, { time: t })
   272→   272→        ctx.restore()
   273→   273→        // per-cell ground line
   274→   274→        ctx.strokeStyle = 'rgba(40,80,110,0.45)'
   275→   275→        ctx.lineWidth = 1
   276→   276→        ctx.beginPath()
   277→   277→        ctx.moveTo(cx - cellW * 0.42, groundY)
   278→   278→        ctx.lineTo(cx + cellW * 0.42, groundY)
   279→   279→        ctx.stroke()
   280→   280→        // foot-contact markers: green = support (near ground), orange = swing
   281→   281→        const hipY = groundY - (lowContact + 2.4) * S
   282→   282→        const markFoot = (j: { x: number; y: number }) => {
   283→   283→          const airborne = (lowContact - j.y) > 4 // clearly lifted vs support
   284→   284→          ctx.fillStyle = airborne ? 'rgba(255,140,30,0.9)' : '#18c25a'
   285→   285→          ctx.beginPath()
   286→   286→          ctx.arc(cx + j.x * S, hipY + j.y * S, 3, 0, Math.PI * 2)
   287→   287→          ctx.fill()
   288→   288→        }
   289→   289→        markFoot(g.ankleR)
   290→   290→        markFoot(g.ankleL)
   291→   291→        // frame tick
   292→   292→        ctx.fillStyle = 'rgba(30,70,100,0.55)'
   293→   293→        ctx.font = '600 9px ui-monospace, monospace'
   294→   294→        ctx.fillText(`f${c}`, cx - 5, baseY - 3)
   295→   295→      }
   296→   296→    }
   297→   297→    ctx.restore()
   298→   298→  }
   299→   299→}

   305→   305→interface QaCheck {
   306→   306→  name: string
   307→   307→  pass: boolean
   308→   308→  detail: string

   311→   311→const sleep = (ms: number) => new Promise<void>((r) => setTimeout(r, ms))
   312→   312→
   313→   313→/** Scripted end-to-end verification. Parks the report on
   314→   314→ * window.__qaResult. Drive from the browser console / agent-browser. */
   315→   315→export async function runQA(game: Game): Promise<void> {
   316→   316→  const checks: QaCheck[] = []
   317→   317→  const log: string[] = []
   318→   318→  const p = game.player
   319→   319→  const input = game.input
   320→   320→
   321→   321→  const samplePose = () => {
   322→   322→    let bad = 0
   323→   323→    for (let i = 0; i < 17; i++) if (!Number.isFinite(p.pose[i])) bad++
   324→   324→    return bad
   325→   325→  }
   326→   326→
   327→   327→  try {
   328→   328→    // ---- setup: controlled empty arena (waves suppressed, player alive) ----
   329→   329→    game.qaMode = true
   330→   330→    game.phase = 'playing'
   331→   331→    game.enemies.length = 0
   332→   332→    game.waves.reset()
   333→   333→    // release every virtual input left held by prior console driving —
   334→   334→    // stale held buttons contaminate the whole scripted sequence
   335→   335→    input.setVirtualAxis(0, 0)
   336→   336→    for (const a of ['left', 'right', 'up', 'down', 'light', 'heavy', 'dash', 'jump', 'parry', 'ability', 'ult', 'execute', 'sheathe'] as const) {
   337→   337→      input.setVirtual(a, false)
   338→   338→    }
   339→   339→    p.reset(400)
   340→   340→    input.setEnabled(true)
   341→   341→    log.push('setup ok')
   342→   342→
   343→   343→    // ---- 1. anatomy invariants (the arms-from-the-neck fix) ----
   344→   344→    {
   345→   345→      const g = p.geom
   346→   346→      const spec = PLAYER_SPEC
   347→   347→      const armLen = Math.hypot(g.elbR.x - g.shoulderR.x, g.elbR.y - g.shoulderR.y)
   348→   348→      // local y grows DOWN: shoulder below neck-top means shoulderR.y GREATER
   349→   349→      const shoulderBelowNeck = g.shoulderR.y > g.neckTop.y + 1.5
   350→   350→      const armFromShoulder = Math.abs(armLen - spec.upperArm) < 0.6
   351→   351→      // neck column rises above chest: neckTop.y LESS (more negative) than chest
   352→   352→      const neckVisible = g.neckTop.y < g.chest.y - 2.5
   353→   353→      checks.push({
   354→   354→        name: 'anatomy: shoulder joints below neck top',
   355→   355→        pass: shoulderBelowNeck,
   356→   356→        detail: `shoulderR.y=${g.shoulderR.y.toFixed(1)} neckTop.y=${g.neckTop.y.toFixed(1)}`,
   357→   357→      })
   358→   358→      checks.push({
   359→   359→        name: 'anatomy: arm chains FROM shoulder (len=upperArm)',
   360→   360→        pass: armFromShoulder,
   361→   361→        detail: `len=${armLen.toFixed(2)} spec=${spec.upperArm}`,
   362→   362→      })
   363→   363→      checks.push({
   364→   364→        name: 'anatomy: neck column visible above chest',
   365→   365→        pass: neckVisible,
   366→   366→        detail: `neckTop.y=${g.neckTop.y.toFixed(1)} chest.y=${g.chest.y.toFixed(1)}`,
   367→   367→      })
   368→   368→    }
   369→   369→
   370→   370→    // ---- 2. walk → run escalation (Condition A, analog speed mapping) ----
   371→   371→    input.setVirtualAxis(0.55, 0)
   372→   372→    await sleep(700)
   373→   373→    const earlyMode = p.loco
   374→   374→    const earlyVx = Math.abs(p.vx)
   375→   375→    input.setVirtualAxis(1, 0)
   376→   376→    await sleep(1500)
   377→   377→    const fullMode = p.loco
   378→   378→    input.setVirtualAxis(0, 0)
   379→   379→    await sleep(500)
   380→   380→    checks.push({
   381→   381→      name: 'locomotion: WALK at partial input',
   382→   382→      pass: earlyMode === 'walk' && earlyVx > 60 && earlyVx < 210,
   383→   383→      detail: `mode@0.7s=${earlyMode} vx=${earlyVx.toFixed(0)}`,
   384→   384→    })
   385→   385→    checks.push({
   386→   386→      name: 'locomotion: RUN at full input',
   387→   387→      pass: fullMode === 'run',
   388→   388→      detail: `mode@2.2s=${fullMode}`,
   389→   389→    })
