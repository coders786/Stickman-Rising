    15→/** circle vs swept blade quad (prev segment + current segment) — prevents tunneling */
    16→function circleHitsSweptQuad(
    17→  cx: number, cy: number, r: number,
    18→  pbx: number, pby: number, ptx: number, pty: number,
    19→  bx: number, by: number, tx: number, ty: number,
    20→): boolean {
    21→  // current + previous blade segments
    22→  if (segHitsCircle(bx, by, tx, ty, cx, cy, r)) return true
    23→  if (segHitsCircle(pbx, pby, ptx, pty, cx, cy, r)) return true
    24→  // swept edges (tip path + base path)
    25→  if (segHitsCircle(ptx, pty, tx, ty, cx, cy, r)) return true
    26→  if (segHitsCircle(pbx, pby, bx, by, cx, cy, r)) return true
    27→  // circle center inside the swept quad (spins / wide arcs)
    28→  if (pointInTri(cx, cy, pbx, pby, ptx, pty, tx, ty)) return true
    29→  if (pointInTri(cx, cy, pbx, pby, tx, ty, bx, by)) return true
    30→  return false
    31→}

    33→function pointInTri(px: number, py: number, x1: number, y1: number, x2: number, y2: number, x3: number, y3: number): boolean {
    34→  const d1 = (px - x2) * (y1 - y2) - (x1 - x2) * (py - y2)
    35→  const d2 = (px - x3) * (y2 - y3) - (x2 - x3) * (py - y3)
    36→  const d3 = (px - x1) * (y3 - y1) - (x3 - x1) * (py - y1)
    37→  const hasNeg = d1 < 0 || d2 < 0 || d3 < 0
    38→  const hasPos = d1 > 0 || d2 > 0 || d3 > 0
    39→  return !(hasNeg && hasPos)
    40→}

    46→const SCALE = 1.32
    47→/** nominal (airborne) hip height = thigh(25)+shin(24) = 49 local × SCALE.
    48→ * Grounded height is NOT fixed — it derives from the planted foot via the
    49→ * foot-ground lock (updateFootLock): idle bent knees, lunges and landing
    50→ * squashes all sit exactly ON the ground line. */
    51→const HIP_H = (25 + 24) * SCALE

    53→/**
    54→ * Enemy body spheres: [centerY offset above ground, radius] in enemy-local
    55→ * units (× e.scale). Traced from the ENEMY_SPEC skeleton — hip at 47u, torso
    56→ * 32u, head top ≈ 94u. The old single circle (y-34, r 26) only covered the
    57→ * lower 2/3 of the body, so chest/head-height slashes passed straight
    58→ * through the sprite without connecting. Three spheres hug the visual body.
    59→ */
    60→const BODY_SPHERES: ReadonlyArray<{ oy: number; r: number }> = [
    61→  { oy: 14, r: 15 }, // legs / hip mass
    62→  { oy: 48, r: 21 }, // torso (the main target)
    63→  { oy: 78, r: 17 }, // chest / head zone
    64→]
    65→
    66→/** blade segment vs an enemy's full visual body (any sphere, swept-aware).
    67→ * `pad` fattens the radius per-attack — a katana edge is a plane; the hit
    68→ * volume must match the visual footprint of the blade edge corridor. */
    69→function bladeHitsBody(
    70→  e: { x: number; y: number; scale: number },
    71→  hasPrev: boolean,
    72→  prev: { bx: number; by: number; tx: number; ty: number },
    73→  cur: { bx: number; by: number; tx: number; ty: number },
    74→  pad = 3,
    75→): boolean {
    76→  const s = e.scale
    77→  for (const sp of BODY_SPHERES) {
    78→    const cy = e.y - sp.oy * s
    79→    const r = sp.r * s + pad
    80→    if (hasPrev
    81→      ? circleHitsSweptQuad(e.x, cy, r, prev.bx, prev.by, prev.tx, prev.ty, cur.bx, cur.by, cur.tx, cur.ty)
    82→      : segHitsCircle(cur.bx, cur.by, cur.tx, cur.ty, e.x, cy, r)
    83→    ) return true
    84→  }
    85→  return false
    86→}

   311→  update(dt: number, w: WorldRef): void {
   312→    this.iframes = Math.max(0, this.iframes - dt)
   313→    this.armorT = Math.max(0, this.armorT - dt)
   314→    this.flashT = Math.max(0, this.flashT - dt * 5)
   315→    this.swingGlow = Math.max(0, this.swingGlow - dt * 5)
   316→    this.riposteT = Math.max(0, this.riposteT - dt)
   317→    this.energy = Math.min(this.maxEnergy, this.energy + dt * 3)
   318→
   319→    switch (this.state) {
   320→      case 'move': this.updateMove(dt, w); break
   321→      case 'attack': this.updateAttack(dt, w); break
   322→      case 'dash': this.updateDash(dt, w); break
   323→      case 'parry': this.updateParry(dt, w); break
   324→      case 'charge': this.updateCharge(dt, w); break
   325→      case 'hurt': this.updateHurt(dt, w); break
   326→      case 'dead': this.updateDead(dt, w); break
   327→      case 'scripted': this.updateScripted(dt); break
   328→    }
   329→
   330→    // physics integration (shared)
   331→    if (this.state !== 'dash') {
   332→      let g: number
   333→      if (this.state === 'attack' && this.action && this.action.def.float) {
   334→        // authored air-combo float overrides the shaping (juggle windows)
   335→        g = GRAVITY * this.action.def.float
   336→      } else if (this.vy < 0) {
   337→        // ANTI-FLOAT: apex decay — no moon hovering
   338→        g = GRAVITY * (this.vy < -260 ? GRAV_RISE : GRAV_APEX)
   339→      } else {
   340→        // heavy fall: the body plummets
   341→        g = GRAVITY * GRAV_FALL
   342→      }
   343→      this.vy += g * dt
   344→      if (this.vy > TERMINAL_VY) this.vy = TERMINAL_VY
   345→    }
   346→    this.x += this.vx * dt
   347→    this.y += this.vy * dt
   348→
   349→    // ground
   350→    const wasAir = !this.grounded
   351→    if (this.y >= 0) {
   352→      this.y = 0
   353→      if (this.vy > 140) {
   354→        this.landSpring = clamp(this.vy / 700, 0, 1)
   355→        if (wasAir) this.onLand(w)
   356→      } else if (wasAir) {
   357→        this.landSpring = 0.4
   358→        this.onLand(w)
   359→      }
   360→      this.vy = 0
   361→      this.grounded = true
   362→      this.jumpsUsed = 0
   363→      this.airDashUsed = false
   364→    } else {
   365→      this.grounded = false
   366→    }
   367→    // arena bounds
   368→    if (this.x < w.arenaMinX + 20) {
   369→      this.x = w.arenaMinX + 20
   370→      if (this.vx < -200) w.fx.dust(this.x - 12, 0, 5)
   371→      this.vx = Math.max(0, this.vx)
   372→    }
   373→    if (this.x > w.arenaMaxX - 20) {
   374→      this.x = w.arenaMaxX - 20
   375→      if (this.vx > 200) w.fx.dust(this.x + 12, 0, 5)
   376→      this.vx = Math.min(0, this.vx)
   377→    }
   378→
   379→    this.dashCd = Math.max(0, this.dashCd - dt)
   380→    this.dashRecharge = Math.max(0, this.dashRecharge - dt)
   381→    if (this.dashRecharge <= 0) this.dashConsec = 0
   382→
   383→    // ability script
   384→    if (this.ability) this.updateAbility(dt, w)
   385→
   386→    // animation + solve
   387→    this.animator.update(dt)
   388→    // Base layer: the clip FULLY owns the pose each frame (write, never
   389→    // accumulate). applyProcedural() is a bounded additive layer on top.
   390→    // Without this copy, procedural += deltas integrate unbounded → spine
   391→    // spin + root sink. (Same contract as Spine's AnimationState.apply /
   392→    // Godot AnimationTree → Skeleton2D.)
   393→    this.pose.set(this.animator.pose)
   394→    this.applyProcedural(dt)
   395→    // procedural layers (bounded, frame-local — never accumulate):
   396→    this.updateBladeDynamics(dt, w) // wrist lag spring + kinetic chain — feeds hitboxes
   397→    this.applyAimAssist(dt, w)      // bounded wrist convergence onto the target center line
   398→    this.applyTwoHandGrip(dt)       // off-hand IK onto the tsuka
   399→    solveSkeleton(this.pose, PLAYER_SPEC, this.geom)
   400→    this.updateFootLock(dt)         // weight: hips sit on the planted foot
   401→    this.updateBladeWorld()
   402→    this.updateSwingScale(dt)
   403→    this.updateBladeZ(dt)
   404→  }

   406→  private onLand(w: WorldRef): void {
   407→    w.audio.play('land', { volume: 0.5 })
   408→    w.fx.dust(this.x, 0, 7)
   409→    // slam impact while attacking
   410→    const a = this.action
   411→    if (a && a.def.slamImpact && !a.impactDone) {
   412→      a.impactDone = true
   413→      this.slamAoe(w, a.def.slamImpact.r, a.def.slamImpact.dmg, a.def.slamImpact.launch)
   414→    }
   415→    if (this.state === 'move' && this.lastGrounded === false) {
   416→      this.animator.play(LAND_CLIP, { blend: 0.06 })
   417→    }
   418→    this.airIndex = 0
   419→  }

   421→  private slamAoe(w: WorldRef, r: number, dmg: number, launch: number): void {
   422→    const info: AttackRuntimeInfo = {
   423→      id: 'slam', dmg: dmg * this.sword.stats.dmg, heavy: true,
   424→      stagger: 0.5, kb: 240, kbY: 0, launch, juggle: 0,
   425→      hitstop: 0.11, shake: 0.45, aoe: r, breakGuard: true, kind: 'heavy',
   426→    }
   427→    const colors = this.sword.colors
   428→    w.fx.shockwave(this.x, 0, r, colors.glow, 0.5, 8)
   429→    w.fx.groundCrack(this.x, 0, r * 0.6)
   430→    w.fx.burst(this.x, -6, 14, colors.glow, 340, 0.5, 'spark')
   431→    w.fx.screenFlash(colors.trail, 0.16, 5)
   432→    w.addHitstop(0.09, 0.06)
   433→    w.cam.addTrauma(0.4)
   434→    w.audio.play('slam', { intensity: 1 })
   435→    const bx = this.x
   436→    for (const e of w.enemies) {
   437→      if (e.dead || e.spawning) continue
   438→      const dx = e.x - bx
   439→      if (Math.abs(dx) < r && e.y > -140) {
   440→        w.hitEnemy(e.id, { bx: bx - 20, by: -20, tx: bx + 20, ty: -20 }, info)
   441→      }
   442→    }
   443→  }

   449→  private updateMove(dt: number, w: WorldRef): void {
   450→    const input = w.input
   451→    const ax = input.axisX()
   452→    const ay = input.axisY()
   453→
   454→    // facing follows input
   455→    if (ax !== 0) this.face = ax > 0 ? 1 : -1
   456→
   457→    // horizontal movement
   458→    if (ax !== 0) {
   459→      const accel = this.grounded ? ACCEL : AIR_ACCEL
   460→      this.vx += ax * accel * dt
   461→      this.vx = clamp(this.vx, -MOVE_MAX, MOVE_MAX)
   462→    } else if (this.grounded) {
   463→      const f = FRICTION * dt
   464→      this.vx = Math.abs(this.vx) <= f ? 0 : this.vx - Math.sign(this.vx) * f
   465→    } else {
   466→      this.vx = damp(this.vx, 0, 0.4, dt)
   467→    }
   468→
   469→    // locomotion clip selection
   470→    const speed = Math.abs(this.vx)
   471→    if (this.grounded) {
   472→      if (speed > 40) {
   473→        this.animator.play(RUN_CLIP, { blend: 0.09, speed: clamp(speed / MOVE_MAX, 0.65, 2.1) })
   474→      } else if (this.vy === 0) {
   475→        this.animator.play(IDLE_CLIP, { blend: 0.16 })
   476→      }
   477→    } else if (this.vy < -60) {
   478→      this.animator.play(JUMP_CLIP, { blend: 0.06 })
   479→    } else {
   480→      this.animator.play(FALL_CLIP, { blend: 0.12 })
   481→    }
   482→
   483→    // ---- actions ----
   484→    if (this.tryCombatInputs(w, ax, ay)) return
   485→  }

   487→  /** returns true if a new state was entered */
   488→  private tryCombatInputs(w: WorldRef, ax: number, ay: number): boolean {
   489→    const input = w.input
   490→
   491→    // dash (anytime, from move)
   492→    if ((input.pressed('dash') || input.buffered('dash')) && this.canDash()) {
   493→      this.startDash(w, ax, ay)
   494→      return true
   495→    }
   496→    // jump (grounded = jump 1, plus one air jump)
   497→    if (input.pressed('jump') || input.buffered('jump', 0.18)) {
   498→      if (this.jumpsUsed < 2) {
   499→        this.doJump(w)
   500→        return false
   501→      }
   502→    }
   503→    // parry
   504→    if (input.pressed('parry')) {
   505→      this.enterParry(w)
   506→      return true
   507→    }
   508→    // ability
   509→    if (input.pressed('ability')) {
   510→      if (this.energy >= this.sword.ability.cost) {
   511→        this.startAbility(w)
   512→        return true
   513→      }
   514→    }
   515→    // ult is handled by Game (cinematic)
   516→
   517→    // heavy: charge vs tap
   518→    if (input.down('heavy') && this.grounded) {
   519→      this.enterCharge(w)
   520→      return true
   521→    }
   522→    if (input.pressed('heavy') || input.buffered('heavy')) {
   523→      input.takeBuffered('heavy')
   524→      this.startAttack(this.pickHeavy(w), w)
   525→      return true
   526→    }
   527→    // light
   528→    if (input.pressed('light') || input.buffered('light')) {
   529→      input.takeBuffered('light')
   530→      // DASH MOMENTUM STRIKE: attacking while the last dash's momentum
   531→      // still carries the body converts into the through-cut — the
   532→      // dash-strike feel (Hades). Without this, the 150ms in-dash input
   533→      // window was brutally tight and dash attacks "randomly" whiffed.
   534→      if (this.grounded
   535→        && w.time - this.dashEndedAt < 0.18
   536→        && Math.abs(this.vx) > 110
   537→        && this.riposteT <= 0) {
   538→        this.startAttack(ATTACKS.dashSlash, w)
   539→        return true
   540→      }
   541→      const id = this.riposteT > 0 ? 'riposte' : this.pickLight(w, ax, ay)
   542→      this.startAttack(ATTACKS[id], w)
   543→      return true
   544→    }
   545→    return false
   546→  }
   547→
   548→  private doJump(w: WorldRef): void {
   549→    w.input.takeBuffered('jump')
   550→    this.jumpsUsed++
   551→    this.vy = this.jumpsUsed === 1 ? JUMP_V : AIR_JUMP_V
   552→    this.grounded = false
   553→    w.audio.play('jump', { volume: 0.4 })
   554→    w.fx.dust(this.x, 0, 4)
   555→    if (this.jumpsUsed === 2) {
   556→      w.fx.burst(this.x, this.y - 20, 6, '#9fb4c4', 160, 0.4, 'dot')
   557→    }
   558→  }

   564→  private canDash(): boolean {
   565→    if (this.dashCd > 0 || this.dashRecharge > 0) return false
   566→    if (!this.grounded && this.airDashUsed) return false
   567→    return true
   568→  }

   570→  startDash(w: WorldRef, ax: number, ay: number): void {
   571→    w.input.takeBuffered('dash')
   572→    let dx = ax
   573→    let dy = ay
   574→    if (dx === 0 && dy === 0) dx = this.face
   575→    const len = Math.hypot(dx, dy) || 1
   576→    dx /= len
   577→    dy /= len
   578→    if (dx !== 0) this.face = dx > 0 ? 1 : -1
   579→    this.dashDx = dx
   580→    this.dashDy = dy
   581→    this.state = 'dash'
   582→    this.dashT = DASH_TIME
   583→    this.dashCd = DASH_CD
   584→    this.dashConsec++
   585→    if (this.dashConsec >= 2) this.dashRecharge = DASH_RECHARGE
   586→    if (!this.grounded) this.airDashUsed = true
   587→    this.iframes = Math.max(this.iframes, 0.12)
   588→    this.action = null
   589→    this.charging = false
   590→    this.vx = dx * DASH_SPEED
   591→    this.vy = dy * DASH_SPEED * (this.grounded && dy < -0.4 ? 0.55 : 1)
   592→    this.animator.play(this.grounded ? DASH_CLIP : DASH_AIR_CLIP, { blend: 0.05 })
   593→    w.audio.play('dash', { pitch: rand(0.95, 1.1) })
   594→    const colors = this.sword.colors
   595→    w.fx.speedStreaks(this.x, this.y - 40, dx * DASH_SPEED * 0.9, 6, rgba(colors.trail, 1))
   596→    w.fx.chroma(0.4) // lens strain on the launch
   597→    w.fx.dust(this.x, this.grounded ? 0 : this.y, 5)
   598→    w.cam.addTrauma(0.06)
   599→  }
   600→
   601→  private updateDash(dt: number, w: WorldRef): void {
   602→    this.dashT -= dt
   603→    // afterimages
   604→    this.afterimgTimer -= dt
   605→    if (this.afterimgTimer <= 0) {
   606→      this.afterimgTimer = 0.026
   607→      w.fx.afterimage(this.pose, this.x, this.hipY, this.face, this.sword.colors.trail, 0.3, SCALE)
   608→    }
   609→    // maintain velocity with slight decay
   610→    const k = 1 - this.dashT / DASH_TIME
   611→    const sp = DASH_SPEED * (1 - k * 0.25)
   612→    this.vx = this.dashDx * sp
   613→    this.vy = this.dashDy * sp * (this.grounded ? (this.dashDy < 0 ? 0.4 : 1) : 1)
   614→
   615→    // dash attack
   616→    const input = w.input
   617→    if (input.pressed('light') || input.buffered('light')) {
   618→      input.takeBuffered('light')
   619→      this.startAttack(ATTACKS.dashSlash, w)
   620→      return
   621→    }
   622→    if (input.pressed('heavy') || input.buffered('heavy')) {
   623→      if (!this.grounded) {
   624→        input.takeBuffered('heavy')
   625→        this.startAttack(ATTACKS.slam, w)
   626→        return
   627→      }
   628→      input.takeBuffered('heavy')
   629→      this.startAttack(ATTACKS.dashHeavy, w)
   630→      return
   631→    }
   632→    if (this.dashT <= 0) {
   633→      this.state = 'move'
   634→      this.dashEndedAt = w.time
   635→      this.vx *= 0.55
   636→      if (!this.grounded) {
   637→        this.vy *= 0.5
   638→      } else if (Math.abs(this.vx) > 260) {
   639→        // THE BRAKING SKID: momentum must shed through a visible physical
   640→        // pose — heel-dig crouch + a dust scrape, never a dead stop.
   641→        w.fx.dust(this.x - this.face * 12, 0, 5)
   642→        this.landSpring = 0.35
   643→        this.animator.play(LAND_CLIP, { blend: 0.05 })
   644→      }
   645→    }
   646→  }

   652→  private pickLight(w: WorldRef, ax: number, ay: number): string {
   653→    // contextual branches
   654→    if (ay < -0.4 && this.grounded) return 'launcher'
   655→    if (ay > 0.4 && this.grounded) return 'spin'
   656→    if (this.grounded && ax !== 0 && Math.sign(ax) !== this.face) return 'retreat'
   657→    if (!this.grounded) {
   658→      if (ay > 0.4) return 'dive'
   659→      this.airIndex = (this.airIndex % 3) + 1
   660→      return `a${this.airIndex}`
   661→    }
   662→    // ground light chain (with time decay)
   663→    if (w.time - this.lastLightAt > 1.1) this.lightIndex = 0
   664→    this.lightIndex = Math.min(this.lightIndex + 1, 5)
   665→    return `l${Math.min(this.lightIndex, 5)}`
   666→  }

   668→  /** Heavy CHAIN: Kiri Oroshi → Tsuki → Grand Slam. Sequential like the
   669→   * light chain, with time decay — heavy attacks finally have combo
   670→   * pairing instead of one animation forever. */
   671→  private pickHeavy(w: WorldRef): AttackDef {
   672→    if (w.time - this.lastHeavyAt > 1.15) this.heavyIndex = 0
   673→    this.heavyIndex = Math.min(this.heavyIndex + 1, 3)
   674→    return ATTACKS[`h${this.heavyIndex}`] ?? ATTACKS.h1
   675→  }

   677→  startAttack(def: AttackDef, w: WorldRef, fromParry = false): void {
   678→    if (!def) return
   679→    // IN-BETWEEN LAW: capture the source state BEFORE switching — chained
   680→    // attacks crossfade slower (0.11s) than fresh entries (0.07s) so every
   681→    // transition between animation instances gets real in-betweens.
   682→    const wasAttack = this.state === 'attack'
   683→    // requirement checks
   684→    if (def.requires === 'ground' && !this.grounded) def = ATTACKS.a1
   685→    if (def.requires === 'air' && this.grounded) def = ATTACKS.l1
   686→
   687→    // face toward held direction at attack start
   688→    const ax = w.input.axisX()
   689→    if (ax !== 0 && !fromParry) {
   690→      this.face = ax > 0 ? 1 : -1
   691→    } else if (!fromParry) {
   692→      // SOFT AIM: no direction held → face the nearest threat. Every swing
   693→      // commits toward a real target — the root of "aimed at different
   694→      // angles": cuts should converge on the enemy, not on empty planes.
   695→      const tgt = this.nearestEnemy(w, 600)
   696→      if (tgt && Math.abs(tgt.x - this.x) > 24) {
   697→        this.face = tgt.x > this.x ? 1 : -1
   698→      }
   699→    }
   700→
   701→    // heavy chain bookkeeping (h1 → h2 → h3 sequencing)
   702→    const hm = /^h([123])$/.exec(def.id)
   703→    if (hm) {
   704→      this.heavyIndex = Number(hm[1])
   705→      this.lastHeavyAt = w.time
   706→    }
   707→    if (/^l[1-5]$/.test(def.id)) this.lastLightAt = w.time
   708→
   709→    // HYPER-ARMOR: the charged release cannot be interrupted (the root
   710→    // cause of "charged never hits" — enemies struck mid-swing cancelled it)
   711→    if (def.id === 'charged') this.armorT = 0.5
   712→
   713→    const trail = w.fx.startTrail(this.sword.colors.trail, this.sword.colors.trail2, this.sword.trailWidth)
   714→    this.hasPrevBlade = false
   715→    this.hasPrevAng = false
   716→    this.action = {
   717→      def,
   718→      animT: 0,
   719→      hitMap: new Map(),
   720→      trail,
   721→      sfxDone: false,
   722→      moveDone: false,
   723→      moveYDone: false,
   724→      impactDone: false,
   725→      overshootDone: false,
   726→      flashDone: !def.drawFlash,
   727→      fromParry,
   728→      aimUsed: 0,
   729→    }
   730→    if (fromParry) this.riposteT = 0
   731→    this.state = 'attack'
   732→    this.charging = false
   733→    // ANIME TIMING: attack bodies play "on twos" (24Hz stepped keys —
   734→    // crisp, keyframed combat beats à la GG Strive) while blade physics,
   735→    // smears, and trails run full-rate on the procedural layers.
   736→    this.animator.play(def.clip, { blend: wasAttack ? 0.11 : 0.07, speed: this.atkSpeed, stepRate: 24 })
   737→    // slight ground friction during attacks
   738→  }

   740→  private updateAttack(dt: number, w: WorldRef): void {
   741→    const a = this.action
   742→    if (!a) {
   743→      this.state = 'move'
   744→      return
   745→    }
   746→    const def = a.def
   747→    a.animT = this.animator.getTime()
   748→    const animT = a.animT
   749→    const spd = this.atkSpeed
   750→
   751→    // ground friction / air drag during attack
   752→    if (this.grounded) {
   753→      const f = 900 * dt
   754→      this.vx = Math.abs(this.vx) <= f ? 0 : this.vx - Math.sign(this.vx) * f
   755→    }
   756→
   757→    // sfx event
   758→    if (!a.sfxDone && animT >= def.swingSfx.t) {
   759→      a.sfxDone = true
   760→      w.audio.play(def.swingSfx.name, { pitch: rand(0.94, 1.08) })
   761→    }
   762→    // iai SNAP-DRAW flash release: the blade clears the saya in one frame —
   763→    // a localized light event + a 2-frame stretch on the steel sells it.
   764→    if (!a.flashDone && animT >= def.trail.start) {
   765→      a.flashDone = true
   766→      const c = this.sword.colors
   767→      w.fx.burst(this.blade.tx, this.blade.ty, 7, c.core, 300, 0.3, 'spark')
   768→      w.fx.shockwave(this.blade.tx, this.blade.ty, 46, c.glow, 0.18, 3)
   769→      w.cam.addTrauma(0.05)
   770→      this.swingScale = Math.max(this.swingScale, 1.3) // velocity stretch
   771→    }
   772→    // forward impulse
   773→    if (!a.moveDone && def.moveX !== undefined && animT >= (def.moveXAt ?? def.hit.start)) {
   774→      a.moveDone = true
   775→      this.vx = this.face * def.moveX * (def.kind === 'dash' ? 1 : 1)
   776→    }
   777→    // vertical impulse (launcher hop / slam plunge)
   778→    if (!a.moveYDone && def.moveY !== undefined && animT >= (def.moveYAt ?? 0)) {
   779→      a.moveYDone = true
   780→      this.vy = def.moveY
   781→      if (def.moveY < 0) {
   782→        this.grounded = false
   783→        w.fx.dust(this.x, 0, 6)
   784→      }
   785→    }
   786→
   787→    // hit window: sample blade segment & test enemies (swept, anti-tunneling)
   788→    const h = def.hit
   789→    if (animT >= h.start && animT <= h.end) {
   790→      this.swingGlow = 1
   791→      const info = makeRuntimeInfo(def, this.sword.stats.dmg)
   792→      // DASH THROUGH-CUT: the iai pass cuts everything the dash swept
   793→      // through — the player crosses the target at 980px/s BEFORE the
   794→      // attack anim even starts, so the blade-only segment points past
   795→      // them. The corridor is the DASH PATH itself: a fixed chest-height
   796→      // band from 124px behind the player to 66px ahead — anchored to the
   797→      // path (not the wandering kissaki, which wraps behind on the
   798→      // follow-through and would collapse the span). This is the swept
   799→      // volume of the through-cut.
   800→      let curSeg = this.blade
   801→      let usePrev = this.hasPrevBlade
   802→      if (def.kind === 'dash') {
   803→        curSeg = {
   804→          bx: this.x - this.face * 124,
   805→          by: this.y - 55,
   806→          tx: this.x + this.face * 66,
   807→          ty: this.y - 55,
   808→        }
   809→        usePrev = false
   810→      }
   811→      for (const e of w.enemies) {
   812→        // spawning enemies within blade reach (landing pods, y > -90) are
   813→        // fair game — the swing connects the moment the body crosses the arc
   814→        if (e.dead || (e.spawning && e.y < -90)) continue
   815→        const last = a.hitMap.get(e.id)
   816→        const reArm = h.multiHit ? (last === undefined || animT - last >= h.multiHit) : last === undefined
   817→        if (!reArm) continue
   818→        if (bladeHitsBody(e, usePrev, this.prevBlade, curSeg, h.pad ?? 3)) {
   819→          a.hitMap.set(e.id, animT)
   820→          w.hitEnemy(e.id, this.blade, info)
   821→          this.onLandedHit(w, info, e.x, e.y - 48 * e.scale)
   822→        }
   823→      }
   824→      // remember this frame's blade for the next sweep test
   825→      this.prevBlade.bx = this.blade.bx
   826→      this.prevBlade.by = this.blade.by
   827→      this.prevBlade.tx = this.blade.tx
   828→      this.prevBlade.ty = this.blade.ty
   829→      this.hasPrevBlade = true
   830→      // aoe shockwave at strike moment for heavy hits — the wave is REAL:
   831→      // enemies inside the radius eat the strike (the charged/h3 promise:
   832→      // the shockwave visual must connect, not just decorate)
   833→      if (h.aoe && h.aoe > 0 && !a.impactDone && this.grounded && animT >= h.start + 0.02) {
   834→        a.impactDone = true
   835→        w.fx.shockwave(this.x + this.face * 30, 0, h.aoe, this.sword.colors.glow, 0.4, 6)
   836→        if (this.grounded) w.fx.groundCrack(this.x + this.face * 34, 0, h.aoe * 0.45)
   837→        for (const e of w.enemies) {
   838→          if (e.dead || e.spawning) continue
   839→          if (a.hitMap.has(e.id) && !h.multiHit) continue
   840→          if (Math.abs(e.x - this.x) < h.aoe && e.y > -140) {
   841→            a.hitMap.set(e.id, animT)
   842→            w.hitEnemy(e.id, this.blade, info)
   843→            this.onLandedHit(w, info, e.x, e.y - 48 * e.scale)
   844→          }
   845→        }
   846→      }
   847→    }
   848→
   849→    // OVERSHOOT (follow-through law): as the active window closes, inject a
   850→    // 2-frame rotational impulse — the tip pushes PAST the stop point and
   851→    // the wrist spring elastic-snaps it back. Stopping dead reads weightless.
   852→    if (!a.overshootDone && animT > h.end && this.armVel !== 0) {
   853→      a.overshootDone = true
   854→      this.bladeLagVel += Math.sign(this.armVel) * 2.4
   855→    }
   856→
   857→    // trail emission
   858→    if (animT >= def.trail.start && animT <= def.trail.end) {
   859→      w.fx.pushTrail(a.trail, this.blade.bx, this.blade.by, this.blade.tx, this.blade.ty)
   860→    } else if (animT > def.trail.end && a.trail.emitting) {
   861→      w.fx.endTrail(a.trail)
   862→    }
   863→
   864→    // cancel window → allow chaining / dash-cancel / jump-cancel
   865→    const dashCancel = animT >= h.end * 0.6
   866→    if (animT >= def.cancel || dashCancel) {
   867→      const input = w.input
   868→      if ((input.pressed('dash') || input.buffered('dash')) && this.canDash()) {
   869→        w.fx.endTrail(a.trail)
   870→        this.startDash(w, input.axisX(), input.axisY())
   871→        return
   872→      }
   873→      if (input.pressed('parry')) {
   874→        w.fx.endTrail(a.trail)
   875→        this.enterParry(w)
   876→        return
   877→      }
   878→      if (animT >= def.cancel) {
   879→        // chain next attack
   880→        const ax = input.axisX()
   881→        const ay = input.axisY()
   882→        let next: string | null = null
   883→        if (input.buffered('light')) {
   884→          input.takeBuffered('light')
   885→          next = this.riposteT > 0 && this.grounded ? 'riposte' : this.pickLight(w, ax, ay)
   886→        } else if (input.buffered('heavy')) {
   887→          input.takeBuffered('heavy')
   888→          if (!this.grounded) {
   889→            next = 'slam'
   890→          } else if (input.down('heavy')) {
   891→            // still holding → convert to charge
   892→            this.enterCharge(w)
   893→            w.fx.endTrail(a.trail)
   894→            return
   895→          } else {
   896→            // advance the heavy CHAIN (h1 → h2 → h3) like the light chain
   897→            const heavyNext = def.chain.find((c) => /^h[123]$/.test(c))
   898→            next = heavyNext ?? (def.kind === 'heavy' ? 'l1' : 'h1')
   899→          }
   900→        }
   901→        if (next) {
   902→          // validate chain (or allow from-parry riposte)
   903→          if (fromChain(def, next, this)) {
   904→            w.fx.endTrail(a.trail)
   905→            this.startAttack(ATTACKS[next], w)
   906→            return
   907→          }
   908→        }
   909→        // jump cancel on ground
   910→        if (this.grounded && (input.pressed('jump') || input.buffered('jump', 0.15))) {
   911→          w.fx.endTrail(a.trail)
   912→          this.doJump(w)
   913→          this.state = 'move'
   914→          return
   915→        }
   916→      }
   917→    }
   918→
   919→    // finish
   920→    if (this.animator.isDone()) {
   921→      w.fx.endTrail(a.trail, 0.18)
   922→      this.action = null
   923→      this.state = 'move'
   924→    }
   925→  }

   927→  private onLandedHit(w: WorldRef, info: AttackRuntimeInfo, ex: number, ey: number): void {

   930→    // elemental on-hit flourishes

   932→    const dir = Math.atan2(ey - this.y + 40, ex - this.x)

   936→        break
   937→      case 'thunder':
   938→        if (Math.random() < 0.3) w.fx.lightningArc(ex, ey, ex + rand(-40, 40), ey + rand(-30, 30), c.trail)
   939→        break
   940→      case 'void':

   943→      case 'gravity':

   948→        break
   949→      default:
   950→        break
   951→    }

  1401→
  1402→  /** FOOT-GROUND LOCK — the hip sits at the planted foot's extension. This
  1403→   * is the weight system: idle bent knees no longer float (measured 9.5px
  1404→   * gap), lunges/crouches stay exactly grounded, landing squashes through
  1405→   * the leg bend instead of sinking into the floor. Airborne → nominal
  1406→   * hip height (physics owns y). */
  1407→  private updateFootLock(dt: number): void {
  1408→    const g = this.geom
  1409→    // lowest contact point: ankles AND toes (a pointed trailing toe grazes
  1410→    // the ground at toe-off — real gait), plus the sole half-width so the
  1411→    // visual foot bottom kisses the line, not the ankle joint center
  1412→    const lowContact = Math.max(g.ankleL.y, g.ankleR.y, g.toeL.y, g.toeR.y) + 2.4
  1413→    const support = lowContact * SCALE
  1414→    const groundedPose = this.grounded && this.state !== 'dead'
  1415→    let target: number
  1416→    if (groundedPose && support >= HIP_H * 0.86) {
  1417→      // SUPPORT phase — the lowest foot is load-bearing (within 14% of full
  1418→      // leg extension): pin the hip so the sole EXACTLY contacts. This is
  1419→      // the weight: no float, no dig, crouches stay crouched.
  1420→      target = support
  1421→    } else {
  1422→      // FLIGHT (both feet swinging/tucked mid-stride) or airborne: physics
  1423→      // owns height — honor the clip's authored bob instead of dragging the
  1424→      // hip down to a tucked toe (which inverted the run's natural bounce)
  1425→      target = HIP_H + this.pose[1] * SCALE
  1426→    }
  1427→    // k=24: tracks a 0.3s stride with ~2px residual — crisp, never mushy
  1428→    this.footLock = damp(this.footLock, target, 24, dt)
  1429→  }
  1430→
  1431→  /** BLADE DYNAMICS — rotational inertia of the katana as a spring-damper
  1432→   * on the wrist channel. When the arm accelerates (windup → cut), the tip
  1433→   * lags behind the drive angle; when the arm decelerates, the stored
  1434→   * energy whips the tip THROUGH the target line with a small overshoot —
  1435→   * the physical furi-otoshi snap real swordsmen feel. Under-damped
  1436→   * (ζ≈0.3): lively, never mushy. Applied BEFORE solve → hitboxes inherit
  1437→   * the physical blade, not just the keyframed one.
  1438→   *
  1439→   * During hit-stop (world timeScale ≈ 0.05) the blade takes a ±5° impact
  1440→   * resistance deflection AGAINST its swing direction — the cut bites into
  1441→   * something with mass (spec: kinetic impact modulation). */
  1442→  private updateBladeDynamics(dt: number, w: WorldRef): void {
  1443→    if (dt <= 1e-5) return
  1444→    const p = this.pose
  1445→    const armSum = p[CH.shR] + p[CH.elR] + p[CH.wrR]
  1446→    const armVel = (armSum - this.prevArmSum) / dt
  1447→    this.armVel = armVel
  1448→    this.prevArmSum = armSum
  1449→    // lag target: proportional to angular velocity (inertia resisting)
  1450→    const target = w.timeScale < 0.5
  1451→      ? clamp(-Math.sign(armVel || this.bladeLag || 1) * 0.09, -0.15, 0.15) // impact resistance ~5°
  1452→      : clamp(-armVel * 0.045, -0.15, 0.15)
  1453→    // spring toward target, under-damped for the whip-through
  1454→    const K = 260
  1455→    const C = 9
  1456→    this.bladeLagVel += (K * (target - this.bladeLag) - C * this.bladeLagVel) * dt
  1457→    this.bladeLag += this.bladeLagVel * dt
  1458→    this.bladeLag = clamp(this.bladeLag, -0.17, 0.17)
  1459→    this.bladeLagVel = clamp(this.bladeLagVel, -6, 6)
  1460→    p[CH.wrR] += this.bladeLag
  1461→
  1462→    // KINETIC CHAIN (hips 30% · torso 50% · shoulders 20%): a slower spring
  1463→    // driven by the same arm whip. Accelerating into a cut → the torso loads
  1464→    // INTO the swing; the deceleration at impact → the body recoils back.
  1465→    // The upper body can never again rotate independently of the hips.
  1466→    const k2 = 90
  1467→    const c2 = 7.5
  1468→    const chainTarget = clamp(-armVel * 0.02, -0.14, 0.14)
  1469→    this.chainLagVel += (k2 * (chainTarget - this.chainLag) - c2 * this.chainLagVel) * dt
  1470→    this.chainLag = clamp(this.chainLag + this.chainLagVel * dt, -0.16, 0.16)
  1471→    p[CH.rootRot] += -this.chainLag * 0.3
  1472→    p[CH.spine] += -this.chainLag * 0.5
  1473→    p[CH.shR] += this.chainLag * 0.2
  1474→  }

  1476→  /** AIM ASSIST — bounded wrist convergence onto the target's center line.
  1477→   * Runs on the pose AFTER blade-dynamics, so the correction steers the
  1478→   * FINAL blade angle (physical blade included). Soft-lock rules:
  1479→   *  · only during the pre-strike chamber + active frames
  1480→   *  · only toward enemies within the attack's reach envelope
  1481→   *  · total correction capped (±0.24 rad) + rate-limited — a nudge onto
  1482→   *    the center line, never a robotic snap. This is what makes every
  1483→   *    sword read as "aimed" at the enemy regardless of chain angle. */
  1484→  private applyAimAssist(dt: number, w: WorldRef): void {
  1485→    const a = this.action
  1486→    if (this.state !== 'attack' || !a) return
  1487→    const h = a.def.hit
  1488→    const animT = a.animT
  1489→    // converge during chamber (weak) → strike (strong) → off after the window
  1490→    if (animT < h.start - 0.06 || animT > h.end + 0.02) return
  1491→    const g = this.geom // last frame's solve — continuous, 1-frame stale
  1492→    const reach = PLAYER_SPEC.blade * this.sword.stats.range * SCALE * this.swingScale + 74
  1493→    let tx = 0
  1494→    let ty = 0
  1495→    let bestD = Infinity
  1496→    for (const e of w.enemies) {
  1497→      if (e.dead || e.spawning) continue
  1498→      const dx = (e.x - this.x) * this.face // in facing space
  1499→      if (dx < 14 || dx > reach) continue
  1500→      if (Math.abs(e.y - this.y) > 170) continue
  1501→      const chestY = e.y - 48 * e.scale
  1502→      const d = Math.abs(dx) + Math.abs(chestY - (this.y - 60)) * 0.5
  1503→      if (d < bestD) {
  1504→        bestD = d
  1505→        tx = (e.x - this.x) * this.face / SCALE // → character-local
  1506→        ty = (chestY - this.hipY) / SCALE
  1507→      }
  1508→    }
  1509→    if (bestD === Infinity) return
  1510→    // desired blade angle from the hand to the target chest
  1511→    const desired = Math.atan2(ty - g.handR.y, tx - g.handR.x)
  1512→    let dAng = desired - g.bladeAngle
  1513→    while (dAng > Math.PI) dAng -= TAU_LIKE
  1514→    while (dAng < -Math.PI) dAng += TAU_LIKE
  1515→    if (Math.abs(dAng) < 0.03) return
  1516→    const strong = animT >= h.start ? 1 : 0.35
  1517→    const maxRate = 3.4 * dt * strong
  1518→    let corr = clamp(dAng * 3.2 * strong * dt, -maxRate, maxRate)
  1519→    // respect the per-attack budget so technique variety survives
  1520→    const room = AIM_BUDGET - Math.abs(a.aimUsed)
  1521→    if (room <= 0) return
  1522→    corr = clamp(corr, -room, room)
  1523→    a.aimUsed += corr
  1524→    // bladeAngle = π/2 − (shR+elR+wrR) → subtracting from wrR ADDS to the angle
  1525→    this.pose[CH.wrR] -= corr
  1526→  }

  1528→  private nearestEnemy(w: WorldRef, range: number): { x: number; y: number } | null {
  1529→    let best: { x: number; y: number } | null = null
  1530→    let bestD = Infinity
  1531→    for (const e of w.enemies) {
  1532→      if (e.dead || e.spawning) continue
  1533→      const d = Math.abs(e.x - this.x)
  1534→      if (d < bestD && d < range) {
  1535→        bestD = d
  1536→        best = { x: e.x, y: e.y }
  1537→      }
  1538→    }
  1539→    return best
  1540→  }

  1542→  /** HYPER-SCALE: the blade grows ×1.22 through the active frames and ×1.12

  1548→    else if (this.state === 'attack' && this.action) {
  1549→      const h = this.action.def.hit

  1570→      }

  1574→  /** TWO-HANDED GRIP — the off hand solves 2-bone IK onto the tsuka butt
  1575→   * (te-no-uchi: left hand at the pommel, right near the guard — how a

  1577→   * with a state-driven weight so running/dashing pumps stay authored. */
  1578→  private applyTwoHandGrip(dt: number): void {
  1579→    // desired grip weight by state
  1580→    let target = 0
  1581→    switch (this.state) {
  1582→      case 'move':
  1583→        target = this.grounded && Math.abs(this.vx) < 60 ? 1 : 0.1
  1584→        break
  1585→      case 'attack':
  1586→        target = this.action ? (this.action.def.grip ?? 0.45) : 0.45
  1587→        break
  1588→      case 'charge':
  1589→      case 'parry':
  1590→        target = 1
  1591→        break
  1592→      case 'scripted':
  1593→        target = 0.5
  1594→        break
  1595→      default:
  1596→        target = 0
  1597→    }
  1598→    this.twoHandW = damp(this.twoHandW, target, 11, dt)
  1599→    const w = this.twoHandW
  1600→    if (w < 0.03) return
  1601→
  1602→    const g = this.geom // last frame's solve — continuous, 1-frame stale
  1603→    const tsuka = PLAYER_SPEC.blade * this.sword.stats.range * 0.3
  1604→    // tsuka butt (pommel end) in local space
  1605→    const bax = Math.cos(g.bladeAngle)
  1606→    const bay = Math.sin(g.bladeAngle)
  1607→    const tx = g.handR.x - bax * tsuka
  1608→    const ty = g.handR.y - bay * tsuka
  1609→    // shoulder anchor = chest (solver attaches arms there)
  1610→    const sx = g.chest.x
  1611→    const sy = g.chest.y
  1612→    const U = PLAYER_SPEC.upperArm
  1613→    const F = PLAYER_SPEC.foreArm
  1614→    let dx = tx - sx
  1615→    let dy = ty - sy
  1616→    let d = Math.hypot(dx, dy)
  1617→    const maxD = (U + F) * 0.98
  1618→    if (d > maxD) {
  1619→      const k = maxD / d
  1620→      dx *= k
  1621→      dy *= k
  1622→      d = maxD
  1623→    }
  1624→    if (d < 2) return
  1625→    // law of cosines: interior angles of the shoulder-elbow-target triangle
  1626→    const cosB = clamp((U * U + d * d - F * F) / (2 * U * d), -1, 1)
  1627→    const B = Math.acos(cosB)
  1628→    const base = Math.atan2(dy, dx)
  1629→    // elbow bends DOWN (anatomical) → rotate the upper arm below the line
  1630→    const upperAng = base + B
  1631→    const ex = sx + Math.cos(upperAng) * U
  1632→    const ey = sy + Math.sin(upperAng) * U
  1633→    // convert elbow/hand to the channel convention (from straight-down, +fwd)
  1634→    const shIk = Math.atan2(ex - sx, ey - sy)
  1635→    const elIk = Math.atan2(tx - ex, ty - ey) - shIk
  1636→    const p = this.pose
  1637→    p[CH.shL] += (shIk - p[CH.shL]) * w
  1638→    p[CH.elL] += (elIk - p[CH.elL]) * w
  1639→  }

  1641→  private applyProcedural(dt: number): void {
  1642→    const p = this.pose
  1643→    // velocity lean (proportional, eased)
  1644→    const speedLean = clamp((this.vx * this.face) / MOVE_MAX, -1, 1)
  1645→    p[2] += speedLean * 0.14

  1647→    if (!this.grounded && this.state !== 'attack') {
  1648→      p[2] += clamp(this.vy / 1400, -0.22, 0.3)
  1649→    }

  1651→    if (this.landSpring > 0) {
  1652→      this.landSpring = Math.max(0, this.landSpring - dt * 5)
  1653→      p[1] += this.landSpring * 6
  1654→    }

  1656→    if (this.charging) {
  1657→      p[0] += rand(-0.8, 0.8) * this.chargeRatio

  1660→    // hurt shake
  1661→    if (this.hurtT > 0) {
  1662→      p[0] += rand(-1.4, 1.4) * this.hurtT
  1663→    }
  1664→  }

  1864→function fromChain(def: AttackDef, next: string, player: Player): boolean {
  1865→  if (next === 'riposte') return true
  1866→  // launcher chains naturally into air attacks
  1867→  if (def.id === 'launcher' && (next.startsWith('a') || next === 'dive' || next === 'slam')) return true
  1868→  if (def.chain.includes(next)) return true
  1869→  // any light may restart the chain if the window allows
  1870→  if (player.grounded && next.startsWith('l') && def.chain.some((c) => c.startsWith('l'))) return true
  1871→  if (!player.grounded && (next.startsWith('a') || next === 'dive' || next === 'slam') && def.chain.some((c) => c.startsWith('a') || c === 'dive' || c === 'slam')) return true
  1872→  return false
  1873→}
