    24→/** circle vs swept blade quad (prev segment + current segment) — prevents tunneling */
    25→function circleHitsSweptQuad(
    26→  cx: number, cy: number, r: number,
    27→  pbx: number, pby: number, ptx: number, pty: number,
    28→  bx: number, by: number, tx: number, ty: number,
    29→): boolean {
    30→  // current + previous blade segments
    31→  if (segHitsCircle(bx, by, tx, ty, cx, cy, r)) return true
    32→  if (segHitsCircle(pbx, pby, ptx, pty, cx, cy, r)) return true
    33→  // swept edges (tip path + base path)
    34→  if (segHitsCircle(ptx, pty, tx, ty, cx, cy, r)) return true
    35→  if (segHitsCircle(pbx, pby, bx, by, cx, cy, r)) return true
    36→  // circle center inside the swept quad (spins / wide arcs)
    37→  if (pointInTri(cx, cy, pbx, pby, ptx, pty, tx, ty)) return true
    38→  if (pointInTri(cx, cy, pbx, pby, tx, ty, bx, by)) return true
    39→  return false
    40→}

    42→function pointInTri(px: number, py: number, x1: number, y1: number, x2: number, y2: number, x3: number, y3: number): boolean {
    43→  const d1 = (px - x2) * (y1 - y2) - (x1 - x2) * (py - y2)
    44→  const d2 = (px - x3) * (y2 - y3) - (x2 - x3) * (py - y3)
    45→  const d3 = (px - x1) * (y3 - y1) - (x3 - x1) * (py - y1)
    46→  const hasNeg = d1 < 0 || d2 < 0 || d3 < 0
    47→  const hasPos = d1 > 0 || d2 > 0 || d3 > 0
    48→  return !(hasNeg && hasPos)
    49→}

    70→const SCALE = 1.32
    71→/** nominal (airborne) hip height = thigh(25)+shin(24) = 49 local × SCALE.
    72→ * Grounded height is NOT fixed — it derives from the planted foot via the
    73→ * foot-ground lock (updateFootLock): idle bent knees, lunges and landing
    74→ * squashes all sit exactly ON the ground line. */
    75→const HIP_H = (25 + 24) * SCALE

    77→/**
    78→ * Enemy body spheres: [centerY offset above ground, radius] in enemy-local
    79→ * units (× e.scale). Traced from the ENEMY_SPEC skeleton — hip at 47u, torso
    80→ * 32u, head top ≈ 94u. The old single circle (y-34, r 26) only covered the
    81→ * lower 2/3 of the body, so chest/head-height slashes passed straight
    82→ * through the sprite without connecting. Three spheres hug the visual body.
    83→ */
    84→const BODY_SPHERES: ReadonlyArray<{ oy: number; r: number }> = [
    85→  { oy: 14, r: 15 }, // legs / hip mass
    86→  { oy: 48, r: 21 }, // torso (the main target)
    87→  { oy: 78, r: 17 }, // chest / head zone
    88→]
    89→
    90→/** blade segment vs an enemy's full visual body (any sphere, swept-aware).
    91→ * `pad` fattens the radius per-attack — a katana edge is a plane; the hit
    92→ * volume must match the visual footprint of the blade edge corridor. */
    93→function bladeHitsBody(
    94→  e: { x: number; y: number; scale: number },
    95→  hasPrev: boolean,
    96→  prev: { bx: number; by: number; tx: number; ty: number },
    97→  cur: { bx: number; by: number; tx: number; ty: number },
    98→  pad = 3,
    99→): boolean {
   100→  const s = e.scale
   101→  for (const sp of BODY_SPHERES) {
   102→    const cy = e.y - sp.oy * s
   103→    const r = sp.r * s + pad
   104→    if (hasPrev
   105→      ? circleHitsSweptQuad(e.x, cy, r, prev.bx, prev.by, prev.tx, prev.ty, cur.bx, cur.by, cur.tx, cur.ty)
   106→      : segHitsCircle(cur.bx, cur.by, cur.tx, cur.ty, e.x, cy, r)
   107→    ) return true
   108→  }
   109→  return false
   110→}

   468→  update(dt: number, w: WorldRef): void {
   469→    this.iframes = Math.max(0, this.iframes - dt)
   470→    this.armorT = Math.max(0, this.armorT - dt)
   471→    this.flashT = Math.max(0, this.flashT - dt * 5)
   472→    this.swingGlow = Math.max(0, this.swingGlow - dt * 5)
   473→    this.riposteT = Math.max(0, this.riposteT - dt)
   474→    this.energy = Math.min(this.maxEnergy, this.energy + dt * 3)
   475→
   476→    // context-system clocks
   477→    this.now += dt
   478→    this.locoT += dt
   479→    if (this.turnT > 0) this.turnT = Math.max(0, this.turnT - dt * 6) // ~0.17s pivot
   480→    if (this.postAttackT > 0) this.postAttackT = Math.max(0, this.postAttackT - dt)
   481→    // the sword-home ritual only lives in the move state — any external
   482→    // transition (hurt / parry / scripted) abandons it
   483→    if (this.state !== 'move' && this.sheathAct) this.sheathAct = null
   484→
   485→    switch (this.state) {
   486→      case 'move': this.updateMove(dt, w); break
   487→      case 'attack': this.updateAttack(dt, w); break
   488→      case 'dash': this.updateDash(dt, w); break
   489→      case 'parry': this.updateParry(dt, w); break
   490→      case 'charge': this.updateCharge(dt, w); break
   491→      case 'hurt': this.updateHurt(dt, w); break
   492→      case 'dead': this.updateDead(dt, w); break
   493→      case 'scripted': this.updateScripted(dt); break
   494→    }
   495→
   496→    // physics integration (shared)
   497→    if (this.state !== 'dash') {
   498→      let g: number
   499→      if (this.state === 'attack' && this.action && this.action.def.float) {
   500→        // authored air-combo float overrides the shaping (juggle windows)
   501→        g = GRAVITY * this.action.def.float
   502→      } else if (this.vy < 0) {
   503→        // ANTI-FLOAT: apex decay — no moon hovering
   504→        g = GRAVITY * (this.vy < -260 ? GRAV_RISE : GRAV_APEX)
   505→      } else {
   506→        // heavy fall: the body plummets
   507→        g = GRAVITY * GRAV_FALL
   508→      }
   509→      this.vy += g * dt
   510→      if (this.vy > TERMINAL_VY) this.vy = TERMINAL_VY
   511→    }
   512→    this.x += this.vx * dt
   513→    this.y += this.vy * dt
   514→
   515→    // ground
   516→    const wasAir = !this.grounded
   517→    if (this.y >= 0) {
   518→      this.y = 0
   519→      if (this.vy > 140) {
   520→        this.landSpring = clamp(this.vy / 700, 0, 1)
   521→        if (wasAir) this.onLand(w)
   522→      } else if (wasAir) {
   523→        this.landSpring = 0.4
   524→        this.onLand(w)
   525→      }
   526→      this.vy = 0
   527→      this.grounded = true
   528→      this.jumpsUsed = 0
   529→      this.airDashUsed = false
   530→    } else {
   531→      this.grounded = false
   532→    }
   533→    // arena bounds
   534→    if (this.x < w.arenaMinX + 20) {
   535→      this.x = w.arenaMinX + 20
   536→      if (this.vx < -200) w.fx.dust(this.x - 12, 0, 5)
   537→      this.vx = Math.max(0, this.vx)
   538→    }
   539→    if (this.x > w.arenaMaxX - 20) {
   540→      this.x = w.arenaMaxX - 20
   541→      if (this.vx > 200) w.fx.dust(this.x + 12, 0, 5)
   542→      this.vx = Math.min(0, this.vx)
   543→    }
   544→
   545→    this.dashCd = Math.max(0, this.dashCd - dt)
   546→    this.dashRecharge = Math.max(0, this.dashRecharge - dt)
   547→    this.retreatLockT = Math.max(0, this.retreatLockT - dt)
   548→    if (this.dashRecharge <= 0) {
   549→      // chain bookkeeping: the counter survives for the chain window after
   550→      // the LAST dash — reset only when the window closes (or the recharge
   551→      // punishment completes), never unconditionally every frame
   552→      this.dashChainT = Math.max(0, this.dashChainT - dt)
   553→      if (this.dashChainT <= 0) this.dashConsec = 0
   554→    }
   555→
   556→    // ability script
   557→    if (this.ability) this.updateAbility(dt, w)
   558→
   559→    // animation + solve
   560→    this.animator.update(dt)
   561→    // Base layer: the clip FULLY owns the pose each frame (write, never
   562→    // accumulate). applyProcedural() is a bounded additive layer on top.
   563→    // Without this copy, procedural += deltas integrate unbounded → spine
   564→    // spin + root sink. (Same contract as Spine's AnimationState.apply /
   565→    // Godot AnimationTree → Skeleton2D.)
   566→    this.pose.set(this.animator.pose)
   567→    this.applyProcedural(dt)
   568→    this.updateBladeZ(dt) // zArc → bladeYaw channel (BEFORE the 3D solve)
   569→    // procedural layers (bounded, frame-local — never accumulate):
   570→    // SWORD-LED ARC first: it rewrites the arm channels, so the blade
   571→    // dynamics spring below sees the ARC's angular velocity — the whip
   572→    // lags/overshoots the real swing, not the raw keyframes.
   573→    this.swordArc.update(dt, {
   574→      pose: this.pose,
   575→      shoulderX: this.geom.shoulderR.x, shoulderY: this.geom.shoulderR.y,
   576→      shoulderZ: this.geom.shoulderR.z,
   577→      handX: this.geom.handR.x, handY: this.geom.handR.y,
   578→      handZ: this.geom.handR.z,
   579→      def: this.state === 'attack' ? this.action?.def ?? null : null,
   580→      animT: this.action ? this.action.animT : 0,
   581→      enemies: w.enemies,
   582→      face: this.face,
   583→      x: this.x,
   584→      hipY: this.hipY,
   585→      reach: PLAYER_SPEC.blade * this.sword.stats.range * SCALE * 1.1 + 130,
   586→    })
   587→    this.updateBladeDynamics(dt, w) // wrist lag spring + kinetic chain — feeds hitboxes
   588→    this.applyTwoHandGrip(dt)       // off-hand IK onto the tsuka
   589→    // 3D SOLVE: sagittal channels + depth channels → projected joints + blade
   590→    solveFighter3D(
   591→      this.pose, PLAYER_SPEC, this.geom,
   592→      PLAYER_SPEC.blade * this.sword.stats.range * 1.06,
   593→      this.swingScale,
   594→    )
   595→    this.updateFootLock(dt)         // weight: hips sit on the planted foot
   596→    this.updateBladeWorld()
   597→    this.updateSwingScale(dt)
   598→  }

   643→  /** Per-frame enemy-formation scan (spec §5): nearest threat + proximity
   644→   * weight (with hysteresis) + surrounded flag. Feeds the locomotion
   645→   * selector AND the procedural posture layers. NEVER takes control away
   646→   * from the player — posture only. */
   647→  private scanEnemies(dt: number, w: WorldRef): void {
   648→    const RADIUS = 430
   649→    let bestD = Infinity
   650→    let ahead = false
   651→    let behind = false
   652→    for (const e of w.enemies) {
   653→      if (e.dead || e.spawning) continue
   654→      const dx = e.x - this.x
   655→      const ad = Math.abs(dx)
   656→      if (ad < RADIUS) {
   657→        if (dx * this.face > 0) ahead = true
   658→        else behind = true
   659→      }
   660→      if (ad < bestD) {
   661→        bestD = ad
   662→        this.threatX = e.x
   663→        this.threatY = e.y
   664→        this.hasThreat = true
   665→      }
   666→    }
   667→    if (bestD === Infinity) this.hasThreat = false
   668→    this.surrounded = ahead && behind && bestD < 300
   669→    // asymmetric damp = hysteresis: fast to raise the guard, slow to relax
   670→    const target = bestD < RADIUS ? 1 : 0
   671→    const k = target > this.enemyNearW ? 7 : 2.2
   672→    this.enemyNearW = damp(this.enemyNearW, target, k, dt)
   673→    // head-tracking weight: quick to look, slower to look away
   674→    const ht = this.hasThreat && bestD < 520 ? 1 : 0
   675→    const hk = ht > this.headTrackW ? 8 : 3
   676→    this.headTrackW = damp(this.headTrackW, ht, hk, dt)
   677→  }

   679→  /** Select the locomotion mode with dwell hysteresis (no flicker): a mode
   680→   * must be held ≥0.1s before another can take over (except brake/turn
   681→   * interrupts). Priority: brake > sprint latch > combat (moving, enemy
   682→   * near) > backpedal (retreating vs facing) > run > walk > combatIdle
   683→   * (threat engaged, standing) > idle (safe, standing).
   684→   *
   685→   * THE IDLE FIX: standing still with a threat engaged is combatIdle — a
   686→   * ROOTED guard (planted feet, weight shifts). The old selector returned
   687→   * the combat SHUFFLE here, so the player visibly marched in place while
   688→   * "idle". Stepping only happens when actually moving. */
   689→  private pickLoco(speed: number, ax: number): LocoMode {
   690→    const fwd = this.vx * this.face
   691→    if (this.braking) return 'brake'
   692→    if (!this.grounded) return this.loco // airborne handled by clips, not modes
   693→    if (speed < 36) {
   694→      return this.enemyNearW > 0.5 || this.surrounded ? 'combatIdle' : 'idle'
   695→    }
   696→    // retreating while a threat is engaged → guarded backpedal
   697→    if (fwd < -60 && this.enemyNearW > 0.4) return 'backpedal'
   698→    // explicit sprint outranks proximity awareness (player intent wins)
   699→    if (this.sprintLatch && speed > 300) return 'sprint'
   700→    // Condition B: combat shuffle while enemies are near (but full-speed
   701→    // committed running still uses the run mechanics)
   702→    if (this.enemyNearW > 0.55 && speed < 310 && ax !== 0) return 'combat'
   703→    if (speed > 210) return 'run'
   704→    return 'walk'
   705→  }
   706→
   707→  private setLoco(mode: LocoMode): void {
   708→    if (mode === this.loco) return
   709→    // dwell hysteresis: hold each mode ≥0.1s (brake interrupts instantly)
   710→    if (this.locoT < 0.1 && mode !== 'brake' && this.loco !== 'brake') return
   711→    this.loco = mode
   712→    this.locoT = 0
   713→  }

   715→  private updateMove(dt: number, w: WorldRef): void {
   716→    const input = w.input
   717→    const ax = input.axisX()
   718→    const ay = input.axisY()
   719→
   720→    // ---- enemy-formation awareness (posture influence ONLY) ----
   721→    this.scanEnemies(dt, w)
   722→
   723→    // ---- SWORD HOME ritual: an active sheathe/draw owns the body (rooted,
   724→    // planted — you do not walk while feeding steel home). Danger inputs
   725→    // interrupt instantly with the combat snap-draw. ----
   726→    if (this.sheathAct) {
   727→      this.updateSheathAct(dt, w, ax, ay)
   728→      return
   729→    }
   730→
   731→    // ---- facing: hold toward an engaged threat at maneuvering speed ----
   732→    // (Condition B: circle/retreat keeps the guard trained on the enemy —
   733→    // backpedal instead of a civilian turn-away. Full-speed escape or
   734→    // open-space movement flips freely — EXCEPT inside the post-retreat
   735→    // guard lock: a dash-back never converts into a fleeing turn.)
   736→    if (ax !== 0 && this.face !== (ax > 0 ? 1 : -1)) {
   737→      const engaged = this.enemyNearW > 0.55 && this.hasThreat
   738→      const slow = Math.abs(this.vx) < 260 && !this.sprintLatch
   739→      if (!(engaged && slow) && this.retreatLockT <= 0) {
   740→        if (Math.abs(this.vx) > 120) {
   741→          this.turnT = 1
   742→          this.turnDir = -Math.sign(ax)
   743→          w.fx.dust(this.x, 0, 4)
   744→        }
   745→        this.face = ax > 0 ? 1 : -1
   746→      }
   747→    }
   748→
   749→    // ---- double-tap direction → sprint latch (DMC-style, event-driven) ----
   750→    // Press stamps come from the keydown EVENTS (input clock), so a rapid
   751→    // tap-release-tap registers even if the release window contained zero
   752→    // game frames (low fps / throttled rAF).
   753→    {
   754→      const sr = w.input.pressStamp('right')
   755→      const sl = w.input.pressStamp('left')
   756→      const newR = sr !== this.stampR && sr >= 0
   757→      const newL = sl !== this.stampL && sl >= 0
   758→      if (newR || newL) {
   759→        const stamp = newR ? sr : sl
   760→        const dir = newR ? 1 : -1
   761→        if (stamp - this.lastTapT < 0.3 && dir === this.lastTapDir) {
   762→          this.sprintLatch = true
   763→        }
   764→        this.lastTapDir = dir
   765→        this.lastTapT = stamp
   766→      }
   767→      this.stampR = sr
   768→      this.stampL = sl
   769→    }
   770→    if (ax === 0) this.sprintLatch = false
   771→
   772→    // ---- horizontal movement (sprint has its own accel curve + ceiling) ----
   773→    const sprinting = this.sprintLatch && this.grounded && ax !== 0 && Math.sign(ax) === Math.sign(this.vx || ax)
   774→    const vMax = sprinting ? SPRINT_MAX : MOVE_MAX
   775→    if (ax !== 0) {
   776→      // input reversal at speed → BRAKE (skid decel + one-shot clip + dust)
   777→      const reversing = Math.sign(ax) !== Math.sign(this.vx) && Math.abs(this.vx) > 240
   778→      if (reversing && this.grounded && !this.braking) {
   779→        this.braking = true
   780→        this.animator.play(BRAKE_CLIP, { blend: 0.04 })
   781→        w.fx.dust(this.x - this.face * 10, 0, 7)
   782→        w.audio.play('land', { volume: 0.35 })
   783→      }
   784→      if (reversing || this.braking) {
   785→        // hard skid deceleration toward zero before re-accelerating
   786→        const d = BRAKE_DECEL * dt
   787→        this.vx = Math.abs(this.vx) <= d ? 0 : this.vx - Math.sign(this.vx) * d
   788→      } else {
   789→        // analog target speed: axis magnitude maps to a target velocity —
   790→        // partial input = partial speed (walk reachable; touch-friendly),
   791→        // full keys still hit the ceiling. Accelerate toward it, never past.
   792→        const targetV = ax * vMax
   793→        const movingToward = Math.sign(this.vx) === Math.sign(targetV) || this.vx === 0
   794→        if (!movingToward || Math.abs(this.vx) < Math.abs(targetV)) {
   795→          const accel = sprinting ? SPRINT_ACCEL : this.grounded ? ACCEL : AIR_ACCEL
   796→          this.vx += ax * accel * dt
   797→          if (Math.sign(this.vx) === Math.sign(targetV) && Math.abs(this.vx) > Math.abs(targetV)) {
   798→            this.vx = targetV
   799→          }
   800→        } else if (Math.sign(this.vx) === Math.sign(targetV) && Math.abs(this.vx) > Math.abs(targetV)) {
   801→          // ease down toward the lower analog target (stick pushed softer)
   802→          const f = FRICTION * dt
   803→          const dv = Math.min(f, Math.abs(this.vx) - Math.abs(targetV))
   804→          this.vx -= Math.sign(this.vx) * dv
   805→        }
   806→      }
   807→      this.vx = clamp(this.vx, -vMax, vMax)
   808→    } else if (this.grounded) {
   809→      const f = FRICTION * dt
   810→      this.vx = Math.abs(this.vx) <= f ? 0 : this.vx - Math.sign(this.vx) * f
   811→    } else {
   812→      this.vx = damp(this.vx, 0, 0.4, dt)
   813→    }
   814→
   815→    // ---- locomotion clip selection (context-aware, hysteresis) ----
   816→    const speed = Math.abs(this.vx)
   817→    if (this.grounded) {
   818→      if (this.braking) {
   819→        if (this.animator.isDone() || speed < 90) this.braking = false
   820→      } else {
   821→        this.setLoco(this.pickLoco(speed, ax))
   822→        switch (this.loco) {
   823→          case 'walk':
   824→            this.animator.play(WALK_CLIP, { blend: 0.18, speed: clamp(speed / 210, 0.7, 1.35) })
   825→            break
   826→          case 'run':
   827→            this.animator.play(RUN_CLIP, { blend: 0.12, speed: clamp(speed / MOVE_MAX, 0.8, 1.5) })
   828→            break
   829→          case 'sprint':
   830→            this.animator.play(SPRINT_CLIP, { blend: 0.1, speed: clamp(speed / SPRINT_MAX, 0.85, 1.25) })
   831→            break
   832→          case 'combat':
   833→            this.animator.play(COMBAT_STEP_CLIP, { blend: 0.16, speed: clamp(speed / 260, 0.7, 1.7) })
   834→            break
   835→          case 'combatIdle':
   836→            this.animator.play(COMBAT_IDLE_CLIP, { blend: 0.18 })
   837→            break
   838→          case 'backpedal':
   839→            this.animator.play(BACKPEDAL_CLIP, { blend: 0.14, speed: clamp(speed / 240, 0.7, 1.6) })
   840→            break
   841→          default:
   842→            this.animator.play(IDLE_CLIP, { blend: 0.16 })
   843→        }
   844→      }
   845→    } else if (this.vy < -60) {
   846→      // rising: the takeoff family (jump / leap / air-jump) is authored AT
   847→      // the press (doJump) — never stomp it the frame after; only default
   848→      // to the plain rise clip when airborne WITHOUT a jump entry this
   849→      // flight (e.g. exiting an air dash or an attack float)
   850→      const cn = this.animator.getClip()?.name
   851→      if (cn !== 'jump' && cn !== 'jumpLeap' && cn !== 'airJump') {
   852→        this.animator.play(JUMP_CLIP, { blend: 0.06 })
   853→      }
   854→    } else {
   855→      this.animator.play(FALL_CLIP, { blend: 0.12 })
   856→    }
   857→
   858→    // ---- actions ----
   859→    if (this.tryCombatInputs(w, ax, ay)) return
   860→
   861→    // ---- AUTO-SHEATHE (the quiet beat): grounded, no threat, standing
   862→    // still with the blade out for 3.5s → the samurai-film ending. Any
   863→    // movement or threat resets the dwell. ----
   864→    if (this.swordDrawn && this.grounded && !this.hasThreat && Math.abs(this.vx) < 24) {
   865→      this.peaceT += dt
   866→      if (this.peaceT > 3.5) this.startSheath(w)
   867→    } else {
   868→      this.peaceT = 0
   869→    }
   870→  }

   872→  /** returns true if a new state was entered */
   873→  private tryCombatInputs(w: WorldRef, ax: number, ay: number): boolean {
   874→    const input = w.input
   875→
   876→    // ---- SWORD HOME toggle (KeyQ): the notō / iai ritual ----
   877→    if (input.pressed('sheathe')) {
   878→      if (this.swordDrawn) this.startSheath(w)
   879→      else this.startDraw(w)
   880→      return true
   881→    }
   882→
   883→    // dash (anytime, from move) — the steel snaps out with it
   884→    if ((input.pressed('dash') || input.buffered('dash')) && this.canDash()) {
   885→      if (!this.swordDrawn) this.snapDraw(w)
   886→      this.startDash(w, ax, ay)
   887→      return true
   888→    }
   889→    // jump (grounded = jump 1, plus one air jump) — HANDS-ONLY: the steel
   890→    // can stay home through the whole airborne family
   891→    if (input.pressed('jump') || input.buffered('jump', 0.18)) {
   892→      if (this.jumpsUsed < 2) {
   893→        this.doJump(w)
   894→        return false
   895→      }
   896→    }
   897→    // parry
   898→    if (input.pressed('parry')) {
   899→      if (!this.swordDrawn) this.snapDraw(w)
   900→      this.enterParry(w)
   901→      return true
   902→    }
   903→    // ability
   904→    if (input.pressed('ability')) {
   905→      if (this.energy >= this.sword.ability.cost) {
   906→        if (!this.swordDrawn) this.snapDraw(w)
   907→        this.startAbility(w)
   908→        return true
   909→      }
   910→    }
   911→    // ult is handled by Game (cinematic)
   912→
   913→    // heavy: charge vs tap
   914→    if (input.down('heavy') && this.grounded) {
   915→      if (!this.swordDrawn) this.snapDraw(w)
   916→      this.enterCharge(w)
   917→      return true
   918→    }
   919→    if (input.pressed('heavy') || input.buffered('heavy')) {
   920→      input.takeBuffered('heavy')
   921→      if (!this.swordDrawn) this.snapDraw(w)
   922→      this.startAttack(this.pickHeavy(w), w)
   923→      return true
   924→    }
   925→    // light
   926→    if (input.pressed('light') || input.buffered('light')) {
   927→      input.takeBuffered('light')
   928→      if (!this.swordDrawn) this.snapDraw(w)
   929→      // DASH MOMENTUM STRIKE: attacking while the last dash's momentum
   930→      // still carries the body converts into the through-cut — the
   931→      // dash-strike feel (Hades). Without this, the 150ms in-dash input
   932→      // window was brutally tight and dash attacks "randomly" whiffed.
   933→      if (this.grounded
   934→        && w.time - this.dashEndedAt < 0.18
   935→        && Math.abs(this.vx) > 110
   936→        && this.riposteT <= 0) {
   937→        this.startAttack(ATTACKS.dashSlash, w)
   938→        return true
   939→      }
   940→      const id = this.riposteT > 0 ? 'riposte' : this.pickLight(w, ax, ay)
   941→      this.startAttack(ATTACKS[id], w)
   942→      return true
   943→    }
   944→    return false
   945→  }

   968→  /** instant combat draw: the ritual is abandoned, steel out NOW — every
   969→   * combat input while sheathed routes through here (iai snap-draw). */
   970→  private snapDraw(w: WorldRef): void {
   971→    this.sheathAct = null

  1098→  /** CONDITIONED VARIANT RESOLUTION: snapshot the full tactical context at

  1212→  private updateDash(dt: number, w: WorldRef): void {
  1213→    this.dashT -= dt
  1214→    // SHADOW-SILHOUETTE afterimages — the dark ghost trail with the backlit
  1215→    // rim (replaces the old neon ghosts + speed-streak clutter). Spacing
  1216→    // tuned per variant (denser on the aggressive).
  1217→    this.afterimgTimer -= dt
  1218→    if (this.afterimgTimer <= 0) {
  1219→      this.afterimgTimer = this.dashCfg.ghost
  1220→      w.fx.shadowGhost(this.pose, this.x, this.hipY, this.face, this.dashDx, this.sword.colors.core, 0.32, SCALE)
  1221→    }
  1222→    // velocity envelope (combat/dash.ts): entry burst → sustain plateau →
  1223→    // quadratic tail bleed. Hunt/pierce HOLD full speed through the mid
  1224→    // dash (the hunger); launch bleeds immediately into the exit handoff.
  1225→    const k = 1 - this.dashT / this.dashDur
  1226→    const sp = this.dashSpd * dashEnvelope(k, this.dashCfg.sustain)
  1227→    this.vx = this.dashDx * sp
  1228→    this.vy = this.dashDy * sp * (this.grounded ? (this.dashDy < 0 ? 0.4 : 1) : 1)
  1229→
  1230→    // dash attack (kept: light = through-cut, heavy = piercing fang)
  1231→    const input = w.input
  1232→    if (input.pressed('light') || input.buffered('light')) {
  1233→      input.takeBuffered('light')
  1234→      this.startAttack(ATTACKS.dashSlash, w)
  1235→      return
  1236→    }
  1237→    if (input.pressed('heavy') || input.buffered('heavy')) {
  1238→      if (!this.grounded) {
  1239→        input.takeBuffered('heavy')
  1240→        this.startAttack(ATTACKS.slam, w)
  1241→        return
  1242→      }
  1243→      input.takeBuffered('heavy')
  1244→      this.startAttack(ATTACKS.dashHeavy, w)
  1245→      return
  1246→    }
  1247→    if (this.dashT <= 0) {
  1248→      this.state = 'move'
  1249→      this.dashEndedAt = w.time
  1250→      // retreat family: commit the facing toward the threat through the
  1251→      // recovery beat (see retreatLockT — the guard never turns its back)
  1252→      if (this.dashVariant === 'retreat' || this.dashVariant === 'carve' || this.dashVariant === 'vault') {
  1253→        this.retreatLockT = 0.45
  1254→      }
  1255→      // ---- MOMENTUM EXIT (the smoothness law): never a dead stop ----
  1256→      const ax = input.axisX()
  1257→      const dirHeld = Math.sign(ax) === Math.sign(this.dashDx) && Math.abs(ax) > 0.25
  1258→      const retreat = this.dashVariant === 'retreat' || this.dashVariant === 'carve'
  1259→      if (this.grounded) {
  1260→        if (dirHeld && Math.abs(this.dashDx) > 0.35) {
  1261→          // HANDOFF: dash speed decays into locomotion. The forward lunge
  1262→          // hands off sprint-worthy speed (the player FEELS carried, not
  1263→          // braked); the retreat variants hand off at BACKPEDAL pace — a
  1264→          // guarded retreat must never convert into a full-speed fleeing
  1265→          // run with the guard dropped (the facing logic would flip the
  1266→          // body away from the threat mid-retreat)
  1267→          const cap = retreat ? 250 : SPRINT_MAX
  1268→          const exitSp = Math.min(Math.abs(this.vx), cap)
  1269→          this.vx = Math.sign(this.dashDx) * exitSp
  1270→          this.dashHandoff = true
  1271→        } else if (retreat) {
  1272→          // retreat brake: a short hop cuts DEEP and settles through the
  1273→          // clip's own guard tail — never the forward landing skid (the
  1274→          // guarded retreat would visually pitch chest-first into a crouch)
  1275→          this.vx *= 0.45
  1276→        } else {
  1277→          // forward brake: soften the cut, then the physical skid takes over
  1278→          this.vx *= 0.72
  1279→          if (Math.abs(this.vx) > 240) {
  1280→            // dark skid streak (shadow language — not a dust puff cloud)
  1281→            w.fx.shadowSmear(this.x - Math.sign(this.vx) * 20, 26, 0.4)
  1282→            this.landSpring = 0.35
  1283→            this.animator.play(LAND_CLIP, { blend: 0.05 })
  1284→          }
  1285→        }
  1286→      } else {
  1287→        this.vy *= 0.5
  1288→      }
  1289→    }
  1290→  }

  1433→    // forward impulse
  1434→    if (!a.moveDone && def.moveX !== undefined && animT >= (def.moveXAt ?? def.hit.start)) {

  1451→      this.swingGlow = 1
  1452→      const info = makeRuntimeInfo(def, this.sword.stats.dmg)
  1453→      // DASH THROUGH-CUT: the iai pass cuts everything the dash swept
  1454→      // through — the player crosses the target at 980px/s BEFORE the
  1455→      // attack anim even starts, so the blade-only segment points past
  1456→      // them. The corridor is the DASH PATH itself: a fixed chest-height
  1457→      // band from 124px behind the player to 66px ahead — anchored to the
  1458→      // path (not the wandering kissaki, which wraps behind on the
  1459→      // follow-through and would collapse the span). This is the swept

  1464→        curSeg = {
  1465→          bx: this.x - this.face * 124,
  1466→          by: this.y - 55,
  1467→          tx: this.x + this.face * 66,
  1468→          ty: this.y - 55,
  1469→        }
  1470→        usePrev = false
  1471→      }
  1472→      for (const e of w.enemies) {
  1473→        // spawning enemies within blade reach (landing pods, y > -90) are
  1474→        // fair game — the swing connects the moment the body crosses the arc
  1475→        if (e.dead || (e.spawning && e.y < -90)) continue

  1477→        const reArm = h.multiHit ? (last === undefined || animT - last >= h.multiHit) : last === undefined

  1486→      this.prevBlade.bx = this.blade.bx
  1487→      this.prevBlade.by = this.blade.by

  1493→      // the shockwave visual must connect, not just decorate)

  1500→          if (a.hitMap.has(e.id) && !h.multiHit) continue

  1503→            w.hitEnemy(e.id, this.blade, info)

  1509→

  1511→    // 2-frame rotational impulse — the tip pushes PAST the stop point and
  1512→    // the wrist spring elastic-snaps it back. Stopping dead reads weightless.
  1513→    if (!a.overshootDone && animT > h.end && this.armVel !== 0) {
  1514→      a.overshootDone = true

  1520→      w.fx.pushTrail(a.trail, this.blade.bx, this.blade.by, this.blade.tx, this.blade.ty)
  1521→    } else if (animT > def.trail.end && a.trail.emitting) {
  1522→      w.fx.endTrail(a.trail)
  1523→    }
  1524→

  1533→      }

  1535→        w.fx.endTrail(a.trail)
  1536→        this.enterParry(w)
  1537→        return
  1538→      }
  1539→      if (animT >= def.cancel) {
  1540→        // chain next attack
  1541→        const ax = input.axisX()
  1542→        const ay = input.axisY()
  1543→        let next: string | null = null
  1544→        if (input.buffered('light')) {
  1545→          input.takeBuffered('light')
  1546→          next = this.riposteT > 0 && this.grounded ? 'riposte' : this.pickLight(w, ax, ay)
  1547→        } else if (input.buffered('heavy')) {
  1548→          input.takeBuffered('heavy')
  1549→          if (!this.grounded) {
  1550→            next = 'slam'
  1551→          } else if (input.down('heavy')) {
  1552→            // still holding → convert to charge
  1553→            this.enterCharge(w)
  1554→            w.fx.endTrail(a.trail)
  1555→            return

  1558→            const heavyNext = def.chain.find((c) => /^h[123]$/.test(c))
  1559→            next = heavyNext ?? (def.kind === 'heavy' ? 'l1' : 'h1')
  1560→          }

  1562→        if (next) {
  1563→          // validate chain (or allow from-parry riposte)
  1564→          if (fromChain(def, next, this)) {
  1565→            w.fx.endTrail(a.trail)
  1566→            this.startAttack(ATTACKS[next], w)
  1567→            return
  1568→          }
  1569→        }
  1570→        // jump cancel on ground
  1571→        if (this.grounded && (input.pressed('jump') || input.buffered('jump', 0.15))) {
  1572→          w.fx.endTrail(a.trail)
  1573→          this.doJump(w)
  1574→          this.state = 'move'
  1575→          return
  1576→        }
  1577→      }
  1578→    }
  1579→
  1580→    // finish — capture the swing's final rotational momentum: the body
  1581→    // carries it into locomotion (post-attack recovery, spec §7)
  1582→    if (this.animator.isDone()) {
  1583→      w.fx.endTrail(a.trail, 0.18)
  1584→      this.postAttackT = 0.34
  1585→      this.postArmVel = this.armVel
  1586→      this.action = null
  1587→      this.state = 'move'
  1588→    }
  1589→  }
